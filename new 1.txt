					WHEN param_shape.range_low_value IS NULL
					 AND param_srv.range_low_value   IS NULL     THEN 1

				END AS match_score
	
			FROM base_euro b

				ON w.warehouse_code = b.warehouse_code
			   AND w.dt_storage_end_month = b.dt_storage_end_month

			-- параметры ветвления услуги
			-- --------------------------------------------------------
			LEFT JOIN dict_dds.settings_and_parameters_sap lifnr_param
				ON lifnr_param.abap_program_code = '/RUSAL/ZLE112M'
			   AND lifnr_param.case_code = 'PERET_FRN'

				base_3.storage_cost_calculated_amount_local
				* coalesce(rates_for_1090.currency_to_multiplier, 1) * case when coalesce(rates_for_1090.currency_rate, 1) > 0 then coalesce(rates_for_1090.currency_rate, 1) else 1 end /
				(case when coalesce(rates_for_1090.currency_rate, 1) < 0 then abs(coalesce(rates_for_1090.currency_rate, 1)) else 1 end * coalesce(currency_from_multiplier, 1))
				end, 0) else 0
		end as storage_calculated_cost_over_365_amount, -- LE.001100

		null as metal_in_china_stock_balance_closing_weight -- LE.001063 Нет логики заполнения

	from pg_temp.base_3 as base_3
	left join dict_dds.currency_rates as rates_for_1090 on
		rates_for_1090.currency_rate_type_code = 'M'
		and rates_for_1090.currency_from_code = base_3.purchase_contract_position_currency_code
		and rates_for_1090.currency_to_code = 'USD'
		and rates_for_1090.dt_currency_rate = base_3.dt_report
		and rates_for_1090.deleted_flag = false
	;
	-- Вторая часть скрипта Заканчивается.



	get diagnostics v_row_cnt = row_count;
	raise notice '% [%] Insert target (%)', to_char(clock_timestamp(),'hh24:mi:ss'), (clock_timestamp() - v_last_step_finish_dttm)::interval, v_row_cnt;	
	v_last_step_finish_dttm :=  clock_timestamp();

	v_row_cnt_agg := v_row_cnt_agg + v_row_cnt;


	if v_row_cnt_agg > 0 then
		analyze dm_calc.storage_sales_bundles_amount;
		get diagnostics v_row_cnt = row_count;
		raise notice '% [%] analyze target (%)', to_char(clock_timestamp(),'hh24:mi:ss'), (clock_timestamp() - v_last_step_finish_dttm)::interval, v_row_cnt;	
		v_last_step_finish_dttm :=  clock_timestamp();
	end if;

	v_calc_time :=  clock_timestamp() - v_start_dttm ;
	raise notice '% FINISH (v_start_dttm = % ;  v_calc_time = %.)', to_char(clock_timestamp(),'hh24:mi:ss'), v_start_dttm, v_calc_time;
