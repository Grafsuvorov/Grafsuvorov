DROP TABLE IF EXISTS pp_base;
DROP TABLE IF EXISTS pp_dates;
DROP TABLE IF EXISTS cr_curr_small;
DROP TABLE IF EXISTS cr_next;
DROP TABLE IF EXISTS mlcd_agg;
DROP TABLE IF EXISTS pp_fact;
DROP TABLE IF EXISTS fact_with_dates;
DROP TABLE IF EXISTS dates;

-- =========================
-- 1) PP: тройка + дата периода
-- =========================
CREATE TEMP TABLE pp_base
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper,2,2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
FROM userdata.ckmlpp_ral p
WHERE p.untper = '000'
DISTRIBUTED BY (kalnr, bdatj, poper);

ANALYZE pp_base;

-- Уникальные даты по материалу
CREATE TEMP TABLE pp_dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT DISTINCT kalnr, dt_valid_from
FROM pp_base
DISTRIBUTED BY (kalnr, dt_valid_from);

ANALYZE pp_dates;

-- =========================
-- 2) CR текущего периода (по тройке)
-- =========================
CREATE TEMP TABLE cr_curr_small
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    vprsv, stprs::double precision AS stprs,
    peinh, pvprs::double precision AS pvprs,
    COALESCE(abprd_o,0)::double precision  AS abprd_o,
    COALESCE(abprd_mo,0)::double precision AS abprd_mo,
    COALESCE(salk3,0)::double precision    AS salk3
FROM userdata.ckmlcr_ral
WHERE untper = '000'
  AND curtp  = '10'
DISTRIBUTED BY (kalnr, bdatj, poper);

ANALYZE cr_curr_small;

-- =========================
-- 3) CR на 1-е след. месяца (только по тройкам из PP)
--    без GROUP BY, дублей нет
-- =========================
CREATE TEMP TABLE cr_next
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    c.kalnr,
    (make_date(c.bdatj::int, substring(c.poper,2,2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    c.salk3
FROM cr_curr_small c
WHERE EXISTS (
  SELECT 1
  FROM pp_base p
  WHERE p.kalnr = c.kalnr
    AND p.bdatj = c.bdatj
    AND p.poper = c.poper
)
DISTRIBUTED BY (kalnr, dt_posting);

ANALYZE cr_next;

-- =========================
-- 4) MLCD: дублей по тройке нет → без агрегации
-- =========================
CREATE TEMP TABLE mlcd_agg
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    COALESCE(salk3,0)::double precision  AS salk3,
    COALESCE(estprd,0)::double precision AS estprd,
    COALESCE(mstprd,0)::double precision AS mstprd
FROM userdata.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
DISTRIBUTED BY (kalnr, bdatj, poper);

ANALYZE mlcd_agg;

-- =========================
-- 5) DATES через LEAD (в одном проходе)
--    дистрибуция по ключу стыка
-- =========================
CREATE TEMP TABLE dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    kalnr,
    dt_valid_from,
    COALESCE(
      LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
FROM pp_dates
DISTRIBUTED BY (kalnr, dt_valid_from);

ANALYZE dates;

-- =========================
-- 6) Собираем «факт» (PP + CR + MLCD) локально по тройке
-- =========================
CREATE TEMP TABLE pp_fact
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    pp.kalnr, pp.dt_valid_from,
    pp.umkumo::double precision AS umkumo,
    pp.abkumo::double precision AS abkumo,
    pp.zukumo::double precision AS zukumo,
    pp.lbkum ::double precision AS lbkum,

    cr.vprsv,
    cr.stprs, cr.peinh, cr.pvprs,
    cr.abprd_o, cr.abprd_mo,
    cr.salk3 AS salk3_curr,

    ma.salk3  AS mlcd_salk3,
    ma.estprd AS mlcd_estprd,
    ma.mstprd AS mlcd_mstprd
FROM pp_base pp
LEFT JOIN cr_curr_small cr
  ON cr.kalnr = pp.kalnr
 AND cr.bdatj = pp.bdatj
 AND cr.poper = pp.poper
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr
 AND ma.bdatj = pp.bdatj
 AND ma.poper = pp.poper
DISTRIBUTED BY (kalnr, dt_valid_from);

ANALYZE pp_fact;

-- =========================
-- 7) Подливаем даты и salk3 следующего месяца (всё коллокировано)
-- =========================
CREATE TEMP TABLE fact_with_dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
    f.*,
    d.dt_valid_to,
    n.salk3 AS salk3_nextmonth
FROM pp_fact f
LEFT JOIN dates d
  ON d.kalnr = f.kalnr
 AND d.dt_valid_from = f.dt_valid_from
LEFT JOIN cr_next n
  ON n.kalnr = f.kalnr
 AND n.dt_posting = f.dt_valid_from
DISTRIBUTED BY (kalnr, dt_valid_from);

ANALYZE fact_with_dates;

-- =========================
-- 8) Финал (один лёгкий LEFT JOIN по kalnr)
-- =========================
-- INSERT INTO userdata.material_price_calculation (..)
SELECT
    t.kalnr AS calculation_code,
    t.dt_valid_from,
    t.dt_valid_to,

    t.vprsv  AS price_valuation_type_code,
    t.stprs  AS standard_price_amount,
    t.peinh  AS price_unit_code,
    t.pvprs  AS moving_average_price_amount,

    /* rule3 */
    CASE
      WHEN (t.umkumo + t.abkumo + t.zukumo) = 0 THEN 0
      ELSE TRUNC(
             CAST(ROUND(
                   (t.stprs / NULLIF(t.peinh,0)) * (t.umkumo + t.abkumo)
                 + COALESCE(t.abprd_o,0) + COALESCE(t.abprd_mo,0)
                 + COALESCE(t.mlcd_salk3,0) + COALESCE(t.mlcd_estprd,0) + COALESCE(t.mlcd_mstprd,0)
             , 2) AS NUMERIC(40,20))
             / NULLIF((t.umkumo + t.abkumo + t.zukumo),0), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE
      WHEN (t.abkumo + t.umkumo + t.zukumo) <> 0
      THEN TRUNC(
             CAST(
                 COALESCE(t.salk3_nextmonth,0)
               + COALESCE(t.mlcd_salk3,0) + COALESCE(t.mlcd_estprd,0) + COALESCE(t.mlcd_mstprd,0)
             AS NUMERIC(40,20))
             / NULLIF((t.abkumo + t.umkumo + t.zukumo),0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE
      WHEN t.lbkum <> 0
      THEN TRUNC(CAST(COALESCE(t.salk3_curr,0) AS NUMERIC(40,20)) / NULLIF(t.lbkum,0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    h.price_rule_code
FROM fact_with_dates t
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = t.kalnr;
