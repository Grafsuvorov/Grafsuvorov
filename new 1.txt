from typing import Any, Sequence, Dict
from airflow.hooks.base import BaseHook
from airflow.models import Variable
from airflow.providers.common.sql.operators.sql import SQLExecuteQueryOperator
from airflow.utils.context import Context

from plugins.help_utility.help_utility import get_parquet_dir_with_ts
import os
import yaml

class SqlMetaMinioOperator(SQLExecuteQueryOperator):
    # FIX: leave only 'sql' in template_fields; other fields don't need templating here
    template_fields: Sequence[str] = ("sql",)
    template_ext: Sequence[str] = (".sql",)
    ui_color = "#89d3f5"

    def __init__(
        self,
        sql: Any,
        conn_id: str = None,
        postgres_conn_id: str = None,   # FIX: bridge for YAML using postgres_conn_id
        minio_conn: str = None,
        secure: bool = False,           # FIX: default bool, not class 'bool'
        **kwargs,
    ):
        # FIX: map postgres_conn_id -> conn_id if needed
        if conn_id is None and postgres_conn_id is not None:
            conn_id = postgres_conn_id

        super().__init__(sql=sql, conn_id=conn_id, **kwargs)
        self.minio_conn = minio_conn
        self.secure = bool(secure)      # FIX: cast to bool
        # FIX: removed self.params.update() (it did nothing)

    # --- helpers -------------------------------------------------------------
    def _build_meta_from_yaml(self, p: Dict[str, Any]) -> Dict[str, Any]:
        """Read meta_data_file.yaml and build columns/settings strings."""
        meta_path = os.path.join(
            os.environ.get("AIRFLOW_HOME", "/opt/airflow"),
            "meta_info", "database", "clickhouse", "schema_name",
            p["table_schema"],
            "table" if p["table_schema"] != "dm_view" else "view",
            p["table_name"],
            "meta_data_file.yaml",
        )
        # Safety: missing file -> empty config
        yaml_data = {}
        if os.path.exists(meta_path):
            with open(meta_path, "r") as f:
                yaml_data = yaml.safe_load(f) or {}

        # FIX: build identifiers WITHOUT single quotes (were strings before)
        # If you truly need quoted identifiers, use double quotes.
        attrs = yaml_data.get("attributes") or []
        cols_ident = ",\n".join(a.get("column_name_click") for a in attrs if a.get("column_name_click"))

        # Optional settings string
        setts = yaml_data.get("settings_external_table") or {}
        settings_external_table = ""
        if setts:
            settings_external_table = "settings " + ", ".join(f"{k} = {v}" for k, v in setts.items())

        return {
            "columns": cols_ident,  # FIX: was a list of single-quoted strings -> now identifiers
            "settings_external_table": settings_external_table,
            "meta_path": meta_path,
        }

    def _get_minio_client(self) -> Minio:
        hook = BaseHook.get_connection(self.minio_conn)
        return Minio(
            f"{hook.host}:{hook.port}",
            access_key=hook.login,
            secret_key=hook.password,
            secure=self.secure,
        )

    def remove_file(self, client: Minio, load_dir: str):
        for removed_object in client.list_objects(
            bucket_name=self.params.get("minio_bucket"),
            prefix=f"{load_dir}",
            recursive=True,
        ):
            self.log.info(f"Delete {removed_object.object_name}")
            client.remove_object(
                bucket_name=self.params.get("minio_bucket"),
                object_name=removed_object.object_name,
            )

    # FIX: compute all required params BEFORE Jinja rendering happens
    def prepare_template_context(self, context: Context) -> dict:
        ctx = super().prepare_template_context(context)
        p = dict(ctx.get("params", {}))

        # Required params guard (fail early with clear message)
        required = ["table_schema", "table_name", "minio_bucket"]
        missing = [k for k in required if not p.get(k)]
        if missing:
            raise ValueError(f"Missing required params for SqlMetaMinioOperator: {missing}")

        # Cluster flags (from Airflow Variable 'cluster')
        cluster_var = Variable.get("cluster", "dev")
        p["cluster"] = "PROD" if cluster_var.lower() == "prod" else "DEV"
        p["IS_PROD"] = (p["cluster"] == "PROD")

        # Timestamped directory used by templates (replacement for parquet_date())
        p["ts_date"] = get_parquet_dir_with_ts()  # e.g., 1759985153440

        # Read YAML -> columns + settings
        meta = self._build_meta_from_yaml(p)
        p.update(meta)

        ctx["params"] = p
        return ctx

    def execute(self, context: Context) -> Any:
        # Minio housekeeping uses params prepared above
        cluster_prefix = "gp_prod" if self.params.get("cluster") == "PROD" else "gp_dev"
        ts_date = self.params.get("ts_date")
        minio_dir = f"{cluster_prefix}/{self.params.get('table_schema')}/{self.params.get('table_name')}/{ts_date}"

        client = self._get_minio_client()
        # drop current partition dir (if exists)
        self.remove_file(client=client, load_dir=minio_dir)

        # keep last two partitions only
        prefix = f"{cluster_prefix}/{self.params.get('table_schema')}/{self.params.get('table_name')}/"
        load_obj = client.list_objects(bucket_name=self.params.get("minio_bucket"), prefix=prefix)
        load_dirs = sorted((x.object_name for x in load_obj), reverse=True)
        if len(load_dirs) > 2:
            for load_dir in load_dirs[2:]:
                self.remove_file(client=client, load_dir=load_dir)

        # Execute SQL via parent (templates are now safe to render)
        return super().execute(context=context)
