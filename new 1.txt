# -*- coding: utf-8 -*-
"""
Быстрая выгрузка SELECT в CSV с разделителем ';' (опционально .gz) через server-side COPY.
Подходит для больших выборок (миллионы строк).
"""

import os
import time
import gzip
import psycopg2

# --- НАСТРОЙКИ ПОД СЕБЯ ---
DB_HOST = "127.0.0.1"
DB_PORT = 5432
DB_NAME = "dwh"
DB_USER = "postgres"
DB_PASSWORD = "0506"

# Куда сохранять
OUTPUT_PATH = r"C:\temp\account_debt_2025.csv"   # или "C:/temp/account_debt_2025.csv"
COMPRESS_GZIP = False                             # True -> запишет .csv.gz (меньше места)

# Сам запрос (можно менять при необходимости)
QUERY = """
SELECT *
FROM dm.account_debt a
WHERE a.dt >= DATE '2025-01-01'
"""

# --- НЕ ТРОГАТЬ НИЖЕ, ЕСЛИ НЕ НУЖНО ---
COPY_SQL = f"""
COPY (
    {QUERY}
) TO STDOUT
WITH CSV
DELIMITER ';'
HEADER
NULL ''
QUOTE '"'
ESCAPE '"'
"""

def main():
    t0 = time.time()
    os.makedirs(os.path.dirname(OUTPUT_PATH), exist_ok=True)

    conn = psycopg2.connect(
        host=DB_HOST,
        port=DB_PORT,
        dbname=DB_NAME,
        user=DB_USER,
        password=DB_PASSWORD,
        application_name="csv_export_account_debt"
    )
    conn.set_session(readonly=True, autocommit=True)
    conn.set_client_encoding("UTF8")

    try:
        with conn.cursor() as cur:
            if COMPRESS_GZIP:
                # Пишем в gzip (txt-режим, чтобы copy_expert писал строки)
                gz_path = OUTPUT_PATH if OUTPUT_PATH.endswith(".gz") else OUTPUT_PATH + ".gz"
                with gzip.open(gz_path, mode="wt", encoding="utf-8", newline="") as f:
                    cur.copy_expert(COPY_SQL, f)
                out_path = gz_path
            else:
                # Обычный CSV
                with open(OUTPUT_PATH, mode="w", encoding="utf-8", newline="") as f:
                    cur.copy_expert(COPY_SQL, f)
                out_path = OUTPUT_PATH
    finally:
        conn.close()

    dt = time.time() - t0
    print(f"[OK] Экспорт завершён: {out_path}")
    print(f"[INFO] Время: {dt:.1f} сек")

if __name__ == "__main__":
    main()
