	'A' as non_liquid_asset_type_code,
	'Основные средства' as non_liquid_asset_type_name,
	fa.dt_report,
	fa.acquisition_cost_cumulative_amount,
	fa.depreciation_typical_cumulative_amount + fa.depreciation_special_cumulative_amount + fa.depreciation_unplanned_cumulative_amount as depreciation_total_cumulative_amount,
	fa.depreciation_typical_cumulative_amount,
	fa.depreciation_special_cumulative_amount,
	fa.depreciation_unplanned_cumulative_amount
from
	(
	
	select
		fao.unit_balance_code,
		fao.asset_main_code,
		fao.asset_sub_code,
		fao.valuation_area_code,
		dt_end_of_month as dt_report,
		coalesce(max(cumulative_acquisition_and_production_cost_amount),0) + coalesce(sum(acquisition_cost_valuation_area_currency_amount),0) as acquisition_cost_cumulative_amount,
		coalesce(max(cumulative_depreciation_typical_amount),0) + coalesce(sum(depreciation_typical_amount),0) as depreciation_typical_cumulative_amount,
		coalesce(max(cumulative_depreciation_special_amount),0) + coalesce(sum(depreciation_special_amount),0) as depreciation_special_cumulative_amount,
		coalesce(max(cumulative_depreciation_unplanned_amount),0) + coalesce(sum(depreciation_unplanned_amount),0) as depreciation_unplanned_cumulative_amount
	from
		dm_calc.unit_balance_operating_periods rt 
	join  dm_calc.fixed_asset_operations fao on
		fao.unit_balance_code = rt.unit_balance_code
	--where 
	and fao.dt_posting <= dt_end_of_month
		and fao.dt_posting >= date_trunc('year',dt_end_of_month)::date
		and dt_end_of_month > date_trunc('year', now()- interval '1 year')::date
		-->'2024-01-01'::date
	group by
		fao.unit_balance_code,
		fao.asset_main_code,
		fao.asset_sub_code,
		fao.valuation_area_code,
		dt_end_of_month
) fa
join dm.fixed_asset_main fam on
	fam.unit_balance_code = fa.unit_balance_code
	and fam.asset_main_code = fa.asset_main_code
	and fam.asset_sub_code = fa.asset_sub_code
	and fam.valuation_area_code = fa.valuation_area_code
	and fa.dt_report between fam.dt_valid_from and fam.dt_valid_to;
