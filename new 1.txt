-- Параметры
-- :base_entity  — ваша сущность (например, 36)
-- :run_date     — интересующая дата в мониторинге
-- примеры: \set base_entity 36
--          \set run_date '2025-10-20'

with start_targets as (
    select distinct d.target_full_name
    from tech_etl.tables_dependencies d
    where d.entity_id = :base_entity
      and d.target_full_name is not null
)
select distinct
    d2.entity_id                        as affected_entity_id,
    coalesce(em.entity_name, '—')       as affected_entity_name,
    d2.target_schema,
    d2.target_table
from start_targets st
join tech_etl.tables_dependencies d2
  on d2.source_full_name = st.target_full_name
left join tech_etl.entities_meta em
  on em.entity_id = d2.entity_id
join tech_monitoring.table_compare tc         -- фильтруем только нужную дату
  on tc.table_schema = d2.target_schema
 and tc.table_name  = d2.target_table
 and tc.date_id     = :run_date
-- если нужно отфильтровать по «окну» времени (пример 03:00–03:30), раскомментируйте:
-- and tc.curr_start_dttm::time <  time '03:30'
-- and coalesce(tc.curr_finish_dttm::time, time '23:59') >= time '03:00'
order by affected_entity_id, affected_entity_name, d2.target_schema, d2.target_table;



-- Параметры
-- :base_entity  — ваша сущность (например, 36)
-- :run_date     — интересующая дата в мониторинге

with start_targets as (
    select distinct d.target_full_name
    from tech_etl.tables_dependencies d
    where d.entity_id = :base_entity
      and d.target_full_name is not null
),
rec as (
    -- уровень 1
    select
        1 as lvl,
        d2.entity_id                 as affected_entity_id,
        d2.target_schema,
        d2.target_table,
        d2.target_full_name
    from start_targets st
    join tech_etl.tables_dependencies d2
      on d2.source_full_name = st.target_full_name
    join tech_monitoring.table_compare tc
      on tc.table_schema = d2.target_schema
     and tc.table_name  = d2.target_table
     and tc.date_id     = :run_date
    -- фильтр по времени — опционально:
    -- and tc.curr_start_dttm::time <  time '03:30'
    -- and coalesce(tc.curr_finish_dttm::time, time '23:59') >= time '03:00'

    union all

    -- уровни 2+
    select
        r.lvl + 1,
        d2.entity_id,
        d2.target_schema,
        d2.target_table,
        d2.target_full_name
    from rec r
    join tech_etl.tables_dependencies d2
      on d2.source_full_name = r.target_full_name
    join tech_monitoring.table_compare tc
      on tc.table_schema = d2.target_schema
     and tc.table_name  = d2.target_table
     and tc.date_id     = :run_date
    -- фильтр по времени — опционально:
    -- and tc.curr_start_dttm::time <  time '03:30'
    -- and coalesce(tc.curr_finish_dttm::time, time '23:59') >= time '03:00'
)
select distinct
    r.lvl,
    r.affected_entity_id,
    coalesce(em.entity_name, '—') as affected_entity_name,
    r.target_schema,
    r.target_table
from rec r
left join tech_etl.entities_meta em
  on em.entity_id = r.affected_entity_id
order by r.lvl, r.affected_entity_id, r.target_schema, r.target_table;
