from typing import Sequence

import clickhouse_driver
from airflow.hooks.base import BaseHook
from airflow.providers.common.sql.operators.sql import SQLExecuteQueryOperator


class ClickhouseOperator(SQLExecuteQueryOperator):
    template_fields: Sequence[str] = ("sql", "parameters", *SQLExecuteQueryOperator.template_fields)
    template_ext: Sequence[str] = (".sql", ".json")

    ui_color = "#Edaaed"

    def __init__(
            self,
            sql,
            conn_id: str,
            **kwargs
    ):
        super().__init__(conn_id=conn_id, sql=sql, **kwargs)

    def execute(self, context):
        self.log.info("Executing: %s", self.sql)
        if not isinstance(self.sql, list):
            self.sql = [self.sql]
        hook = BaseHook.get_connection(self.conn_id)
        with clickhouse_driver.connect(
                database=hook.schema,
                user=hook.login,
                password=hook.password,
                host=hook.host,
                port=hook.port,
                secure=None,
        ) as conn:
            with conn.cursor() as cur:
                for v_sql in self.sql:
                    self.logger().info(v_sql)
                    cur.execute(v_sql)
