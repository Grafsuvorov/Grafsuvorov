drop table if exists pg_temp_bill_step1;
create temporary table pg_temp_bill_step1 as (
    select
        a1.delivery_code,
        a2.bill_of_lading_code
    from (
        select
            perw.delivery_code,
            sbdr1.sales_bundle_code
        from dds.transportation_service_invoice_subposition as perw
        inner join dds.transportation_service_invoice_header as perh
            on
                perw.plant_code = perh.plant_code
                and perw.transportation_service_invoice_code = perh.transportation_service_invoice_code
        left join dds.bill_of_lading as vbsk_src
            on
                vbsk_src.bill_of_lading_code = perw.bill_of_lading_code
        inner join dds.sales_bundle_and_delivery_relationship as sbdr1
            on
                sbdr1.delivery_code = perw.delivery_code
        where
            (vbsk_src.bill_of_lading_type_code = 'Й' or perw.bill_of_lading_code is null)
            and perh.dt_transportation_service_invoice >= '2023-01-01'
            and sbdr1.is_deleted_code is null
        group by
            perw.delivery_code,
            sbdr1.sales_bundle_code
    ) as a1
    inner join (
        select
            sbdr2.sales_bundle_code,
            sbdr2.delivery_code,
            vbsk_trg.bill_of_lading_code
        from dds.sales_bundle_and_delivery_relationship as sbdr2		--по химиям ищем полный список поставок
        inner join dds.delivery_document_header as likp
            on
                likp.delivery_code = sbdr2.delivery_code
                and likp.delivery_type_code = 'EL'
        inner join dds.logistics_document_and_delivery_relationship as vbss
            on
                vbss.delivery_code = likp.delivery_code
        inner join dds.bill_of_lading as vbsk_trg
            on
                vbsk_trg.bill_of_lading_code = vbss.logistics_document_code
                and vbsk_trg.bill_of_lading_type_code = 'Y'
        where sbdr2.is_deleted_code is null
        group by
            sbdr2.sales_bundle_code,
            sbdr2.delivery_code,
            vbsk_trg.bill_of_lading_code
    ) as a2
        on
            a1.sales_bundle_code = a2.sales_bundle_code
    group by
        a1.delivery_code,
        a2.bill_of_lading_code
)
distributed by (bill_of_lading_code);

drop table if exists pg_temp_bill_step2;
create temporary table pg_temp_bill_step2 as (
    select
        bill_step1.delivery_code,
        bill_step1.bill_of_lading_code,
        vbsk_trg.bill_of_lading_number,
        nom.vessel_number,
        tv.vessel_name,
        row_number() over (partition by bill_step1.delivery_code order by bill_step1.bill_of_lading_code) as rwn
    from pg_temp_bill_step1 as bill_step1
    left join dds.bill_of_lading as vbsk_trg
        on
            vbsk_trg.bill_of_lading_code = bill_step1.bill_of_lading_code
    left join dds.nomination as nom
        on
            nom.nomination_code = vbsk_trg.nomination_code
    left join dict_dds.transport_vehicle_texts as tv on
        tv.vessel_code = nom.vessel_number
        and tv.language_code = 'R'
)
distributed by (delivery_code);

drop table if exists pg_temp_bill_search;
create temporary table pg_temp_bill_search as (
    with cte as (
        select
            delivery_code,
            string_agg(bill_of_lading_code, ',' order by rwn) as bill_of_lading_compound_code,
            string_agg(bill_of_lading_number, ',' order by rwn) as bill_of_lading_compound_number,
            vessel_number,
            vessel_name,
            row_number() over (partition by delivery_code order by vessel_number) as rwn_nom
        from pg_temp_bill_step2
        group by
            delivery_code,
            vessel_number,
            vessel_name
    )
 
    select
        delivery_code,
        string_agg(bill_of_lading_compound_code, ',' order by rwn_nom) as bill_of_lading_compound_code,
        string_agg(bill_of_lading_compound_number, ',' order by rwn_nom) as bill_of_lading_compound_number,
        string_agg(vessel_number, ',' order by rwn_nom) as vessel_compound_number,
        string_agg(vessel_name, ',' order by rwn_nom) as vessel_compound_name
    from cte
    group by
        delivery_code
)
distributed by (delivery_code);
