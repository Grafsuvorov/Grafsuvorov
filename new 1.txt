-- 1. Таблица в ClickHouse
DROP TABLE IF EXISTS dm.sales_alverse_mlc2 ON CLUSTER dwhcluster SYNC;

CREATE TABLE IF NOT EXISTS dm.sales_alverse_mlc2 ON CLUSTER dwhcluster (
    batch String COMMENT '...',
    sales_order_in_shipment String NULL,
    plant_producer_code String NULL,
    plant_producer_name String NULL,
    port_of_loading_name String NULL,
    dt_shipment Date NULL,
    dt_arrival_by_railway Date NULL,
    dt_forwarder Date NULL,
    material_aggr_name String NULL,
    material_group_code String NULL,
    forwarder_name String NULL,
    dt_warehouse Date NULL,
    transport_railcar_type_name String NULL,
    weight_gross Decimal(13,3) NULL,
    weight_net Decimal(13,3) NULL,
    weight_net_with_wirerod Decimal(13,3) NULL,
    customer_name String NULL,
    contract_name String NULL,
    dt_bill_of_lading Date NULL
)
ENGINE = ReplicatedMergeTree('/clickhouse/tables/{shard}/dm/"sales_alverse_mlc2"', '{replica}')
ORDER BY (batch)
COMMENT 'MLC (Alverse)';

-- 2. WRITABLE external table в Greenplum
DROP EXTERNAL TABLE IF EXISTS raw_ext.sales_alverse_mlc_write2;

CREATE WRITABLE EXTERNAL TABLE raw_ext.sales_alverse_mlc_write2 (
    batch text,
    sales_order_in_shipment text,
    plant_producer_code text,
    plant_producer_name text,
    port_of_loading_name text,
    dt_shipment date,
    dt_arrival_by_railway date,
    dt_forwarder date,
    material_aggr_name text,
    material_group_code text,
    forwarder_name text,
    dt_warehouse date,
    transport_railcar_type_name text,
    weight_gross numeric,
    weight_net numeric,
    weight_net_with_wirerod numeric,
    customer_name text,
    contract_name text,
    dt_bill_of_lading date
)
LOCATION (
  'pxf://dm.sales_alverse_mlc2
    ?PROFILE=JDBC
    &SERVER=clickhouse_prod
    &QUOTE_COLUMNS=true
    &BATCH_SIZE=10000
    &POOL_SIZE=6'
)
ON ALL
FORMAT 'CUSTOM' (FORMATTER='pxfwritable_export')
ENCODING 'UTF8';

-- 3. Загрузка данных из dm.sales_alverse_mlc в ClickHouse через external table
INSERT INTO raw_ext.sales_alverse_mlc_write2 (
    batch,
    sales_order_in_shipment,
    plant_producer_code,
    plant_producer_name,
    port_of_loading_name,
    dt_shipment,
    dt_arrival_by_railway,
    dt_forwarder,
    material_aggr_name,
    material_group_code,
    forwarder_name,
    dt_warehouse,
    transport_railcar_type_name,
    weight_gross,
    weight_net,
    weight_net_with_wirerod,
    customer_name,
    contract_name,
    dt_bill_of_lading
)
SELECT
    batch::text,
    sales_order_in_shipment::text,
    plant_producer_code::text,
    plant_producer_name::text,
    port_of_loading_name::text,
    tech_etl.validate_date(dt_shipment),
    tech_etl.validate_date(dt_arrival_by_railway),
    tech_etl.validate_date(dt_forwarder),
    material_aggr_name::text,
    material_group_code::text,
    forwarder_name::text,
    tech_etl.validate_date(dt_warehouse),
    transport_railcar_type_name::text,
    weight_gross,
    weight_net,
    weight_net_with_wirerod,
    customer_name::text,
    contract_name::text,
    tech_etl.validate_date(dt_bill_of_lading)
FROM dm.sales_alverse_mlc
WHERE 1 = 1;
