console.log("🎄 Holiday theme JS loaded");

function initSnow() {
    if (!document.body || document.querySelector(".snowflakes")) {
        return;
    }

    const snow = document.createElement("div");
    snow.className = "snowflakes";
    snow.style.position = "fixed";
    snow.style.top = "0";
    snow.style.left = "0";
    snow.style.width = "100vw";
    snow.style.height = "100vh";
    snow.style.pointerEvents = "none";
    snow.style.zIndex = "9999";

    document.body.appendChild(snow);

    for (let i = 0; i < 60; i++) {
        const flake = document.createElement("div");
        flake.className = "snowflake";
        flake.innerHTML = "❄";
        snow.appendChild(flake);
    }

    console.log("❄️ Snow initialized");
}

// Airflow React UI → ждём
setTimeout(initSnow, 1500);


systemctl restart airflow-webserver
