-- 3.1. Уникальный мэппинг address_code -> country_code
CREATE TEMP TABLE adrc
ON COMMIT DROP AS
SELECT
    address_code,
    MIN(country_code) AS country_code
FROM dict_dds.address
GROUP BY address_code
DISTRIBUTED REPLICATED;

-- 3.2. Хабы с географией (страна + регион рынка)
CREATE TEMP TABLE dim_hub_geo
ON COMMIT DROP AS
SELECT
    th.transport_hub_code,
    a.country_code,
    ct.country_full_name,
    mr.market_region1_name
FROM dict_dds.transport_hub th
LEFT JOIN adrc a
       ON a.address_code = th.address_code
LEFT JOIN dict_dds.country_texts ct
       ON ct.country_code   = a.country_code
      AND ct.language_code  = 'R'
LEFT JOIN dict_dds.country c
       ON c.country_code = a.country_code
LEFT JOIN dict_dds.market_region1_texts mr
       ON mr.market_region1_code = c.market_region1_code
      AND mr.language_code       = 'R'
DISTRIBUTED REPLICATED;

-- 3.3. Агрегированные тексты хабов (R/E в одну строку)
CREATE TEMP TABLE dim_hub_texts_agg
ON COMMIT DROP AS
SELECT
    transport_hub_code,
    MAX(CASE WHEN language_code = 'R' THEN transport_hub_name END) AS name_r,
    MAX(CASE WHEN language_code = 'E' THEN transport_hub_name END) AS name_e
FROM dict_dds.transport_hub_texts
GROUP BY transport_hub_code
DISTRIBUTED REPLICATED;

-- 3.4. Справочник материалов (R)
CREATE TEMP TABLE dim_material_texts
ON COMMIT DROP AS
SELECT
    material_code,
    MAX(material_name) AS material_name
FROM dict_dds.material_texts
WHERE language_code = 'R'
GROUP BY material_code
DISTRIBUTED REPLICATED;

-- 3.5. Справочник формы металла по material_code (через спецификацию)
CREATE TEMP TABLE dim_material_shape
ON COMMIT DROP AS
SELECT
    ms.material_code,
    MAX(mf.material_shape_full_name) AS material_shape_full_name
FROM dict_dds.material_specification ms
JOIN dict_dds.material_shape_texts mf
      ON mf.shape_code    = ms.shape_code
     AND mf.language_code = 'R'
GROUP BY ms.material_code
DISTRIBUTED REPLICATED;

-- 3.6. Тип подвижного состава (R)
CREATE TEMP TABLE dim_transport_type
ON COMMIT DROP AS
SELECT
    transport_transfer_type_code,
    MAX(transport_transfer_type_name) AS transport_transfer_type_name
FROM dict_dds.transport_transfer_type_texts
WHERE language_code = 'R'
GROUP BY transport_transfer_type_code
DISTRIBUTED REPLICATED;

-- 3.7. Тексты услуг транспортировки (R)
CREATE TEMP TABLE dim_service_texts
ON COMMIT DROP AS
SELECT
    transportation_service_code AS service_code,
    MAX(transportation_service_name) AS transportation_service_name
FROM dict_dds.transportation_service_texts
WHERE language_code = 'R'
GROUP BY transportation_service_code
DISTRIBUTED REPLICATED;

-- 3.8. Собственник металла по поставке хранения
--     (Берём plant_producer_code по позиции 000010, агрегируем MIN)
CREATE TEMP TABLE dim_storage_owner
ON COMMIT DROP AS
SELECT
    delivery_code,
    MIN(plant_producer_code) AS plant_code
FROM dds.delivery_document_position
WHERE delivery_position_line_item_code = '000010'
GROUP BY delivery_code
DISTRIBUTED REPLICATED;

                                                           AS service_search_name,

    base.purchase_contract_position_uom_code,

    -- Кредитор по инструкции
    addr_8.address_full_name                               AS creditor_in_shipment_instruction_name,
    base.creditor_in_shipment_instruction_code || '-' || addr_8.address_full_name
                                                           AS creditor_in_shipment_instruction_search_name,

    -- Параметры длительности хранения
    base.storage_duration_free_by_contract_calendar_days,
    base.storage_duration_free_by_delivery_calendar_days,
    base.storage_duration_payable_calendar_days,

    -- Удалённый склад
    base.remote_warehouse_code,
    CASE
        WHEN hub_remote.country_full_name = 'Россия'
            THEN COALESCE(ht_remote.name_r, ht_remote.name_e, 'Не определено')
        ELSE COALESCE(ht_remote.name_e, ht_remote.name_r, 'Не определено')
    END                                                      AS remote_warehouse_name,
    hub_remote.country_code                                  AS country_of_remote_warehouse_code,

    -- Даты этапов
    base.dt_transportation_stage_start_p,
    base.dt_transportation_stage_start_r,
    base.dt_shipment_from_foreign_warehouse,

    -- Стоимость хранения
    base.storage_cost_calculated_amount,
    base.storage_cost_calculated_amount_local,

    -- Объем платного хранения по диапазонам
    base.storage_payable_001_030_days_gross_weight,
    base.storage_payable_031_060_days_gross_weight,
    base.storage_payable_061_090_days_gross_weight,
    base.storage_payable_091_180_days_gross_weight,
    base.storage_payable_181_365_days_gross_weight,
    base.storage_payable_over_365_days_gross_weight,

    -- Признак порта РФ
    base.is_stored_in_russian_port_code,

    -- Тип склада / тип фрахта
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN 'Терминал порта'
        WHEN base.storage_cost_calculation_type_name = 'Логика СВХ'
            THEN 'Склад СВХ'
        ELSE 'Не определено'
    END                                                      AS type_warehouse_name,
    CASE
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-контейнеры' THEN 'Контейнеры'
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-балк'       THEN 'Балк'
        ELSE 'Не определено'
    END                                                      AS type_freight_name,

    -- Собственник металла для отчётности (через dim_storage_owner)
    pas_1124.plant_short_name || ' ' || pas_1124.plant_full_name
                                                           AS metal_owner_for_reporting_name,

    -- Стоимости по диапазонам
    base.storage_calculated_cost_001_030_amount,
    base.storage_calculated_cost_031_060_amount,
    base.storage_calculated_cost_061_090_amount,
    base.storage_calculated_cost_091_180_amount,
    base.storage_calculated_cost_181_365_amount,
    base.storage_calculated_cost_over_365_amount,

    -- Китайский остаток
    base.metal_in_china_stock_balance_closing_weight
FROM dm_calc.storage_sales_bundles_amount AS base
    -- Материалы / форма
    LEFT JOIN dim_material_texts   AS mat
           ON mat.material_code = base.material_code
    LEFT JOIN dim_material_shape   AS dms
           ON dms.material_code = base.material_code

    -- ЕТСНГ
    LEFT JOIN dict_dds.etsng_texts AS ette
           ON ette.etsng_code    = base.etsng_code
          AND ette.language_code = 'R'

    -- Тип ПС
    LEFT JOIN dim_transport_type  AS dtt
           ON dtt.transport_transfer_type_code = base.transport_type_code

    -- Услуги
    LEFT JOIN dim_service_texts   AS svc
           ON svc.service_code = base.service_code

    -- Завод поставки
    LEFT JOIN dict_dds.plant_and_subsidiary AS pas
           ON pas.plant_code = base.receiving_plant_code

    -- Собственник металла через поставку хранения
    LEFT JOIN dim_storage_owner AS so
           ON so.delivery_code = base.delivery_for_storage_calculation_code
    LEFT JOIN dict_dds.plant_and_subsidiary AS pas_1124
           ON pas_1124.plant_code = so.plant_code

    -- Гео по складу
    LEFT JOIN dim_hub_geo AS hub_wh
           ON hub_wh.transport_hub_code = base.warehouse_code

    -- Гео по удалённому складу
    LEFT JOIN dim_hub_geo AS hub_remote
           ON hub_remote.transport_hub_code = base.remote_warehouse_code

    -- Тексты хабов (агрегированные)
    LEFT JOIN dim_hub_texts_agg AS ht_dep
           ON ht_dep.transport_hub_code = base.transport_departure_hub_code
    LEFT JOIN dim_hub_texts_agg AS ht_dest
           ON ht_dest.transport_hub_code = base.transport_destination_hub_code
    LEFT JOIN dim_hub_texts_agg AS ht_wh
           ON ht_wh.transport_hub_code = base.warehouse_code
    LEFT JOIN dim_hub_texts_agg AS ht_area
           ON ht_area.transport_hub_code = base.warehouse_storage_area_code
    LEFT JOIN dim_hub_texts_agg AS ht_remote
           ON ht_remote.transport_hub_code = base.remote_warehouse_code
    LEFT JOIN dim_hub_texts_agg AS ht_port
           ON ht_port.transport_hub_code = base.port_of_discharge_code

    -- Экспедитор
    LEFT JOIN dict_dds.counterparty AS counter_forwarder
           ON counter_forwarder.counterparty_code = base.forwarder_code

    -- Кредитор (для имени)
    LEFT JOIN dict_dds.counterparty AS counter_creditor
           ON counter_creditor.counterparty_code = base.creditor_in_shipment_instruction_code
    LEFT JOIN dict_dds.address AS addr_8
           ON addr_8.address_code = counter_creditor.address_code
          AND addr_8.international_display_format_code IS NULL

    -- Этап перевозки
    LEFT JOIN dict_dds.transportation_stage_texts AS trst
           ON trst.transportation_stage_code = base.transportation_stage_code
          AND trst.language_code            = 'R'
;
