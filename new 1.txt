WITH RECURSIVE lineage AS (
    SELECT
        d.parent_table_id,
        d.child_table_id,
        1 AS depth
    FROM tech_etl.table_dependencies d
    WHERE d.parent_table = 'dm.sb_wuc'

    UNION ALL

    SELECT
        d.parent_table_id,
        d.child_table_id,
        l.depth + 1
    FROM tech_etl.table_dependencies d
    JOIN lineage l
      ON l.child_table_id = d.parent_table_id
    WHERE l.depth < 10
)
SELECT DISTINCT
    cm.table_schema AS child_schema,
    pm.table_schema AS parent_schema
FROM lineage l
JOIN tech_etl.tables_meta pm ON pm.table_id = l.parent_table_id
JOIN tech_etl.tables_meta cm ON cm.table_id = l.child_table_id
WHERE cm.table_schema <> pm.table_schema   -- 🔥 ВАЖНО
ORDER BY
    CASE cm.table_schema
        WHEN 'raw' THEN 1
        WHEN 'raw_ext' THEN 1
        WHEN 'stg' THEN 2
        WHEN 'dict_stg' THEN 2
        WHEN 'ods' THEN 3
        WHEN 'dds' THEN 4
        WHEN 'dict_dds' THEN 4
        WHEN 'dm_calc' THEN 5
        WHEN 'dm' THEN 6
        ELSE 99
    END;
