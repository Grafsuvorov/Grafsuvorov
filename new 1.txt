CREATE TEMP TABLE for_1079_map
ON COMMIT DROP AS
WITH fb AS (
    -- Предфильтруем for_1079_base максимально рано
    SELECT
        purchase_contract_code,
        position_line_item_code,
        bukrs,
        lifnr,
        zzport,
        werks,
        zzsrvpos,
        zzforma,
        zzet_tarif,
        sdabw,
        dt_valid_from,
        dt_valid_to,
        datab,
        datbi,
        konws,
        service_code,
        price_uom_code,
        currency_dogovor,
        purchase_document_subtype_code,
        tmp_1,
        tmp_2,
        order_rf,
        order_asia,
        order_euro_svh,
        kopos,
        kbetr,
        netpr,
        zzterminal,
        vzzzone
    FROM for_1079_base
    WHERE (
        bukrs IN ('1120','1121')  -- РФ/Европа/Азия/СВХ
        OR bukrs IS NULL
    )
),
bsw AS (
    SELECT *
    FROM base_sw
    WHERE storage_cost_calculation_type_name IN
        ('Логика РФ-контейнеры',
         'Логика РФ-балк',
         'Логика Азия',
         'Логика Европа',
         'Логика СВХ')
),
match_rules AS (
    SELECT
        bsw.rn,
        fb.purchase_contract_code,
        fb.position_line_item_code,
        fb.konws,
        fb.service_code,
        fb.price_uom_code,
        fb.currency_dogovor,
        fb.datab,
        fb.datbi,

        /* EURO FIELDS — только для Европы */
        CASE
            WHEN bsw.storage_cost_calculation_type_name = 'Логика Европа'
                THEN CASE
                        WHEN EXTRACT(day FROM bsw.dt_storage_end) < EXTRACT(day FROM bsw.dt_storage_start)
                             THEN (EXTRACT(year FROM bsw.dt_storage_end) - EXTRACT(year FROM bsw.dt_storage_start)) * 12
                                  + (EXTRACT(month FROM bsw.dt_storage_end) - EXTRACT(month FROM bsw.dt_storage_start))
                        ELSE (EXTRACT(year FROM bsw.dt_storage_end) - EXTRACT(year FROM bsw.dt_storage_start)) * 12
                             + (EXTRACT(month FROM bsw.dt_storage_end) - EXTRACT(month FROM bsw.dt_storage_start)) + 1
                     END
        END AS euro_month_difference_2,

        CASE
            WHEN bsw.storage_cost_calculation_type_name = 'Логика Европа'
                THEN CASE
                        WHEN bsw.storage_cost_special_calculation_type_code = '4'
                             THEN CEIL(bsw.storage_duration_total_calendar_days / 7.0)
                        WHEN bsw.storage_cost_special_calculation_type_code = '5'
                             THEN CEIL(bsw.storage_duration_total_calendar_days / 30.0)
                     END
        END AS euro_special,

        /* Приоритеты логик */
        CASE
            WHEN bsw.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'
                THEN 10
            WHEN bsw.storage_cost_calculation_type_name = 'Логика РФ-балк'
                AND fb.zzsrvpos = bsw.service_number THEN 20
            WHEN bsw.storage_cost_calculation_type_name = 'Логика РФ-балк'
                THEN 30
            WHEN bsw.storage_cost_calculation_type_name = 'Логика Азия'
                THEN 40
            WHEN bsw.storage_cost_calculation_type_name = 'Логика Европа'
                THEN 50
            WHEN bsw.storage_cost_calculation_type_name = 'Логика СВХ'
                THEN 60
        END AS priority,

        /* Флаг совпадений условий для сортировки */
        (
            (CASE WHEN fb.zzport = bsw.port_of_discharge_code OR fb.zzport IS NULL THEN 1 ELSE 0 END) +
            (CASE WHEN fb.zzforma = bsw.material_shape_code OR fb.zzforma IS NULL THEN 1 ELSE 0 END) +
            (CASE WHEN fb.zzet_tarif = bsw.etsng_code OR fb.zzet_tarif IS NULL THEN 1 ELSE 0 END) +
            (CASE WHEN fb.sdabw = bsw.transport_type_code OR fb.sdabw IS NULL THEN 1 ELSE 0 END) +
            (CASE WHEN fb.zzterminal = bsw.warehouse_code OR fb.zzterminal IS NULL THEN 1 ELSE 0 END) +
            (CASE WHEN fb.vzzzone = bsw.warehouse_storage_area_code OR fb.vzzzone IS NULL THEN 1 ELSE 0 END)
        ) AS match_score

    FROM bsw
    JOIN fb
        ON fb.lifnr = bsw.forwarder_code
       AND bsw.dt_for_price_search_in_purchase_contract
              BETWEEN fb.dt_valid_from AND fb.dt_valid_to
       AND bsw.dt_for_price_search_in_purchase_contract
              BETWEEN fb.datab AND fb.datbi

       /* Логики */
       AND (
            (bsw.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'
             AND fb.bukrs = '1120'
             AND fb.werks = '1121'
             AND fb.zzsrvpos = 'S49')

            OR (bsw.storage_cost_calculation_type_name = 'Логика РФ-балк'
                AND fb.bukrs = '1120'
                AND fb.werks = '1121')

            OR (bsw.storage_cost_calculation_type_name = 'Логика Азия'
                AND fb.bukrs = bsw.unit_balance_code
                AND fb.werks = bsw.plant_code)

            OR (bsw.storage_cost_calculation_type_name = 'Логика Европа'
                AND fb.purchase_document_subtype_code = 'ZDTL'
                AND fb.bukrs = bsw.unit_balance_code
                AND fb.werks = bsw.plant_code)

            OR (bsw.storage_cost_calculation_type_name = 'Логика СВХ'
                AND fb.bukrs = '1120'
                AND fb.werks = '1121'
                AND fb.purchase_document_subtype_code = 'ZDTL')
       )
)
SELECT DISTINCT ON (rn)
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    euro_month_difference_2,
    euro_special
FROM match_rules
ORDER BY rn, priority, match_score DESC, konws DESC, datab DESC
DISTRIBUTED BY (rn);
