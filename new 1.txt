INSERT INTO userdata.material_price_calculation
WITH
pp AS (                         -- базовый PP: 1 строка на (kalnr,bdatj,poper)
  SELECT
    (kalnr)::text AS kalnr_text,
    bdatj::text   AS bdatj,
    poper::text   AS poper,
    to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD')        AS dt_valid_from,
    (to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD') + INTERVAL '1 month')::date AS dt_next_cal  -- 1-е след. месяца (календарно)
  FROM userdata.ckmlpp_ral
  WHERE untper='000'
),
/* быстрый путь: попытаться взять СЛЕДУЮЩИЙ КАЛЕНДАРНЫЙ период как следующий */
pp_next_cal AS (                -- есть ли у этого kalnr_period след. календарный месяц?
  SELECT p1.kalnr_text, p1.dt_valid_from,
         p2.dt_valid_from AS dt_next_found
  FROM pp p1
  LEFT JOIN pp p2
    ON p2.kalnr_text   = p1.kalnr_text
   AND p2.dt_valid_from = p1.dt_next_cal
),
/* медленный, но редкий фоллбэк: если календарного нет, берём ближайший ДОСТУПНЫЙ будущий */
pp_next_fallback AS (           -- MIN будущей даты > текущей в рамках того же kalnr
  SELECT p1.kalnr_text, p1.dt_valid_from,
         MIN(p2.dt_valid_from) AS dt_next_found
  FROM pp p1
  JOIN pp p2
    ON p2.kalnr_text   = p1.kalnr_text
   AND p2.dt_valid_from > p1.dt_valid_from
  GROUP BY p1.kalnr_text, p1.dt_valid_from
),
/* единый маппер следующего периода: сначала календарный, иначе фоллбэк */
pp_ranges AS (
  SELECT
    p.kalnr_text,
    p.bdatj, p.poper,
    p.dt_valid_from,
    /* старая логика:
       - если есть следующий период → dt_valid_to = next - 1 day
       - иначе → 2299-12-31
    */
    COALESCE(nc.dt_next_found, nf.dt_next_found, DATE '2299-12-31') - INTERVAL '1 day' AS dt_valid_to
  FROM pp p
  LEFT JOIN pp_next_cal      nc ON nc.kalnr_text = p.kalnr_text AND nc.dt_valid_from = p.dt_valid_from
  LEFT JOIN pp_next_fallback nf ON nf.kalnr_text = p.kalnr_text AND nf.dt_valid_from = p.dt_valid_from
),

/* один проход по CKMLCR: текущий месяц + «дату постинга следующего» */
cr AS (
  SELECT
    (kalnr)::text AS kalnr_text,
    bdatj::text   AS bdatj,
    poper::text   AS poper,
    to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD')                         AS dt_start,
    (to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD') + INTERVAL '1 month')::date AS dt_next,  -- 1-е след. месяца
    stprs, peinh, pvprs, abprd_o, abprd_mo, vprsv, salk3
  FROM userdata.ckmlcr_ral
  WHERE untper='000' AND curtp='10'
),

/* единственная агрегация: MLCD */
mlcd_agg AS (
  SELECT
    (kalnr)::text AS kalnr_text,
    bdatj::text   AS bdatj,
    poper::text   AS poper,
    SUM(salk3)  AS salk3,
    SUM(estprd) AS estprd,
    SUM(mstprd) AS mstprd
  FROM userdata.mlcd_ral
  WHERE curtp='10' AND categ IN ('ZU','VP','PC')
  GROUP BY 1,2,3
)

SELECT
  r.kalnr_text                          AS calculation_code,
  r.dt_valid_from,
  r.dt_valid_to::date                   AS dt_valid_to,          -- совпадёт со старым

  c_curr.vprsv                          AS price_valuation_type_code,
  c_curr.stprs                          AS standard_price_amount,
  c_curr.peinh                          AS price_unit_code,
  c_curr.pvprs                          AS moving_average_price_amount,

  /* rule3 (без изменений) */
  CASE WHEN (pp_src.umkumo + pp_src.abkumo + pp_src.zukumo)=0 THEN 0
       ELSE TRUNC(
              CAST(ROUND(
                    (c_curr.stprs / NULLIF(c_curr.peinh,0)) * (pp_src.umkumo + pp_src.abkumo)
                  + c_curr.abprd_o + c_curr.abprd_mo
                  + COALESCE(m.salk3 + m.estprd + m.mstprd,0)
              ,2) AS NUMERIC(40,20))
              / NULLIF((pp_src.umkumo + pp_src.abkumo + pp_src.zukumo),0), 15)
  END AS average_weighted_stock_price_rule3_amount,

  /* rule2_prev_date — строго как раньше:
     берём salk3 той строки CR, у которой dt_next == dt_valid_from текущего PP
     (это «сальдо на 1-е число текущего месяца», посчитанное предыдущим периодом)
  */
  CASE WHEN (pp_src.abkumo + pp_src.umkumo + pp_src.zukumo)<>0
       THEN TRUNC(
              CAST(
                COALESCE(c_prev.salk3,0)
                + COALESCE(m.salk3 + m.estprd + m.mstprd,0)
              AS NUMERIC(40,20))
              / NULLIF((pp_src.abkumo + pp_src.umkumo + pp_src.zukumo),0), 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_prev_date_amount,

  /* rule2_next_date — как раньше: salk3 текущего CR / lbkum */
  CASE WHEN pp_src.lbkum<>0
       THEN TRUNC(CAST(c_curr.salk3 AS NUMERIC(40,20)) / NULLIF(pp_src.lbkum,0), 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_next_date_amount,

  h.price_rule_code
FROM pp_ranges r
JOIN pp         pp_src
  ON pp_src.kalnr_text    = r.kalnr_text
 AND pp_src.dt_valid_from = r.dt_valid_from

LEFT JOIN cr c_curr
  ON  c_curr.kalnr_text = pp_src.kalnr_text
  AND c_curr.bdatj      = pp_src.bdatj
  AND c_curr.poper      = pp_src.poper

LEFT JOIN cr c_prev
  ON  c_prev.kalnr_text = pp_src.kalnr_text
  AND c_prev.dt_next    = pp_src.dt_valid_from   -- как в старом ckmlcr_2 → join по dt_posting

LEFT JOIN mlcd_agg m
  ON  m.kalnr_text = pp_src.kalnr_text
  AND m.bdatj      = pp_src.bdatj
  AND m.poper      = pp_src.poper

LEFT JOIN dds.material_ledger_header h
  ON  h.calculation_code = pp_src.kalnr_text
;
