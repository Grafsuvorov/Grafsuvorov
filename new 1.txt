
  #!/usr/bin/bash
  set -euo pipefail

  unset http_proxy
  unset HTTPS_PROXY
  unset https_proxy
  unset HTTP_PROXY
  export GIT_SSL_NO_VERIFY=1

  echo 'gogogogo'

  # git repo есть только тут
  git -C /root/git_dir/etl -c http.sslVerify=false pull

  # директории в контейнере
  mkdir -p /app/meta_info
  mkdir -p /app/config_files/meta

  # копируем meta_info
  rm -rf /app/meta_info/*
  cp -a /root/git_dir/etl/meta_info/. /app/meta_info/

  # копируем config_files/meta
  rm -rf /app/config_files/meta/*
  cp -a /root/git_dir/meta/. /app/config_files/meta/

  echo 'Закончили упражнение'
  echo 'Как же ты хорош!'

  ## 2) docker-compose.yml — тома и env

  Должно быть так:

  services:
    api:
      volumes:
        - /root/table-dependency-viewer/scripts:/app/scripts:ro
        - /root/table-dependency-viewer/meta_info:/app/meta_info
        - /root/table-dependency-viewer/config_files:/app/config_files
        - /root/git_dir:/root/git_dir
      environment:
        - META_PARENT_DIR=/app/meta_info/database/greenplum/schema_name/tech_etl/etl_loads_entity
        - CLICK_META_DIR=/app/config_files/meta
        - ADMIN_CICD_SCRIPT=/app/scripts/ci_cd.sh

  ## 3) Применить

  chmod +x /root/table-dependency-viewer/scripts/ci_cd.sh
  docker compose up -d --force-recreate
