-- 1) PP: нормализация + dt_valid_from  → ключи
CREATE TEMP TABLE pp
ON COMMIT DROP AS
SELECT
  kalnr,
  bdatj,
  poper,
  make_date(bdatj::int, substring(poper,2,2)::int, 1) AS dt_valid_from,
  MAX(umkumo) AS umkumo,
  MAX(abkumo) AS abkumo,
  MAX(zukumo) AS zukumo,
  MAX(lbkum)  AS lbkum
FROM ods.ckmlpp_ral
WHERE untper='000'
GROUP BY kalnr,bdatj,poper
DISTRIBUTED BY (kalnr);

ANALYZE pp;

CREATE TEMP TABLE pp_keys
ON COMMIT DROP AS
SELECT DISTINCT kalnr,bdatj,poper
FROM pp
DISTRIBUTED BY (kalnr);

ANALYZE pp_keys;

-- 2) PP_W: границы действия (одно окно, без самоджойнов)
CREATE TEMP TABLE pp_w
ON COMMIT DROP AS
SELECT
  p.*,
  COALESCE(
    LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
    DATE '2299-12-31'
  )::date AS dt_valid_to
FROM pp p
DISTRIBUTED BY (kalnr);

ANALYZE pp_w;

-- 3) Набор нужных «первых чисел месяца» для следующего периода
CREATE TEMP TABLE pp_postings
ON COMMIT DROP AS
SELECT DISTINCT kalnr, dt_valid_from AS dt_posting
FROM pp_w
DISTRIBUTED BY (kalnr);

ANALYZE pp_postings;

-- 4) CR_CURR: CKMLCR текущего периода — только для ключей из pp
CREATE TEMP TABLE cr_curr
ON COMMIT DROP AS
SELECT
  c.kalnr, c.bdatj, c.poper,
  MAX(stprs)    AS stprs,
  MAX(peinh)    AS peinh,
  MAX(pvprs)    AS pvprs,
  MAX(abprd_o)  AS abprd_o,
  MAX(abprd_mo) AS abprd_mo,
  MAX(vprsv)    AS vprsv,
  MAX(salk3)    AS salk3    -- если это поток, замени на SUM(...)
FROM ods.ckmlcr_ral c
JOIN pp_keys k
  ON k.kalnr=c.kalnr AND k.bdatj=c.bdatj AND k.poper=c.poper
WHERE c.untper='000' AND c.curtp='10'
GROUP BY c.kalnr,c.bdatj,c.poper
DISTRIBUTED BY (kalnr);

ANALYZE cr_curr;

-- 5) CR_NEXT: salk3 на 1-е число следующего месяца — только нужные даты
CREATE TEMP TABLE cr_next
ON COMMIT DROP AS
SELECT
  c.kalnr,
  (make_date(c.bdatj::int, substring(c.poper,2,2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
  MAX(c.salk3) AS salk3
FROM ods.ckmlcr_ral c
JOIN pp_postings p
  ON p.kalnr=c.kalnr
 AND p.dt_posting = (make_date(c.bdatj::int, substring(c.poper,2,2)::int, 1) + INTERVAL '1 month')::date
WHERE c.untper='000' AND c.curtp='10'
GROUP BY c.kalnr, (make_date(c.bdatj::int, substring(c.poper,2,2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);

ANALYZE cr_next;

-- 6) MLCD: агрегаты только по ключам из pp
CREATE TEMP TABLE mlcd_agg
ON COMMIT DROP AS
SELECT
  m.kalnr, m.bdatj, m.poper,
  SUM(m.salk3)  AS salk3,
  SUM(m.estprd) AS estprd,
  SUM(m.mstprd) AS mstprd
FROM ods.mlcd_ral m
JOIN pp_keys k USING (kalnr,bdatj,poper)
WHERE m.curtp='10' AND m.categ IN ('ZU','VP','PC')
GROUP BY m.kalnr,m.bdatj,m.poper
DISTRIBUTED BY (kalnr);

ANALYZE mlcd_agg;

-- 7) Финальный селект
SELECT
  p.kalnr AS calculation_code,
  p.dt_valid_from,
  p.dt_valid_to,

  cr_curr.vprsv AS price_valuation_type_code,
  cr_curr.stprs AS standard_price_amount,
  cr_curr.peinh AS price_unit_code,
  cr_curr.pvprs AS moving_average_price_amount,

  CASE WHEN (p.umkumo + p.abkumo + p.zukumo)=0 THEN 0
       ELSE TRUNC(
              CAST(ROUND(
                    (cr_curr.stprs / cr_curr.peinh) * (p.umkumo + p.abkumo)
                  + cr_curr.abprd_o + cr_curr.abprd_mo
                  + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
              ,2) AS NUMERIC(40,20))
              / (p.umkumo + p.abkumo + p.zukumo), 15)
  END AS average_weighted_stock_price_rule3_amount,

  CASE WHEN (p.abkumo + p.umkumo + p.zukumo)<>0
       THEN TRUNC(
              CAST(cr_next.salk3 + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd,0) AS NUMERIC(40,20))
              / (p.abkumo + p.umkumo + p.zukumo), 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_prev_date_amount,

  CASE WHEN p.lbkum<>0
       THEN TRUNC(CAST(cr_curr.salk3 AS NUMERIC(40,20)) / p.lbkum, 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_next_date_amount,

  ckmlhd.price_rule_code
FROM pp_w p
LEFT JOIN cr_curr  ON cr_curr.kalnr=p.kalnr AND cr_curr.bdatj=p.bdatj AND cr_curr.poper=p.poper
LEFT JOIN cr_next  ON cr_next.kalnr=p.kalnr AND cr_next.dt_posting=p.dt_valid_from
LEFT JOIN mlcd_agg ON mlcd_agg.kalnr=p.kalnr AND mlcd_agg.bdatj=p.bdatj AND mlcd_agg.poper=p.poper
LEFT JOIN dds.material_ledger_header ckmlhd
  ON ckmlhd.calculation_code=p.kalnr;
