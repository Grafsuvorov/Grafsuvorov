import json
  import requests
  import urllib3

  urllib3.disable_warnings()

  issue = "DWH-10867"
  YOUTRACK_URL = "https://yt.rusal.ru"
  YOUTRACK_TOKEN = "..."

  fields = (
      "activities("
      "timestamp,"
      "author(name,login),"
      "field(name),"
      "added(name,fullName,login,text,presentation),"
      "removed(name,fullName,login,text,presentation),"
      "target(id,idReadable)"
      ")"
  )

  r = requests.get(
      f"{YOUTRACK_URL}/api/issues/{issue}/activitiesPage",
      headers={
          "Authorization": f"Bearer {YOUTRACK_TOKEN}",
          "Accept": "application/json",
      },
      params={"fields": fields},
      verify=False,
      timeout=60,
  )

  print(r.status_code)
  print(r.text[:500])

  r.raise_for_status()
  data = r.json()

  for row in data.get("activities", []):
