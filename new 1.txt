Вот обновлённый ci_cd.sh и фрагмент compose.

  ### scripts/ci_cd.sh

  #!/usr/bin/bash
  set -euo pipefail

  unset http_proxy
  unset HTTPS_PROXY
  unset https_proxy
  unset HTTP_PROXY

  export GIT_SSL_NO_VERIFY=1

  echo 'gogogogo'

  cd /root/git_dir/etl
  git -c http.sslVerify=false pull

  cd /root/git_dir/meta
  git -c http.sslVerify=false pull

  mkdir -p /app/meta_info
  mkdir -p /app/config_files/meta

  rm -rf /app/meta_info/*
  cp -a /root/git_dir/etl/meta_info/. /app/meta_info/

  rm -rf /app/config_files/meta/*
  cp -a /root/git_dir/meta/. /app/config_files/meta/

  echo 'Закончили упражнение'
  echo 'Как же ты хорош!'
