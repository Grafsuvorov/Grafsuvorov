
CREATE TABLE tech_etl.release_log (
	release_id text NOT NULL,
	release_type text NOT NULL,
	initiated_by text NOT NULL,
	started_at timestamp NOT NULL,
	finished_at timestamp NULL,
	status text NOT NULL,
	total_objects int4 NOT NULL,
	created_at timestamp DEFAULT now() NULL,
	ready_to_release bool NULL
)
DISTRIBUTED BY (release_id);

CREATE TABLE tech_etl.release_objects (
	release_id text NOT NULL,
	task_id text NOT NULL,
	target_system text NOT NULL,
	schema_name text NOT NULL,
	table_name text NOT NULL,
	entity_id int4 NOT NULL,
	entity_name text NOT NULL,
	change_type text NOT NULL,
	final_status text NOT NULL,
	attempts_count int4 NOT NULL,
	created_at timestamp DEFAULT now() NULL,
	failed_objects bool DEFAULT false NULL,
	attempt_no int4 NOT NULL,
	error_message text NULL,
	error_stacktrace text NULL,
	last_updated timestamp NULL
)
DISTRIBUTED BY (release_id);
