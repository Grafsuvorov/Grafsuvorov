with cf as (
    select
        issue_id,

        max(case when field_name = 'Приоритет' then field_value end) as priority,
        max(case when field_name = 'Срок' then field_value end) as due_date_txt,
        max(case when field_name = 'Дата релиза' then field_value end) as release_date_txt,
        max(case when field_name = 'Команда (Подсистема)' then field_value end) as team,
        max(case when field_name = 'Направление' then field_value end) as direction,

        max(case when field_name = 'Оценка трудозатрат' then field_value end) as estimate_txt,
        max(case when field_name = 'Фактические трудозатраты' then field_value end) as fact_txt,
        max(case when field_name = 'Экономия за счет ИИ' then field_value end) as ai_saving_txt

    from tech_etl.yt_issue_custom_field
    group by issue_id
),

worklog as (
    select
        issue_id,
        sum(minutes) / 60.0 as worklog_hours
    from tech_etl.yt_issue_worklog
    group by issue_id
),

final as (
    select
        s.issue_id,
        s.summary,
        s.description,
        s.project_name,
        s.project_key,
        s.created_by,
        s.assignee,
        s.created_at,
        s.updated_at,
        s.resolved_at,
        s.current_state,

        cf.priority,
        cf.team,
        cf.direction,

        cf.due_date_txt,
        cf.release_date_txt,
        cf.estimate_txt,
        cf.fact_txt,
        cf.ai_saving_txt,

        case
            when cf.due_date_txt ~ '^\d{4}-\d{2}-\d{2}'
                then left(cf.due_date_txt, 10)::date
            when cf.due_date_txt ~ '^\d{2}\.\d{2}\.\d{4}'
                then to_date(left(cf.due_date_txt, 10), 'DD.MM.YYYY')
            else null
        end as due_date,

        case
            when cf.release_date_txt ~ '^\d{4}-\d{2}-\d{2}'
                then left(cf.release_date_txt, 10)::date
            when cf.release_date_txt ~ '^\d{2}\.\d{2}\.\d{4}'
                then to_date(left(cf.release_date_txt, 10), 'DD.MM.YYYY')
            else null
        end as release_date,

        (
            coalesce(nullif(substring(cf.estimate_txt from '([0-9]+)\s*ч'), '')::numeric, 0) +
            coalesce(nullif(substring(cf.estimate_txt from '([0-9]+)\s*м'), '')::numeric, 0) / 60.0
        ) as estimate_hours,

        (
            coalesce(nullif(substring(cf.fact_txt from '([0-9]+)\s*ч'), '')::numeric, 0) +
            coalesce(nullif(substring(cf.fact_txt from '([0-9]+)\s*м'), '')::numeric, 0) / 60.0
        ) as fact_hours,

        (
            coalesce(nullif(substring(cf.ai_saving_txt from '([0-9]+)\s*ч'), '')::numeric, 0) +
            coalesce(nullif(substring(cf.ai_saving_txt from '([0-9]+)\s*м'), '')::numeric, 0) / 60.0
        ) as ai_saving_hours,

        coalesce(w.worklog_hours, 0) as worklog_hours,

        case
            when s.resolved_at is null
             and (
                case
                    when cf.due_date_txt ~ '^\d{4}-\d{2}-\d{2}'
                        then left(cf.due_date_txt, 10)::date
                    when cf.due_date_txt ~ '^\d{2}\.\d{2}\.\d{4}'
                        then to_date(left(cf.due_date_txt, 10), 'DD.MM.YYYY')
                    else null
                end
             ) < current_date
            then 1 else 0
        end as is_overdue,

        case
            when s.assignee is null or s.assignee = ''
            then 1 else 0
        end as is_without_assignee,

        case
            when cf.estimate_txt is null or cf.estimate_txt = ''
            then 1 else 0
        end as is_without_estimate,

        extract(day from coalesce(s.resolved_at, now()) - s.created_at) as age_days

    from tech_etl.yt_issue_snapshot s
    left join cf
        on s.issue_id = cf.issue_id
    left join worklog w
        on s.issue_id = w.issue_id
    where s.project_key = 'КХД'
)

select *
from final;
