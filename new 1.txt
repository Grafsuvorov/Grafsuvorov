insert into dds.kaizen_personnel_involvement (
	dt_report,
	kaizen_counterparty_rims_code,
	kaizen_counterparty_mdm_code,
	kaizen_personnel_involved_actual_quantity
)
select
	t.dt_report,
	t.kaizen_counterparty_rims_code,
	t.kaizen_counterparty_mdm_code,
	t.kaizen_personnel_involved_actual_quantity
	from (
with cte as (
	select
		c.counterparty_mdm_code,
		c.counterparty_rims_code,
		c.counterparty_code,
		c.counterparty_full_name,
		c.counterparty_truncated_code,
		c.counterparty_search_name,
		c.is_deleted
		from (
			select
				counterparty_mdm_code,
				counterparty_rims_code,
				counterparty_code,
				counterparty_full_name,
				counterparty_truncated_code,
				counterparty_search_name,
				is_deleted,
				DENSE_RANK () OVER (partition by counterparty_mdm_code order by is_deleted desc, counterparty_account_group_code asc) as ranks
				from dict_dds.counterparty ) c
		where c.ranks = 1
),
hr as (
	select 
		dt_report, 
		kaizen_counterparty_rims_code,
		count(dt_report) as kaizen_personnel_involved_actual_quantity
		from (
			select
				us.userid,
				res.companyid as kaizen_counterparty_rims_code,
				min(datecreated) as dt_report,
				res.datecreated_yyyy
				from (
					select
						userid,
						kaizenid
						from ods.sapxi_web_kaizen_authors
					union all
					select
						userid,
						kaizenid
						from ods.sapxi_web_kaizen_performers
				) us,
				( 	select
						ks.kaizenid,
				        k.companyid,
						min(ks.datecreated::date) as datecreated,
						extract(year from ks.datecreated::date) as datecreated_yyyy
						from ods.sapxi_web_kaizen_states as ks
						join ods.sapxi_web_kaizens as k on k.id = ks.kaizenid
					 	where ks.state = '4' and ks.datedeleted is null and k.isproject = '0'
					 	group by ks.kaizenid, k.companyid, extract(year from ks.datecreated::date)
				) res
				where us.kaizenid = res.kaizenid
				group by us.userid, res.companyid, res.datecreated_yyyy ) t
		group by dt_report, kaizen_counterparty_rims_code
)
select
	hr.dt_report,
	hr.kaizen_counterparty_rims_code,
	cte.counterparty_mdm_code as kaizen_counterparty_mdm_code,
	hr.kaizen_personnel_involved_actual_quantity
	from hr
	left join cte on hr.kaizen_counterparty_rims_code = cte.counterparty_rims_code
) t;
