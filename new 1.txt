#!/usr/bin/env python3
  import json
  import os
  import sys

  import requests
  import urllib3

  urllib3.disable_warnings()

  YOUTRACK_URL = os.environ["YOUTRACK_URL"].rstrip("/")
  YOUTRACK_TOKEN = os.environ["YOUTRACK_TOKEN"]

  ISSUE_ID = sys.argv[1] if len(sys.argv) > 1 else "DWH-10867"

  HEADERS = {
      "Authorization": f"Bearer {YOUTRACK_TOKEN}",
      "Accept": "application/json",
  }


  def api_get(path, params=None):
      r = requests.get(
          f"{YOUTRACK_URL}{path}",
          headers=HEADERS,
          params=params or {},
          verify=False,
          timeout=60,
      )
      r.raise_for_status()
      return r.json()


  def dump(title, payload):
      print(f"\n===== {title} =====")
      if isinstance(payload, list):
          for row in payload:
              print(json.dumps(row, ensure_ascii=False))
      else:
          print(json.dumps(payload, ensure_ascii=False, indent=2))


  def main():
      issue = api_get(
          f"/api/issues/{ISSUE_ID}",
          {
              "fields": ",".join(
                  [
                      "id",
                      "idReadable",
                      "summary",
                      "project(id,shortName,name)",
                      "customFields(id,name,$type,projectCustomField(field(id,name),$type),value(id,name,login,fullName,text,presentation,$type))",
                  ]
              )
          },
      )
      dump("ISSUE", issue)

      custom_fields = api_get(
          f"/api/issues/{ISSUE_ID}/customFields",
          {
              "fields": ",".join(
                  [
                      "id",
                      "name",
                      "$type",
                      "projectCustomField(field(id,name),$type)",
                      "value(id,name,login,fullName,text,presentation,$type)",
                  ]
              )
          },
      )
      dump("ISSUE CUSTOM FIELDS", custom_fields)

      project_id = issue["project"]["id"]

      project_fields = api_get(
          f"/api/admin/projects/{project_id}/customFields",
          {
              "fields": ",".join(
                  [
                      "id",
                      "$type",
                      "field(id,name)",
                      "bundle(id,name)",
                      "canBeEmpty",
                      "emptyFieldText",
                  ]
              )
          },
      )
      dump("PROJECT CUSTOM FIELDS", project_fields)

      print("\n===== FILTERED BY NAME =====")
      targets = {"Направление", "Дашборд КХД/Направление", "Дашборд SD"}
      for row in custom_fields:
          if row.get("name") in targets:
              print(json.dumps(row, ensure_ascii=False, indent=2))
      print("\n===== PROJECT FIELD NAMES =====")
      for row in project_fields:
          field = row.get("field") or {}
          name = field.get("name")
          if name in targets:
              print(json.dumps(row, ensure_ascii=False, indent=2))


  if __name__ == "__main__":
      main()

  Как запускать:

  python3 scripts/debug_ytrack_fields.py DWH-10867

  Что он покажет:

  - сам issue;
  - все customFields задачи;
  - все custom fields проекта;
  - отдельно отфильтрует:
      - Направление
      - Дашборд КХД/Направление
      - Дашборд SD

  На что смотреть:

  - есть ли Направление в ISSUE CUSTOM FIELDS
  - есть ли Направление в PROJECT CUSTOM FIELDS
  - совпадает ли field.id у старой и новой схемы
