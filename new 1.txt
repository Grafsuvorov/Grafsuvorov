DROP TABLE IF EXISTS date_for_vitr_tmp;

CREATE TEMP TABLE date_for_vitr_tmp AS
SELECT 
    dt::date AS dt_balance
FROM generate_series(
        current_date - interval '1 year',
        current_date,
        interval '1 day'
     ) AS gs(dt)
DISTRIBUTED REPLICATED;

SELECT
    vbss.delivery_code AS vbeln,
    MIN(vbsk.dt_end_of_free_storage) AS dt_end_of_free_storage
FROM dds.logistics_document_and_delivery_relationship vbss
LEFT JOIN dds.release vbsk
       ON vbsk.release_code = vbss.logistics_document_code
      AND vbsk.release_type_code = 'X'
LEFT JOIN dict_dds.settings_and_parameters_sap frel_kod
       ON vbsk.release_template_code = frel_kod.range_low_value

  AND frel_kod.range_sign_code = 'I'
  AND frel_kod.range_option_code = 'EQ'

GROUP BY vbss.delivery_code

DROP TABLE IF EXISTS base;

CREATE TEMP TABLE base AS
WITH
-- границы периода (чтобы не дергать MIN/MAX в WHERE)
date_bounds AS (
    SELECT 
        MIN(dt_balance)::date AS dt_min,
        MAX(dt_balance)::date AS dt_max
    FROM date_for_vitr_tmp
),

-- приоритеты складов (Азия / Европа)
forwapd_1 AS (
    SELECT 
        transport_hub_code,
        MIN(market_region1_code) AS market_region1_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE market_region1_code IN ('03','04')
      AND (
            is_terminal_code          IS NOT NULL
         OR is_foreign_warehouse_code IS NOT NULL
         OR is_temporary_warehouse_code IS NOT NULL
      )
    GROUP BY transport_hub_code
),

-- российский порт
forwapd_2 AS (
    SELECT DISTINCT transport_hub_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE is_russian_port_code IS NOT NULL
),

-- стадии для Европы
stage_eu AS (
    SELECT range_low_value
    FROM dict_dds.settings_and_parameters_sap
    WHERE abap_program_code = '/RUSAL/SD2973M_2'
      AND parameter_code     = 'STAGE_EU'
      AND range_low_value IS NOT NULL
),

-- стадии для Азии
stage_a AS (
    SELECT range_low_value
    FROM dict_dds.settings_and_parameters_sap
    WHERE abap_program_code = '/RUSAL/SD2973M_2'
      AND parameter_code     = 'STAGE_A'
      AND range_low_value IS NOT NULL
),

-- связи для Европы (отфильтрованы)
sabadr_03 AS (
    SELECT s.*
    FROM dds.sales_bundle_and_delivery_relationship s
    JOIN stage_eu se
      ON se.range_low_value = s.transportation_stage_code
    WHERE s.is_deleted_code IS NULL
),

-- связи для Азии (отфильтрованы)
sabadr_04 AS (
    SELECT s.*
    FROM dds.sales_bundle_and_delivery_relationship s
    JOIN stage_a sa
      ON sa.range_low_value = s.transportation_stage_code
    WHERE s.is_deleted_code IS NULL
)

SELECT
    /* ключи и базовые поля */
    t4.sales_bundle_code,
    lake.SD_000003  AS plant_producer_delivery_code,
    lake.SD_000110  AS plant_producer_delivery_position_code,
    lake.SD_000001  AS initial_delivery_code,
    lake.SD_000143  AS material_code,
    lake.SD_000010  AS dt_shipment,
    lake.SD_000013  AS railcar_code,
    lake.SD_000014  AS transport_bill_code,
    lake.SD_000151  AS uni,
    maspe.shape_code AS material_shape_code,
    lake.SD_000028  AS transport_type_code,
    mater.etsng_code,
    sabu.sales_bundle_net_weight  / 1000 AS sales_bundle_net_weight,
    sabu.sales_bundle_gross_weight / 1000 AS sales_bundle_gross_weight,
    lake.SD_000004 AS batch_code,

    /* Дата прибытия */
    t4.dt_transportation_stage_start_r AS dt_arrival,

    /* Завод продажной поставки */
    dedopo_1.plant_producer_code AS receiving_plant_code,

    lake.delivery_number_sales AS sales_delivery_code,

    /* Входящая тех. поставка по складской логике */
    CASE 
        WHEN fw1.transport_hub_code IS NOT NULL 
        THEN t4.delivery_code_le_p 
    END AS transportation_inbound_delivery_code,

    /* Поставка расчёта хранения (LE.001016) */
    CASE
        WHEN fw1.market_region1_code = '03' AND sab03.delivery_code IS NOT NULL THEN sab03.delivery_code
        WHEN fw1.market_region1_code = '04' AND sab04.delivery_code IS NOT NULL THEN sab04.delivery_code
        WHEN fw2.transport_hub_code  IS NOT NULL THEN lake.delivery_number_sales
    END AS delivery_for_storage_calculation_code,

    /* Позиция тех. поставки входа */
    t4.transportation_inbound_delivery_position_code,

    /* Позиция поставки расчета хранения */
    CASE
        WHEN fw1.market_region1_code IN ('03','04')
         AND (sab03.delivery_code IS NOT NULL OR sab04.delivery_code IS NOT NULL)
        THEN t4.transportation_inbound_delivery_position_code
        ELSE '10'
    END AS delivery_position_for_storage_calculation_code,

    /* Экспедитор */
    COALESCE(ascr_1.supplier_code, ascr_2.supplier_code) AS forwarder_code,

    /* Тип логики хранения */
    CASE
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 = 'TL04' THEN 'Логика РФ-контейнеры'
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 <> 'TL04' THEN 'Логика РФ-балк'

             AND asia.counterparty_code IS NULL THEN 'Логика Азия'
        WHEN likp.delivery_type_code = 'ZLE' AND euro_1.counterparty_code IS NOT NULL THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE' AND euro_2.counterparty_code IS NOT NULL 
             AND euro_log.range_low_value IS NOT NULL THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE' AND svh.counterparty_code IS NOT NULL THEN 'Логика СВХ'

    /* Особая логика LE.001065 */

        WHEN (
            likp.delivery_type_code = 'ZLE'
            AND euro_1.counterparty_code IS NOT NULL

            likp.delivery_type_code = 'ZLE'

    sbudr_base.transport_route_destination_hub_code,
    sbudr_base.is_final_transportation_stage_code,
    sbudr_base.transportation_stage_code,

    for_LE_001035.cnt AS storage_calculation_bundle_quantity,

    /* Разное */
    lake.SD_000482 AS dt_final_release,
    lake.SD_000041 AS bill_of_lading_number,
    lake.SD_000042 AS dt_bill_of_lading,
    lake.SD_000047 AS SD_000047,
    lake.SD_000048 AS SD_000048,
    lake.SD_000054 AS SD_000054,
    lake.SD_000049 AS SD_000049,
    lake.SD_000067 AS incoterms_code,
    lake.SD_000649,
    lake.SD_000044,
    lake.SD_000040,
    t4.delivery_code_le_r                 AS SD_000516,
    t4.transportation_outbound_delivery_position_code AS SD_000897,

    /* NET / BRUTTO */
    likp.weight_net              AS ntgew,
    likp.weight_net_with_wirerod AS btgew,

    /* Удалённый склад */
    t4.knote AS remote_warehouse_code,

    /* Дата прибытия в РФ порт */
    CASE
        WHEN fw1.market_region1_code IN ('03','04')
         AND (sab03.delivery_code IS NOT NULL OR sab04.delivery_code IS NOT NULL)
        THEN NULL
        ELSE lake.SD_000024
    END AS dt_arrival_to_russian_port,

    /* Интервалы */
    t4.dt_transportation_stage_start_p,
    COALESCE(t4.dt_transportation_stage_start_r, current_date) AS dt_transportation_stage_start_r,

    /* ID */
    CONCAT(t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code) AS id_for_unique

FROM dm_calc.sales_bundle_transport_hub_turnover_sdt0004 t4
CROSS JOIN date_bounds

LEFT JOIN lake
       ON t4.sales_bundle_code = lake.sales_bundle_code

LEFT JOIN dict_dds.material mater
       ON mater.material_code = lake.SD_000143

LEFT JOIN dds.sales_bundle sabu
       ON sabu.sales_bundle_code = t4.sales_bundle_code

LEFT JOIN forwapd_1 fw1
       ON fw1.transport_hub_code = t4.knote

LEFT JOIN forwapd_2 fw2
       ON fw2.transport_hub_code = t4.knote

LEFT JOIN sabadr_03 sab03
       ON sab03.sales_bundle_code              = t4.sales_bundle_code
      AND sab03.transport_route_destination_hub_code = t4.knote
      AND fw1.market_region1_code             = '03'

LEFT JOIN sabadr_04 sab04
       ON sab04.sales_bundle_code              = t4.sales_bundle_code
      AND sab04.transport_route_destination_hub_code = t4.knote
      AND fw1.market_region1_code             = '04'

LEFT JOIN dds.sales_bundle_and_delivery_relationship sbudr_base
       ON sbudr_base.sales_bundle_code      = t4.sales_bundle_code
      AND sbudr_base.delivery_code          = t4.delivery_code_le_p
      AND sbudr_base.delivery_position_code = t4.transportation_inbound_delivery_position_code
      AND sbudr_base.is_deleted_code IS NULL

LEFT JOIN dds.delivery_document_position dedopo_1
       ON dedopo_1.delivery_code = lake.delivery_number_sales

LEFT JOIN dds.sales_document_counterparty_role ascr_1
       ON ascr_1.sales_document_code   = t4.delivery_code_le_p
      AND ascr_1.counterparty_role_code = 'XR'

LEFT JOIN dds.sales_document_counterparty_role ascr_2
       ON ascr_2.sales_document_code   = t4.delivery_code_le_p
      AND ascr_2.counterparty_role_code = 'ZU'

LEFT JOIN dds.delivery_document_header likp
       ON likp.delivery_code =
          CASE
             WHEN fw1.market_region1_code = '03' AND sab03.delivery_code IS NOT NULL THEN sab03.delivery_code
             WHEN fw1.market_region1_code = '04' AND sab04.delivery_code IS NOT NULL THEN sab04.delivery_code
             WHEN fw2.transport_hub_code  IS NOT NULL THEN lake.delivery_number_sales
          END

LEFT JOIN asia 
       ON asia.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN euro_1 
       ON euro_1.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN euro_2 
       ON euro_2.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN svh    
       ON svh.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN asia_log 
       ON asia_log.range_low_value = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN euro_log 
       ON euro_log.range_low_value = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN dict_dds.material_specification maspe
       ON maspe.material_code = lake.SD_000143

LEFT JOIN for_LE_001035
       ON for_LE_001035.delivery_code = t4.delivery_code_le_p
      AND for_LE_001035.batch_code   = lake.SD_000004

/* Спец-логика для расчёта хранения */
LEFT JOIN dict_dds.transportation_storage_calculation_method trscm
       ON trscm.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

WHERE
    t4.dt_transportation_stage_start_p <= date_bounds.dt_max
    AND COALESCE(t4.dt_transportation_stage_start_r, current_date) >= date_bounds.dt_min
    AND t4.dt_transportation_stage_start_p <> COALESCE(t4.dt_transportation_stage_start_r, DATE '2299-12-31')
    AND t4.transport_route_destination_hub_code LIKE 'G%'
    AND (
        CASE
            WHEN fw1.market_region1_code = '03' AND sab03.delivery_code IS NOT NULL THEN sab03.delivery_code
            WHEN fw1.market_region1_code = '04' AND sab04.delivery_code IS NOT NULL THEN sab04.delivery_code
            WHEN fw2.transport_hub_code  IS NOT NULL THEN lake.delivery_number_sales
        END
    ) IS NOT NULL

DISTRIBUTED BY (sales_bundle_code);

CREATE TEMP TABLE base_1_5
--ON COMMIT DROP
as
SELECT
    b.*,
    d.dt_balance AS dt_report,
    row_number() OVER (
        PARTITION BY b.id_for_unique, d.dt_balance
        ORDER BY b.sales_bundle_code
    ) AS rn
FROM base b
JOIN date_for_vitr_tmp d
    ON d.dt_balance BETWEEN b.dt_transportation_stage_start_p
                        AND COALESCE(b.dt_transportation_stage_start_r, CURRENT_DATE)
DISTRIBUTED BY (sales_bundle_code, dt_report);


select count(*) from base_1_5


CREATE TEMP TABLE base_final --9-30
AS
WITH normalized AS (
    SELECT 
        b1.*,

        /* начало хранения = дата старта этапа */
        b1.dt_transportation_stage_start_p AS dt_storage_start,

        /* конец хранения = дата отчёта */
        b1.dt_report AS dt_storage_end,

        /* длительность хранения */
        (b1.dt_report - b1.dt_transportation_stage_start_p) + 1 
            AS storage_duration_total_calendar_days,

        /* номер услуги = просто ранжирование по bundle/день */
        row_number() OVER (
            PARTITION BY b1.sales_bundle_code, b1.dt_report
            ORDER BY b1.rn
        ) AS service_number
    FROM base_1_5 b1
)

SELECT *
FROM normalized
DISTRIBUTED BY (sales_bundle_code, dt_report);
