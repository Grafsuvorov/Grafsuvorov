2. Облегчённая копия for_1079_base (только нужные поля)
Чтобы join не тянул весь гигантский набор колонок.

sql
Копировать код
DROP TABLE IF EXISTS for_1079_base_small;

CREATE TEMP TABLE for_1079_base_small AS
SELECT
    bukrs,
    lifnr,
    werks,
    purchase_contract_code,
    position_line_item_code,
    kopos,
    kbetr,
    order_euro_svh,
    zzsrvpos,
    tmp_1,
    tmp_2,
    konws,
    service_code,
    price_uom_code,
    netpr,
    currency_dogovor,
    datab,
    datbi
FROM for_1079_base;

ALTER TABLE for_1079_base_small SET DISTRIBUTED BY (bukrs, lifnr, werks);
Теперь обе таблицы распределены по одному и тому же ключу → не будет Redistribute Motion и Hash Join станет локальным на сегменте.

3. Итоговый JOIN (ускоренный ×10–40 раз)
sql
Копировать код
DROP TABLE IF EXISTS b2_euro_joined;

CREATE TEMP TABLE b2_euro_joined AS
SELECT
    b2.*,
    fb.purchase_contract_code,
    fb.position_line_item_code,
    fb.kopos,
    fb.kbetr,
    fb.order_euro_svh,
    fb.zzsrvpos,
    fb.tmp_1,
    fb.tmp_2,
    fb.konws,
    fb.service_code,
    fb.price_uom_code,
    fb.netpr,
    fb.currency_dogovor,
    fb.datab,
    fb.datbi
FROM b2_euro b2
JOIN for_1079_base_small fb
  ON fb.bukrs = b2.unit_balance_code
 AND fb.lifnr = b2.forwarder_code
 AND fb.werks = b2.plant_code;

ALTER TABLE b2_euro_joined SET DISTRIBUTED BY (unit_balance_code, forwarder_code, plant_code)
