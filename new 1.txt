 -- CTE dates: сгенерированный календарь дат (dt_report) от 01.01.2026 до текущей даты
with dates as (
    select generate_series(
        date '2026-01-01',
        current_date,
        '1 day'::interval
    )::date as dt_report
),
 
-------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- CTE dates_with_timezone: формирование ключевых полей (rw_part, kpr_home) и дат
-- (прибытие/убытие с завода, станции, окончание выгрузки) с корректировкой на time_zone завода 
dates_with_timezone as (
    select
        rwcar_date_control.rw_part,
        date(
            (
                (rwcar_date_control.dt_in_plant at time zone 'UTC')
                at time zone past.time_zone_iana_code
            ) - interval '3 hours'
        ) as dt_in_plant,											-- Дата прибытия на завод
        date(
            (
                (rwcar_date_control.dt_out_plant at time zone 'UTC')
                at time zone past.time_zone_iana_code
            ) - interval '3 hours'
        ) as dt_out_plant,											-- Дата убытия с завода
        date(
            (
                (rwcar_date_control.dt_in_gdc at time zone 'UTC')
                at time zone past.time_zone_iana_code
            ) - interval '3 hours'
        ) as dt_in_gdc,											-- Дата прибытия на станцию
        date(
            (
                (rwcar_date_control.dt_out_gdc at time zone 'UTC')
                at time zone past.time_zone_iana_code
            ) - interval '3 hours'
        ) as dt_out_gdc,											-- Дата убытия со станции
        date(
            (
                (rwcar_date_control.dt_end_unload at time zone 'UTC')
                at time zone past.time_zone_iana_code
            ) - interval '3 hours'
        ) as dt_end_unload,										-- Дата окончания разгрузки
        rwcar_date_control.kpr_home												-- Код завода	
    from ods.asuzdc_rwcar_date_control as rwcar_date_control
    left join dict_dds.plant_and_subsidiary_td as past
        on past.asuzdc_plant_code = case
            when rwcar_date_control.kpr_home = '15'
                then '2'
            else rwcar_date_control.kpr_home
        end
    where past.asuzdc_plant_code is not null
),
 
-------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- Вычисление кода субагента
plant_id as (
    select
        pas.asuzdc_plant_code,																																																		-- Идентификатор завода АСУЖДЦ
        pas.plant_code,																																																			-- Код завода
        pas.plant_short_name,																																																		-- Наименование завода
        case
            when pas.plant_code = '5201' then 1
            when pas.plant_code = '5101' then 2
            when pas.plant_code = '5401' then 3
            when pas.plant_code = '5301' then 4
            when pas.plant_code = '6100' then 5
            when pas.plant_code = '5601' then 6
            when pas.plant_code = '6300' then 7
            when pas.plant_code = '6800' then 8
            when pas.plant_code = '5701' then 9
            else 100
        end as subsequent_code										-- Субкод завода
    from dict_dds.plant_and_subsidiary as pas
    left join dict_dds.plant_and_subsidiary_texts as past
        on
            past.plant_code = pas.plant_code
            and past.language_code = 'R'
    where
        asuzdc_plant_code is not null
        and pas.deleted_flag = false
        and past.deleted_flag = false
),

-- Агрегация нормативов с начала месяца / года отчетной даты dt_report
normative_period as (
    select
        d.dt_report,
        aurp.consignee_mdm_code as id_mdm,																																														-- Код МДМ Грузополучателя
        aurp.railcar_type_asuzdc_code as car_type,																																													-- Тип т/с
        aurp.package_type_asuzdc_code as pack_id,																																												-- Тип тары (Код)
        aurp.is_unloaded_outside_plant_code as pr_avto,																																							-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        past.asuzdc_plant_code as kpr_home,																																														-- Индикатор завода
        -- MTD: накопление с начала месяца до (dt_report - 1)
        sum(case
            when
                cal.dt_report >= date_trunc('month', d.dt_report)
                and cal.dt_report <= d.dt_report
                then aurp.unload_daily_normative_quantity
            else 0
        end) as railcar_at_plant_unload_normative_quantity_mtd,		-- Агрегация норматива по разгрузке вагонов за месяц
        sum(case
            when
                cal.dt_report >= date_trunc('month', d.dt_report)
                and cal.dt_report <= d.dt_report
                then aurp.unload_daily_normative_weight
            else 0
        end) as railcar_at_plant_unload_normative_weight_mtd,			-- Агрегация норматива по разгрузке массы глинозема за месяц
        -- YTD: накопление с начала года до (dt_report - 1)
        sum(case
            when
                cal.dt_report >= date_trunc('year', d.dt_report)
                and cal.dt_report <= d.dt_report
                then aurp.unload_daily_normative_quantity
            else 0
        end) as railcar_at_plant_unload_normative_quantity_ytd,		-- Агрегация норматива по разгрузке вагонов за год
        sum(case
            when
                cal.dt_report >= date_trunc('year', d.dt_report)
                and cal.dt_report <= d.dt_report
                then aurp.unload_daily_normative_weight
            else 0
        end) as railcar_at_plant_unload_normative_weight_ytd			-- Агрегация норматива по разгрузке массы глинозема за год
    from dates as d
    cross join dates as cal
    left join dict_dds.raw_material_unload_at_plant_normative_td as aurp
        on
            cal.dt_report >= aurp.dt_valid_from
            and (cal.dt_report <= aurp.dt_valid_to or aurp.dt_valid_to is null)
            and aurp.cargo_group_asuzdc_code = '1'
    left join dict_dds.plant_and_subsidiary_td as past on past.plant_code = aurp.plant_code
    where cal.dt_report <= d.dt_report
 
    group by
        d.dt_report,
        aurp.consignee_mdm_code,
        aurp.railcar_type_asuzdc_code,
        aurp.package_type_asuzdc_code,
        aurp.is_unloaded_outside_plant_code,
        past.asuzdc_plant_code
),

-- CTE sbor: подготовка данных для основной витрины —
-- формирование ключевых полей + промежуточных расчётов метрик
sbor as (
    select
        dt_report,																																																						-- Дата отчета
        qt.id_mdm,																																																						-- Код МДМ Грузополучателя
        qt.car_type,																																																					-- Тип т/с	
        qt.pr_avto,																																																					-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        case
            when qt.rw_sort in ('23', '24', '25', '94', '9 ') and ugm.cargo_group_asuzdc_code = '1' then '3'
            when qt.cargo_places = 5 and ugm.cargo_group_asuzdc_code = '1' then '1'
            when qt.cargo_places = 1 and ugm.cargo_group_asuzdc_code = '1' then '2'
            when qt.cargo_places > 5 and ugm.cargo_group_asuzdc_code = '1' then '4'
            when qt.cargo_places = 0 and (qt.rw_sort = '6 ' or qt.rw_sort = '68') and ugm.cargo_group_asuzdc_code = '1' then '2'
            when ugm.cargo_group_asuzdc_code in ('2', '3', '4') then '0'
            else '6'
        end as pack_id,																																																				-- LE.500019 Тип тары (Код)
        dt_in_plant,																																																					-- Дата прибытия на завод
        dt_out_plant,																																																					-- Дата убытия с завода
        dt_in_gdc,																																																						-- Дата прибытия на станцию
        dt_out_gdc,																																																					-- Дата убытия со станции
        dt_end_unload,																																																					-- Дата окончания разгрузки
        cargo_weight,  											-- Вес по документу																																																					-- 
        case
            when
                dt_in_plant is not null
                and dt_in_plant <= dt_report - 1
                and (dt_out_plant is null or dt_out_plant > dt_report - 1)
                and (dt_end_unload is null or dt_end_unload > dt_report - 1)
                then 1
            else 0
        end as in_plant_sbor,
        case
            when
                dt_in_gdc <= dt_report - 1
                and (dt_out_gdc is null or dt_out_gdc > dt_report - 1)
                and (dt_end_unload is null or dt_end_unload > dt_report - 1)
                --and (dt_in_plant is null or dt_in_plant > dt_report - 1)
                then 1
            else 0
        end as in_gdc_sbor,
        case
            when
                dt_in_plant > dt_report - 1
                and dt_in_plant <= dt_report
                then 1
            else 0
        end as in_plant_day_prihod,										-- LE.500028 Поступило  вагонов с глиноземом на завод факт
        case
            when
                extract(month from dt_in_plant) = extract(month from dt_report)
                and (dt_in_plant <= dt_report - 1)
                then 1
            else 0
        end as in_plant_month_prihod,									-- LE.500044 Поступило вагонов с глиноземом на завод факт с начала месяца MTD
        case
            when
                extract(year from dt_in_plant) = extract(year from dt_report)
                and (dt_in_plant <= dt_report - 1)
                then 1
            else 0
        end as in_plant_year_prihod,										-- LE.500028 Поступило  вагонов с глиноземом на завод факт
        case
            when dt_end_unload >= dt_report and dt_end_unload <= dt_report
                then 1
            else 0
        end as dt_end_unload_razgruzca
    from (
        select
            dt_report,
            coalesce(docs.id_mdm_p, docs1.id_mdm_p) as id_mdm,													-- Код МДМ Грузополучателя
            rwcar_parts.car_type,																																																		-- Тип т/с	 
            case
                when coalesce(docs.rw_st_p, docs1.rw_st_p) = '924906'
                    then 1
                else 0
            end as pr_avto,													-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
            coalesce(docs_cars1.cargo_places, docs_cars2.cargo_places) as cargo_places,												-- Количество мест
            coalesce(coalesce(docs_cars1.rw_kg, docs_cars2.rw_kg), docs_cars.rw_kg) as rw_kg,													-- Код груза по документу 
            hrw_cars.railcar_type_asuzdc_code as rw_sort,													-- Род вагона
            coalesce(docs_cars1.cargo_weight, docs_cars2.cargo_weight) as cargo_weight,  											-- Вес по документу
            dwz.dt_in_plant,																																																			-- Дата прибытия на завод
            dwz.dt_out_plant,																																																			-- Дата убытия с завода
            dwz.dt_in_gdc,																																																				-- Дата прибытия на станцию
            dwz.dt_out_gdc,																																																			-- Дата убытия со станции
            dwz.dt_end_unload																																																			-- Дата окончания разгрузки
        from dates
        left join dates_with_timezone as dwz
            on (
                extract(year from dwz.dt_in_plant) = extract(year from dt_report)
                or extract(year from dwz.dt_in_gdc) = extract(year from dt_report)
                or extract(year from dwz.dt_end_unload) = extract(year from dt_report)
            )
            --and (dwz.kpr_home != '15' or (dwz.kpr_home = '15' and (dwz.dt_in_plant is null or dwz.dt_in_plant >= dt_report - 1)))
            --and (dwz.kpr_home != '15' or (dwz.kpr_home = '15' and dwz.dt_in_plant is null))
            -- and (dwz.kpr_home != '15' or (dwz.kpr_home = '15' and dwz.dt_in_plant is null))
            --and (dwz.kpr_home not in ('2', '15') or ((dwz.kpr_home = '15' and (dwz.dt_in_plant is null or dwz.dt_in_plant >= dt_report - 1))) or ((dwz.kpr_home = '2' and dwz.dt_in_plant <= dt_report - 1)))
            --and (dwz.kpr_home not in ('2', '15') or ((dwz.kpr_home = '15' and (dwz.dt_in_plant is null or dwz.dt_in_plant >= dt_report))) or ((dwz.kpr_home = '2' and dwz.dt_in_plant < dt_report)))
            and (dwz.kpr_home != '15' or (dwz.kpr_home = '15' and dwz.dt_in_plant is null))
        left join ods.asuzdc_rwcar_parts as rwcar_parts
            on
                dwz.rw_part = rwcar_parts.rw_part
                and rwcar_parts.car_type in ('2', '3')
        left join ods.asuzdc_docs_cars as docs_cars
            on
                docs_cars.rw_part = rwcar_parts.rw_part
                and docs_cars.doc_type = '1'
        left join ods.asuzdc_docs_cars as docs_cars1
            on
                docs_cars1.rw_part = rwcar_parts.rw_part
                and docs_cars1.doc_type = '2'
        left join ods.asuzdc_docs as docs
            on docs.rn_doc::varchar = docs_cars1.rn_doc
        left join ods.asuzdc_docs_cars as docs_cars2
            on docs_cars2.rw_part = dwz.rw_part and docs_cars2.doc_type = '10'
        left join ods.asuzdc_docs as docs1
            on docs1.rn_doc::varchar = docs_cars2.rn_doc
        left join dict_dds.railcar_certificate as hrw_cars
            on hrw_cars.railcar_code = rwcar_parts.car_num
    --		and hrw_cars.transport_type_code = rwcar_parts.car_type
    ) as qt
    left join dict_dds.etsng as ugm on qt.rw_kg = ugm.etsng_code
    where
        ugm.cargo_group_asuzdc_code = '1'
        and qt.id_mdm is not null
),

-- основная расчетная cte для высисления значений на день
sbor_day as (
    select
        sbor.dt_report,
        sbor.id_mdm,																																																					-- Код МДМ Грузополучателя
        sbor.car_type,																																																					-- Тип т/с	
        sbor.pr_avto,																																																					-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        sbor.pack_id,																																																					-- LE.500019 Тип тары (Код)
        sum(in_plant_sbor) as railcar_at_plant_actual_quantity,							-- LE.500025 Наличие вагонов с глиноземом на заводе факт
        sum(in_gdc_sbor) - sum(in_plant_sbor) as railcar_at_station_actual_quantity,						-- LE.500026 Наличие вагонов с глиноземом на станции факт
        sum(case when in_plant_sbor = 1 then cargo_weight / 1000 else 0 end) as railcar_at_plant_actual_weight,							-- LE.500035 Наличие тонн глинозема на заводе
        sum(case when in_gdc_sbor = 1 then cargo_weight / 1000 else 0 end)
        -
        sum(case when in_plant_sbor = 1 then cargo_weight / 1000 else 0 end) as railcar_at_station_actual_weight,							-- LE.500036 Наличие тонн глинозема на станции
        sum(in_plant_day_prihod) as railcar_arrived_to_plant_actual_quantity,					-- LE.500028 Поступило  вагонов с глиноземом на завод факт
        sum(case when in_plant_day_prihod = 1 then cargo_weight / 1000 else 0 end) as railcar_arrived_to_plant_actual_weight,					-- LE.500038 Поступило тонн глинозема в ПС на завод факт
        sum(dt_end_unload_razgruzca) as railcar_at_plant_unloaded_actual_quantity,				-- LE.500031 Вагоны с глиноземом разгружено на заводе факт
        sum(case when dt_end_unload_razgruzca = 1 then cargo_weight / 1000 else 0 end) as railcar_at_plant_unloaded_actual_weight					-- LE.500035 Наличие тонн глинозема на заводе
    from sbor
    group by
        sbor.dt_report,																																																				-- Дата очтета
        sbor.id_mdm,																																																					-- Код МДМ Грузополучателя
        sbor.car_type,																																																					-- Тип т/с	
        sbor.pr_avto,																																																					-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        pack_id																																																						-- LE.500019 Тип тары (Код)				
),
 
-- основная расчетная cte для высисления значений на месяц
sbor_month as (
    select
        dates.dt_report,																																																			-- Дата отчета
        sbor.id_mdm,																																																				-- Код МДМ Грузополучателя
        sbor.car_type,																																																				-- LE.500022 Тип т/с	
        sbor.pr_avto,																																																				-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        sbor.pack_id,																																																				-- LE.500019 Тип тары (Код)
        sum(in_plant_day_prihod) as railcar_arrived_to_plant_actual_mtd_quantity,				-- LE.500044 Поступило вагонов с глиноземом на завод факт с начала месяца MTD	
        sum(dt_end_unload_razgruzca) as railcar_at_plant_unloaded_actual_mtd_quantity, 			-- LE.500045 Вагоны с глиноземом под разгрузку на заводе факт  с начала месяца MTD		
        sum(case when dt_end_unload_razgruzca = 1 then cargo_weight / 1000 else 0 end) as railcar_at_plant_unloaded_actual_mtd_weight,				-- LE.500053 Тонн глинозема в вагонах разгружено на заводе факт  с начала месяца MTD
        sum(case when in_plant_day_prihod = 1 then cargo_weight / 1000 else 0 end) as railcar_arrived_to_plant_actual_mtd_weight				-- LE.500050 Поступило тонн глинозема в ПС на завод факт с начала месяца
    from dates
    left join sbor
        on
            dates.dt_report > sbor.dt_report - 1
            and extract(month from sbor.dt_report) = extract(month from dates.dt_report)
    group by
        dates.dt_report,																																																				-- Дата отчета
        sbor.id_mdm,																																																					-- Код МДМ Грузополучателя
        sbor.car_type,																																																					-- LE.500022 Тип т/с	
        sbor.pr_avto,																																																					-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        sbor.pack_id																																																					-- LE.500019 Тип тары (Код)		
),
 
-- основная расчетная cte для высисления значений на год
sbor_year as (
    select
        dates.dt_report,																																																			-- Дата отчета
        sbor.id_mdm,																																																				-- Код МДМ Грузополучателя
        sbor.car_type,																																																				-- LE.500022 Тип т/с	
        sbor.pr_avto,																																																				-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        sbor.pack_id,																																																				-- LE.500019 Тип тары (Код)
        sum(in_plant_day_prihod) as railcar_arrived_to_plant_actual_ytd_quantity,				-- LE.500056 Поступило вагонов с глиноземом на завод факт с начала года YTD
        sum(dt_end_unload_razgruzca) as railcar_at_plant_unloaded_actual_ytd_quantity, 			-- LE.500059 Вагоны с глиноземом разгружено на заводе факт  с начала года YTD	
        sum(case when dt_end_unload_razgruzca = 1 then cargo_weight / 1000 else 0 end) as railcar_at_plant_unloaded_actual_ytd_weight,				-- LE.500065 Тонн глинозема в вагонах разгружено на заводе факт  с начала года YTD
        sum(case when in_plant_day_prihod = 1 then cargo_weight / 1000 else 0 end) as railcar_arrived_to_plant_actual_ytd_weight				-- LE.500062 Поступило тонн глинозема в ПС на завод факт с начала года YTD
    from dates
    left join sbor
        on
            dates.dt_report > sbor.dt_report - 1
            and extract(year from sbor.dt_report) = extract(year from dates.dt_report)
    group by
        dates.dt_report,																																																				-- Дата отчета
        sbor.id_mdm,																																																					-- Код МДМ Грузополучателя
        sbor.car_type,																																																					-- LE.500022 Тип т/с	
        sbor.pr_avto,																																																					-- Признак разгрузки на станции примыкания и доставки на завод автотранспортом
        sbor.pack_id																																																					-- LE.500019 Тип тары (Код)		
)
 
select
    dt_report,																																																						-- LE.500001 Дата отчета
    plant_code,																																																					-- LE.500002 Завод (код)
    plant_name, 																																																					-- LE.500003 Завод (наименование)
    plant_search_name,																																																				-- LE.500004 Завод (связка)
    subsequent_code,																																																				-- LE.500017 Субкод завода для сортировки
    --	, plant_short_name																																																				-- LE.500018 Короткое название завода
    package_type_code,																																																				-- LE.500019 Тип тары (Код)
    package_type_name,																																																				-- LE.500020 Тип тары (Наименование)
    package_type_search,																																																			-- LE.500021 Тип тары (Связка)
    transport_type_code,																																																			-- LE.500022 Тип ПС (Код)
    transport_type_name,																																																			-- LE.500023 Тип ПС (Наименование)
    transport_type_search,																																																			-- LE.500024 Тип ПС (Связка)
 
    -- сборка день
    sum(railcar_at_plant_actual_quantity),																																															-- LE.500025 Наличие вагонов с глиноземом на заводе факт
