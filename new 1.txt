 SELECT
      r.run_uuid,
      r.schema_name,
      s.table_name,
      MIN(s.start_dttm) AS start_dttm,
      MAX(s.end_dttm) AS end_dttm,
      SUM(EXTRACT(EPOCH FROM s.duration)) AS actual_duration_seconds
  FROM tech_etl.click_fact_table_load_run r
  JOIN tech_etl.click_fact_table_load_stage s
    ON s.run_uuid = r.run_uuid
  WHERE s.stage_name IN ('UPLOAD_TO_S3', 'CLICKHOUSE_LOAD')
    AND r.schema_name = 'dm'
    AND s.table_name = 'counter_party'
  GROUP BY
      r.run_uuid,
      r.schema_name,
      s.table_name
  ORDER BY start_dttm;
