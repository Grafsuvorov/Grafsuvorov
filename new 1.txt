-- 0. Подготовка: адреса без дублей (как у тебя, но явно и чисто)

DROP TABLE IF EXISTS adrc;

CREATE TEMP TABLE adrc
ON COMMIT DROP AS
SELECT
    address_code,
    MIN(country_code) AS country_code
FROM dict_dds.address
GROUP BY
    address_code
DISTRIBUTED REPLICATED;

ANALYZE adrc;

----------------------------------------------------------------------
-- 1. DIM: Склад (warehouse_code) → регион + страна (строковое имя)
--    (ранее: base -> tvkn -> adrc_2 -> country -> market_region1_texts)
----------------------------------------------------------------------

DROP TABLE IF EXISTS tmp_wh_geo;

CREATE TEMP TABLE tmp_wh_geo
ON COMMIT DROP AS
SELECT
    tvkn.transport_hub_code AS warehouse_code,

    /* region_of_remote_warehouse_name (LE.001061) */
    CASE
        WHEN t25a1.market_region1_name = 'Турция' THEN 'Западная Азия'
        WHEN t25a1.market_region1_name = 'Средний Восток и Афр' THEN 'Ближний Восток'
        WHEN t25a1.market_region1_name IN ('Азия', 'Китай') THEN 'Восточная Азия'
        ELSE t25a1.market_region1_name
    END AS region_of_remote_warehouse_name,

    /* country_of_remote_warehouse_name (через country_texts.cotex) */
    CASE
        WHEN cotex.country_full_name = 'Корея, республика' THEN 'Южная Корея'
        WHEN cotex.country_full_name = 'Тайвань (Китай)'   THEN 'Тайвань'
        ELSE cotex.country_full_name
    END AS country_of_remote_warehouse_name,

    cotex.country_full_name AS warehouse_country_full_name
FROM dict_dds.transport_hub tvkn
LEFT JOIN adrc       adrc_2 ON adrc_2.address_code = tvkn.address_code
LEFT JOIN dict_dds.country r_sd556
       ON r_sd556.country_code = adrc_2.country_code
LEFT JOIN dict_dds.market_region1_texts t25a1
       ON t25a1.market_region1_code = r_sd556.market_region1_code
      AND t25a1.language_code = 'R'
LEFT JOIN dict_dds.country_texts cotex
       ON cotex.country_code = adrc_2.country_code
      AND cotex.language_code = 'R'
GROUP BY
    tvkn.transport_hub_code,
    t25a1.market_region1_name,
    cotex.country_full_name
DISTRIBUTED REPLICATED;

ANALYZE tmp_wh_geo;

----------------------------------------------------------------------
-- 2. DIM: Имя склада (warehouse_code → warehouse_name)
--    (ранее: выбор языка для trht_3 по стране, cotex.country_full_name)
----------------------------------------------------------------------

DROP TABLE IF EXISTS tmp_wh_name;

CREATE TEMP TABLE tmp_wh_name
ON COMMIT DROP AS
SELECT
    tvkn.transport_hub_code AS warehouse_code,
    CASE
        WHEN cotex.country_full_name = 'Россия'
            THEN trht_r.transport_hub_name
        ELSE trht_e.transport_hub_name
    END AS warehouse_name
FROM dict_dds.transport_hub tvkn
LEFT JOIN adrc       adrc_2  ON adrc_2.address_code = tvkn.address_code
LEFT JOIN dict_dds.country_texts cotex
       ON cotex.country_code = adrc_2.country_code
      AND cotex.language_code = 'R'
LEFT JOIN dict_dds.transport_hub_texts trht_r
       ON trht_r.transport_hub_code = tvkn.transport_hub_code
      AND trht_r.language_code   = 'R'
LEFT JOIN dict_dds.transport_hub_texts trht_e
       ON trht_e.transport_hub_code = tvkn.transport_hub_code
      AND trht_e.language_code   = 'E'
GROUP BY
    tvkn.transport_hub_code,
    cotex.country_full_name,
    trht_r.transport_hub_name,
    trht_e.transport_hub_name
DISTRIBUTED REPLICATED;

ANALYZE tmp_wh_name;

----------------------------------------------------------------------
-- 3. DIM: Удалённый склад (remote_warehouse_code → name + country_code)
--    (ранее: base -> trhu_2 -> adrc_3 -> country_texts cotex_2 -> trht_5)
----------------------------------------------------------------------

DROP TABLE IF EXISTS tmp_remote_wh_dim;

CREATE TEMP TABLE tmp_remote_wh_dim
ON COMMIT DROP AS
SELECT
    trhu_2.transport_hub_code AS remote_warehouse_code,
    /* выбор языка для имени склада по стране (как у тебя для trht_5) */
    CASE
        WHEN cotex_2.country_full_name = 'Россия'
            THEN trht_r.transport_hub_name
        ELSE trht_e.transport_hub_name
    END AS remote_warehouse_name,
    adrc_3.country_code AS country_of_remote_warehouse_code
FROM dict_dds.transport_hub trhu_2
LEFT JOIN adrc       adrc_3  ON adrc_3.address_code = trhu_2.address_code
LEFT JOIN dict_dds.country_texts cotex_2
       ON cotex_2.country_code = adrc_3.country_code
      AND cotex_2.language_code = 'R'
LEFT JOIN dict_dds.transport_hub_texts trht_r
       ON trht_r.transport_hub_code = trhu_2.transport_hub_code
      AND trht_r.language_code = 'R'
LEFT JOIN dict_dds.transport_hub_texts trht_e
       ON trht_e.transport_hub_code = trhu_2.transport_hub_code
      AND trht_e.language_code = 'E'
GROUP BY
    trhu_2.transport_hub_code,
    adrc_3.country_code,
    cotex_2.country_full_name,
    trht_r.transport_hub_name,
    trht_e.transport_hub_name
DISTRIBUTED REPLICATED;

ANALYZE tmp_remote_wh_dim;
