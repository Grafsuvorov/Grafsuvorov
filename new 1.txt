-- витрина зависимостей: одна строка = сущность × таблица (источник или таргет)
DROP TABLE IF EXISTS tech_etl.entity_table_deps_v2;

CREATE TABLE tech_etl.entity_table_deps_v2 (
    date_id                  date            NOT NULL,
    entity_id                int             NOT NULL,
    entity_name              text            NOT NULL,
    -- таблица (схема.имя), о которой идёт речь в строке:
    table_schema             text            NOT NULL,
    table_name               text            NOT NULL,
    -- флаг: TRUE = эта (schema.table) является целевой таблицей сущности (INSERT INTO),
    -- FALSE = это источник (читаемая таблица)
    target                   boolean         NOT NULL DEFAULT false,
    -- опционально: тип ребра, если нужно
    edge_type                text            NULL,    -- 'insert_target' | 'select_source'

    -- тайминг из мониторинга (если найден)
    curr_start_dttm          timestamp       NULL,
    curr_finish_dttm         timestamp       NULL,
    prev_start_dttm          timestamp       NULL,
    prev_finish_dttm         timestamp       NULL,

    -- служебное
    load_dttm                timestamp       NOT NULL DEFAULT now()
);

CREATE INDEX ON tech_etl.entity_table_deps_v2 (date_id);
CREATE INDEX ON tech_etl.entity_table_deps_v2 (entity_id);
CREATE INDEX ON tech_etl.entity_table_deps_v2 (table_schema, table_name);
CREATE INDEX ON tech_etl.entity_table_deps_v2 (target);


-- параметр дня мониторинга
WITH params AS (
    SELECT DATE '2025-10-20' AS p_date
),

-- 2.1 parsed: результат парсинга query_insert_init (ИСТОЧНИК ДАННЫХ!)
parsed AS (
    SELECT
        lower(t.schema)  AS table_schema,
        lower(t.table)   AS table_name,
        t.source_type    AS edge_type,          -- 'insert_target' | 'select_source'
        t.meta_name      AS entity_name_raw
    FROM tech_etl.tables_meta_parse t
),

-- 2.2 entities: нормализуем названия и подтягиваем entity_id
entities AS (
    SELECT
        em.entity_id,
        em.entity_name,
        em.entity_last_load,     -- расписание/последний запуск (если нужно)
        em.entity_load_mode
    FROM tech_etl.entities_meta em
),

-- 2.3 связка парсинга с entities по имени сущности
parsed_with_entity AS (
    SELECT
        p.table_schema,
        p.table_name,
        p.edge_type,
        COALESCE(e.entity_id, -1)        AS entity_id,        -- -1 если не нашли
        COALESCE(e.entity_name, p.entity_name_raw) AS entity_name
    FROM parsed p
    LEFT JOIN entities e
      ON e.entity_name = p.entity_name_raw
),

-- 2.4 мониторинг за нужный день: по сущности (entity_name) возьмём её тайминги всех таблиц
mon_today AS (
    SELECT
        tc.entity_id,
        tc.entity_name,
        tc.table_schema,
        tc.table_name,
        tc.curr_start_dttm,
        tc.curr_finish_dttm,
        tc.prev_start_dttm,
        tc.prev_finish_dttm
    FROM tech_monitoring.table_compare tc
    JOIN params p ON tc.date_id = p.p_date
)

-- 2.5 финальная загрузка
INSERT INTO tech_etl.entity_table_deps_v2 (
    date_id, entity_id, entity_name,
    table_schema, table_name,
    target, edge_type,
    curr_start_dttm, curr_finish_dttm, prev_start_dttm, prev_finish_dttm
)
SELECT
    (SELECT p_date FROM params)              AS date_id,
    pwe.entity_id,
    pwe.entity_name,
    pwe.table_schema,
    pwe.table_name,
    CASE WHEN pwe.edge_type = 'insert_target' THEN TRUE ELSE FALSE END AS target,
    pwe.edge_type,
    mt.curr_start_dttm,
    mt.curr_finish_dttm,
    mt.prev_start_dttm,
    mt.prev_finish_dttm
FROM parsed_with_entity pwe
LEFT JOIN mon_today mt
  ON  mt.entity_name  = pwe.entity_name
  AND mt.table_schema = pwe.table_schema
  AND mt.table_name   = pwe.table_name
-- уберём мусор: без схемы/имени
WHERE pwe.table_schema IS NOT NULL
  AND pwe.table_schema <> ''
  AND pwe.table_name   IS NOT NULL
  AND pwe.table_name   <> '';


SELECT
  date_id, table_schema, table_name, entity_id, entity_name,
  target, edge_type, curr_start_dttm, curr_finish_dttm
FROM tech_etl.entity_table_deps_v2
WHERE date_id = DATE '2025-10-20'
ORDER BY table_schema, table_name, entity_id;


-- любые пересечения по таблицам (обычно смотрим источники)
SELECT
  table_schema,
  table_name,
  COUNT(DISTINCT entity_id)                            AS entities_cnt,
  array_agg(DISTINCT entity_name ORDER BY entity_id)   AS entities
FROM tech_etl.entity_table_deps_v2
WHERE date_id = DATE '2025-10-20'
  AND target = FALSE              -- только источники; уберите строку, если хотите вообще любые совпадения
GROUP BY 1,2
HAVING COUNT(DISTINCT entity_id) > 1
ORDER BY table_schema, table_name;

Шаг A: какие таргеты делает 36?

Шаг B: какие другие сущности используют эти таргеты как источники?

WITH my_targets AS (
  SELECT table_schema, table_name
  FROM tech_etl.entity_table_deps_v2
  WHERE date_id = DATE '2025-10-20'
    AND entity_id = 36
    AND target = TRUE
)
SELECT
  s.entity_id,
  s.entity_name,
  s.table_schema,
  s.table_name,
  s.curr_start_dttm, s.curr_finish_dttm
FROM tech_etl.entity_table_deps_v2 s
JOIN my_targets t
  ON  s.table_schema = t.table_schema
  AND s.table_name   = t.table_name
WHERE s.date_id = DATE '2025-10-20'
  AND s.target = FALSE             -- они читают наши таргеты
  AND s.entity_id <> 36
ORDER BY s.table_schema, s.table_name, s.entity_id;

3.4 Что читает сама entity_id = 36 (её входы), и кто ещё это читает (конкуренты за те же источники)
WITH my_sources AS (
  SELECT table_schema, table_name
  FROM tech_etl.entity_table_deps_v2
  WHERE date_id = DATE '2025-10-20'
    AND entity_id = 36
    AND target = FALSE
)
SELECT
  s.table_schema, s.table_name,
  COUNT(DISTINCT s.entity_id) AS consumers_cnt,
  array_agg(DISTINCT s.entity_name ORDER BY s.entity_id) AS consumers
FROM tech_etl.entity_table_deps_v2 s
JOIN my_sources ms
  ON  s.table_schema = ms.table_schema
  AND s.table_name   = ms.table_name
WHERE s.date_id = DATE '2025-10-20'
  AND s.target = FALSE
GROUP BY 1,2
ORDER BY table_schema, table_name;

Примечания/замены под вашу среду

Если у вас парсер пишет в другую таблицу (не tech_etl.tables_meta_parse) — просто поменяйте CTE parsed.

Если связь parsed.meta_name → entities_meta.entity_name иногда расходится по регистру/пробелам — добавьте lower/trim по обе стороны.

Если в мониторинге table_compare имена отличаются (table_schema NULL и всё хранится в table_name как schema.table) — разделите на две колонки перед JOIN.
