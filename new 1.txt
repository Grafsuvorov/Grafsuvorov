CREATE TABLE meta.table_dependencies (
    parent_table      varchar(200) NOT NULL,   -- кто зависит (DM, витрина)
    child_table       varchar(200) NOT NULL,   -- от кого зависит (DDS/ODS/STG)
    dependency_level  int          NOT NULL,   -- глубина от parent
    parent_last_load  timestamp    NULL,        -- время загрузки parent
    child_last_load   timestamp    NULL,        -- время загрузки child
    source            varchar(50)  NULL,        -- YAML, SQL, мануал, авто
    CONSTRAINT uq_dep UNIQUE (parent_table, child_table)
)
DISTRIBUTED BY (parent_table);
