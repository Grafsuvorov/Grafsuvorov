CREATE TEMP  TABLE for_1079_base_norm AS
WITH base AS (
    SELECT 
        purchase_contract_code,
        position_line_item_code,
        bukrs,
        lifnr,
        zzport,
        werks,
        zzsrvpos,
        zzforma,
        zzet_tarif,
        sdabw,
        zzterminal,
        vzzzone,
        konws,
        service_code,
        price_uom_code,
        netpr,
        kbetr,
        order_asia,
        currency_dogovor,
        datab,
        datbi,
        purchase_document_subtype_code,

        LEAD(datab) OVER (
            PARTITION BY purchase_contract_code, position_line_item_code
            ORDER BY datab
        ) AS next_datab

    FROM for_1079_base
)
SELECT *,
       COALESCE(next_datab - INTERVAL '1 day', datbi) AS effective_datbi
FROM base
DISTRIBUTED REPLICATED;


CREATE TEMP TABLE for_1079_asia AS
SELECT 
    sw.rn,
    fb.purchase_contract_code,
    fb.position_line_item_code,
    fb.konws,
    fb.service_code,
    fb.price_uom_code,
    fb.currency_dogovor,
    fb.datab,
    fb.effective_datbi AS datbi
FROM dm_calc.storage_sales_bundles_weight sw
JOIN for_1079_base_norm fb
    ON fb.bukrs = sw.unit_balance_code
   AND fb.lifnr = sw.forwarder_code
   AND (fb.zzport = sw.port_of_discharge_code OR fb.zzport IS NULL)
   AND fb.werks = sw.plant_code
   AND fb.zzsrvpos = 'S49'
   AND (fb.zzforma = sw.material_shape_code OR fb.zzforma IS NULL)
   AND (fb.zzet_tarif = sw.etsng_code OR fb.zzet_tarif IS NULL)
   AND (fb.sdabw = sw.transport_type_code OR fb.sdabw IS NULL)
   AND (fb.zzterminal = sw.warehouse_code OR fb.zzterminal IS NULL)
   AND (fb.vzzzone = sw.warehouse_storage_area_code OR fb.vzzzone IS NULL)
   AND fb.purchase_document_subtype_code = 'ZDTL'
   AND sw.dt_for_price_search_in_purchase_contract 
           BETWEEN fb.datab AND fb.effective_datbi
WHERE sw.storage_cost_calculation_type_name = 'Логика Азия'
DISTRIBUTED BY (rn);
