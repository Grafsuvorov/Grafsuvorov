#!/usr/bin/bash
  set -euo pipefail

  unset http_proxy
  unset https_proxy
  unset HTTP_PROXY
  unset HTTPS_PROXY
  export GIT_SSL_NO_VERIFY=1

  REPO_ROOT="/root/git_dir/etl"
  APP_ROOT="/app"
  BRANCH="main"

  echo "[ci_cd] start $(date -u +"%Y-%m-%dT%H:%M:%SZ")"

  git -C "${REPO_ROOT}" fetch origin
  git -C "${REPO_ROOT}" checkout "${BRANCH}"
  git -C "${REPO_ROOT}" reset --hard "origin/${BRANCH}"
  git -C "${REPO_ROOT}" clean -fd

  mkdir -p "${APP_ROOT}/meta_info"
  mkdir -p "${APP_ROOT}/config_files/meta/dm"
  mkdir -p "${APP_ROOT}/config_files/meta/dm_view"

  rm -rf "${APP_ROOT}/meta_info"/*
  rm -rf "${APP_ROOT}/config_files/meta/dm"/*
  rm -rf "${APP_ROOT}/config_files/meta/dm_view"/*

  cp -a "${REPO_ROOT}/meta_info/." "${APP_ROOT}/meta_info/"
  cp -a "${REPO_ROOT}/config_files/meta/dm/." "${APP_ROOT}/config_files/meta/dm/"
  cp -a "${REPO_ROOT}/config_files/meta/dm_view/." "${APP_ROOT}/config_files/meta/dm_view/"

  echo "[ci_cd] synced meta_info and config_files/meta from origin/${BRANCH}"
  echo "[ci_cd] done"

  docker exec -it table-dependency-viewer-api-1 bash -x /app/scripts/ci_cd.sh
  docker exec -it table-dependency-viewer-api-1 sh -c 'find /app/config_files/meta -iname "*sb_wuc*"'
