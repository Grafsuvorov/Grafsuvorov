Начинаем с двух витрин:

детальная — одна строка на одну задачу в релизе;
сводная — одна строка на один запуск релиза.

Так графики не будут раздуваться из-за нескольких объектов и повторных попыток.

1. Детальная витрина по задачам релиза
create or replace view tech_etl.v_release_task_analytics as
with release_objects_last as (
    select
        ro.release_id,
        ro.task_id,
        ro.target_system,
        ro.schema_name,
        ro.table_name,
        ro.entity_id,
        ro.entity_name,
        ro.change_type,
        ro.final_status,
        ro.attempts_count,
        ro.failed_objects,
        ro.attempt_no,
        ro.error_message,
        ro.last_updated,
        row_number() over (
            partition by
                ro.release_id,
                ro.task_id,
                ro.target_system,
                ro.schema_name,
                ro.table_name,
                ro.entity_id
            order by
                ro.attempt_no desc,
                ro.last_updated desc nulls last,
                ro.created_at desc
        ) as rn
    from tech_etl.release_objects ro
),

release_task_objects as (
    select
        release_id,
        task_id,

        count(*) as objects_cnt,
        count(*) filter (
            where coalesce(failed_objects, false)
        ) as failed_objects_cnt,

        count(*) filter (
            where lower(coalesce(final_status, '')) in (
                'success',
                'successful',
                'completed',
                'done'
            )
        ) as successful_objects_cnt,

        max(attempts_count) as max_attempts_count,
        avg(attempts_count::numeric) as avg_attempts_count,

        string_agg(
            distinct target_system,
            ', '
            order by target_system
        ) as target_systems,

        string_agg(
            distinct schema_name || '.' || table_name,
            ', '
            order by schema_name || '.' || table_name
        ) as changed_tables

    from release_objects_last
    where rn = 1
    group by
        release_id,
        task_id
),

task_fields as (
    select
        issue_id,

        max(
            case
                when field_name = 'Направление'
                then nullif(field_value, '')
            end
        ) as direction,

        max(
            case
                when field_name in ('Subsystem', 'Команда (Подсистема)')
                then nullif(field_value, '')
            end
        ) as subsystem,

        max(
            case
                when field_name = 'Дашборд КХД/Направление'
                then nullif(field_value, '')
            end
        ) as khd_dashboard

    from tech_etl.yt_issue_custom_field
    group by issue_id
),

base as (
    select
        rl.release_id,
        lower(trim(rl.release_type)) as release_type,
        rl.initiated_by,
        rl.started_at,
        rl.finished_at,
        rl.started_at::date as release_date,
        rl.status as release_status,
        rl.total_objects as release_total_objects,
        rl.ready_to_release,

        rto.task_id,
        s.summary,
        s.description,
        s.project_name,
        s.project_key,
        s.created_by,
        s.assignee,
        s.created_at as task_created_at,
        s.updated_at as task_updated_at,
        s.resolved_at,
        s.current_state,

        tf.direction,
        tf.subsystem,
        tf.khd_dashboard,

        rto.objects_cnt,
        rto.failed_objects_cnt,
        rto.successful_objects_cnt,
        rto.max_attempts_count,
        rto.avg_attempts_count,
        rto.target_systems,
        rto.changed_tables,

        case
            when s.created_at is not null
            then rl.started_at::date - s.created_at::date
        end as days_before_release

    from tech_etl.release_log rl
    join release_task_objects rto
        on rto.release_id = rl.release_id
    left join tech_etl.yt_issue_snapshot s
        on s.issue_id = rto.task_id
    left join task_fields tf
        on tf.issue_id = rto.task_id
)

select
    b.*,

    case
        when b.days_before_release is null then 'Нет даты задачи'
        when b.days_before_release < 0 then 'После старта'
        when b.days_before_release = 0 then 'В день релиза'
        when b.days_before_release = 1 then 'За 1 день'
        when b.days_before_release between 2 and 3 then 'За 2–3 дня'
        when b.days_before_release between 4 and 7 then 'За 4–7 дней'
        when b.days_before_release between 8 and 14 then 'За 8–14 дней'
        when b.days_before_release between 15 and 30 then 'За 15–30 дней'
        else 'Более чем за 30 дней'
    end as lead_time_bucket,

    case
        when b.days_before_release < 0 then 0
        when b.days_before_release = 0 then 1
        when b.days_before_release = 1 then 2
        when b.days_before_release between 2 and 3 then 3
        when b.days_before_release between 4 and 7 then 4
        when b.days_before_release between 8 and 14 then 5
        when b.days_before_release between 15 and 30 then 6
        when b.days_before_release > 30 then 7
        else 99
    end as lead_time_bucket_order,

    case
        when b.days_before_release between 0 and 1 then 1
        else 0
    end as is_last_1_day,

    case
        when b.days_before_release between 0 and 3 then 1
        else 0
    end as is_last_3_days,

    case
        when b.days_before_release between 0 and 7 then 1
        else 0
    end as is_last_7_days,

    case
        when b.failed_objects_cnt > 0 then 1
        else 0
    end as task_has_failures

from base b;

2. Сводная витрина: одна строка на релиз
create or replace view tech_etl.v_release_summary as
select
    release_id,
    release_date,
    release_type,
    initiated_by,
    started_at,
    finished_at,
    release_status,
    ready_to_release,

    count(distinct task_id) as tasks_cnt,
    sum(objects_cnt) as objects_cnt,
    sum(failed_objects_cnt) as failed_objects_cnt,

    count(distinct case
        when task_has_failures = 1 then task_id
    end) as failed_tasks_cnt,

    round(avg(days_before_release)::numeric, 1)
        as avg_days_before_release,

    count(distinct case
        when is_last_1_day = 1 then task_id
    end) as tasks_last_1_day,

    count(distinct case
        when is_last_3_days = 1 then task_id
    end) as tasks_last_3_days,

    count(distinct case
        when is_last_7_days = 1 then task_id
    end) as tasks_last_7_days,

    round(
        100.0
        * count(distinct case
            when is_last_3_days = 1 then task_id
          end)
        / nullif(count(distinct task_id), 0),
        1
    ) as tasks_last_3_days_pct,

    round(
        100.0
        * sum(failed_objects_cnt)
        / nullif(sum(objects_cnt), 0),
        2
    ) as failed_objects_pct,

    round(
        sum(objects_cnt)::numeric
        / nullif(count(distinct task_id), 0),
        1
    ) as objects_per_task

from tech_etl.v_release_task_analytics
group by
    release_id,
    release_date,
    release_type,
    initiated_by,
    started_at,
    finished_at,
    release_status,
    ready_to_release;

С какого графика начать

Начинаем с «Размер релизов во времени» на датасете tech_etl.v_release_summary.

Не используй release_id как ось X: у тебя он технический и длинный. Используй дату.

Настройка в Superset
Добавь v_release_summary как новый датасет.
Создай график типа Столбчатая диаграмма ECharts.
Настройки:
Ось X: release_date
Единица времени: День
Показатель: SUM(tasks_cnt)
Группировка: release_type
Сортировка по оси X: по возрастанию
Временной диапазон: Без фильтра / весь период
Режим накопления: Пусто

Почему SUM(tasks_cnt): если в один день было несколько запусков одного типа, они будут корректно сложены. При наведении покажется дата, тип и количество задач.

Чтобы график был читаемым

Во вкладке Кастомизация:

Тип ряда: Столбцы
Показывать значения: включить
Легенда: сверху
Поворот подписей X: 45°
Формат даты: %d.%m
Минимальная ширина графика: на всю строку дашборда

Название:

Размер и тип релизов во времени

Этот график покажет общую картину, но сильная аналитика начнется следующим графиком: доля задач, добавленных за последние три дня до релиза. Его построим по той же v_release_summary:

Ось X: release_date
Показатель: AVG(tasks_last_3_days_pct)
Группировка: release_type
Тип: линейный график

Так будет сразу видно не просто размер релиза, а какие релизы готовились в последний момент.
