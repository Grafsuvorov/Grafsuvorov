------------------------------------------------------------
-- DIM: владелец металла по поставке хранения
------------------------------------------------------------
DROP TABLE IF EXISTS dim_storage_owner;

CREATE TEMP TABLE dim_storage_owner
ON COMMIT DROP AS
SELECT
    deldp.delivery_code                            AS delivery_code,
    MIN(deldp.plant_producer_code)                 AS plant_producer_code
FROM dds.delivery_document_position deldp
JOIN (
    SELECT DISTINCT delivery_for_storage_calculation_code
    FROM dm_calc.storage_sales_bundles_amount
    WHERE delivery_for_storage_calculation_code IS NOT NULL
) b
  ON b.delivery_for_storage_calculation_code = deldp.delivery_code
WHERE deldp.delivery_position_line_item_code = '000010'
  AND deldp.deleted_flag = FALSE
GROUP BY deldp.delivery_code
DISTRIBUTED REPLICATED;

-- (Не обязательно, но очень полезно для плана)
ALTER TABLE dim_storage_owner
ADD CONSTRAINT pk_dim_storage_owner UNIQUE (delivery_code);


-- Завод-получатель (как и было)
LEFT JOIN dict_dds.plant_and_subsidiary pas
       ON pas.plant_code = base.receiving_plant_code

-- DIM-владелец металла из поставки хранения
LEFT JOIN dim_storage_owner ow
       ON ow.delivery_code = base.delivery_for_storage_calculation_code

-- Завод-собственник металла (наименование)
LEFT JOIN dict_dds.plant_and_subsidiary op
       ON op.plant_code = ow.plant_producer_code
