    CREATE temporary table pg_temp.calendar_periods with (appendonly = true,orientation = column,compresstype = zstd,compresslevel = 3) as(
        SELECT
            dt,
            '1' as period_comparison_rule_code,
            (dt - INTERVAL '1 day')::date as dt_comparison
        FROM dict_dds.calendar
        WHERE calendar_year BETWEEN 2000 AND 2050
 
        UNION ALL
 
        -- 2. Предыдущая неделя (понедельник → понедельник)
        SELECT
            dt,
            '2',
            (dt - INTERVAL '1 week')::date
        FROM dict_dds.calendar
        WHERE day_of_week = 1
          AND calendar_year BETWEEN 2000 AND 2050
 
        UNION ALL
 
        -- 3. Предыдущий месяц (последний день месяца)
        SELECT
            dt,
            '3',
            (date_trunc('month', dt)::date - 1)
        FROM dict_dds.calendar
        WHERE calendar_day = days_in_month
          AND calendar_year BETWEEN 2000 AND 2050
 
        UNION ALL
 
        -- 4. Предыдущий квартал (последний день квартала)
        SELECT
            dt,
            '4',
            (dt + INTERVAL '1 day' - INTERVAL '3 month' - INTERVAL '1 day')::date
        FROM dict_dds.calendar
        WHERE date_part('quarter', dt) <> date_part('quarter', dt + INTERVAL '1 day')
          AND calendar_year BETWEEN 2000 AND 2050
 
        UNION ALL
 
        -- 5. Предыдущий год (31 декабря)
        SELECT
            dt,
            '5',
            (dt - INTERVAL '1 year')::date
        FROM dict_dds.calendar
        WHERE calendar_month = 12
          AND calendar_day = 31
          AND calendar_year BETWEEN 2000 AND 2050
    ) distributed replicated;
