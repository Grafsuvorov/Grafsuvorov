create temporary table match_candidates on commit drop as
select
    sw.rn,
    sw.storage_cost_calculation_type_name,
    fb.purchase_contract_code,
    fb.position_line_item_code,
    fb.datab,
    fb.datbi,
    fb.price_uom_code,
    fb.service_code,
    fb.konws,
    fb.currency_dogovor,

    -- совпадения (булево → 1 / 0)
    case when fb.bukrs = sw.unit_balance_code then 1 else 0 end as m_bukrs,
    case when fb.lifnr = sw.forwarder_code
               or fb.lifnr = sw.creditor_in_shipment_instruction_code then 1 else 0 end as m_lifnr,
    case when fb.werks = sw.plant_code then 1 else 0 end as m_werks,
    case when fb.service_code = sw.service_number then 1 else 0 end as m_service,
    case when fb.zzport = sw.port_of_discharge_code or fb.zzport is null then 1 else 0 end as m_port,
    case when fb.zzforma = sw.material_shape_code or fb.zzforma is null then 1 else 0 end as m_shape,
    case when fb.zzet_tarif = sw.etsng_code or fb.zzet_tarif is null then 1 else 0 end as m_etsng,
    case when fb.sdabw = sw.transport_type_code or fb.sdabw is null then 1 else 0 end as m_transport,
    case when fb.zzterminal = sw.warehouse_code or fb.zzterminal is null then 1 else 0 end as m_terminal,
    case when fb.vzzzone = sw.warehouse_storage_area_code or fb.vzzzone is null then 1 else 0 end as m_area,

    -- дата
    case when sw.dt_for_price_search_in_purchase_contract between fb.dt_valid_from and fb.dt_valid_to then 1 else 0 end as m_docdate,
    case when sw.dt_for_price_search_in_purchase_contract between fb.datab and fb.datbi then 1 else 0 end as m_cond_date

from dm_calc.storage_sales_bundles_weight sw
join for_1079_base_clean fb
     on sw.dt_for_price_search_in_purchase_contract between fb.datab and fb.datbi
distributed replicated;

create temporary table match_scored on commit drop as
select
    *,
    -- единый score
    (
        m_bukrs + m_lifnr + m_werks + m_service +
        m_port + m_shape + m_etsng + m_transport +
        m_terminal + m_area + m_docdate + m_cond_date
    ) as score,

    case
        when storage_cost_calculation_type_name = 'Логика РФ-контейнеры' then 10
        when storage_cost_calculation_type_name = 'Логика РФ-балк' then 20
        when storage_cost_calculation_type_name = 'Логика Азия' then 30
        when storage_cost_calculation_type_name = 'Логика Европа' then 40
        when storage_cost_calculation_type_name = 'Логика СВХ' then 50
    end as priority
from match_candidates;

create temporary table match_1079_final on commit drop as
select *
from (
    select
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        service_code,
        price_uom_code,
        currency_dogovor,
        datab,
        datbi,

        row_number() over (
            partition by rn
            order by priority, score desc
        ) as rn_num

    from match_scored
) t
where rn_num = 1
distributed replicated;


create temporary table base_3 on commit drop as
with match as (
    select
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        service_code,
        price_uom_code,
        currency_dogovor,
        datab,
        datbi
    from match_1079_final
)
select
    sw.*,

    ---------------------------------------
    -- 1079 / 1080 / 1081 / 1083 / 1091
    ---------------------------------------
    m.purchase_contract_code,
    m.position_line_item_code,
    m.konws as purchase_contract_position_currency_code,
    m.service_code,
    m.price_uom_code as purchase_contract_position_uom_code,

    ---------------------------------------
    -- 1087 — free days by contract
    ---------------------------------------
    greatest(f1087.price_scale_limit_from_quantity - 1, 0)
        as storage_duration_free_by_contract_calendar_days,

    ---------------------------------------
    -- 1088 — free days by delivery
    ---------------------------------------
    least(
        sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0)
    ) as storage_duration_free_by_delivery_calendar_days,

    ---------------------------------------
    -- 1089 — payable days
    ---------------------------------------
    sw.storage_duration_total_calendar_days
      - least(
          sw.storage_duration_total_calendar_days,
          greatest(f1087.price_scale_limit_from_quantity - 1, 0)
        ) as storage_duration_payable_calendar_days,

    ---------------------------------------
    -- 1090 — local currency amount
    ---------------------------------------
    (case
        /* special type 4/5 */
        when sw.storage_cost_special_calculation_type_code = '2'
            then coalesce(f1090_1.konp_kbetr, f1090_2.konp_kbetr)
                 * sw.sales_bundle_gross_weight

        /* price-scale logic: PB00 step model */
        when f1090_1.evrtn is not null
            then f1090_1.konp_kbetr
                 * (sw.storage_duration_total_calendar_days
                    - least(sw.storage_duration_total_calendar_days,
                            greatest(f1087.price_scale_limit_from_quantity - 1, 0)))
                 * m.currency_dogovor

        when f1090_2.evrtn is not null
            then (coalesce(f1090_2.presum, 0)
                  + greatest(
                        sw.storage_duration_total_calendar_days
                        - f1090_2.kstbm + 1, 0
                    ) * f1090_2.konm_kbetr
                 ) * m.currency_dogovor
    end) as storage_cost_calculated_amount_local,

    ---------------------------------------
    -- weight buckets (free/paid segmentation)
    ---------------------------------------
    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0))
        between 1 and 30
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_001_030_days_gross_weight,

    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0))
        between 31 and 60
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_031_060_days_gross_weight,

    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0))
        between 61 and 90
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_061_090_days_gross_weight,

    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0))
        between 91 and 180
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_091_180_days_gross_weight,

    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0))
        between 181 and 365
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_181_365_days_gross_weight,

    case when sw.storage_duration_total_calendar_days - least(sw.storage_duration_total_calendar_days,
        greatest(f1087.price_scale_limit_from_quantity - 1, 0)) > 365
        then sw.sales_bundle_gross_weight else 0 end
        as storage_payable_over_365_days_gross_weight

from dm_calc.storage_sales_bundles_weight sw

left join match m
       on m.rn = sw.rn

left join for_1087 f1087
       on f1087.purchase_contract_code = m.purchase_contract_code
      and f1087.purchase_contract_position_code = m.position_line_item_code
      and f1087.dt_price_condition_valid_from = m.datab
      and f1087.dt_price_condition_valid_to = m.datbi

left join for_1090 f1090_1
       on f1090_1.evrtn = m.purchase_contract_code
      and f1090_1.evrtp = m.position_line_item_code
      and f1090_1.dt_price_condition_valid_from = m.datab
      and f1090_1.dt_price_condition_valid_to = m.datbi
      and f1090_1.price_scale_basis_code is null

left join for_1090 f1090_2
       on f1090_2.evrtn = m.purchase_contract_code
      and f1090_2.evrtp = m.position_line_item_code
      and f1090_2.dt_price_condition_valid_from = m.datab
      and f1090_2.dt_price_condition_valid_to = m.datbi
      and f1090_2.price_scale_basis_code = 'C'
      and sw.storage_duration_total_calendar_days
          between f1090_2.kstbm and f1090_2.kstbm_next

distributed by (sales_bundle_code);
