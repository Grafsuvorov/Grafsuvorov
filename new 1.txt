CREATE TABLE tech_etl.table_dependencies (
    parent_table_id int,
    parent_table      varchar(200) NOT NULL, 
    child_table_id    int NOT NULL, 
    child_table       varchar(200) NOT NULL,    dependency_level  int          NOT NULL,   -- глубина от parent
    parent_last_load  timestamp    NULL,    child_last_load   timestamp    NULL       
)
DISTRIBUTED BY (parent_table);
