# gp_copy_schema.py
# -*- coding: utf-8 -*-
"""
Копирование ВСЕХ таблиц указанной схемы из прод-Greenplum в препрод потоком через COPY.
- Никаких CLI-параметров: при запуске спросит только имя схемы.
- Для стабильности делает REPEATABLE READ-снимок на источнике (единый на всю схему).
- На приёмнике по каждой таблице: TRUNCATE -> COPY FROM STDIN -> ANALYZE.
- Порядок колонок берётся из целевой таблицы (препрод), чтобы точно совпало.

Зависимости: psycopg2-binary
pip install psycopg2-binary
"""

import os
import sys
import queue
import threading
import psycopg2


# --- НАСТРОЙКИ ПОД ТЕБЯ ---------------------------------------------
# Можно задать тут, а можно через переменные окружения GP_SRC_DSN / GP_DST_DSN
SRC_DSN = os.getenv("GP_SRC_DSN", "postgresql://user:pass@prod-host:5432/prod_db")
DST_DSN = os.getenv("GP_DST_DSN", "postgresql://user:pass@preprod-host:5432/preprod_db")

# Если какие-то таблицы копировать не надо — добавь сюда (имена без схемы)
EXCLUDE_TABLES = set([
    # "big_log_table",
])

# Включить/выключить ANALYZE после копирования каждой таблицы
DO_ANALYZE = True
# --------------------------------------------------------------------


def qi(ident: str) -> str:
    return '"' + ident.replace('"', '""') + '"'


def qname(schema: str, table: str) -> str:
    return f"{qi(schema)}.{qi(table)}"


def list_tables(dsn: str, schema: str) -> list[str]:
    sql = """
    SELECT table_name
    FROM information_schema.tables
    WHERE table_schema = %s AND table_type = 'BASE TABLE'
    ORDER BY table_name
    """
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(sql, (schema,))
        return [r[0] for r in cur.fetchall()]


def dest_has_table(dsn: str, schema: str, table: str) -> bool:
    sql = """
    SELECT 1
    FROM information_schema.tables
    WHERE table_schema = %s AND table_name = %s AND table_type='BASE TABLE'
    """
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(sql, (schema, table))
        return cur.fetchone() is not None


def get_dest_columns(dsn: str, schema: str, table: str) -> list[str]:
    sql = """
    SELECT column_name
    FROM information_schema.columns
    WHERE table_schema = %s AND table_name = %s
    ORDER BY ordinal_position
    """
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(sql, (schema, table))
        rows = cur.fetchall()
        if not rows:
            raise RuntimeError(f"Целевая таблица {schema}.{table} не найдена.")
        return [r[0] for r in rows]


def truncate_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"TRUNCATE TABLE {qname(schema, table)}")
        conn.commit()


def analyze_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"ANALYZE {qname(schema, table)}")
        conn.commit()


class QueueWriter:
    """file-like для COPY TO STDOUT (источник)"""
    def __init__(self, q: queue.Queue):
        self.q = q

    def write(self, b):
        if isinstance(b, str):
            b = b.encode("utf-8")
        self.q.put(b)

    def flush(self):  # noqa
        pass


class QueueReader:
    """file-like для COPY FROM STDIN (приёмник)"""
    def __init__(self, q: queue.Queue, sentinel):
        self.q = q
        self.buf = bytearray()
        self.sentinel = sentinel
        self.closed = False

    def read(self, n=-1):
        while not self.closed and (n < 0 or len(self.buf) < n):
            chunk = self.q.get()
            if chunk is self.sentinel:
                self.closed = True
                break
            self.buf.extend(chunk)
        if n < 0:
            n = len(self.buf)
        out = bytes(self.buf[:n])
        del self.buf[:n]
        return out


def copy_one_table(src_conn, dst_dsn: str, schema: str, table: str, cols: list[str]):
    """
    Копирует одну таблицу: TRUNCATE -> COPY -> ANALYZE (опционально).
    src_conn — ОБЩЕЕ подключение к источнику (уже в REPEATABLE READ).
    """
    print(f"  → {schema}.{table} …", flush=True)

    # Очередь и поток для приёмника
    q = queue.Queue(maxsize=64)
    sentinel = object()

    def dest_copy():
        reader = QueueReader(q, sentinel)
        copy_in_sql = (
            f"COPY {qname(schema, table)} ({', '.join(qi(c) for c in cols)}) "
            "FROM STDIN WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
        )
        with psycopg2.connect(dst_dsn) as conn, conn.cursor() as cur:
            cur.execute("SET client_encoding TO 'UTF8'")
            cur.copy_expert(copy_in_sql, reader)
            conn.commit()

    # TRUNCATE на приёмнике
    truncate_table(dst_dsn, schema, table)

    # Старт приёмника
    t = threading.Thread(target=dest_copy, daemon=True)
    t.start()

    # Источник: COPY (SELECT …) TO STDOUT
    select_cols = ", ".join(qi(c) for c in cols)
    copy_out_sql = (
        f"COPY (SELECT {select_cols} FROM {qname(schema, table)}) TO STDOUT "
        "WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
    )
    with src_conn.cursor() as cur:
        cur.execute("SET client_encoding TO 'UTF8'")
        writer = QueueWriter(q)
        cur.copy_expert(copy_out_sql, writer)
        q.put(sentinel)  # сигнал конца

    t.join()

    if DO_ANALYZE:
        analyze_table(dst_dsn, schema, table)

    print(f"  ✓ {schema}.{table} — done", flush=True)


def main():
    print("=== GP Schema COPY (prod → preprod) ===")
    print(f"SRC: {SRC_DSN}")
    print(f"DST: {DST_DSN}")
    schema = input("Введите имя схемы для копирования: ").strip()
    if not schema:
        print("Схема не указана.", file=sys.stderr)
        sys.exit(2)

    # Список таблиц в источнике
    src_tables = list_tables(SRC_DSN, schema)
    if not src_tables:
        print(f"В источнике нет таблиц в схеме {schema}.", file=sys.stderr)
        sys.exit(3)

    # Открываем единое соединение к источнику и берём снимок
    with psycopg2.connect(SRC_DSN) as src_conn, src_conn.cursor() as cur:
        src_conn.autocommit = False
        cur.execute("BEGIN")
        cur.execute("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ")
        print("Источник: REPEATABLE READ snapshot установлен.")

        for tbl in src_tables:
            if tbl in EXCLUDE_TABLES:
                print(f"  • {schema}.{tbl} — пропущена (в EXCLUDE_TABLES).")
                continue
            if not dest_has_table(DST_DSN, schema, tbl):
                print(f"  ! {schema}.{tbl} — нет в приёмнике, пропуск.", flush=True)
                continue
            try:
                cols = get_dest_columns(DST_DSN, schema, tbl)
                if not cols:
                    print(f"  ! {schema}.{tbl} — нет колонок на приёмнике, пропуск.")
                    continue
                copy_one_table(src_conn, DST_DSN, schema, tbl, cols)
            except Exception as e:
                print(f"  ✗ {schema}.{tbl} — ошибка: {e}", file=sys.stderr)

        # фиксация не обязательна (чтение), но завершим чисто
        src_conn.commit()

    print("=== Готово. ===")


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\nInterrupted.", file=sys.stderr)
        sys.exit(130)
    except Exception as e:
        print(f"ERROR: {e}", file=sys.stderr)
        sys.exit(1)
