from pathlib import Path
  from minio import Minio

  host = 'res-s-khs3.resource.local'
  port = 9000
  login = 'gpetl'
  password = 'pass'
  secure = True

  bucket_name = 'dbt-zp-prod'
  load_dir = 'dbt_run_manual/2026-04-10/2026-04-10 15:57:17.494487/artifacts'
  save_dir = Path('./minio_download')
  needed_files = {'manifest.json', 'run_results.json'}

  client = Minio(
      f'{host}:{port}',
      access_key=login,
      secret_key=password,
      secure=secure,
      cert_check=False,
  )

  save_dir.mkdir(parents=True, exist_ok=True)

  for obj in client.list_objects(bucket_name=bucket_name, prefix=load_dir, recursive=True):
      file_name = Path(obj.object_name).name
      if file_name in needed_files:
          local_path = save_dir / file_name
          print(f'Download: {obj.object_name} -> {local_path}')
          client.fget_object(
              bucket_name=bucket_name,
              object_name=obj.object_name,
              file_path=str(local_path),
          )

  print('Done')
