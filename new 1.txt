-- (предполагается, что pp_base, pp_keys3, pp_postings, pp_dates, cr_curr, cr_next, mlcd_agg уже созданы как раньше)
-- ключевая правка: pp_dates ДОЛЖНА быть распределена по (kalnr, dt_valid_from)
-- если у вас было DISTRIBUTED BY (kalnr) — пересоздайте:
DROP TABLE IF EXISTS pp_dates;
CREATE TEMP TABLE pp_dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
  kalnr,
  dt_valid_from,
  COALESCE(LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
           DATE '2299-12-31')::date AS dt_valid_to
FROM (SELECT kalnr, dt_valid_from FROM pp_base GROUP BY kalnr, dt_valid_from) s
DISTRIBUTED BY (kalnr, dt_valid_from);
ANALYZE pp_dates(kalnr, dt_valid_from);

----------------------------------------------------------------
-- A) curr-ветка: (kalnr,bdatj,poper) — все джоины локальные
----------------------------------------------------------------
DROP TABLE IF EXISTS curr_branch;
CREATE TEMP TABLE curr_branch
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
  b.kalnr,
  b.bdatj,
  b.poper,
  b.dt_valid_from,
  b.umkumo, b.abkumo, b.zukumo, b.lbkum,
  c.vprsv, c.stprs, c.peinh, c.pvprs, c.abprd_o, c.abprd_mo, c.salk3 AS salk3_curr,
  m.salk3 AS m_salk3, m.estprd AS m_estprd, m.mstprd AS m_mstprd,

  /* rule3 */
  CASE WHEN (b.umkumo + b.abkumo + b.zukumo) = 0 THEN 0
       ELSE TRUNC(
              CAST(ROUND(
                    (c.stprs / c.peinh) * (b.umkumo + b.abkumo)
                  + c.abprd_o + c.abprd_mo
                  + COALESCE(m.salk3 + m.estprd + m.mstprd, 0)
              , 2) AS NUMERIC(40,20))
              / (b.umkumo + b.abkumo + b.zukumo), 15)
  END AS price_rule3,

  /* rule2_next */
  CASE WHEN b.lbkum <> 0
       THEN TRUNC(CAST(COALESCE(c.salk3,0) AS NUMERIC(40,20)) / b.lbkum, 15)
       ELSE 0
  END AS price_rule2_next

FROM pp_base b
LEFT JOIN cr_curr   c ON c.kalnr=b.kalnr AND c.bdatj=b.bdatj AND c.poper=b.poper
LEFT JOIN mlcd_agg  m ON m.kalnr=b.kalnr AND m.bdatj=b.bdatj AND m.poper=b.poper
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE curr_branch(kalnr,bdatj,poper,dt_valid_from);

----------------------------------------------------------------
-- B) mlcd в "координатах постинга" для prev-ветки: (kalnr,dt_valid_from)
----------------------------------------------------------------
DROP TABLE IF EXISTS mlcd_by_posting;
CREATE TEMP TABLE mlcd_by_posting
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
  b.kalnr,
  b.dt_valid_from,   -- сопоставили тройку с постинговой датой
  SUM(COALESCE(m.salk3,0))  AS salk3,
  SUM(COALESCE(m.estprd,0)) AS estprd,
  SUM(COALESCE(m.mstprd,0)) AS mstprd
FROM pp_base b
LEFT JOIN mlcd_agg m
  ON m.kalnr=b.kalnr AND m.bdatj=b.bdatj AND m.poper=b.poper
GROUP BY b.kalnr, b.dt_valid_from
DISTRIBUTED BY (kalnr, dt_valid_from);
ANALYZE mlcd_by_posting(kalnr,dt_valid_from);

----------------------------------------------------------------
-- C) prev-ветка: (kalnr,dt_valid_from) — локально с cr_next
----------------------------------------------------------------
DROP TABLE IF EXISTS prev_branch;
CREATE TEMP TABLE prev_branch
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
ON COMMIT DROP AS
SELECT
  p.kalnr,
  p.dt_valid_from,
  -- нам нужны знаменатели из PP, поэтому подтащим их сюда из pp_base по (kalnr,dt_valid_from)
  MAX(b.abkumo) AS abkumo,
  MAX(b.umkumo) AS umkumo,
  MAX(b.zukumo) AS zukumo,
  COALESCE(n.salk3,0)                                       AS salk3_next,
  COALESCE(mp.salk3,0) + COALESCE(mp.estprd,0) + COALESCE(mp.mstprd,0) AS mlcd_total,

  /* rule2_prev */
  CASE WHEN (MAX(b.abkumo) + MAX(b.umkumo) + MAX(b.zukumo)) <> 0
       THEN TRUNC(
              CAST( COALESCE(n.salk3,0) + COALESCE(mp.salk3,0) + COALESCE(mp.estprd,0) + COALESCE(mp.mstprd,0)
              AS NUMERIC(40,20))
              / (MAX(b.abkumo) + MAX(b.umkumo) + MAX(b.zukumo)), 15)
       ELSE 0
  END AS price_rule2_prev
FROM pp_postings p
JOIN cr_next n
  ON n.kalnr=p.kalnr AND n.dt_posting=p.dt_posting    -- локально (оба по (kalnr,dt_posting))
JOIN mlcd_by_posting mp
  ON mp.kalnr=p.kalnr AND mp.dt_valid_from=p.dt_posting
JOIN pp_base b
  ON b.kalnr=p.kalnr AND b.dt_valid_from=p.dt_posting
GROUP BY p.kalnr, p.dt_valid_from, n.salk3, mp.salk3, mp.estprd, mp.mstprd
DISTRIBUTED BY (kalnr, dt_valid_from);
ANALYZE prev_branch(kalnr,dt_valid_from);

----------------------------------------------------------------
-- D) финальная сборка: сначала сшиваем ветки по (kalnr,dt_valid_from),
-- потом добавляем границы (pp_dates) и справочник.
----------------------------------------------------------------
INSERT INTO dm.material_price_calculation
SELECT
  cb.kalnr                                       AS calculation_code,
  cb.dt_valid_from,
  d.dt_valid_to,
  cb.vprsv                                       AS price_valuation_type_code,
  cb.stprs                                       AS standard_price_amount,
  cb.peinh                                       AS price_unit_code,
  cb.pvprs                                       AS moving_average_price_amount,
  cb.price_rule3                                 AS average_weighted_stock_price_rule3_amount,
  pb.price_rule2_prev                            AS average_weighted_stock_price_rule2_prev_date_amount,
  cb.price_rule2_next                            AS average_weighted_stock_price_rule2_next_date_amount,
  h.price_rule_code
FROM (
  SELECT DISTINCT kalnr, dt_valid_from FROM pp_base
) base
JOIN curr_branch cb
  ON cb.kalnr=base.kalnr AND cb.dt_valid_from=base.dt_valid_from
JOIN prev_branch pb
  ON pb.kalnr=base.kalnr AND pb.dt_valid_from=base.dt_valid_from
JOIN pp_dates d
  ON d.kalnr=base.kalnr AND d.dt_valid_from=base.dt_valid_from  -- локально, если pp_dates по (kalnr, dt_valid_from)
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code=base.kalnr;
