WITH RECURSIVE lineage AS (
    -- стартуем от витрины
    SELECT
        d.parent_table_id,
        d.parent_table,
        d.child_table_id,
        d.child_table,
        1 AS depth
    FROM tech_etl.table_dependencies d
    WHERE d.parent_table = 'dm.sb_wuc'

    UNION ALL

    SELECT
        d.parent_table_id,
        d.parent_table,
        d.child_table_id,
        d.child_table,
        l.depth + 1
    FROM tech_etl.table_dependencies d
    JOIN lineage l
      ON l.child_table_id = d.parent_table_id
    WHERE l.depth < 15
),
-- берём минимальную глубину для каждой таблицы
dedup AS (
    SELECT
        child_table_id,
        child_table,
        MIN(depth) AS min_depth
    FROM lineage
    GROUP BY child_table_id, child_table
)
SELECT
    d.min_depth               AS depth,
    tm.table_schema,
    tm.table_name,
    tm.table_last_load
FROM dedup d
JOIN tech_etl.tables_meta tm
  ON tm.table_id = d.child_table_id
ORDER BY
    d.min_depth,
    CASE tm.table_schema
        WHEN 'raw' THEN 1
        WHEN 'raw_ext' THEN 1
        WHEN 'stg' THEN 2
        WHEN 'dict_stg' THEN 2
        WHEN 'ods' THEN 3
        WHEN 'dds' THEN 4
        WHEN 'dict_dds' THEN 4
        WHEN 'dm_calc' THEN 5
        WHEN 'dm' THEN 6
        ELSE 99
    END,
    tm.table_name;
