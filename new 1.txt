CREATE TEMP TABLE for_1079_euro AS
SELECT *
FROM (
    SELECT
        p.rn,
        fb.purchase_contract_code,
        fb.position_line_item_code,
        fb.konws,
        fb.service_code,
        fb.price_uom_code,
        p.warehouse_weight_month_end,
        CASE
            WHEN extract(day from p.dt_storage_end) < extract(day from p.dt_storage_start)
            THEN (extract(year from p.dt_storage_end) - extract(year from p.dt_storage_start)) * 12 +
                 (extract(month from p.dt_storage_end) - extract(month from p.dt_storage_start))
            ELSE (extract(year from p.dt_storage_end) - extract(year from p.dt_storage_start)) * 12 +
                 (extract(month from p.dt_storage_end) - extract(month from p.dt_storage_start)) + 1
        END AS month_difference_2,
        fb.netpr,
        CASE
            WHEN p.storage_cost_special_calculation_type_code = '4'
                THEN ceil(p.storage_duration_total_calendar_days / 7.0)
            WHEN p.storage_cost_special_calculation_type_code = '5'
                THEN ceil(p.storage_duration_total_calendar_days / 30.0)
        END AS special,
        fb.currency_dogovor,
        fb.datab,
        fb.effective_datbi,
        ROW_NUMBER() OVER (
            PARTITION BY p.rn
            ORDER BY fb.order_euro_svh,
                     fb.kopos,
                     fb.kbetr,
                     fb.purchase_contract_code,
                     fb.position_line_item_code
        ) AS rn_ord
    FROM storage_europe_prep p
    JOIN for_1079_base_norm fb
      ON fb.bukrs = p.unit_balance_code
     AND fb.lifnr = p.forwarder_code
     AND fb.werks = p.plant_code
     AND (fb.zzport = p.port_of_discharge_code OR fb.zzport IS NULL)
     AND (fb.zzforma = p.material_shape_code OR fb.zzforma IS NULL)
     AND (fb.zzet_tarif = p.etsng_code OR fb.zzet_tarif IS NULL)
     AND (fb.sdabw = p.transport_type_code OR fb.sdabw IS NULL)
     AND (fb.zzterminal = p.warehouse_code OR fb.zzterminal IS NULL)
     AND (fb.vzzzone = p.warehouse_storage_area_code OR fb.vzzzone IS NULL)
     AND p.dt_for_price_search_in_purchase_contract
         BETWEEN fb.datab AND fb.effective_datbi
     AND fb.purchase_document_subtype_code = 'ZDTL'
) t
WHERE rn_ord = 1
DISTRIBUTED BY (rn);
