# gp_copy_table_align_where_unlogged.py
# -*- coding: utf-8 -*-
"""
Перелив одной таблицы prod -> preprod через COPY:
- Без information_schema: имена колонок забираем через SELECT * LIMIT 0.
- Копируем ИНТЕРСЕКЦИЮ колонок в порядке приёмника (preprod).
- Можно задать WHERE_CLAUSE (например, dt_report >= DATE '2025-01-01').
- На приёмнике можно сделать таблицу UNLOGGED перед заливкой (меньше WAL).
"""

import os
import sys
import queue
import threading
import psycopg2

# ===== НАСТРОЙКИ =====
SRC_DSN    = os.getenv("GP_SRC_DSN", "postgresql://user:pass@prod-host:5432/prod_db")
DST_DSN    = os.getenv("GP_DST_DSN", "postgresql://user:pass@preprod-host:5432/preprod_db")
TABLE_FQTN = "dm.sales_stock_balance_with_forecast"   # <<< укажи schema.table

# Пример фильтра: только записи с 2025-01-01 и позже
WHERE_CLAUSE = "dt_report >= DATE '2025-01-01'"       # либо None, чтобы копировать всё

DO_TRUNCATE       = True   # чистим целевую таблицу перед заливкой
DO_ANALYZE        = True   # ANALYZE после заливки
QUEUE_SIZE        = 64
STRICT_MATCH      = False  # True -> падать, если в источнике нет какой-либо колонки приёмника
MAKE_UNLOGGED     = True   # перед заливкой: ALTER TABLE ... SET UNLOGGED
REVERT_TO_LOGGED  = False  # после заливки вернуть LOGGED (если надо)
# =====================


def qi(ident: str) -> str:
    return '"' + ident.replace('"', '""') + '"'


def parse_fqtn(s: str):
    if not s or "." not in s:
        raise ValueError("TABLE_FQTN должен быть в формате schema.table")
    a, b = s.split(".", 1)
    a, b = a.strip(), b.strip()
    if not a or not b:
        raise ValueError("Неверный TABLE_FQTN")
    return a, b


def get_cols_via_limit0(dsn: str, schema: str, table: str):
    sql = f"SELECT * FROM {qi(schema)}.{qi(table)} LIMIT 0"
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(sql)
        return [d.name for d in cur.description]


def validate_where_clause_on_source(dsn: str, schema: str, table: str, where_clause: str):
    """Быстрая проверка, что WHERE корректен для источника (не падает по синтаксису/колонкам)."""
    sql = f"SELECT 1 FROM {qi(schema)}.{qi(table)} WHERE {where_clause} LIMIT 0"
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(sql)  # если что-то не так — тут упадёт


def truncate_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"TRUNCATE TABLE {qi(schema)}.{qi(table)}")
        conn.commit()


def analyze_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"ANALYZE {qi(schema)}.{qi(table)}")
        conn.commit()


def set_logged_mode(dsn: str, schema: str, table: str, *, unlogged: bool):
    mode = "UNLOGGED" if unlogged else "LOGGED"
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"ALTER TABLE {qi(schema)}.{qi(table)} SET {mode}")
        conn.commit()


class QueueWriter:
    def __init__(self, q: queue.Queue):
        self.q = q
    def write(self, b):
        if isinstance(b, str):
            b = b.encode("utf-8")
        self.q.put(b)
    def flush(self):
        pass


class QueueReader:
    def __init__(self, q: queue.Queue, sentinel):
        self.q = q
        self.buf = bytearray()
        self.sentinel = sentinel
        self.closed = False
    def read(self, n=-1):
        while not self.closed and (n < 0 or len(self.buf) < n):
            chunk = self.q.get()
            if chunk is self.sentinel:
                self.closed = True
                break
            self.buf.extend(chunk)
        if n < 0:
            n = len(self.buf)
        out = bytes(self.buf[:n])
        del self.buf[:n]
        return out


def copy_with_aligned_columns(schema: str, table: str, where_clause: str | None):
    # 1) Собираем перечень колонок
    dst_cols = get_cols_via_limit0(DST_DSN, schema, table)  # порядок приёмника
    src_cols = get_cols_via_limit0(SRC_DSN, schema, table)
    src_set = set(src_cols)
    common = [c for c in dst_cols if c in src_set]          # пересечение в порядке приёмника

    if STRICT_MATCH and len(common) != len(dst_cols):
        missing = [c for c in dst_cols if c not in src_set]
        raise RuntimeError(f"В источнике нет колонок приёмника: {missing}")

    if not common:
        raise RuntimeError("Не найдено общих колонок между источником и приёмником.")

    skipped = [c for c in dst_cols if c not in src_set]
    if skipped:
        print(f"[WARN] Пропускаем колонки приёмника (нет в источнике): {skipped}")

    print(f"[INFO] Будем копировать колонки ({len(common)} шт.): {common}")

    # 2) Валидируем WHERE на источнике (если задан)
    if where_clause:
        print(f"[INFO] WHERE: {where_clause}")
        validate_where_clause_on_source(SRC_DSN, schema, table, where_clause)

    # 3) Настраиваем пайп
    q = queue.Queue(maxsize=QUEUE_SIZE)
    sentinel = object()
    thread_err: list[BaseException] = []

    # 4) Поток-приёмник
    def dest_copy():
        try:
            copy_in_sql = (
                f"COPY {qi(schema)}.{qi(table)} "
                f"({', '.join(qi(c) for c in common)}) "
                "FROM STDIN WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
            )
            with psycopg2.connect(DST_DSN) as conn, conn.cursor() as cur:
                cur.execute("SET client_encoding TO 'UTF8'")
                reader = QueueReader(q, sentinel)
                cur.copy_expert(copy_in_sql, reader)
                conn.commit()
        except BaseException as e:
            thread_err.append(e)
            raise

    # 5) Подготовка приёмника: UNLOGGED/LOGGED, TRUNCATE
    if MAKE_UNLOGGED:
        print("[DST] ALTER TABLE … SET UNLOGGED")
        set_logged_mode(DST_DSN, schema, table, unlogged=True)

    if DO_TRUNCATE:
        print("[DST] TRUNCATE…")
        truncate_table(DST_DSN, schema, table)

    print("[DST] COPY FROM STDIN — старт…")
    t = threading.Thread(target=dest_copy, daemon=True)
    t.start()

    # 6) Источник: COPY (SELECT ... WHERE ...) TO STDOUT
    select_cols = ", ".join(qi(c) for c in common)
    src_sql = f"SELECT {select_cols} FROM {qi(schema)}.{qi(table)}"
    if where_clause:
        src_sql += f" WHERE {where_clause}"

    copy_out_sql = (
        f"COPY ({src_sql}) "
        "TO STDOUT WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
    )
    print("[SRC] COPY TO STDOUT…")
    with psycopg2.connect(SRC_DSN) as src_conn, src_conn.cursor() as cur:
        cur.execute("SET client_encoding TO 'UTF8'")
        writer = QueueWriter(q)
        cur.copy_expert(copy_out_sql, writer)
        q.put(sentinel)

    # 7) Дожидаемся приёмника и пробрасываем ошибку, если была
    t.join()
    if thread_err:
        raise thread_err[0]

    print("[OK] COPY завершён.")

    # 8) Логгирование обратно (опционально) и ANALYZE
    if REVERT_TO_LOGGED:
        print("[DST] ALTER TABLE … SET LOGGED")
        set_logged_mode(DST_DSN, schema, table, unlogged=False)

    if DO_ANALYZE:
        print("[DST] ANALYZE…")
        analyze_table(DST_DSN, schema, table)
        print("[OK] ANALYZE done.")


def main():
    try:
        schema, table = parse_fqtn(TABLE_FQTN)
        print("=== COPY prod → preprod (aligned cols, WHERE, UNLOGGED) ===")
        print(f"SRC:   {SRC_DSN}")
        print(f"DST:   {DST_DSN}")
        print(f"TABLE: {schema}.{table}")
        print("===========================================================")
        copy_with_aligned_columns(schema, table, WHERE_CLAUSE)
    except Exception as e:
        print(f"ERROR: {e}", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    main()
