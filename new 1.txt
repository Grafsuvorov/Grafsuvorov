drop view if exists dm_view.dq_transportation_vehicle_capacity_average_utilization on cluster '{cluster}' sync;
CREATE view dm_view.dq_transportation_vehicle_capacity_average_utilization  on cluster '{cluster}' AS
with last_check_dttm as (
    select distinct on (error_code, model_name)
        error_code,
        model_name,
        start_dttm as check_dttm
    from dm_view.check_result
    where error_code in ('dq_le0001', 'dq_le0002', 'dq_le0003')
    order by error_code asc, model_name asc, check_dttm desc
)
select
    d.error_code,															-- ID проверки
    dqe.error_short_name,													-- Краткое наименование
    CASE
        WHEN d.error_code = 'dq_le0001'
            THEN concat('Отсутствие плановой загрузки нетто в схеме погрузки ', 
                       JSONExtractString(d.check_result, 'cargo_arrangement_scheme_external_number'), 
                       ' (ошибка завода при регистрации схемы погрузки)')
        WHEN d.error_code = 'dq_le0002'
            THEN concat('Отсутствие плановой загрузки брутто в схеме погрузки ', 
                       JSONExtractString(d.check_result, 'cargo_arrangement_scheme_external_number'), 
                       ' (ошибка завода при регистрации схемы погрузки)')
        WHEN d.error_code = 'dq_le0003'
            THEN concat('Отсутствие схемы погрузки в поставке ', 
                       JSONExtractString(d.check_result, 'initial_delivery_code'))
    END AS error_full_name,    dqe.business_area_code,
    dqe.error_type_code,													-- Полное наименование (текст ошибки)
    dqe.severity_type_code,													-- Функциональная область
    dqe.table_log_code,
    dqe.table_source_code,
    dqe.error_description_text,													-- Тип проверки (1-бизнес/2-технический)
    dqe.error_algorithm_text,													-- Степень критичности (3-info|2-warning|1-critical)
    concat ('Необходимо обратиться в отдел оформления отгрузки завода ', nullif(JSONExtractString(d.check_result, 'plant_producer_search_name'), 'None')) as error_solution_text,												-- Бизнес-смысл
    nullif(JSONExtractString(d.check_result, 'plant_producer_search_name'), 'None') AS plant_producer_search_name,
    nullif(JSONExtractString(d.check_result, 'transport_type_search_name'), 'None') AS transport_type_search_name,
    nullif(JSONExtractString(d.check_result, 'transport_subtype_search_name'), 'None') AS transport_subtype_search_name,
    nullif(JSONExtractString(d.check_result, 'shape_search_name'), 'None') AS shape_search_name,
    nullif(JSONExtractString(d.check_result, 'cargo_arrangement_scheme_external_number'), 'None') AS cargo_arrangement_scheme_external_number,    d.check_dttm                                       as dttm_inserted,
    d.check_dttm                                       as dttm_updated,
    'dbt_dq'                                           as job_name,
    false                                              as deleted_flag
from dm_view.check_result_detail                       as d
inner join last_check_dttm                             as ls
    on
        ls.error_code = d.error_code
        and ls.check_dttm = d.check_dttm
        and ls.model_name = d.model_name
left join dm_view.dq_error                            as dqe
    on
        dqe.error_code = d.error_code
where d.error_code in ('dq_le0001', 'dq_le0002', 'dq_le0003');
