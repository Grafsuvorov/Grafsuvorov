class IndentDumper(yaml.Dumper):

def create_meta_clickhouse_from_db(greenplum_table_schema: str = '',
                                   greenplum_table_name: str = '',

                                   clickhouse_table_name: str = '',

                                   order_by: list = None,

    Arguments:
    :param greenplum_table_schema:   схема таблицы в гринпламе, по умолчанию ''. Заполняется для создания YAML для одной таблицы
    :param greenplum_table_name:     имя таблицы в гринпламе, по умолчанию ''. Заполняется, когда имена таблицы
                                     в гринпламе и в кликхаусе отличаются. Иначе достаточно указать имя в "clickhouse_table_name"
    :param clickhouse_table_schema:  схема таблицы в кликхаусе, по умолчанию ''. Заполняется для создания YAML для одной таблицы
    :param clickhouse_table_name:    имя таблицы в кликхаусе, по умолчанию ''. Заполняется для создания YAML для одной таблицы

with tables_info as(

        end) data_type,

        'data_type_gp',
        concat(c.udt_name,

            when character_maximum_length is not null

            else false

        'default',

        'description',
        pd.description
        ) json_data,
        pc.reloptions,
        pg_catalog.pg_get_table_distributedby(pc.oid) distributedby,
        v.view_definition,
        regexp_replace(split_part(v.view_definition, 'FROM', 2), '[[:space:];]+', '', 'g') as view_base,

	    on pc.relname = c.table_name        
	    and nsp.nspname = c.table_schema    
    left join pg_description pd

        and pd.objoid = pc.oid
    left join information_schema."views" v        
        on v.table_schema = nsp.nspname       
        and v.table_name = pc.relname
    where {f"nsp.nspname = '{greenplum_table_schema}' and pc.relname = '{object_name}'" if greenplum_table_schema != '' and clickhouse_table_name != '' else "em.flag_active"}
    and {"pc.relkind = 'v'" if greenplum_table_schema == 'dm_view' else "pc.relkind != 'v'"}

select
    k.table_type,
    k.table_schema,
    k.table_name,
    k.entity_id,
    k.entity_name,

            when k.view_definition like '%WITH%' then null

        end, '.', 1) as view_base_schema,
    split_part(
        case

        end, '.', 2) as view_base_name,

from tables_info k
group by

    k.entity_name,
    k.reloptions,

        print(err)
        raise err


def get_metadata_clickhouse(schema, table):
    click_schema = schema if schema != '' else 'dm'
    sql_query = f"""
SELECT
    table_name,
    database_name,
    groupArray(user_json) AS columns,
    primary_key,
    engine,
    partitions,
    order_by,
    settings,
    table_comment
FROM (
    SELECT
        t.name AS table_name,
        t.database AS database_name,
        map('column_name_click', c.name,
            'data_type_click',
            CASE WHEN c.type like 'Nullable(%'
                then replaceOne(replace(c.type, 'Nullable(', ''), ')', '')
                else c.type
            END AS data_type_click,
            'position', c.position::text,
            'is_nullable', hasToken(c.type, 'Nullable')::bool::text,
            'description', comment,
            'default', default_expression
        ) AS user_json,
        c.comment AS comment,
        t.primary_key,
        t.engine,
        t.partition_key as partitions,
        t.sorting_key as order_by,
        splitByString('SETTINGS', engine_full)[-1] AS settings,
        t.comment as table_comment
    FROM system.columns c
    JOIN system.tables t
        ON c.database = t.database AND c.table = t.name
    WHERE c.database = '{click_schema}' and c.table = '{table}')
GROUP BY table_name,
    database_name,

def get_template_content(
        template_name: str,
        templates_path: str = None,
        encoding: str = 'utf-8',
        **kwargs
)->str:
    """
    Функция подлежит удалению. В дальнейшем использовать будет нельзя.
    Args:
        template_name: наименование файла или полный путь до файла.
        templates_path: путь к файлу.
        encoding: Кодировка файла.
        **kwargs:

    Returns: возвращает отрендеренный шаблон.

    """
    from jinja2 import Template
    if templates_path and os.path.exists(os.path.join(templates_path, template_name)):
        content = Path(os.path.join(templates_path, template_name)).read_text(encoding=encoding)
    elif os.path.exists(template_name):
        content = Path(template_name).read_text(encoding=encoding)
    else:
        content = template_name
    template = Template(content)
    rendered_content = template.render(**kwargs)
    return rendered_content

def get_template_sql(
        template_name: str,
        templates_path: str = None,
        encoding: str = 'utf-8',
        **kwargs
)->str:
    """
    Функция подлежит удалению. В дальнейшем использовать будет нельзя.
    Args:
        template_name: наименование файла или полный путь до файла.
        templates_path: путь к файлу.
        encoding: Кодировка файла.
        **kwargs:

    Returns: возвращает отрендеренный шаблон.

    """
    from jinja2 import Template
    if templates_path and os.path.exists(os.path.join(templates_path, template_name)):
        sql = Path(os.path.join(templates_path, template_name)).read_text()
    elif os.path.exists(template_name):
        sql = Path(template_name).read_text(encoding=encoding)
    else:
        sql = template_name
    template = Template(source=sql)
    sql_text = template.render(**kwargs)
    return sql_text

def get_parquet_dir_with_ts() -> int:
    from airflow.operators.python import get_current_context
    in_datetime = get_current_context()['ts']
    if '+' in in_datetime:
        if len(in_datetime) != 32:
            return int(datetime.strptime(in_datetime[:-6], '%Y-%m-%dT%H:%M:%S').timestamp() * 1000)
        else:
            return int(datetime.strptime(in_datetime[:-6], '%Y-%m-%dT%H:%M:%S.%f').timestamp() * 1000)
    return int(datetime.strptime(in_datetime[:-6], '%Y-%m-%dT%H:%M:%S.%f%z').timestamp() * 1000)

def set_dag_status(ti, dag_id, task_id, message_error, pulse_url, upstream_task_ids=None, **kwargs):
    from airflow.utils.state import State
    from plugins.help_utility.pulse_release_notification import send_release_finished_message
    logger = kwargs['dag'].logger()
    release_user = kwargs.get('params', {}).get('release_user')
    
    dag_run = kwargs.get('dag_run')
    failed_upstream_tasks = []
    
    if dag_run and upstream_task_ids:
        for upstream_task_id in upstream_task_ids:
            upstream_ti = dag_run.get_task_instance(task_id=upstream_task_id)
            if upstream_ti:
                logger.info(f"Task {upstream_task_id} state: {upstream_ti.state}")
                if upstream_ti.state in [State.FAILED, State.UPSTREAM_FAILED]:
                    failed_upstream_tasks.append(upstream_task_id)
    
    if ti.xcom_pull(
            dag_id=dag_id,
            task_ids=task_id,
    ):
        raise Exception(message_error)
    
    try:
        if failed_upstream_tasks:
            logger.error(f"Failed upstream tasks: {failed_upstream_tasks}")
            send_release_finished_message(
                event_type='bad_finished',
                logger=logger,
                user_name=release_user,
                pulse_url=pulse_url
            )
        else:
            send_release_finished_message(
                event_type='finished',
                logger=logger,
                user_name=release_user,
                pulse_url=pulse_url
            )
    except Exception as e:
        logger.error(f'PULSE NOTIFICATION EXCEPTION: {e}')
        
def send_message_pulse(
        header: Union[str, bytes],
        msg: Union[str, list, dict],
        pulse_url: str='pulse',
        return_response=True
):
    import requests
    pulse_url = f'http://10.66.129.239/{pulse_url}'
    try:
        if isinstance(msg, str):
            data = msg.replace('"', '\\"').replace('\n','\\n')
        elif isinstance(msg, list):
            data =  "\\n".join(str(msg).replace('"', '\\"'))
        elif isinstance(msg, dict):
            data = ''
            for sub_header, sub_data in msg.items():
                if sub_data:
                    if isinstance(sub_data, str):
                        sub_data = sub_data.replace('"', '\\"')
                    data += f"\\n_**{sub_header}**_ :\\n{sub_data}\\n"
        else:
            data = 'No data'
        message =  f"**{header}**\\n{data}"
        if len(message)>2000:
            message = f"{message[:1900]} \\n...\\nПолный отчёт смотри в почте"
        response = requests.post(url=pulse_url, data=message.encode('utf-8'))
        response.raise_for_status()
        if return_response:
            return response
        return None
    except Exception as e:
        print('Error in send_message_pulse:', str(e))
        print(f"Status Code: {response.status_code}")
        print(f"Response JSON: {response.json()}")

def pulse_format_text(
        data,
        warp_rows: str = '||',
        rows_delimiter: str = '\\n',
        column_delimiter: str = '|',
) -> str:
    return f'{rows_delimiter}'.join([f'''{warp_rows}{f"{column_delimiter}".join(row)}{warp_rows}''' for row in data])

def util_text_to_null_validation(column: str, source_alias: str = None):
    col = f'{source_alias}."{column}"' if source_alias else f'"{column}"'
    return f'''case when nullif(trim({col}), '') is not null then {col} else null end'''

def util_text_to_date_validation(column: str, calendar_alias: str, source_alias: str):
    return f'''case
        when {calendar_alias}.date_text is not null
            then {calendar_alias}.date_as_date
        when {source_alias}."{column}" !~ '^\d{{8}}$'
            then to_date('19000101'::text, 'YYYYMMDD')
        when {source_alias}."{column}" = '00000000'
            then null
        when {source_alias}."{column}" < '22991231'
            then to_date('19000101'::text, 'YYYYMMDD')
        when {source_alias}."{column}" >= '22991231'
            then to_date('22991231'::text, 'YYYYMMDD')
    end::date'''

def util_text_to_time_validation(column: str, source_alias: str = None):
    col = f'{source_alias}."{column}"' if source_alias else f'"{column}"'
    return f'''case
        when {col} is null
            then null
        when {col}::text !~ '^\d{{6}}$'
            then null
        when substring({col}::text FROM 1 FOR 2)::int between 0 and 23
            and substring({col}::text FROM 3 FOR 2)::int between 0 and 59
            and substring({col}::text FROM 5 FOR 2)::int between 0 and 59
            then {col}
        else '00:00:00'
    end'''

def util_text_to_timestamp_validation(
        column_date: str,
        column_time: str,
        calendar_alias: str,
        source_alias: str
    ):
    return f'''case
        when {util_text_to_date_validation(
            column=column_date,
            calendar_alias=calendar_alias,
            source_alias=source_alias
        )} is null
            then to_timestamp('1900-01-01 00:00:00', 'YYYY-MM-DD HH24:MI:SS')
        when {util_text_to_time_validation(
            column=column_time,
            source_alias=source_alias,
        )} is null
            then to_timestamp('1900-01-01 00:00:00', 'YYYY-MM-DD HH24:MI:SS')
        else concat({util_text_to_date_validation(
            column=column_date,
            calendar_alias=calendar_alias,
            source_alias=source_alias
        )}, ' ', coalesce({util_text_to_time_validation(
            column=column_time,
            source_alias=source_alias,
        )}, '00:00:00'))::timestamp
    end'''

def update_release_branches(gitlab_conn_id: str, **kwargs):
    import gitlab
    from airflow.hooks.base import BaseHook
    from time import sleep
    
    logger = kwargs['dag'].logger()
    gitlab_hook = BaseHook.get_connection(gitlab_conn_id)
    gitlab_host, gitlab_token = gitlab_hook.host, gitlab_hook.password
    with gitlab.Gitlab(gitlab_host, private_token = gitlab_token) as gs:
        project = gs.projects.get(18)
        branch_list = project.branches.list(get_all=True)
        logger.info(f'Getting branches list for project {project.name}')

        for branch in branch_list:
            if 'release' in branch.name or 'hotfix' in branch.name:
                logger.info(f'Updating branch {branch.name}')
                try:
                    mr = project.mergerequests.create(
                        {
                            'source_branch': 'main',
                            'target_branch': branch.name,
                            'title': f'Update {branch.name} from main',
                        }
                    )
                    mrn = project.mergerequests.get(mr.iid)
                    sleep(5)
                    logger.info(f'Checking diffs for branch {branch.name}')
                    diffs = mrn.diffs.list(get_all=True)
                    if len(diffs) > 0:
                        logger.info(f'Updating branch {branch.name} from main branch')
                        mrn.merge(merge_when_pipeline_succeeds=True)
                    else:
                        logger.info('No diffs found. Deleting Merge Request')
                        mrn.delete()
                except Exception as e:
                    logger.error('Error -', str(e))

def get_sensor_run(date_cron, is_prev: bool = True):
    import pytz
    from croniter import croniter
    from datetime import timedelta
    from pendulum import DateTime
    base = datetime.now() - timedelta(days=1)
    base = base.astimezone(tz=pytz.timezone('Europe/Moscow'))
    iter_cron = croniter(date_cron, base)
    if is_prev:
        run = iter_cron.get_prev(DateTime)
    else:
        run = iter_cron.get_next(DateTime)
    return run
