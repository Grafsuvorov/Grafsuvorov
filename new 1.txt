# gp_copy_table_min.py
# -*- coding: utf-8 -*-
"""
Ручной перенос одной таблицы из prod-Greenplum в preprod потоком через COPY.
— Меняешь TABLE_FQTN = "schema.table" и запускаешь.
— Без REPEATABLE READ и без information_schema.
— Делает TRUNCATE целевой таблицы, COPY и ANALYZE (настраивается флагами ниже).

Зависимости: pip install psycopg2-binary
"""

import os
import sys
import queue
import threading
import psycopg2


# ===== НАСТРОЙКИ =====
SRC_DSN    = os.getenv("GP_SRC_DSN", "postgresql://user:pass@prod-host:5432/prod_db")
DST_DSN    = os.getenv("GP_DST_DSN", "postgresql://user:pass@preprod-host:5432/preprod_db")
TABLE_FQTN = "football.stats_match_fbref_v2"   # <<< поменяй на нужную таблицу schema.table

DO_TRUNCATE = True   # очистить целевую таблицу перед заливкой
DO_ANALYZE  = True   # ANALYZE после заливки
QUEUE_SIZE  = 64     # размер буфера стрима
# =====================


def qi(ident: str) -> str:
    """Quote identifier."""
    return '"' + ident.replace('"', '""') + '"'


def parse_fqtn(s: str):
    if not s or "." not in s:
        raise ValueError("TABLE_FQTN должен быть в формате schema.table")
    schema, table = s.split(".", 1)
    schema, table = schema.strip(), table.strip()
    if not schema or not table:
        raise ValueError("Неверный TABLE_FQTN. Нужен schema.table")
    return schema, table


class QueueWriter:
    """file-like writer для COPY TO STDOUT (источник)"""
    def __init__(self, q: queue.Queue):
        self.q = q
    def write(self, b):
        if isinstance(b, str):
            b = b.encode("utf-8")
        self.q.put(b)
    def flush(self):  # noqa
        pass


class QueueReader:
    """file-like reader для COPY FROM STDIN (приёмник)"""
    def __init__(self, q: queue.Queue, sentinel):
        self.q = q
        self.buf = bytearray()
        self.sentinel = sentinel
        self.closed = False
    def read(self, n=-1):
        while not self.closed and (n < 0 or len(self.buf) < n):
            chunk = self.q.get()
            if chunk is self.sentinel:
                self.closed = True
                break
            self.buf.extend(chunk)
        if n < 0:
            n = len(self.buf)
        out = bytes(self.buf[:n])
        del self.buf[:n]
        return out


def truncate_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"TRUNCATE TABLE {qi(schema)}.{qi(table)}")
        conn.commit()


def analyze_table(dsn: str, schema: str, table: str):
    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        cur.execute(f"ANALYZE {qi(schema)}.{qi(table)}")
        conn.commit()


def copy_table(schema: str, table: str):
    print("=== COPY prod → preprod ===")
    print(f"SRC:   {SRC_DSN}")
    print(f"DST:   {DST_DSN}")
    print(f"TABLE: {schema}.{table}")
    print("===========================")

    q = queue.Queue(maxsize=QUEUE_SIZE)
    sentinel = object()

    # Поток-приёмник (preprod)
    def dest_copy():
        copy_in_sql = (
            f"COPY {qi(schema)}.{qi(table)} "
            "FROM STDIN WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
        )
        with psycopg2.connect(DST_DSN) as conn, conn.cursor() as cur:
            cur.execute("SET client_encoding TO 'UTF8'")
            reader = QueueReader(q, sentinel)
            cur.copy_expert(copy_in_sql, reader)
            conn.commit()

    if DO_TRUNCATE:
        print("[DST] TRUNCATE…")
        truncate_table(DST_DSN, schema, table)

    print("[DST] COPY FROM STDIN — старт…")
    t = threading.Thread(target=dest_copy, daemon=True)
    t.start()

    # Источник (prod): COPY всей таблицы TO STDOUT (без колонок; структура должна совпадать)
    copy_out_sql = (
        f"COPY {qi(schema)}.{qi(table)} "
        "TO STDOUT WITH (FORMAT csv, DELIMITER ',', QUOTE '\"', ESCAPE '\"', NULL '\\N')"
    )
    print("[SRC] COPY TO STDOUT…")
    with psycopg2.connect(SRC_DSN) as src_conn, src_conn.cursor() as cur:
        cur.execute("SET client_encoding TO 'UTF8'")
        writer = QueueWriter(q)
        cur.copy_expert(copy_out_sql, writer)
        q.put(sentinel)  # завершение потока

    t.join()
    print("[OK] COPY завершён.")

    if DO_ANALYZE:
        print("[DST] ANALYZE…")
        analyze_table(DST_DSN, schema, table)
        print("[OK] ANALYZE done.")


def main():
    try:
        schema, table = parse_fqtn(TABLE_FQTN)
        copy_table(schema, table)
    except Exception as e:
        print(f"ERROR: {e}", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    main()
