import requests
import json
import urllib3

urllib3.disable_warnings()

YOUTRACK_URL = ""
YOUTRACK_TOKEN = ""
ISSUE_ID = "DWH-10152"

HEADERS = {
    "Authorization": f"Bearer {YOUTRACK_TOKEN}",
    "Accept": "application/json",
}


def api_get(path, params=None):
    r = requests.get(
        f"{YOUTRACK_URL}{path}",
        headers=HEADERS,
        params=params or {},
        verify=False,
        timeout=60,
    )
    print(f"\n>>> GET {path} | status={r.status_code}")
    if r.status_code != 200:
        print(r.text)
    r.raise_for_status()
    return r.json()


# -------------------------
# 1. ISSUE (основной источник)
# -------------------------
issue = api_get(
    f"/api/issues/{ISSUE_ID}",
    {
        "fields": "idReadable,project(id,name),customFields(name,value(name,text,presentation,$type))"
    },
)

print("\n=== ISSUE CUSTOM FIELDS ===")
direction_issue = None

for f in issue.get("customFields", []):
    print(f["name"], "=>", f.get("value"))
    if f["name"] == "Направление":
        direction_issue = f.get("value")

print("\n=== RESULT FROM ISSUE ===")
print("Direction:", direction_issue)


# -------------------------
# 2. ISSUE CUSTOM FIELDS (альтернативный endpoint)
# -------------------------
custom_fields = api_get(
    f"/api/issues/{ISSUE_ID}/customFields",
    {
        "fields": "name,value(name,text,presentation,$type)"
    },
)

print("\n=== ISSUE /customFields ===")
direction_cf = None

for f in custom_fields:
    print(f["name"], "=>", f.get("value"))
    if f["name"] == "Направление":
        direction_cf = f.get("value")

print("\n=== RESULT FROM /customFields ===")
print("Direction:", direction_cf)


# -------------------------
# 3. PROJECT FIELD (есть ли вообще)
# -------------------------
project_id = issue["project"]["id"]

project_fields = api_get(
    f"/api/admin/projects/{project_id}/customFields",
    {
        "fields": "field(name,id),$type"
    },
)

print("\n=== PROJECT FIELD CHECK ===")

for f in project_fields:
    field = f.get("field", {})
    if field.get("name") == "Направление":
        print("FOUND PROJECT FIELD:", json.dumps(f, ensure_ascii=False, indent=2))


# -------------------------
# 4. ACTIVITIES (история)
# -------------------------
print("\n=== ACTIVITIES RAW ===")

activities = api_get(
    f"/api/issues/{ISSUE_ID}/activitiesPage",
    {
        "fields": "activities"
    },
)

direction_history = None

for a in activities.get("activities", []):
    field = (a.get("field") or {}).get("name")

    if field == "Направление":
        print(json.dumps(a, ensure_ascii=False, indent=2))

        if a.get("added"):
            direction_history = a["added"].get("name")

print("\n=== RESULT FROM HISTORY ===")
print("Direction:", direction_history)


# -------------------------
# 5. TYPE CHECK
# -------------------------
print("\n=== TYPE CHECK ===")

for f in issue.get("customFields", []):
    if f["name"] == "Тип карточки":
        print("Тип:", f.get("value"))


# -------------------------
# 6. FINAL SUMMARY
# -------------------------
print("\n=== FINAL SUMMARY ===")

print("FROM ISSUE:", direction_issue)
print("FROM /customFields:", direction_cf)
print("FROM HISTORY:", direction_history)

if not direction_issue and direction_history:
    print("\n⚠️ ВАЖНО: поле есть только в истории, но НЕ в текущем issue")

if not direction_issue and not direction_history:
    print("\n❌ Поле полностью отсутствует (даже в истории)")
