SELECT
    base.*,

    -- материалы
    matet.material_name,
    mf.material_shape_full_name,

    -- ЕТСНГ
    ette.etsng_name,

    -- услуги
    zsrvpost.transportation_service_name,

    -- завод поставки
    pas.plant_short_name  AS recv_plant_short_name,
    pas.plant_full_name   AS recv_plant_full_name,

    -- собственник металла (завод из договора)
    deldp.plant_producer_code          AS owner_plant_code,
    pas_1124.plant_short_name          AS owner_plant_short_name,
    pas_1124.plant_full_name           AS owner_plant_full_name,

    -- этап перевозки
    trst.transportation_stage_name,

    -- экспедитор
    counter.counterparty_short_name    AS forwarder_short_name,

    -- кредитор
    counter_2.counterparty_short_name  AS creditor_short_name,
    addr_8.address_full_name          AS creditor_address_full_name,

    -- тип ПС
    ipd.transport_transfer_type_name
FROM dm_calc.storage_sales_bundles_amount base
    -- материалы
    LEFT JOIN tmp_material_texts matet
        ON matet.material_code = base.material_code
    LEFT JOIN tmp_material_shape mf
        ON mf.material_code = base.material_code

    -- услуги
    LEFT JOIN tmp_service_texts zsrvpost
        ON zsrvpost.transportation_service_code = base.service_code

    -- ЕТСНГ
    LEFT JOIN tmp_etsng_texts ette
        ON ette.etsng_code = base.etsng_code

    -- заводы
    LEFT JOIN tmp_plant pas
        ON pas.plant_code = base.receiving_plant_code

    -- собственник металла (завод из договора хранения)
    LEFT JOIN tmp_delivery_plant deldp
        ON deldp.delivery_code = base.delivery_for_storage_calculation_code
    LEFT JOIN tmp_plant pas_1124
        ON pas_1124.plant_code = deldp.plant_producer_code

    -- этап перевозки
    LEFT JOIN tmp_transportation_stage trst
        ON trst.transportation_stage_code = base.transportation_stage_code

    -- экспедитор
    LEFT JOIN tmp_counterparty counter
        ON counter.counterparty_code = base.forwarder_code

    -- кредитор + адрес
    LEFT JOIN tmp_counterparty counter_2
        ON counter_2.counterparty_code = base.creditor_in_shipment_instruction_code
    LEFT JOIN tmp_creditor_address addr_8
        ON addr_8.address_code = counter_2.address_code

    -- тип ПС (через temp, а не напрямую dict_dds)
    LEFT JOIN tmp_transport_type ipd
        ON ipd.transport_transfer_type_code = base.transport_type_code;
