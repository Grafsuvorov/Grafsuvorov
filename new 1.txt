SELECT run_uuid, start_dttm, end_dttm, status
  FROM click_fact_table_load_run
  WHERE schema_name = 'dm'
    AND table_name = 'otif_daily'
  ORDER BY start_dttm DESC
  LIMIT 5;

  -- 2) есть ли нужные стадии?
  SELECT run_uuid, stage_name, start_dttm, end_dttm, status
  FROM click_fact_table_load_stage
  WHERE run_uuid IN (
    SELECT run_uuid FROM click_fact_table_load_run
    WHERE schema_name = 'dm' AND table_name = 'otif_daily'
  )
  ORDER BY start_dttm DESC
  LIMIT 20;

  -- 3) попадание в окно (пример)
  SELECT run_uuid, stage_name, start_dttm, end_dttm
  FROM click_fact_table_load_stage
  WHERE stage_name IN ('UPLOAD_TO_S3','CLICKHOUSE_LOAD')
    AND start_dttm BETWEEN '2026-03-06 04:30:00' AND '2026-03-06 05:00:00'
    AND run_uuid IN (
      SELECT run_uuid FROM click_fact_table_load_run
      WHERE schema_name='dm' AND table_name='otif_daily'
    )
  ORDER BY start_dttm;
