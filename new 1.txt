-- === общие сеттинги на сессию ===
SET gp_workfile_compression = on;

-- ==============================================
-- ШАГ 1: cr_next — saldo на 1-е число след. месяца
-- выражения в GROUP BY УБРАЛИ, но dt_posting считаем сразу
-- ==============================================
SET LOCAL statement_mem = '3GB';
-- при спиллах HashAgg можно на время отключить:
-- SET LOCAL enable_hashagg = off;

DROP TABLE IF EXISTS cr_next;
CREATE TEMP TABLE cr_next ON COMMIT DROP AS
SELECT
  kalnr,
  (make_date(bdatj::int, substring(poper,2,2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
  MAX(salk3) AS salk3
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, (make_date(bdatj::int, substring(poper,2,2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);

ANALYZE cr_next;
-- RESET enable_hashagg;

-- ==============================================
-- ШАГ 2: mlcd_agg — агрегаты по тройке
-- ==============================================
SET LOCAL statement_mem = '2GB';
-- SET LOCAL enable_hashagg = off;  -- только если есть спиллы

DROP TABLE IF EXISTS mlcd_agg;
CREATE TEMP TABLE mlcd_agg ON COMMIT DROP AS
SELECT
  kalnr, bdatj, poper,
  SUM(salk3)  AS salk3,
  SUM(estprd) AS estprd,
  SUM(mstprd) AS mstprd
FROM ods.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);

ANALYZE mlcd_agg;
-- RESET enable_hashagg;

-- ==============================================
-- ШАГ 3: финал — меньше дублей и без выражений в Hash Join
-- ==============================================
SET LOCAL statement_mem = '8GB';
SET LOCAL optimizer_enable_broadcast = on;

WITH
pp AS (
  /* сразу убираем дубли по дате: DISTINCT по (kalnr, bdatj, poper) */
  SELECT DISTINCT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper,2,2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
  FROM ods.ckmlpp_ral p
  WHERE p.untper = '000'
),
dates AS (
  /* окно уже над "ужатым" pp — сорт/окно меньше и дешевле */
  SELECT
    kalnr,
    dt_valid_from,
    COALESCE(
      LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
  FROM pp
)
-- INSERT INTO dm.material_price_calculation
SELECT
  pp.kalnr AS calculation_code,
  pp.dt_valid_from,
  d.dt_valid_to,

  cr.vprsv AS price_valuation_type_code,
  cr.stprs AS standard_price_amount,
  cr.peinh AS price_unit_code,
  cr.pvprs AS moving_average_price_amount,

  /* rule3 */
  CASE WHEN (pp.umkumo + pp.abkumo + pp.zukumo) = 0 THEN 0
       ELSE TRUNC(
              CAST(ROUND(
                    (cr.stprs / cr.peinh) * (pp.umkumo + pp.abkumo)
                  + cr.abprd_o + cr.abprd_mo
                  + COALESCE(ma.salk3 + ma.estprd + ma.mstprd, 0)
              , 2) AS NUMERIC(40,20))
              / (pp.umkumo + pp.abkumo + pp.zukumo), 15)
  END AS average_weighted_stock_price_rule3_amount,

  /* rule2_prev_date */
  CASE WHEN (pp.abkumo + pp.umkumo + pp.zukumo) <> 0
       THEN TRUNC(
              CAST(
                  COALESCE(n.salk3,0)
                + COALESCE(ma.salk3,0) + COALESCE(ma.estprd,0) + COALESCE(ma.mstprd,0)
              AS NUMERIC(40,20))
              / (pp.abkumo + pp.umkumo + pp.zukumo), 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_prev_date_amount,

  /* rule2_next_date */
  CASE WHEN pp.lbkum <> 0
       THEN TRUNC(CAST(COALESCE(cr.salk3,0) AS NUMERIC(40,20)) / pp.lbkum, 15)
       ELSE 0
  END AS average_weighted_stock_price_rule2_next_date_amount,

  h.price_rule_code
FROM pp
LEFT JOIN ods.ckmlcr_ral cr
  ON cr.kalnr = pp.kalnr
 AND cr.bdatj = pp.bdatj
 AND cr.poper = pp.poper
 AND cr.untper = '000'
 AND cr.curtp  = '10'
LEFT JOIN cr_next n
  ON n.kalnr = pp.kalnr
 AND n.dt_posting = pp.dt_valid_from      -- БЕЗ выражений в ON
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr
 AND ma.bdatj = pp.bdatj
 AND ma.poper = pp.poper
LEFT JOIN dates d
  ON d.kalnr = pp.kalnr
 AND d.dt_valid_from = pp.dt_valid_from
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = pp.kalnr
;
