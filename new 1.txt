/* ===============================
   ГЛОБАЛЬНЫЕ РЕКОМЕНДАЦИИ (локально на сессию)
   =============================== */
-- сжатие workfile'ов почти всегда помогает на спиллах
SET gp_workfile_compression = on;

/* =========================================================
   ШАГ 1: cr_next  — сальдо на 1-е число следующего месяца
   ---------------------------------------------------------
   ВАЖНО: агрегируем по (kalnr,bdatj,poper) — без выражения!
   dt_posting посчитаем динамически в JOIN финального шага,
   чтобы избежать Redistribute до агрегации.
   ========================================================= */
SET LOCAL statement_mem = '3GB';
-- Если видите "32 batches; spill groups" в HashAgg — можно так:
-- SET LOCAL enable_hashagg = off;  -- использовать сорт-аггрегат

DROP TABLE IF EXISTS cr_next;
CREATE TEMP TABLE cr_next ON COMMIT DROP AS
SELECT
  kalnr,
  bdatj,
  poper,
  MAX(salk3) AS salk3           -- защита от дублей
FROM ods.ckmlcr_ral
WHERE untper = '000'
  AND curtp  = '10'
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);

ANALYZE cr_next;

-- RESET enable_hashagg;  -- если включали выше


/* =========================================================
   ШАГ 2: mlcd_agg — агрегаты по тройке (ZU, VP, PC)
   ---------------------------------------------------------
   Неминуемый Redistribute по (kalnr,bdatj,poper) — ок.
   Даем памяти, чтобы сократить батчи/спиллы.
   ========================================================= */
SET LOCAL statement_mem = '2GB';
-- Если всё же льётся в HashAgg — разрешённый прием:
-- SET LOCAL enable_hashagg = off;

DROP TABLE IF EXISTS mlcd_agg;
CREATE TEMP TABLE mlcd_agg ON COMMIT DROP AS
SELECT
  kalnr, bdatj, poper,
  SUM(salk3)  AS salk3,
  SUM(estprd) AS estprd,
  SUM(mstprd) AS mstprd
FROM ods.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);

ANALYZE mlcd_agg;

-- RESET enable_hashagg;  -- если включали


/* =========================================================
   ШАГ 3: финальный INSERT (или SELECT)
   ---------------------------------------------------------
   - Узкая проекция в CTE pp
   - Окно по датам — на DISTINCT (узкий набор)
   - dt_posting считаем прямо в ON при JOIN с cr_next
   - Коллокация по kalnr сохраняется
   ========================================================= */
SET LOCAL statement_mem = '6GB';
SET LOCAL optimizer_enable_broadcast = on;  -- пусть мелкие таблицы уйдут в broadcast при необходимости

WITH
pp AS (
  SELECT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper, 2, 2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
  FROM ods.ckmlpp_ral p
  WHERE p.untper = '000'
),
dates AS (
  SELECT
    kalnr,
    dt_valid_from,
    COALESCE(
      LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
  FROM (SELECT kalnr, dt_valid_from FROM pp GROUP BY kalnr, dt_valid_from) s
)

-- Вариант 1: сразу загрузка в витрину
-- INSERT INTO dm.material_price_calculation
-- Вариант 2: посмотреть план/результат
-- EXPLAIN (ANALYZE, TIMING OFF)
SELECT
  pp.kalnr        AS calculation_code,
  pp.dt_valid_from,
  d.dt_valid_to,

  cr.vprsv        AS price_valuation_type_code,
  cr.stprs        AS standard_price_amount,
  cr.peinh        AS price_unit_code,
  cr.pvprs        AS moving_average_price_amount,

  /* rule3 */
  CASE
    WHEN (pp.umkumo + pp.abkumo + pp.zukumo) = 0 THEN 0
    ELSE TRUNC(
      CAST(ROUND(
            (cr.stprs / cr.peinh) * (pp.umkumo + pp.abkumo)
          + cr.abprd_o + cr.abprd_mo
          + COALESCE(ma.salk3 + ma.estprd + ma.mstprd, 0)
      , 2) AS NUMERIC(40,20))
      / (pp.umkumo + pp.abkumo + pp.zukumo), 15)
  END AS average_weighted_stock_price_rule3_amount,

  /* rule2_prev_date (равная дата) */
  CASE
    WHEN (pp.abkumo + pp.umkumo + pp.zukumo) <> 0
    THEN TRUNC(
      CAST(
        COALESCE(n.salk3, 0)
        + COALESCE(ma.salk3, 0)
        + COALESCE(ma.estprd, 0)
        + COALESCE(ma.mstprd, 0)
      AS NUMERIC(40,20)) / (pp.abkumo + pp.umkumo + pp.zukumo), 15)
    ELSE 0
  END AS average_weighted_stock_price_rule2_prev_date_amount,

  /* rule2_next_date (не равная дата) */
  CASE
    WHEN pp.lbkum <> 0
    THEN TRUNC(CAST(COALESCE(cr.salk3, 0) AS NUMERIC(40,20)) / pp.lbkum, 15)
    ELSE 0
  END AS average_weighted_stock_price_rule2_next_date_amount,

  h.price_rule_code
FROM pp
LEFT JOIN ods.ckmlcr_ral cr
  ON cr.kalnr = pp.kalnr
 AND cr.bdatj = pp.bdatj
 AND cr.poper = pp.poper
 AND cr.untper = '000'
 AND cr.curtp  = '10'
LEFT JOIN cr_next n
  ON n.kalnr = pp.kalnr
 -- считаем dt_posting на лету: 1-е число следующего месяца
 AND (make_date(n.bdatj::int, substring(n.poper, 2, 2)::int, 1) + INTERVAL '1 month')::date = pp.dt_valid_from
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr
 AND ma.bdatj = pp.bdatj
 AND ma.poper = pp.poper
LEFT JOIN dates d
  ON d.kalnr = pp.kalnr
 AND d.dt_valid_from = pp.dt_valid_from
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = pp.kalnr
;
