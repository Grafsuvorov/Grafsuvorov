DROP TABLE IF EXISTS order_union          ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS params_z_matkl      ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS params_bwrt         ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS params_auart        ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS params_autyp        ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS mg_base             ON COMMIT PRESERVE ROWS;
DROP TABLE IF EXISTS pt1_material_sum    ON COMMIT PRESERVE ROWS;

-- ===========================
-- 1) order_union — большой temp, распределяем по ключу джойна
-- ===========================
CREATE TEMPORARY TABLE order_union (
    order_type_code      varchar(4),
    order_code           varchar(12),
    order_category_code  varchar
)
WITH (
    appendonly = true,
    orientation = column,
    compresstype = zstd,
    compresslevel = 3
)
DISTRIBUTED BY (order_code);

INSERT INTO order_union
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_toro
UNION ALL
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_production
UNION ALL
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_controlling;

ANALYZE order_union;

-- ===========================
-- 2) Параметры — микротаблицы (чтобы не строить большие hash join)
-- ===========================
CREATE TEMP TABLE params_z_matkl AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code = 'ZMM2968M_BW'
  AND parameter_code    = 'Z_MATKL'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_bwrt AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code = '/RUSAL/CO312K'
  AND parameter_code    = 'BWRT_ZKP'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_auart AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code = '/RUSAL/CO706M'
  AND parameter_code    = 'AUART_BU'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_autyp AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code = '/RUSAL/CO706M'
  AND parameter_code    = 'AUTYP_BU'
DISTRIBUTED RANDOMLY;

ANALYZE params_z_matkl;
ANALYZE params_bwrt;
ANALYZE params_auart;
ANALYZE params_autyp;

-- ===========================
-- 3) База движений (mg_base) — только нужные колонки + правильное DISTRIBUTED BY
--    Ключи под дальнейшие join: (plant_code, material_code)
-- ===========================
CREATE TEMP TABLE mg_base
WITH (
    appendonly = true,
    orientation = column,
    compresstype = zstd,
    compresslevel = 3
)
DISTRIBUTED BY (plant_code, material_code)
AS
SELECT
    date_trunc('month', mg.dt_posting)  AS mon,
    mg.plant_code,
    mg.material_code,
    mg.uom_code,
    mg.debit_or_credit_code,
    mg.quantity,
    mg.operation_type_code,
    mg.receipt_type_code,
    mg.movement_type_code,
    mg.order_code
FROM dds.material_movement_document mg
WHERE mg.posting_rule_code IS NOT NULL
  AND mg.sales_document_number_of_valuated_stock_code IS NULL
  AND mg.wbs_element_of_valuated_stock_code = '00000000';

ANALYZE mg_base;

-- ===========================
-- 4) Итоговая таблица суммы (TEMP) — как в исходнике, но без двойной агрегации
-- ===========================
CREATE TEMPORARY TABLE pt1_material_sum (
    spmon    varchar,
    werks    varchar,
    matnr    varchar,
    MENGE11  numeric(28,10),
    MENGE22  numeric(28,10),
    meins    varchar,
    ekgrp    varchar,
    dispo    varchar
)
WITH (
    appendonly = true,
    orientation = column,
    compresstype = zstd,
    compresslevel = 3
)
DISTRIBUTED RANDOMLY;

INSERT INTO pt1_material_sum
SELECT
    to_char(b.mon, 'YYYY-MM')                 AS spmon,
    b.plant_code                              AS werks,
    b.material_code                           AS matnr,
    SUM(CASE
            WHEN b.debit_or_credit_code = 'H' THEN b.quantity * -1
            ELSE b.quantity
        END)                                  AS MENGE11,
    SUM(CASE
            WHEN b.order_code IS NOT NULL
                 AND NOT EXISTS (SELECT 1 FROM params_autyp t WHERE t.range_low_value = ou.order_category_code)
                 AND EXISTS     (SELECT 1 FROM params_auart a WHERE a.range_low_value = ou.order_type_code)
              THEN 0
            WHEN b.debit_or_credit_code = 'H'
                 AND b.operation_type_code = 'WE'
                 AND b.receipt_type_code IS NULL
                 AND NOT EXISTS (SELECT 1 FROM params_bwrt x WHERE x.range_low_value = b.movement_type_code)
              THEN b.quantity * -1
            WHEN b.debit_or_credit_code <> 'H'
                 AND b.operation_type_code = 'WE'
                 AND b.receipt_type_code IS NULL
                 AND NOT EXISTS (SELECT 1 FROM params_bwrt x WHERE x.range_low_value = b.movement_type_code)
              THEN b.quantity
            ELSE 0
        END)                                  AS MENGE22,
    b.uom_code                                AS meins,
    mbypl.purchase_group_code                 AS ekgrp,
    mbypl.material_planning_controller_code   AS dispo
FROM mg_base b
JOIN dict_dds.material_by_plant mbypl
  ON mbypl.material_code = b.material_code
 AND mbypl.plant_code    = b.plant_code
JOIN dict_dds.material ma
  ON ma.material_code = mbypl.material_code
LEFT JOIN order_union ou
  ON ou.order_code = b.order_code
LEFT JOIN params_z_matkl pm
  ON pm.range_low_value = ma.material_group_code
WHERE pm.range_low_value IS NULL      -- эквивалентно вашему фильтру param_m.abap_program_code IS NULL
GROUP BY
    b.mon,
    b.plant_code,
    b.material_code,
    b.uom_code,
    mbypl.purchase_group_code,
    mbypl.material_planning_controller_code
;

ANALYZE pt1_material_sum;
