-- 0) Безопасные дефолты для сеанса (по ситуации).
-- SET optimizer=on; -- ORCA по умолчанию ON
-- SET statement_mem='2GB'; -- если у вас принято это трогать для тяжёлых запросов

--------------------------
-- 1) CKMLPP: предвычисляем дату, убираем лишние поля
--------------------------
CREATE TEMP TABLE pp ON COMMIT DROP AS
SELECT
    kalnr,
    bdatj,
    poper,
    make_date(bdatj::int, substring(poper, 2, 2)::int, 1) AS dt_valid_from,
    umkumo, abkumo, zukumo, lbkum
FROM ods.ckmlpp_ral
WHERE untper = '000'
DISTRIBUTED BY (kalnr);
ANALYZE pp;

--------------------------
-- 2) CKMLCR (текущий период): гарантируем 1 строку на (kalnr,bdatj,poper)
--    Если в источнике встречаются дубляжи, выберите бизнес-правило:
--    SUM(...) / MAX(...) / LAST_VALUE по нужному полю (пример ниже: MAX для “параметров”, SUM для потоков).
--------------------------
CREATE TEMP TABLE cr_curr ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    -- при необходимости заменить на агрегаты, согласованные с бизнесом
    MAX(stprs)    AS stprs,
    MAX(peinh)    AS peinh,
    MAX(pvprs)    AS pvprs,
    MAX(abprd_o)  AS abprd_o,
    MAX(abprd_mo) AS abprd_mo,
    MAX(vprsv)    AS vprsv,
    MAX(salk3)    AS salk3   -- или SUM(...) если это поток/движение, а не срез
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);
ANALYZE cr_curr;

--------------------------
-- 3) CKMLCR (следующий месяц): salk3 на dt_posting = 1-е число следующего месяца (DATE!)
--------------------------
CREATE TEMP TABLE cr_next ON COMMIT DROP AS
SELECT
    kalnr,
    (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    -- если бывают дубли на один и тот же dt_posting — также агрегируйте
    MAX(salk3) AS salk3
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);
ANALYZE cr_next;

--------------------------
-- 4) MLCD агрегаты
--------------------------
CREATE TEMP TABLE mlcd_agg ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    SUM(salk3)  AS salk3,
    SUM(estprd) AS estprd,
    SUM(mstprd) AS mstprd
FROM ods.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);
ANALYZE mlcd_agg;

--------------------------
-- 5) Один WindowAgg для границ действия
--------------------------
WITH ranges AS (
    SELECT
        kalnr,
        dt_valid_from,
        COALESCE(
            LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
            DATE '2299-12-31'
        )::date AS dt_valid_to
    FROM pp
)
SELECT
    r.kalnr AS calculation_code,
    r.dt_valid_from,
    r.dt_valid_to,
    cr_curr.vprsv AS price_valuation_type_code,
    cr_curr.stprs  AS standard_price_amount,
    cr_curr.peinh  AS price_unit_code,
    cr_curr.pvprs  AS moving_average_price_amount,

    /* rule3 */
    CASE WHEN (pp.umkumo + pp.abkumo + pp.zukumo) = 0 THEN 0
         ELSE TRUNC(
                CAST(ROUND(
                      (cr_curr.stprs / cr_curr.peinh) * (pp.umkumo + pp.abkumo)
                    + cr_curr.abprd_o + cr_curr.abprd_mo
                    + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                , 2) AS NUMERIC(40,20))
                / (pp.umkumo + pp.abkumo + pp.zukumo), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE WHEN (pp.abkumo + pp.umkumo + pp.zukumo) <> 0
         THEN TRUNC(
                CAST(
                    cr_next.salk3 + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                AS NUMERIC(40,20))
                / (pp.abkumo + pp.umkumo + pp.zukumo), 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE WHEN pp.lbkum <> 0
         THEN TRUNC(CAST(cr_curr.salk3 AS NUMERIC(40,20)) / pp.lbkum, 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    ckmlhd.price_rule_code
FROM ranges r
JOIN pp
  ON pp.kalnr = r.kalnr
 AND pp.dt_valid_from = r.dt_valid_from
LEFT JOIN cr_curr
  ON cr_curr.kalnr = pp.kalnr
 AND cr_curr.bdatj = pp.bdatj
 AND cr_curr.poper = pp.poper
LEFT JOIN cr_next
  ON cr_next.kalnr = pp.kalnr
 AND cr_next.dt_posting = pp.dt_valid_from
LEFT JOIN mlcd_agg
  ON mlcd_agg.kalnr = pp.kalnr
 AND mlcd_agg.bdatj = pp.bdatj
 AND mlcd_agg.poper = pp.poper
LEFT JOIN dds.material_ledger_header ckmlhd
  ON ckmlhd.calculation_code = pp.kalnr
;

Что вынести на презентацию (каркас слайдов)

Исходная проблема
▸ Большие объемы (сотни млн строк), тяжёлые окна, множественные JOIN’ы.
▸ Симптомы: два WindowAgg/Sort, смешение типов, избыточные движения данных.

Быстрые победы
▸ Один LEAD с единым ORDER BY и COALESCE(... - 1 day).
▸ make_date(...) вместо строк.
▸ Приведение dt_posting к date.

Коллокация и распределение
▸ Почему достаточно DISTRIBUTED BY (kalnr) при PARTITION BY kalnr.
▸ Где и почему возникает Redistribute Motion, как его уменьшили.

Кардинальность и дедупликация
▸ Взрыв строк из-за неуникальных (kalnr,bdatj,poper).
▸ TEMP-агрегации cr_curr/cr_next, выбор корректных агрегатов.

Итоги по людям (конструктивно)

Костя: правильно упростил окно (-1 WindowAgg/Sort), но оставил лишний partition key и склейку дат.

Настя: верно вынесла препроцессинг в TEMP и распределение по (kalnr,bdatj,poper), но допустила логическую ошибку с dt_valid_to и не устранила многократность ckmlcr_2 → взрыв кардинальности.

Результат и метрики
▸ Снижение Total Cost (покажите «до/после»), уменьшение Motion, один WindowAgg.
▸ Рекомендации: фиксировать агрегаты для cr_* в коде ETL, добавить unit-тест «единственности» ключей.


Почему именно (kalnr)

Окно без Motion.
У тебя LEAD(...) OVER (PARTITION BY kalnr ORDER BY dt_valid_from).
Чтобы окно считалось локально на сегментах, все строки одного kalnr должны лежать на одном сегменте. Это гарантирует DISTRIBUTED BY (kalnr). Если распределить по (kalnr, bdatj, poper), строки одного kalnr окажутся на разных сегментах → перед WindowAgg придётся делать Redistribute/Gather.

Коллокация джойнов.
Джойнимся по kalnr, bdatj, poper, но ключевой для коллокации — общий хэш-ключ. Если все участвующие TEMP’ы распределены по одному и тому же ключу (kalnr) и в ON есть ... A.kalnr = B.kalnr ..., то Greenplum держит все «одинаковые kalnr» на одном сегменте и выполняет оставшиеся предикаты (bdatj, poper) уже локально, без Motion.
Если одну таблицу распределить по (kalnr,bdatj,poper), а другую — по (kalnr), хэши будут разными → ORCA чаще вставит Redistribute.

Простая симметрия.
Один и тот же distkey на всех TEMP’ах уменьшает шанс непредсказуемых Motion при разных планах.

Когда стоит думать о (kalnr, bdatj, poper)

Только если у тебя сильный перекос по kalnr (пару «жирных» kalnr, у которых миллионы строк). Тогда можно сделать «двухфазный» подход:

Фаза 1 (окно): считать ranges на таблице, распределённой по kalnr (как сейчас).

Фаза 2 (тяжёлые джойны): материализовать результат окна во временную таблицу, перераспределив по тройке, и все большие стороны джойна (cr_curr, mlcd_agg) тоже CTAS-ом перегнать на ту же тройку. Тогда джойны по полной тройке пройдут локально, но окно останется быстрым.

Пример «гибрида», если вдруг понадобится:

-- уже посчитали ranges + pp при DISTRIBUTED BY (kalnr)

CREATE TEMP TABLE pp3 ON COMMIT DROP AS
SELECT * FROM pp
DISTRIBUTED BY (kalnr, bdatj, poper);

CREATE TEMP TABLE cr_curr3 ON COMMIT DROP AS
SELECT * FROM cr_curr
DISTRIBUTED BY (kalnr, bdatj, poper);

CREATE TEMP TABLE mlcd3 ON COMMIT DROP AS
SELECT * FROM mlcd_agg
DISTRIBUTED BY (kalnr, bdatj, poper);

-- теперь большие джойны по тройке без Motion:
... JOIN cr_curr3 USING (kalnr,bdatj,poper)
... JOIN mlcd3   USING (kalnr,bdatj,poper);

Как понять, нужен ли «гибрид»

Проверь перекос по сегментам после CTAS по kalnr:

SELECT gp_segment_id, count(*) 
FROM pp 
GROUP BY 1 ORDER BY 2 DESC;


Если топ-сегмент сильно тяжелее (в разы) — есть смысл подумать о второй фазе с перераспределением по тройке или о «салтинге» горячих kalnr.
