with cte as 
	(select 
		t.unit_balance_code,
		t.asset_main_code,
		t.asset_sub_code, 
		t.depreciation_area_code,
		t.dt_from,
		row_number() over (partition by t.unit_balance_code, t.asset_main_code, t.asset_sub_code, t.depreciation_area_code 
		order by t.dt_from) as rn 
		from  (
			select 
				fix.unit_balance_code as unit_balance_code,
				fix.asset_main_code as asset_main_code,
				fix.asset_sub_code as asset_sub_code,
				quot.depreciation_area_code,
				fix.dt_from as dt_from
			from dict_dds.fixed_asset_td_v2 as fix 
			left join dict_dds.fixed_asset_depreciation_parameters_td_v2 as quot
				on quot.unit_balance_code = fix.unit_balance_code
				and quot.asset_main_code = fix.asset_main_code
				and quot.asset_sub_code = fix.asset_sub_code
			union all
			select 
				 quot.unit_balance_code as unit_balance_code,
				quot.asset_main_code as asset_main_code,
				 quot.asset_sub_code as asset_sub_code,
				quot.depreciation_area_code,
				quot.dt_valid_from as dt_from
			from dict_dds.fixed_asset_depreciation_parameters_td_v2 as quot 
			left join dict_dds.fixed_asset_td_v2 as conv
				on quot.unit_balance_code = conv.unit_balance_code
				and quot.asset_main_code = conv.asset_main_code
				and quot.asset_sub_code = conv.asset_sub_code
				--where conv.unit_balance_code is null
			union all
			select
			coalesce(liq.unit_balance_code, quot.unit_balance_code) as unit_balance_code,
			coalesce(liq.asset_main_code, quot.asset_main_code) as asset_main_code,
			coalesce(liq.asset_sub_code, quot.asset_sub_code) as asset_sub_code,
			quot.depreciation_area_code,
			coalesce(liq.dt_valid_from, quot.dt_valid_from) as dt_from
			from dict_dds.fixed_asset_depreciation_parameters_td_v2 as quot
			left join
				(select 
					t1.unit_balance_code,
					t1.asset_main_code,
					t1.asset_sub_code,
					t1.asset_inventory_number,
					t2.dt_valid_from,
					t2.dt_valid_to 
				from 
					dict_dds.fixed_asset_td_v2 as t1
				join dict_dds.fixed_asset_non_liquid_technical_state_td as t2 on
				 	t2.unit_balance_code = t1.unit_balance_code 
					and t2.asset_inventory_code = t1.asset_inventory_number)	as liq on
					quot.unit_balance_code = liq.unit_balance_code
				and quot.asset_main_code = liq.asset_main_code
				and quot.asset_sub_code = liq.asset_sub_code
		) as t
		group by 
			t.unit_balance_code,
			t.asset_main_code,
			t.asset_sub_code,
			t.depreciation_area_code,
			t.dt_from), 
sep_int as (
	select 
		t1.unit_balance_code, 
		t1.asset_main_code, 
		t1.asset_sub_code,
		t1.depreciation_area_code,
		case 
			when t1.dt_from is null then to_date('19000101'::text, 'YYYYMMDD') 
			else  t1.dt_from end as dt_from,
		case 
			when t1.dt_from is null and t2.dt_from is null then (tcf_test.dt_from - '1day'::interval)::date  
			when t2.dt_from is null then to_date('22991231'::text, 'YYYYMMDD') 
			else  (t2.dt_from - '1day'::interval)::date end as dt_to,
		t1.rn
	from cte as t1 
	left join cte as t2
		on t1.unit_balance_code = t2.unit_balance_code 
		and t1.asset_main_code = t2.asset_main_code 
		and t1.asset_sub_code = t2.asset_sub_code
		and t1.depreciation_area_code = t2.depreciation_area_code
		and t1.rn = t2.rn-1
	left join cte as tcf_test 
		on tcf_test.unit_balance_code = t1.unit_balance_code 
		and tcf_test.asset_main_code = t1.asset_main_code 
		and tcf_test.asset_sub_code = t1.asset_sub_code
		and tcf_test.depreciation_area_code = t1.depreciation_area_code
		and tcf_test.rn = t1.rn - 1
	where t1.unit_balance_code is not null
		and t1.asset_main_code is not null
		and t1.asset_sub_code is not null)
