/* ===========================
   1. База условий договора 1079
   =========================== */

DROP TABLE IF EXISTS for_1079_base;

CREATE TEMP TABLE for_1079_base AS
WITH base AS (
    SELECT
        ekpo.purchase_contract_code,
        ekpo.position_line_item_code,

        ekko.unit_balance_code                    AS bukrs,
        ekko.supplier_code                        AS lifnr,
        ekko.port_code                            AS zzport,
        ekko.dt_valid_from                        AS kdatb,
        ekko.dt_valid_to                          AS kdate,
        ekpo.plant_code                           AS werks,
        ekpo.service_code                         AS zzsrvpos,
        ekpo.shape_code                           AS zzforma,
        ekpo.etsng_code                           AS zzet_tarif,
        ekpo.transport_type_code                  AS sdabw,

        pr.dt_price_condition_valid_to            AS datbi,
        pr.dt_price_condition_valid_from          AS datab,

        ekko.dt_valid_to,
        ekko.dt_valid_from,
        ekko.purchase_document_subtype_code,

        pr.price_condition_position_code          AS kopos,
        pr.price_condition_position_currency_code AS konwa,

        ekpo.terminal_code                        AS zzterminal,
        ekpo.warehouse_storage_area_code          AS vzzzone,

        COALESCE(pr.price_scale_currency_code,
                 pr.price_condition_position_currency_code) AS konws,

        ekpo.service_code,
        ekpo.price_uom_code,
        ekpo.price_vat_excluded_document_currency_amount AS netpr,
        pr.price_rate                             AS kbetr,

        /* ----- ORDER_WEIGHTS FOR RF / ASIA / EURO ----- */
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END) AS order_rf,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.warehouse_storage_area_code IS NULL THEN 1 ELSE 0 END) AS order_asia,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END) AS order_euro_svh,

        /* флаг "специфичного" контракта:
           1, если есть ХОТЯ БЫ ОДИН заполненный параметр детализации */
        CASE
            WHEN ekko.port_code IS NOT NULL
              OR ekpo.shape_code IS NOT NULL
              OR ekpo.etsng_code IS NOT NULL
              OR ekpo.transport_type_code IS NOT NULL
              OR ekpo.terminal_code IS NOT NULL
              OR ekpo.warehouse_storage_area_code IS NOT NULL
            THEN 1 ELSE 0
        END AS is_specific,

        /* множитель валюты договора */
        CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor,

        pr.is_price_condition_position_deleted_code,

        COUNT(*) OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code
        ) AS cnt,

        ROW_NUMBER() OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code,
                         pr.dt_price_condition_valid_from,
                         pr.dt_price_condition_valid_to,
                         pr.price_scale_basis_code
            ORDER BY pr.price_scale_basis_consecutive_code
        ) AS rwn

    FROM dds.purchase_contract_price_conditions pr
    JOIN dds.purchase_contract_header ekko
         ON ekko.purchase_contract_code = pr.purchase_contract_code
    JOIN dds.purchase_contract_position ekpo
         ON ekpo.purchase_contract_code = ekko.purchase_contract_code
        AND ekpo.position_line_item_code = pr.purchase_contract_position_code

    LEFT JOIN dict_dds.currency_decimal_place_ral curdpr
           ON curdpr.currency_code = COALESCE(pr.price_scale_currency_code,
                                              pr.price_condition_position_currency_code)
          AND curdpr.decimal_place_number = '0'

    WHERE
        ekko.purchase_document_subtype_code IN ('ZDTL','ZDT','ZDE','ZDP','ZDF','ZDS')
        AND ekko.paydox_contract_type_code IN ('1','6')
        AND ekpo.is_deleted IS NULL
        AND pr.is_price_condition_position_deleted_code IS NULL
        AND pr.price_condition_type_code = 'PB00'
        AND (pr.price_scale_basis_code = 'C' OR pr.price_scale_basis_code IS NULL)
)
SELECT *
FROM base
WHERE rwn = 1
DISTRIBUTED BY (purchase_contract_code, position_line_item_code);

/* ===========================
   4. Европа
   =========================== */

DROP TABLE IF EXISTS for_1079_euro;

CREATE TEMP TABLE for_1079_euro AS
WITH euro_base AS (
    SELECT
        b.rn,
        f.purchase_contract_code,
        f.position_line_item_code,
        f.konws,
        f.service_code,
        f.price_uom_code,
        f.netpr,
        f.currency_dogovor,
        f.datab,
        f.datbi,

        /* Суммарный вес по складу и месяцу окончания хранения */
        SUM(b.sales_bundle_gross_weight) OVER (
            PARTITION BY b.warehouse_code,
                         date_trunc('month', b.dt_storage_end)
        ) AS warehouse_weight_month_end,

        /* Разница в месяцах между началом и концом хранения (month_difference_2) */
        CASE
            WHEN EXTRACT(day FROM b.dt_storage_end) < EXTRACT(day FROM b.dt_storage_start) THEN
                (EXTRACT(year  FROM b.dt_storage_end) - EXTRACT(year  FROM b.dt_storage_start)) * 12 +
                (EXTRACT(month FROM b.dt_storage_end) - EXTRACT(month FROM b.dt_storage_start))
            ELSE
                (EXTRACT(year  FROM b.dt_storage_end) - EXTRACT(year  FROM b.dt_storage_start)) * 12 +
                (EXTRACT(month FROM b.dt_storage_end) - EXTRACT(month FROM b.dt_storage_start)) + 1
        END AS month_difference_2,

        /* special — количество недель или месяцев, в зависимости от типа логики */
        CASE
            WHEN b.storage_cost_special_calculation_type_code = '4'
                THEN CEIL(b.storage_duration_total_calendar_days / 7::numeric)
            WHEN b.storage_cost_special_calculation_type_code = '5'
                THEN CEIL(b.storage_duration_total_calendar_days / 30::numeric)
        END AS special,

        ROW_NUMBER() OVER (
            PARTITION BY b.rn
            ORDER BY
                f.order_euro_svh,
                f.kopos,
                f.kbetr,
                CASE
                    WHEN param_sap.range_low_value IS NOT NULL
                     AND param_srv_1.range_low_value IS NOT NULL THEN 1
                    WHEN param_sap.range_low_value IS NULL
                     AND param_srv_1.range_low_value IS NULL THEN 1
                    ELSE 0
                END DESC,
                f.purchase_contract_code,
                f.position_line_item_code
        ) AS row_num
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN for_1079_base f
      ON f.bukrs   = '1520'
     AND f.lifnr   = b.forwarder_code
     AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
     AND f.werks   = '1521'
     AND (f.zzforma    = b.material_shape_code OR f.zzforma    IS NULL)
     AND (f.zzet_tarif = b.etsng_code          OR f.zzet_tarif IS NULL)
     AND (f.sdabw      = b.transport_type_code OR f.sdabw      IS NULL)
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.datab AND f.datbi
     AND b.storage_cost_calculation_type_name IN ('Логика Европа')
     AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
     AND f.purchase_document_subtype_code = 'ZDTL'

    JOIN dict_dds.settings_and_parameters_sap s_eu
      ON s_eu.range_low_value    = f.zzsrvpos
     AND s_eu.range_low_value LIKE 'EU%'
     AND s_eu.abap_program_code  = '/RUSAL/SD4553M'
     AND s_eu.parameter_code     = 'S_USLST'

    LEFT JOIN dict_dds.settings_and_parameters_sap param_sap
      ON param_sap.range_low_value = b.material_shape_code
     AND param_sap.abap_program_code = '/RUSAL/SD4553M'
     AND param_sap.parameter_code    = 'FORMSTEU'

    LEFT JOIN dict_dds.settings_and_parameters_sap param_srv_1
      ON param_srv_1.range_low_value = s_eu.range_low_value
     AND param_srv_1.abap_program_code = '/RUSAL/SD4553M'
     AND param_srv_1.parameter_code    = 'S_USLSTC'
)
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    warehouse_weight_month_end,
    month_difference_2,
    netpr,
    special,
    currency_dogovor,
    datab,
    datbi
FROM euro_base
WHERE row_num = 1
DISTRIBUTED BY (rn);

/* ===========================
   8. for_1090 — интервальные шкалы
   =========================== */

DROP TABLE IF EXISTS for_1090;

CREATE TEMP TABLE for_1090 AS
WITH tmp AS (
    SELECT
        pcpc.price_condition_code                AS knumh,
        pcpc.purchase_contract_code              AS evrtn,
        pcpc.purchase_contract_position_code     AS evrtp,
        pcpc.price_scale_basis_consecutive_code,
        pcpc.price_condition_position_code,
        pcpc.price_rate                          AS konp_kbetr,
        pcpc.price_scale_rate                    AS konm_kbetr,
        pcpc.price_scale_limit_from_quantity     AS kstbm,
        pcpc.price_scale_limit_to_quantity       AS kstbm_next,
        pcpc.price_scale_basis_code,
        pcpc.dt_price_condition_valid_to,
        pcpc.dt_price_condition_valid_from,

        COUNT(pcpc.price_scale_limit_from_quantity)
            OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code) AS cnt,

        ROW_NUMBER() OVER (
            PARTITION BY pcpc.purchase_contract_code,
                         pcpc.purchase_contract_position_code
            ORDER BY pcpc.price_scale_basis_consecutive_code
        ) AS rn,

        MIN(price_scale_limit_from_quantity) OVER (
            PARTITION BY pcpc.purchase_contract_code,
                         pcpc.purchase_contract_position_code,
                         dt_price_condition_valid_from,
                         dt_price_condition_valid_to
        ) AS min_price_scale_limit_from_quantity,

        (pcpc.price_scale_limit_from_quantity -
            LAG(pcpc.price_scale_limit_from_quantity)
                OVER (PARTITION BY pcpc.purchase_contract_code,
                                   pcpc.purchase_contract_position_code
                      ORDER BY pcpc.price_scale_basis_consecutive_code)
        )
        * LAG(pcpc.price_scale_rate)
            OVER (PARTITION BY pcpc.purchase_contract_code,
                               pcpc.purchase_contract_position_code
                  ORDER BY pcpc.price_scale_basis_consecutive_code)
        AS lagging

    FROM dds.purchase_contract_price_conditions pcpc
    WHERE pcpc.price_condition_type_code = 'PB00'
      AND (pcpc.price_scale_basis_code = 'C' OR pcpc.price_scale_basis_code IS NULL)
)
SELECT
    knumh,
    evrtn,
    evrtp,
    dt_price_condition_valid_to,
    dt_price_condition_valid_from,
    price_scale_basis_consecutive_code,
    price_condition_position_code,
    konp_kbetr,
    konm_kbetr,
    kstbm,
    kstbm_next,
    cnt,
    rn,
    price_scale_basis_code,
    min_price_scale_limit_from_quantity,
    SUM(lagging) OVER (
        PARTITION BY evrtn, evrtp
        ORDER BY price_scale_basis_consecutive_code
    ) AS presum
FROM tmp
DISTRIBUTED BY (evrtn, evrtp);
