DROP TABLE IF EXISTS base_1_5;

CREATE TEMP TABLE base_1_5
--ON COMMIT DROP 
AS
WITH bounds AS (
    -- границы периода: управляем диапазоном через date_for_vitr_tmp
    SELECT 
        MIN(dt_balance)::date AS dt_min,
        MAX(dt_balance)::date AS dt_max
    FROM date_for_vitr_tmp
)
SELECT
    -- все поля из base
    b.*,

    -- дата отчёта (будущее поле dt_report в витрине)
    gs.dt_balance AS dt_report,

    -- тех. ключ на случай дублей
    row_number() OVER (
        PARTITION BY b.id_for_unique, gs.dt_balance
        ORDER BY b.sales_bundle_code
    ) AS rn
FROM base b
CROSS JOIN bounds bd
-- для КАЖДОЙ строки base генерим только свои дни хранения
JOIN LATERAL generate_series(
        GREATEST(b.dt_transportation_stage_start_p, bd.dt_min),                      -- начало
        LEAST(COALESCE(b.dt_transportation_stage_start_r, current_date), bd.dt_max), -- конец
        INTERVAL '1 day'
     ) AS gs(dt_balance)
DISTRIBUTED BY (sales_bundle_code, dt_report);
