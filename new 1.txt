-- 1. Отбираем только те b, у которых НЕТ уже найденного "полного" контракта
DROP TABLE IF EXISTS t_b_left;
CREATE TEMP TABLE t_b_left
ON COMMIT DROP AS
SELECT
    b.rn,
    b.storage_cost_calculation_type_name,
    b.forwarder_code,
    b.dt_for_price_search_in_purchase_contract,
    b.port_of_discharge_code,
    b.material_shape_code,
    b.etsng_code,
    b.transport_type_code
FROM dm_calc.storage_sales_bundles_weight b
LEFT JOIN t_contract_spec_full f
       ON f.rn = b.rn
WHERE f.rn IS NULL
      -- при желании можешь сузить по типу логики, если только для них нужны 1079:
      -- AND b.storage_cost_calculation_type_name IN (
      --   'Логика РФ-контейнеры',
      --   'Логика РФ-балк',
      --   'Логика Азия',
      --   'Логика Европа',
      --   'Логика СВХ'
      -- );

-- 2. Для этих rn ищем "частичные" (с null-ами) контракты из t_1079_spec_part
DROP TABLE IF EXISTS t_contract_spec_part;
CREATE TEMP TABLE t_contract_spec_part
ON COMMIT DROP AS
SELECT DISTINCT ON (b.rn)
    b.rn,
    t.purchase_contract_code,
    t.position_line_item_code,
    t.konws,
    t.price_uom_code,
    t.service_code,
    t.currency_dogovor,
    t.datab,
    t.datbi
FROM t_b_left b
JOIN t_1079_spec_part t
  ON t.lifnr = b.forwarder_code
 AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab        AND t.datbi
 AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
 -- условия-ограничения с "подстановками"
 AND (t.zzport     = b.port_of_discharge_code   OR t.zzport  IS NULL)
 AND (t.zzforma    = b.material_shape_code      OR t.zzforma IS NULL)
 AND (t.zzet_tarif = b.etsng_code               OR t.zzet_tarif IS NULL)
 AND (t.sdabw      = b.transport_type_code      OR t.sdabw   IS NULL)
ORDER BY
    b.rn,
    CASE b.storage_cost_calculation_type_name
        WHEN 'Логика РФ-контейнеры' THEN t.order_rf
        WHEN 'Логика РФ-балк'       THEN t.order_rf
        WHEN 'Логика Азия'          THEN t.order_asia
        WHEN 'Логика Европа'        THEN t.order_euro_svh
        WHEN 'Логика СВХ'           THEN t.order_euro_svh
        ELSE 999999
    END,
    t.kopos,
    t.kbetr,
    t.purchase_contract_code,
    t.position_line_item_code;



-- Контроль: не пересекаются ли rn
SELECT
  COUNT(*) AS all_rows,
  COUNT(DISTINCT rn) AS distinct_rn,
  COUNT(*) - COUNT(DISTINCT rn) AS duplicates
FROM (
  SELECT rn FROM t_contract_spec_full
  UNION ALL
  SELECT rn FROM t_contract_spec_part
) s;
