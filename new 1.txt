class IndentDumper(yaml.Dumper):
    def increase_indent(self, flow=False, indentless=False):
        return super(IndentDumper, self).increase_indent(flow, False)


mapping_gp_to_click = {
    'date': 'Date32',
    'timestamp': 'DateTime',
    'varchar': 'String',
    'text': 'String',
    'bpchar': 'String',
    'numeric': 'Decimal',
    'decimal': 'Decimal',
    'int8': 'Int32',
    'int4': 'Int32',
    'int2': 'Int32',
    'int': 'Int32',
    'bigint': 'Int64',
    'bool': 'UInt8',
    'json': 'String',
    'time': 'String',
    'float8': 'Decimal(32,10)'
}


def create_meta_clickhouse_from_db(greenplum_table_schema: str = '',
                                   greenplum_table_name: str = '',
                                   clickhouse_table_schema: str = '',
                                   clickhouse_table_name: str = '',
                                   create_mode: bool = False,
                                   order_by: list = None):
    """
    Функция создания yaml файлов для загрузки данных из Greenplum в Clickhouse через S3
    
    Arguments:
    :param greenplum_table_schema:   схема таблицы в гринпламе, по умолчанию ''. Заполняется для создания YAML для одной таблицы
    :param greenplum_table_name:     имя таблицы в гринпламе, по умолчанию ''. Заполняется, когда имена таблицы
                                     в гринпламе и в кликхаусе отличаются. Иначе достаточно указать имя в "clickhouse_table_name"
    :param clickhouse_table_schema:  схема таблицы в кликхаусе, по умолчанию ''. Заполняется для создания YAML для одной таблицы
    :param clickhouse_table_name:    имя таблицы в кликхаусе, по умолчанию ''. Заполняется для создания YAML для одной таблицы
    :param create_mode:              режим принудительного создания мета-ямл файла по структуре гринплама, по умолчанию False.
    :param order_by:                 lсписок колонок для указания в DDL кликхауса в части ORDER BY, по умолчанию [].
    """
    
    root_path = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    new_path = os.path.join(root_path, 'config_files', 'meta')
    print('Target_path', new_path)
    greenplum_table_schema = greenplum_table_schema.strip()
    clickhouse_table_schema = 'dm' if clickhouse_table_schema == '' else clickhouse_table_schema.strip()
    object_name = greenplum_table_name.strip() if greenplum_table_name != '' else clickhouse_table_name.strip()
    
    ################## Connection to Greenplum ##################
    # host = os.environ.get('gp_prod_host')
    # port = '5432'
    # database = 'dwh'
    # user_name = os.environ.get('gp_prod_login')
    # password = os.environ.get('gp_prod_password')
    
    host = os.environ.get('gp_dev_host')
    port = '5432'
    database = 'dwh'
    user_name = os.environ.get('gp_dev_login')
    password = os.environ.get('gp_dev_password')
    #############################################################
    
    querry = f'''
with tables_info as(
	select
        case
	        when t.table_type = 'BASE TABLE' then 'table'
	        when t.table_type = 'VIEW' then 'view'
        else t.table_type
    	end table_type,
        t.table_schema,
        t.table_name,
        c.column_name,
        c.ordinal_position,
        concat(c.udt_name,
        case
            when character_maximum_length is not null
                then concat('(', c.character_maximum_length, ')')
            when numeric_precision is not null and c.data_type = 'numeric'
                then concat('(', c.numeric_precision, ',', c.numeric_scale, ')')
        end) data_type,
        json_build_object(
        'column_name_gp',
        case when c.column_name = lower(c.column_name) then c.column_name else concat('"', c.column_name, '"' ) end,
        'data_type_gp',
        concat(c.udt_name,
        case
            when character_maximum_length is not null
                then concat('(', c.character_maximum_length, ')')
            when numeric_precision is not null and c.data_type = 'numeric'
                then concat('(', c.numeric_precision, ',', c.numeric_scale, ')')
        end),
        'is_nullable',
        case
            when is_nullable = 'YES'
                then true
            else false
        end,
        'default',
        column_default,
        'description',
        pd.description
        ) json_data,
        pc.reloptions,
        pg_catalog.pg_get_table_distributedby(pc.oid) distributedby,
        v.view_definition,
        regexp_replace(split_part(v.view_definition, 'FROM', 2), '[[:space:];]+', '', 'g') as view_base,
        ent.entity_last_load::time + interval '20 minutes' as entity_last_load
    from information_schema.columns c
 	left join {' tech_etl.tables_meta' if greenplum_table_schema != '' and clickhouse_table_name != '' else '(select distinct table_schema, table_name, entity_id, flag_active from tech_etl.tables_meta_clickhouse_upload)'} em
        on em.table_name = c.table_name
        and em.table_schema = c.table_schema
    left join tech_etl.entities_meta ent
       on ent.entity_id = em.entity_id
    left join information_schema.tables t
        on t.table_schema = c.table_schema
        and t.table_name = c.table_name
    left join pg_catalog.pg_statio_all_tables p
        on c.table_schema = p.schemaname
        and c.table_name = p.relname
    left join pg_catalog.pg_description pd
        on c.ordinal_position = pd.objsubid
        and p.relid = pd.objoid
    left join pg_catalog.pg_class pc
        on pc.relname = t.table_name
        and p.relid = pc.oid
    left join information_schema."views" v
        on v.table_schema = t.table_schema
        and v.table_name = t.table_name
    where {f"t.table_schema = '{greenplum_table_schema}' and t.table_name = '{object_name}'" if greenplum_table_schema != '' and clickhouse_table_name != '' else "em.flag_active"}
), query as (
select
    k.table_type,
    k.table_schema,
    k.table_name,
    k.reloptions,
    k.distributedby distr,
    array_agg(k.json_data order by k.ordinal_position) column_info,
    k.view_definition,
    split_part(
	    case
	    	when view_base like '%JOIN%' then null
	    	when k.view_definition like '%WITH%' then null
	    	else split_part(view_base, 'WHERE', 1)
	    end, '.', 1) as view_base_schema,
	split_part(
	    case
	    	when view_base like '%JOIN%' then null
	    	when k.view_definition like '%WITH%' then null
	    	else split_part(view_base, 'WHERE', 1)
	    end, '.', 2) as view_base_name,
    case when k.table_schema = 'dict_dds'
    	then '0 22 * * *'
    	else concat(extract('minutes' from k.entity_last_load), ' ',  extract('hour' from k.entity_last_load), ' * * *')
   	end as entity_last_load
from tables_info k
group by
    k.table_type,
    k.table_schema,
    k.table_name,
    k.reloptions,
    k.distributedby,
    k.view_definition,
    k.entity_last_load,
    view_base_schema,
    view_base_name
)
select
    qq.table_type,
    qq.table_schema,
    qq.table_name,
    qq.reloptions,
	concat(qq.distr, pg_catalog.pg_get_table_distributedby(pc.oid)) as distributedby,
	qq.column_info,
	qq.view_definition,
	qq.entity_last_load,
	obj_description('{greenplum_table_schema}.{object_name}'::regclass, 'pg_class') as table_comment
from query qq
left join pg_catalog.pg_statio_all_tables p
        on qq.view_base_schema = p.schemaname
        and qq.view_base_name = p.relname
left join (select distinct objoid from pg_catalog.pg_description) pd
		on p.relid = pd.objoid
left join pg_catalog.pg_class pc
        on pc.relname = qq.view_base_name
        and pc.oid = p.relid
order by
    qq.table_schema,
    qq.table_name
;
'''
    try:
        with (psycopg2.connect(
                host=host,
                port=port,
                dbname=database,
                user=user_name,
                password=password
        ) as conn):
            with conn.cursor() as cur:
                print(f'!!!!\n{querry}\n!!!')
                cur.execute(querry)
                result = cur.fetchall()
                count = len(result)
                if not result:
                    print('Error: Greenplum query returned no results. Incorrect schema or table name')
                for object_type, table_schema, table_name, reloptions, distributedby, columns_info, view_def, entity_last_load, table_comment in result:
                    mapping_dict = {}
                    cnt = 0
                    if not create_mode:
                        click = get_metadata_clickhouse(clickhouse_table_schema, table_name)
                        if not click.result_rows:
                            print(f'Error: Table {table_name} doesn\'t exist in clickhouse')
                            count -= 1
                            break
                        else:
                            for column_info in click.column_names:
                                mapping_dict[column_info] = cnt
                                cnt += 1
                            cleared_distributed = None
                            first_row = click.result_rows[0]
                            yaml_data = {'schema_name_gp': table_schema,
                                         'schema_name_click': first_row[mapping_dict['database_name']],
                                         'object_name': table_name}
                            if greenplum_table_name != '':
                                yaml_data['greenplum_table_name'] = greenplum_table_name
                            yaml_data['object_type'] = object_type
                            yaml_data['clickhouse_cluster'] = '{cluster}'
                            yaml_data['engine'] = first_row[mapping_dict['engine']]
                            yaml_data['order_by'] = first_row[mapping_dict['order_by']].split(', ')
                            if distributedby:
                                if distributedby == 'DISTRIBUTED RANDOMLY':
                                    cleared_distributed = ['RANDOMLY']
                                elif distributedby == 'DISTRIBUTED REPLICATED':
                                    cleared_distributed = ['REPLICATED']
                                elif str(distributedby).startswith('DISTRIBUTED BY'):
                                    cleared_distributed = str(distributedby).lstrip('DISTRIBUTED BY (').rstrip(
                                        ')').replace('\"', '').split(', ')
                                else:
                                    print(distributedby)
                                yaml_data['distributed'] = cleared_distributed
                            yaml_data['partitions'] = None if not first_row[mapping_dict['partitions']] else first_row[
                                mapping_dict['partitions']]
                            yaml_data['table_settings'] = (first_row[mapping_dict['settings']])[1:]
                            yaml_data['table_comment'] = None if first_row[mapping_dict['table_comment']] == '' else \
                                first_row[mapping_dict['table_comment']]
                            yaml_data['settings_external_table'] = {'max_threads': 20,
                                                                    'max_insert_threads': 20,
                                                                    'input_format_parallel_parsing': 0}
                            yaml_data['load_type'] = 'full'
                            yaml_data['recreate_mode'] = 'drop_create'
                            yaml_data['truncate_mode_on'] = True if table_name != 'account_turnover' else False
                            yaml_data['postgres_conn_id'] = 'gp_connection'
                            yaml_data['clickhouse_conn_id'] = 'clickhouse'
                            yaml_data['dag_name'] = None
                            yaml_data['dag_schedule_interval'] = {
                                'PROD': None if entity_last_load == '  * * *' else entity_last_load,
                                'DEV': None,
                            }
                            yaml_data['dag_tags'] = []
                            yaml_data['task_pool'] = 'dm_pool'
                            yaml_data['task_pool_slots'] = 1
                            yaml_data['attributes'] = []
                            
                            if len(columns_info) >= len(first_row[mapping_dict['columns']]):
                                print(f'Table_name - {table_schema}.{table_name}')
                                for i in range(0, len(first_row[mapping_dict['columns']])):
                                    data = {
                                        'column_name_click': first_row[mapping_dict['columns']][i]['column_name_click'],
                                        'column_name_gp': first_row[mapping_dict['columns']][i]['column_name_click'] if
                                        first_row[mapping_dict['columns']][i]['column_name_click'] ==
                                        first_row[mapping_dict['columns']][i]['column_name_click'].lower() else
                                        f''' '"{first_row[mapping_dict['columns']][i]['column_name_click']}"' ''',
                                        'data_type_click': first_row[mapping_dict['columns']][i]['data_type_click'],
                                        'data_type_gp': next((item['data_type_gp'] for item in columns_info if
                                                              item['column_name_gp'] ==
                                                              first_row[mapping_dict['columns']][i][
                                                                  'column_name_click']),
                                                             None),
                                        'is_nullable': 'NULL' if first_row[mapping_dict['columns']][i][
                                                                     'is_nullable'].lower() == 'true' else 'NOT NULL',
                                        'description': first_row[mapping_dict['columns']][i]['description'].replace(
                                            '\'', '\'\'').replace('\n','') if first_row[mapping_dict['columns']][i][
                                            'description'] else None,
                                    }
                                    if object_type != 'view':
                                        if data['data_type_gp'] == 'date' and data[
                                            'column_name_click'] not in cleared_distributed:
                                            data[
                                                'column_name_gp'] = f'''tech_etl.validate_date("{data['column_name_gp']}")'''
                                        if data['data_type_gp'] == 'timestamp' and data['column_name_gp'] not in (
                                                'dttm_inserted', 'dttm_updated') and data[
                                            'column_name_click'] not in cleared_distributed:
                                            data[
                                                'column_name_gp'] = f'''tech_etl.validate_timestamp("{data['column_name_gp']}"::text)'''
                                    if first_row[mapping_dict['columns']][i].get('default'):
                                        data['default'] = first_row[mapping_dict['columns']][i]['default']
                                    yaml_data['attributes'].append(data)
                                if not os.path.exists(os.path.join(new_path, yaml_data['schema_name_click'])):
                                    os.makedirs(os.path.join(new_path, yaml_data['schema_name_click']))
                                with open(os.path.join(new_path, yaml_data['schema_name_click'],
                                                       f"{table_schema}_{table_name}_meta.yaml"),
                                          'w', encoding='utf-8') as new_file:
                                    yaml.dump(yaml_data,
                                              new_file,
                                              Dumper=IndentDumper,
                                              default_flow_style=False,
                                              sort_keys=False,
                                              encoding='utf-8',
                                              allow_unicode=True,
                                              width=float("inf"))
                            
                            else:
                                print(f'Table_name - {table_schema}.{table_name}')
                                print('Clickhouse columns count:', len(first_row[mapping_dict['columns']]))
                                print('Greenplum  columns count:', len(columns_info))
                                print('Error: clickhouse table has more columns than Greenplum table')
                                count -= 1
                    else:
                        if not order_by:
                            raise Exception('Не указаны колонки в order_by!')
                        if order_by[0] != 'tuple()':
                            check = null_checking(order_by, table_schema, table_name)
                            cur.execute(check)
                            check_rest = cur.fetchall()
                            col_name = [desc[0] for desc in cur.description]
                            rest = []
                            for i in range(0, len(check_rest[0])):
                                if check_rest[0][i] != 0:
                                    rest.append(col_name[i])
                            if rest:
                                raise Exception(f'Колонки - {rest} указанные в order_by не могут быть NULL!')
                        print('Create_mode activated, creating from Greenplum table')
                        yaml_data = {'schema_name_gp': table_schema,
                                     'schema_name_click': clickhouse_table_schema,
                                     'object_name': table_name}
                        if greenplum_table_name != '':
                            yaml_data['greenplum_table_name'] = greenplum_table_name
                        yaml_data['object_type'] = object_type
                        yaml_data['clickhouse_cluster'] = '{cluster}'
                        yaml_data['engine'] = 'ReplicatedMergeTree'
                        yaml_data['order_by'] = order_by
                        if not order_by:
                            count -= 1
                            raise Exception(f'Error: Order_by is EMPTY. Add list of columns to "order_by" parameter')
                        cleared_distributed = None
                        if distributedby:
                            if distributedby == 'DISTRIBUTED RANDOMLY':
                                cleared_distributed = ['RANDOMLY']
                            elif distributedby == 'DISTRIBUTED REPLICATED':
                                cleared_distributed = ['REPLICATED']
                            elif str(distributedby).startswith('DISTRIBUTED BY'):
                                cleared_distributed = str(distributedby).lstrip('DISTRIBUTED BY (').rstrip(')').replace(
                                    '\"', '').split(', ')
                            else:
                                print(distributedby)
                            yaml_data['distributed'] = cleared_distributed
                        yaml_data['partitions'] = None
                        yaml_data['table_settings'] = 'index_granularity = 8192'
                        yaml_data['table_comment'] = None if table_comment == '' else table_comment
                        yaml_data['settings_external_table'] = {'max_threads': 20,
                                                                'max_insert_threads': 20,
                                                                'input_format_parallel_parsing': 0}
                        yaml_data['load_type'] = 'full'
                        yaml_data['recreate_mode'] = 'drop_create'
                        yaml_data['truncate_mode_on'] = True if table_name != 'account_turnover' else False
                        yaml_data['postgres_conn_id'] = 'gp_connection'
                        yaml_data['clickhouse_conn_id'] = 'clickhouse'
                        yaml_data['dag_name'] = None
                        yaml_data['dag_schedule_interval'] = {
                            'PROD': None if entity_last_load == '  * * *' else entity_last_load,
                            'DEV': None,
                        }
                        yaml_data['dag_tags'] = []
                        yaml_data['task_pool'] = 'dm_pool'
                        yaml_data['task_pool_slots'] = 1
                        yaml_data['attributes'] = []
                        print(f'Table_name - {table_schema}.{table_name}')
                        print('greenplum columns count:', len(columns_info))
                        print('clickhouse new')
                        print(columns_info)
                        print()
                        for i in range(0, len(columns_info)):
                            if '(' in columns_info[i]['data_type_gp'] and columns_info[i]['data_type_gp'].split('(')[
                                0] == 'numeric':
                                numeric_type = mapping_gp_to_click[
                                                   columns_info[i]['data_type_gp'].split('(')[0]] + '(' + \

                            data = {'column_name_click': columns_info[i]['column_name_gp'],

                                    'data_type_gp': columns_info[i]['data_type_gp'],

                                'column_name_click'] not in cleared_distributed and object_type != 'view':

                            if data['data_type_gp'] == 'timestamp' and data['column_name_gp'] not in ('dttm_inserted',
                                                                                                      'dttm_updated'):
                                data[
                                    'column_name_gp'] = f'''tech_etl.validate_timestamp("{data['column_name_gp']}"::text)'''
                            if columns_info[i].get('default'):
                                data['default'] = columns_info[i]['default']
                            yaml_data['attributes'].append(data)
                        if not os.path.exists(os.path.join(new_path, yaml_data['schema_name_click'])):
                            os.makedirs(os.path.join(new_path, yaml_data['schema_name_click']))
                        with open(os.path.join(new_path,

                            yaml.dump(yaml_data,
                                      new_file,
                                      Dumper=IndentDumper,
                                      sort_keys=False,
                                      encoding='utf-8',
                                      allow_unicode=True,
                                      width=float("inf"))
                print('Created meta files for', count, 'tables')
                
    except UndefinedColumn as e:

        raise e
    
    except Exception as err:
        print('Zalupa_tigra')
        print(err)
        raise err


def get_metadata_clickhouse(schema, table):
    click_schema = schema if schema != '' else 'dm'
    sql_query = f"""
SELECT
	table_name,
	database_name,
	groupArray(user_json) AS columns,
	primary_key,
	engine,
	partitions,
	order_by,
	settings,
	table_comment
FROM (
	SELECT
	    t.name AS table_name,
	    t.database AS database_name,
	    map('column_name_click', c.name,
	        'data_type_click',
	        CASE WHEN c.type like 'Nullable(%'
		        then replaceOne(replace(c.type, 'Nullable(', ''), ')', '')
		        else c.type
	    	END AS data_type_click,
	        'position', c.position::text,
	        'is_nullable', hasToken(c.type, 'Nullable')::bool::text,
	        'description', comment,
	        'default', default_expression
	    ) AS user_json,
	    c.comment AS comment,
	    t.primary_key,
	    t.engine,
	    t.partition_key as partitions,
	    t.sorting_key as order_by,
	    splitByString('SETTINGS', engine_full)[-1] AS settings,
	    t.comment as table_comment
	FROM system.columns c
	JOIN system.tables t
	    ON c.database = t.database AND c.table = t.name
	WHERE c.database = '{click_schema}' and c.table = '{table}')
GROUP BY table_name,
	database_name,
	primary_key,
	engine,
	partitions,
	order_by,
	settings,
	table_comment;"""
    
    try:
        ##################### Connection to Clickhouse ######################
        # client = clickhouse_connect.get_client(
        #     host=os.environ.get('click_prod_host'),
        #     port=443,
        #     database='dm',
        #     user_name=os.environ.get('click_prod_login'),
        #     password=os.environ.get('click_prod_password')
        #)
        client = clickhouse_connect.get_client(
            host=os.environ.get('click_dev_host'),
            port=8443,
            secure=True,
            verify=False,
            database='dm',
            user_name=os.environ.get('click_dev_login'),
            password=os.environ.get('click_dev_password')
        )
        #####################################################################
        result = client.query(sql_query)
        return result
    except Exception as err:
        print(err)
        return None

def null_checking(order_by: list, schema, table):
    template = f'''
SELECT
    '''
    for item in order_by[:-1]:
        template += f'''count({item}) - count(*) as {item},
    '''
    template += f'''count({order_by[-1]}) - count(*) as {order_by[-1]}
'''
    template += f'''FROM {schema}.{table};'''
    return template
