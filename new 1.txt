SELECT
  currency_from_code,
  dt_currency_rate,
  count(*)
FROM dict_dds.currency_rates
WHERE currency_rate_type_code = 'M'
  AND currency_to_code = 'USD'
  AND NOT deleted_flag
GROUP BY 1,2
HAVING count(*) > 1;


WITH rates AS (
    SELECT DISTINCT ON (
        currency_from_code,
        dt_currency_rate
    )
        currency_from_code,
        dt_currency_rate,
        currency_rate,
        currency_from_multiplier,
        currency_to_multiplier
    FROM dict_dds.currency_rates
    WHERE currency_rate_type_code = 'M'
      AND currency_to_code = 'USD'
      AND deleted_flag = false
)
