drop table if exists items;
create temporary table items as (
    select
        flow_id,
        source_system,
        record_id,
        uuid,
        dt_insert,
        unnest(xpath('//Item', document_xml)) as xml_item
    from
        landing."INPUT_DATA_FROM_SAPXI_IN"
    where
        flow_id = 'SI_MaintenaceOrder_AI'
        and dt_insert >= current_date - INTERVAL '30 days'
)
distributed by (uuid);
