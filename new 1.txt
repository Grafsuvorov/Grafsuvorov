WITH cdpr AS (
  SELECT currency_code, coalesce(decimal_place_number, 2) AS dplaces
  FROM dict_dds.currency_decimal_place_ral
)
SELECT
  NULLIF(btrim(e."EBELN"), '') AS ebeln,
  NULLIF(btrim(e."CHARG"), '') AS charg,
  NULLIF(btrim(e."MATNR"), '') AS matnr,
  NULLIF(btrim(e."WERKS"), '') AS werks,
  NULLIF(btrim(e."EBELP"), '') AS ebelp,
  NULLIF(btrim(e."BELNR"), '') AS belnr,
  NULLIF(btrim(e."GJAHR"), '') AS gjahr,
  NULLIF(btrim(e."BUZEI"), '') AS buzei,
  NULLIF(btrim(e."ZEKKN"), '') AS zekkn,

  -- коэффициент считаем ОДИН раз и правильно
  -- numeric, чтобы не ловить int-арифметику
  e."DMBTR" * power(10::numeric, (2 - coalesce(c.dplaces, 2))) AS dmbtr,
  e."WRBTR" * power(10::numeric, (2 - coalesce(c.dplaces, 2))) AS wrbtr,
  e."MENGE" AS menge,
  e."AREWR" * power(10::numeric, (2 - coalesce(c.dplaces, 2))) AS arewr,

  NULLIF(btrim(e."LFBNR"), '') AS lfbnr,
  NULLIF(btrim(e."LFGJA"), '') AS lfgja,
  NULLIF(btrim(e."LFPOS"), '') AS lfpos,
  NULLIF(btrim(e."VGABE"), '') AS vgabe,
  NULLIF(btrim(e."SHKZG"), '') AS shkzg,

  CASE
    WHEN e."BUDAT" IS NULL OR btrim(e."BUDAT") IN ('', '00000000') THEN NULL
    WHEN e."BUDAT" ~ '^\d{8}$'
         AND e."BUDAT" >= '19000101' AND e."BUDAT" < '22991231'
      THEN to_date(e."BUDAT",'YYYYMMDD')
    ELSE DATE '1900-01-01'
  END AS budat

FROM stg."EKBE" e
LEFT JOIN cdpr c
  ON c.currency_code = e."WAERS"
WHERE e."MANDT" = '400';


-- строка -> NULL, если пусто
CREATE OR REPLACE FUNCTION tech_etl.nullif_blank(text)
RETURNS text
LANGUAGE sql IMMUTABLE AS
$$ SELECT NULLIF(btrim($1),'') $$;

-- дата YYYYMMDD с граничными условиями
CREATE OR REPLACE FUNCTION tech_etl.to_date_yyyymmdd_safe(text)
RETURNS date
LANGUAGE sql IMMUTABLE AS
$$
SELECT CASE
  WHEN $1 IS NULL OR btrim($1) IN ('', '00000000') THEN NULL
  WHEN $1 ~ '^\d{8}$' THEN
    GREATEST(DATE '1900-01-01',
      LEAST(DATE '2299-12-31', to_date($1,'YYYYMMDD')))
  ELSE DATE '1900-01-01'
END
$$;
