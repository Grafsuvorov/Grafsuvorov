DROP TABLE IF EXISTS for_1079_euro;

CREATE TEMPORARY TABLE for_1079_euro
ON COMMIT DROP
AS
WITH joined AS (
    SELECT
        b2.rn,
        fb.purchase_contract_code,
        fb.position_line_item_code,
        fb.konws,
        fb.service_code,
        fb.price_uom_code,
        fb.netpr,
        fb.currency_dogovor,
        fb.datab,
        fb.datbi,
        fb.order_euro_svh,
        fb.kopos,
        fb.kbetr,

        b2.warehouse_code,
        b2.storage_cost_special_calculation_type_code,
        b2.dt_storage_start,
        b2.dt_storage_end,
        b2.storage_duration_total_calendar_days,
        b2.sales_bundle_gross_weight,

        -- для month_difference_2
        CASE
            WHEN EXTRACT(DAY FROM b2.dt_storage_end)
                 < EXTRACT(DAY FROM b2.dt_storage_start)
                THEN (EXTRACT(YEAR  FROM b2.dt_storage_end)
                      - EXTRACT(YEAR  FROM b2.dt_storage_start)) * 12
                     + (EXTRACT(MONTH FROM b2.dt_storage_end)
                        - EXTRACT(MONTH FROM b2.dt_storage_start))
            ELSE (EXTRACT(YEAR  FROM b2.dt_storage_end)
                  - EXTRACT(YEAR  FROM b2.dt_storage_start)) * 12
                 + (EXTRACT(MONTH FROM b2.dt_storage_end)
                    - EXTRACT(MONTH FROM b2.dt_storage_start)) + 1
        END AS month_difference_2,

        -- спец-количество
        CASE
            WHEN b2.storage_cost_special_calculation_type_code = '4'
                THEN ceil(b2.storage_duration_total_calendar_days / 7::numeric)
            WHEN b2.storage_cost_special_calculation_type_code = '5'
                THEN ceil(b2.storage_duration_total_calendar_days / 30::numeric)
        END AS special,

        -- флаги для приоритета (как в плане)
        CASE
            WHEN param_sap.range_low_value IS NOT NULL
             AND param_srv_1.range_low_value IS NOT NULL THEN 1
            WHEN param_sap.range_low_value IS NULL
             AND param_srv_1.range_low_value IS NULL THEN 1
            ELSE 0
        END AS flag_param_pair
    FROM dm_calc.storage_sales_bundles_weight AS b2
    -- только Европа, сразу отфильтровали
    JOIN for_1079_base fb
      ON b2.storage_cost_calculation_type_name = 'Логика Европа'
     AND fb.purchase_document_subtype_code     = 'ZDTL'
     AND (
            (for_1082_1.range_low_value IS NOT NULL AND fb.zzsrvpos = 'S49')
         OR (for_1082_1.range_low_value IS NULL
             AND for_1082_2.range_low_value IS NOT NULL
             AND fb.tmp_1 IS NOT NULL)
         OR (for_1082_1.range_low_value IS NULL
             AND for_1082_2.range_low_value IS NULL
             AND fb.tmp_2 IS NOT NULL)
         )
     AND fb.bukrs = b2.unit_balance_code
     AND fb.lifnr = b2.forwarder_code
     AND fb.werks = b2.plant_code
     AND (fb.zzport = b2.port_of_discharge_code OR fb.zzport IS NULL)
     AND (fb.zzforma = b2.material_shape_code   OR fb.zzforma IS NULL)
     AND (fb.zzet_tarif = b2.etsng_code         OR fb.zzet_tarif IS NULL)
     AND (fb.sdabw = b2.transport_type_code     OR fb.sdabw IS NULL)
     AND b2.dt_for_price_search_in_purchase_contract
            BETWEEN fb.dt_valid_from AND fb.dt_valid_to
     AND b2.dt_for_price_search_in_purchase_contract
            BETWEEN fb.datab       AND fb.datbi
     AND (fb.zzterminal = b2.warehouse_code OR fb.zzterminal IS NULL)

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_1
      ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
     AND for_1082_1.case_code        = 'PERET_FRN'
     AND for_1082_1.parameter_code   = 'S_LIFNRA'
     AND for_1082_1.range_low_value  = b2.forwarder_code

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_2
      ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
     AND for_1082_2.parameter_code    = 'S_FORMSTEU'
     AND for_1082_2.range_low_value   = b2.material_shape_code

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_3
      ON fb.tmp_2 IS NOT NULL
     AND for_1082_3.range_low_value LIKE 'EU%'
     AND for_1082_3.abap_program_code = '/RUSAL/SD4553M'
     AND for_1082_3.parameter_code    = 'S_USLST'

    LEFT JOIN dict_dds.settings_and_parameters_sap AS param_sap
      ON param_sap.abap_program_code = '/RUSAL/SD4553M'
     AND param_sap.parameter_code    = 'FORMSTEU'
     AND param_sap.range_low_value   = b2.material_shape_code

    LEFT JOIN dict_dds.settings_and_parameters_sap AS param_srv_1
      ON param_srv_1.abap_program_code = '/RUSAL/SD4553M'
     AND param_srv_1.parameter_code    = 'S_USLSTC'
     AND param_srv_1.range_low_value   = for_1082_3.range_low_value
),
with_windows AS (
    SELECT
        j.*,
        -- сумма по складу/месяцу
        SUM(j.sales_bundle_gross_weight) OVER (
            PARTITION BY j.warehouse_code,
                         date_trunc('month', j.dt_storage_end)
        ) AS warehouse_weight_month_end,
        -- выбор лучшего контракта
        ROW_NUMBER() OVER (
            PARTITION BY j.rn
            ORDER BY
                j.order_euro_svh,
                j.kopos,
                j.kbetr,
                j.flag_param_pair DESC,
                j.purchase_contract_code,
                j.position_line_item_code
        ) AS row_num
    FROM joined j
)
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    warehouse_weight_month_end,
    month_difference_2,
    netpr,
    special,
    currency_dogovor,
    datab,
    datbi
FROM with_windows
WHERE row_num = 1
;

-- расставляем нормальное распределение для дальнейшего join по rn
ALTER TABLE for_1079_euro
  SET DISTRIBUTED BY (rn);


DROP TABLE IF EXISTS for_1079_base;

CREATE TEMPORARY TABLE for_1079_base
ON COMMIT DROP
AS
SELECT
    for_1079_base_calc.*,
    tmp_1.range_low_value AS tmp_1,
    tmp_2.range_low_value AS tmp_2
FROM (
    SELECT
        ekpo.purchase_contract_code,
        ekpo.position_line_item_code,
        ekko.unit_balance_code    AS bukrs,
        ekko.supplier_code        AS lifnr,
        ekko.port_code            AS zzport,
        ekko.dt_valid_from        AS kdatb,
        ekko.dt_valid_to          AS kdate,
        ekpo.plant_code           AS werks,
        ekpo.service_code         AS zzsrvpos,
        ekpo.shape_code           AS zzforma,
        ekpo.etsng_code           AS zzet_tarif,
        ekpo.transport_type_code  AS sdabw,
        ekpo_price.dt_price_condition_valid_to   AS datbi,
        ekpo_price.dt_price_condition_valid_from AS datab,
        ekko.dt_valid_to,
        ekko.dt_valid_from,
        ekko.purchase_document_subtype_code,
        ekpo_price.price_condition_position_code          AS kopos,
        ekpo_price.price_condition_position_currency_code AS konwa,
        ekpo.terminal_code                                AS zzterminal,
        ekpo.warehouse_storage_area_code                  AS vzzzone,
        COALESCE(ekpo_price.price_scale_currency_code,
                 ekpo_price.price_condition_position_currency_code) AS konws,
        ekpo.service_code,
        ekpo.price_uom_code,
        ekpo.price_vat_excluded_document_currency_amount AS netpr,
        ekpo_price.price_rate                            AS kbetr,

        -- веса для разных логик
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END) AS order_rf,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.warehouse_storage_area_code IS NULL THEN 1 ELSE 0 END) AS order_asia,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END) AS order_euro_svh,

        CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor,
        ekpo_price.is_price_condition_position_deleted_code,
        COUNT(*) OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code
        ) AS cnt,
        ROW_NUMBER() OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code,
                         ekpo_price.dt_price_condition_valid_from,
                         ekpo_price.dt_price_condition_valid_to,
                         price_scale_basis_code
            ORDER BY price_scale_basis_consecutive_code
        ) AS rwn
    FROM dds.purchase_contract_price_conditions AS ekpo_price
    JOIN dds.purchase_contract_header AS ekko
      ON ekko.purchase_contract_code = ekpo_price.purchase_contract_code
    JOIN dds.purchase_contract_position AS ekpo
      ON ekpo.purchase_contract_code     = ekko.purchase_contract_code
     AND ekpo.position_line_item_code    = ekpo_price.purchase_contract_position_code
    LEFT JOIN dict_dds.currency_decimal_place_ral AS curdpr
      ON curdpr.currency_code      = COALESCE(ekpo_price.price_scale_currency_code,
                                              ekpo_price.price_condition_position_currency_code)
     AND curdpr.decimal_place_number = '0'
    WHERE ekko.purchase_document_subtype_code IN ('ZDTL','ZDT','ZDE','ZDP','ZDF','ZDS')
      AND ekko.paydox_contract_type_code IN ('1','6')
      AND ekpo.is_deleted IS NULL
      AND ekpo_price.is_price_condition_position_deleted_code IS NULL
      AND ekpo_price.price_condition_type_code = 'PB00'
      AND (ekpo_price.price_scale_basis_code = 'C'
           OR ekpo_price.price_scale_basis_code IS NULL)
) AS for_1079_base_calc
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_1
  ON tmp_1.abap_program_code = '/RUSAL/SD4553M'
 AND tmp_1.parameter_code    = 'S_USLSTC'
 AND tmp_1.range_low_value   = for_1079_base_calc.zzsrvpos
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_2
  ON tmp_2.abap_program_code = '/RUSAL/SD4553M'
 AND tmp_2.parameter_code    = 'S_USLST'
 AND tmp_2.range_low_value LIKE 'EU%'
 AND tmp_2.range_low_value   = for_1079_base_calc.zzsrvpos
WHERE for_1079_base_calc.rwn = 1
;
