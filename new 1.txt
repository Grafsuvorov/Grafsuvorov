-- Основа для различных логик договора/позиции хранения (LE.001079/LE.001080)
CREATE TEMPORARY TABLE for_1079_base ON COMMIT DROP AS (
    SELECT
        for_1079_base_calc.purchase_contract_code,
        for_1079_base_calc.position_line_item_code,
        for_1079_base_calc.bukrs,
        for_1079_base_calc.lifnr,
        for_1079_base_calc.zzport,
        for_1079_base_calc.kdatb,
        for_1079_base_calc.kdate,
        for_1079_base_calc.werks,
        for_1079_base_calc.zzsrvpos,
        tmp_1.range_low_value AS tmp_1,
        tmp_2.range_low_value AS tmp_2,
        for_1079_base_calc.zzforma,
        for_1079_base_calc.zzet_tarif,
        for_1079_base_calc.sdabw,
        for_1079_base_calc.datbi,
        for_1079_base_calc.datab,
        for_1079_base_calc.dt_valid_to,
        for_1079_base_calc.dt_valid_from,
        for_1079_base_calc.purchase_document_subtype_code,
        for_1079_base_calc.kopos,
        for_1079_base_calc.konwa,
        for_1079_base_calc.zzterminal,
        for_1079_base_calc.vzzzone,
        for_1079_base_calc.konws,
        for_1079_base_calc.service_code,
        for_1079_base_calc.price_uom_code,
        for_1079_base_calc.netpr,
        for_1079_base_calc.kbetr,
        for_1079_base_calc.order_rf,
        for_1079_base_calc.order_asia,
        for_1079_base_calc.order_euro_svh,
        for_1079_base_calc.currency_dogovor
    FROM (
        SELECT
            ekpo.purchase_contract_code,
            ekpo.position_line_item_code,
            ekko.unit_balance_code                           AS bukrs,
            ekko.supplier_code                               AS lifnr,
            ekko.port_code                                   AS zzport,
            ekko.dt_valid_from                               AS kdatb,
            ekko.dt_valid_to                                 AS kdate,
            ekpo.plant_code                                  AS werks,
            ekpo.service_code                                AS zzsrvpos,
            ekpo.shape_code                                  AS zzforma,
            ekpo.etsng_code                                  AS zzet_tarif,
            ekpo.transport_type_code                         AS sdabw,
            ekpo_price.dt_price_condition_valid_to           AS datbi,
            ekpo_price.dt_price_condition_valid_from         AS datab,
            ekko.dt_valid_to,
            ekko.dt_valid_from,
            ekko.purchase_document_subtype_code,
            ekpo_price.price_condition_position_code         AS kopos,
            ekpo_price.price_condition_position_currency_code AS konwa,
            ekpo.terminal_code                               AS zzterminal,
            ekpo.warehouse_storage_area_code                 AS vzzzone,
            COALESCE(ekpo_price.price_scale_currency_code,
                     ekpo_price.price_condition_position_currency_code) AS konws,
            ekpo.service_code,
            ekpo.price_uom_code,
            ekpo.price_vat_excluded_document_currency_amount AS netpr,
            ekpo_price.price_rate                            AS kbetr,

            -- for_1079_rf
            (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END
            ) AS order_rf,

            -- for_1079_asia
            (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.warehouse_storage_area_code IS NULL THEN 1 ELSE 0 END
            ) AS order_asia,

            -- for_1079_euro_svh
            (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
             CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END
            ) AS order_euro_svh,

            CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor,
            ekpo_price.is_price_condition_position_deleted_code,
            COUNT(*) OVER (
                PARTITION BY ekpo.purchase_contract_code, ekpo.position_line_item_code
            ) AS cnt,
            ROW_NUMBER() OVER (
                PARTITION BY ekpo.purchase_contract_code,
                             ekpo.position_line_item_code,
                             ekpo_price.dt_price_condition_valid_from,
                             ekpo_price.dt_price_condition_valid_to,
                             ekpo_price.price_scale_basis_code
                ORDER BY ekpo_price.price_scale_basis_consecutive_code
            ) AS rwn
        FROM dds.purchase_contract_price_conditions AS ekpo_price
        JOIN dds.purchase_contract_header AS ekko
          ON ekko.purchase_contract_code = ekpo_price.purchase_contract_code
        JOIN dds.purchase_contract_position AS ekpo
          ON ekpo.purchase_contract_code   = ekko.purchase_contract_code
         AND ekpo.position_line_item_code  = ekpo_price.purchase_contract_position_code
        LEFT JOIN dict_dds.currency_decimal_place_ral AS curdpr
          ON curdpr.currency_code       = COALESCE(ekpo_price.price_scale_currency_code,
                                                   ekpo_price.price_condition_position_currency_code)
         AND curdpr.decimal_place_number = '0'
        WHERE ekko.purchase_document_subtype_code IN ('ZDTL', 'ZDT', 'ZDE', 'ZDP', 'ZDF', 'ZDS')
          AND ekko.paydox_contract_type_code       IN ('1', '6')
          AND ekpo.is_deleted IS NULL
          AND ekpo_price.is_price_condition_position_deleted_code IS NULL
          AND ekpo_price.price_condition_type_code = 'PB00'
          AND (ekpo_price.price_scale_basis_code = 'C' OR ekpo_price.price_scale_basis_code IS NULL)
    ) AS for_1079_base_calc
    LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_1
      ON tmp_1.abap_program_code = '/RUSAL/SD4553M'
     AND tmp_1.parameter_code    = 'S_USLSTC'
     AND tmp_1.range_low_value   = for_1079_base_calc.zzsrvpos
    LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_2
      ON tmp_2.abap_program_code = '/RUSAL/SD4553M'
     AND tmp_2.parameter_code    = 'S_USLST'
     AND tmp_2.range_low_value   LIKE 'EU%'
     AND tmp_2.range_low_value   = for_1079_base_calc.zzsrvpos
    WHERE for_1079_base_calc.rwn = 1
)
DISTRIBUTED REPLICATED;  -- оставляем репликацию: таблица относительно небольшая

CREATE TEMPORARY TABLE for_1079_rf_cont ON COMMIT DROP AS (
    SELECT
        dfdf.rn,
        dfdf.purchase_contract_code,
        dfdf.position_line_item_code,
        dfdf.konws,
        dfdf.service_code,
        dfdf.price_uom_code,
        dfdf.currency_dogovor,
        dfdf.datab,
        dfdf.datbi
    FROM (
        SELECT
            base_2.rn,
            for_1079_base.purchase_contract_code,
            for_1079_base.position_line_item_code,
            ROW_NUMBER() OVER (
                PARTITION BY base_2.rn
                ORDER BY for_1079_base.order_rf,
                         for_1079_base.kopos,
                         for_1079_base.kbetr,
                         for_1079_base.purchase_contract_code,
                         for_1079_base.position_line_item_code
            ) AS row_num,
            for_1079_base.konws,
            for_1079_base.service_code,
            for_1079_base.price_uom_code,
            for_1079_base.currency_dogovor,
            for_1079_base.datab,
            for_1079_base.datbi
        FROM dm_calc.storage_sales_bundles_weight AS base_2
        JOIN for_1079_base
          ON for_1079_base.bukrs = '1120'
         AND for_1079_base.lifnr = base_2.creditor_in_shipment_instruction_code
         AND (for_1079_base.zzport = base_2.port_of_discharge_code OR for_1079_base.zzport IS NULL)
         AND for_1079_base.werks = '1121'
         AND for_1079_base.zzsrvpos = 'S49'
         AND (for_1079_base.zzforma   = base_2.material_shape_code OR for_1079_base.zzforma   IS NULL)
         AND (for_1079_base.zzet_tarif = base_2.etsng_code          OR for_1079_base.zzet_tarif IS NULL)
         AND (for_1079_base.sdabw     = base_2.transport_type_code  OR for_1079_base.sdabw     IS NULL)
         AND base_2.dt_for_price_search_in_purchase_contract
               BETWEEN for_1079_base.dt_valid_from AND for_1079_base.dt_valid_to
         AND base_2.dt_for_price_search_in_purchase_contract
               BETWEEN for_1079_base.datab        AND for_1079_base.datbi
         AND base_2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры')
    ) AS dfdf
    WHERE dfdf.row_num = 1
)
DISTRIBUTED BY (rn);   -- ключ соединения с base_2_5

-- Логика Европа
-----------------------------------------------------------------------
CREATE TEMPORARY TABLE for_1079_euro ON COMMIT DROP AS (
    SELECT
        dfdf.rn,
        dfdf.purchase_contract_code,
        dfdf.position_line_item_code,
        dfdf.konws,
        dfdf.service_code,
        dfdf.price_uom_code,
        dfdf.warehouse_weight_month_end,
        dfdf.month_difference_2,
        dfdf.netpr,
        dfdf.special,
        dfdf.currency_dogovor,
        dfdf.datab,
        dfdf.datbi
    FROM (
        SELECT
            base_2.rn,
            for_1079_base.purchase_contract_code,
            for_1079_base.position_line_item_code,
            ROW_NUMBER() OVER (
                PARTITION BY base_2.rn
                ORDER BY
                    for_1079_base.order_euro_svh,
                    for_1079_base.kopos,
                    for_1079_base.kbetr,
                    CASE
                        WHEN param_sap.range_low_value IS NOT NULL
                             AND param_srv_1.range_low_value IS NOT NULL THEN 1
                        WHEN param_sap.range_low_value IS NULL
                             AND param_srv_1.range_low_value IS NULL THEN 1
                        ELSE 0
                    END DESC,
                    for_1079_base.purchase_contract_code,
                    for_1079_base.position_line_item_code
            ) AS row_num,
            for_1079_base.konws,
            for_1079_base.service_code,
            for_1079_base.price_uom_code,
            SUM(base_2.sales_bundle_gross_weight) OVER (
                PARTITION BY base_2.warehouse_code,
                             date_trunc('month', base_2.dt_storage_end)
            ) AS warehouse_weight_month_end,
            CASE
                WHEN EXTRACT(DAY FROM base_2.dt_storage_end)
                     < EXTRACT(DAY FROM base_2.dt_storage_start)
                    THEN (EXTRACT(YEAR  FROM base_2.dt_storage_end) -
                          EXTRACT(YEAR  FROM base_2.dt_storage_start)) * 12 +
                         (EXTRACT(MONTH FROM base_2.dt_storage_end) -
                          EXTRACT(MONTH FROM base_2.dt_storage_start))
                ELSE (EXTRACT(YEAR  FROM base_2.dt_storage_end) -
                      EXTRACT(YEAR  FROM base_2.dt_storage_start)) * 12 +
                     (EXTRACT(MONTH FROM base_2.dt_storage_end) -
                      EXTRACT(MONTH FROM base_2.dt_storage_start)) + 1
            END AS month_difference_2,
            for_1079_base.netpr,
            CASE
                WHEN base_2.storage_cost_special_calculation_type_code = '4'
                    THEN ceil(base_2.storage_duration_total_calendar_days / 7::numeric)
                WHEN base_2.storage_cost_special_calculation_type_code = '5'
                    THEN ceil(base_2.storage_duration_total_calendar_days / 30::numeric)
            END AS special,
            for_1079_base.currency_dogovor,
            for_1079_base.datab,
            for_1079_base.datbi
        FROM dm_calc.storage_sales_bundles_weight AS base_2
        LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_1
          ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
         AND for_1082_1.case_code        = 'PERET_FRN'
         AND for_1082_1.parameter_code   = 'S_LIFNRA'
         AND for_1082_1.range_low_value  = base_2.forwarder_code
        LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_2
          ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
         AND for_1082_2.parameter_code    = 'S_FORMSTEU'
         AND for_1082_2.range_low_value   = base_2.material_shape_code
        JOIN for_1079_base
          ON base_2.storage_cost_calculation_type_name = 'Логика Европа'
         AND for_1079_base.purchase_document_subtype_code = 'ZDTL'
         AND (
                (for_1082_1.range_low_value IS NOT NULL AND for_1079_base.zzsrvpos = 'S49')
             OR (for_1082_1.range_low_value IS NULL
                 AND for_1082_2.range_low_value IS NOT NULL
                 AND for_1079_base.tmp_1 IS NOT NULL)
             OR (for_1082_1.range_low_value IS NULL
                 AND for_1082_2.range_low_value IS NULL
                 AND for_1079_base.tmp_2 IS NOT NULL)
             )
         AND for_1079_base.bukrs = base_2.unit_balance_code
         AND for_1079_base.lifnr = base_2.forwarder_code
         AND for_1079_base.werks = base_2.plant_code
         AND (for_1079_base.zzport = base_2.port_of_discharge_code OR for_1079_base.zzport IS NULL)
         AND (for_1079_base.zzforma = base_2.material_shape_code     OR for_1079_base.zzforma IS NULL)
         AND (for_1079_base.zzet_tarif = base_2.etsng_code           OR for_1079_base.zzet_tarif IS NULL)
         AND (for_1079_base.sdabw = base_2.transport_type_code       OR for_1079_base.sdabw IS NULL)
         AND base_2.dt_for_price_search_in_purchase_contract
               BETWEEN for_1079_base.dt_valid_from AND for_1079_base.dt_valid_to
         AND base_2.dt_for_price_search_in_purchase_contract
               BETWEEN for_1079_base.datab        AND for_1079_base.datbi
         AND (for_1079_base.zzterminal = base_2.warehouse_code OR for_1079_base.zzterminal IS NULL)
        LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_3
          ON for_1079_base.tmp_2 IS NOT NULL
         AND for_1082_3.range_low_value LIKE 'EU%'
         AND for_1082_3.abap_program_code = '/RUSAL/SD4553M'
         AND for_1082_3.parameter_code    = 'S_USLST'
        LEFT JOIN dict_dds.settings_and_parameters_sap AS param_sap
          ON param_sap.range_low_value   = base_2.material_shape_code
         AND param_sap.abap_program_code = '/RUSAL/SD4553M'
         AND param_sap.parameter_code    = 'FORMSTEU'
        LEFT JOIN dict_dds.settings_and_parameters_sap AS param_srv_1
          ON param_srv_1.range_low_value   = for_1082_3.range_low_value
         AND param_srv_1.abap_program_code = '/RUSAL/SD4553M'
         AND param_srv_1.parameter_code    = 'S_USLSTC'
    ) AS dfdf
    WHERE dfdf.row_num = 1
)
DISTRIBUTED BY (rn);

-- LE.001090: расчёт цены хранения (for_1090)
-----------------------------------------------------------------------
CREATE TEMPORARY TABLE for_1090 ON COMMIT DROP AS (
    WITH tmp AS (
        SELECT
            pcpc.price_condition_code           AS knumh,
            pcpc.purchase_contract_code         AS evrtn,
            pcpc.purchase_contract_position_code AS evrtp,
            pcpc.price_scale_basis_consecutive_code,
            pcpc.price_condition_position_code,
            pcpc.price_rate                     AS konp_kbetr,
            pcpc.price_scale_rate               AS konm_kbetr,
            pcpc.price_scale_limit_from_quantity AS kstbm,
            pcpc.price_scale_limit_to_quantity   AS kstbm_next,
            pcpc.price_scale_basis_code,
            pcpc.dt_price_condition_valid_to,
            pcpc.dt_price_condition_valid_from,
            COUNT(pcpc.price_scale_limit_from_quantity) OVER (
                PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
            ) AS cnt,
            ROW_NUMBER() OVER (
                PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
                ORDER BY pcpc.price_scale_basis_consecutive_code
            ) AS rn,
            MIN(price_scale_limit_from_quantity) OVER (
                PARTITION BY pcpc.purchase_contract_code,
                             pcpc.purchase_contract_position_code,
                             dt_price_condition_valid_from,
                             dt_price_condition_valid_to
            ) AS min_price_scale_limit_from_quantity,
            (pcpc.price_scale_limit_from_quantity -
             LAG(pcpc.price_scale_limit_from_quantity) OVER (
                 PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
                 ORDER BY pcpc.price_scale_basis_consecutive_code
             )
            ) * LAG(pcpc.price_scale_rate) OVER (
                 PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
                 ORDER BY pcpc.price_scale_basis_consecutive_code
            ) AS lagging
        FROM dds.purchase_contract_price_conditions AS pcpc
        WHERE pcpc.price_condition_type_code = 'PB00'
          AND (pcpc.price_scale_basis_code = 'C' OR pcpc.price_scale_basis_code IS NULL)
    )
    SELECT
        tmp.knumh,
        tmp.evrtn,
        tmp.evrtp,
        tmp.dt_price_condition_valid_to,
        tmp.dt_price_condition_valid_from,
        tmp.price_scale_basis_consecutive_code,
        tmp.price_condition_position_code,
        tmp.konp_kbetr,
        tmp.konm_kbetr,
        tmp.kstbm,
        tmp.kstbm_next,
        tmp.cnt,
        tmp.rn,
        tmp.price_scale_basis_code,
        tmp.min_price_scale_limit_from_quantity,
        SUM(tmp.lagging) OVER (
            PARTITION BY tmp.evrtn, tmp.evrtp
            ORDER BY tmp.price_scale_basis_consecutive_code
        ) AS presum
    FROM tmp
)
DISTRIBUTED REPLICATED;
