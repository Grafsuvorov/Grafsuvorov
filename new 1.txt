    WHERE abap_program_code = '/RUSAL/ZLE112M'

-- ------------------------------------------------------------------
-- Мэппинг "поставка -> список К/Н + номера + судно" (бывш. bill_search)
-- ------------------------------------------------------------------
a1 AS (  -- исходная постановка: взять поставки, где К/Н типа 'Й' или нет К/Н, связать с sales_bundle
    SELECT
        perw.delivery_code,
        sbdr1.sales_bundle_code
    FROM dds.transportation_service_invoice_subposition perw
    JOIN dds.transportation_service_invoice_header perh
      ON perw.plant_code = perh.plant_code
     AND perw.transportation_service_invoice_code = perh.transportation_service_invoice_code
    LEFT JOIN dds.bill_of_lading vbsk_src
      ON vbsk_src.bill_of_lading_code = perw.bill_of_lading_code
    JOIN dds.sales_bundle_and_delivery_relationship sbdr1
      ON sbdr1.delivery_code = perw.delivery_code
    WHERE perh.dt_transportation_service_invoice >= DATE '2023-01-01'
      AND (vbsk_src.bill_of_lading_type_code = 'Й' OR perw.bill_of_lading_code IS NULL)
      AND sbdr1.is_deleted_code IS NULL
    GROUP BY perw.delivery_code, sbdr1.sales_bundle_code
),
a2 AS (  -- по найденным sales_bundle получить полные поставки и их К/Н (тип 'Y')
    SELECT
        sbdr2.sales_bundle_code,
        sbdr2.delivery_code,
        vbsk_trg.bill_of_lading_code
    FROM dds.sales_bundle_and_delivery_relationship sbdr2
    JOIN dds.delivery_document_header likp
      ON likp.delivery_code = sbdr2.delivery_code
     AND likp.delivery_type_code = 'EL'
    JOIN dds.logistics_document_and_delivery_relationship vbss
      ON vbss.delivery_code = likp.delivery_code
    JOIN dds.bill_of_lading vbsk_trg
      ON vbsk_trg.bill_of_lading_code = vbss.logistics_document_code
     AND vbsk_trg.bill_of_lading_type_code = 'Y'
    WHERE sbdr2.is_deleted_code IS NULL
    GROUP BY sbdr2.sales_bundle_code, sbdr2.delivery_code, vbsk_trg.bill_of_lading_code
),
bill_search AS (   -- агрегируем до поставки
    SELECT
        j.delivery_code,
        STRING_AGG(DISTINCT vbsk.bill_of_lading_code, ',')                              AS bill_of_lading_compound_code,
        STRING_AGG(DISTINCT vbsk.bill_of_lading_number, ',')                            AS bill_of_lading_compound_number,
        STRING_AGG(DISTINCT COALESCE(tv.vessel_name, ''), ',') FILTER (WHERE tv.vessel_name IS NOT NULL) AS vessel_compound_name
    FROM (
        SELECT a1.delivery_code, a2.bill_of_lading_code
        FROM a1
        JOIN a2 ON a1.sales_bundle_code = a2.sales_bundle_code
        GROUP BY a1.delivery_code, a2.bill_of_lading_code
    ) j
    LEFT JOIN dds.bill_of_lading vbsk
      ON vbsk.bill_of_lading_code = j.bill_of_lading_code
    LEFT JOIN dds.nomination nom
      ON nom.nomination_code = vbsk.nomination_code
    LEFT JOIN dict_dds.transport_vehicle_texts tv
      ON tv.vessel_code = nom.vessel_number
     AND tv.language_code = 'R'
    GROUP BY j.delivery_code
),

-- --------------------------
-- perw_likp: поставки по perw
-- --------------------------
perw_likp AS (
    SELECT
        p.plant_code,
        p.transportation_service_invoice_code,
        p.delivery_code,
        likp.delivery_type_code,
        bs.bill_of_lading_compound_code,
        bs.bill_of_lading_compound_number,
        bs.vessel_compound_name,
        tvdlv.vessel_name
    FROM (
        SELECT DISTINCT
            perw.plant_code,
            perw.transportation_service_invoice_code,
            perw.delivery_code
        FROM dds.transportation_service_invoice_subposition perw
        JOIN dds.transportation_service_invoice_header perh
          ON perw.plant_code = perh.plant_code
         AND perw.transportation_service_invoice_code = perh.transportation_service_invoice_code
        WHERE perh.dt_transportation_service_invoice >= DATE '2023-01-01'
          AND perw.delivery_code IS NOT NULL
    ) p
    JOIN dds.delivery_document_header likp
      ON likp.delivery_code = p.delivery_code
    LEFT JOIN bill_search bs
      ON bs.delivery_code = p.delivery_code
    LEFT JOIN dict_dds.transport_vehicle_texts tvdlv
      ON tvdlv.vessel_code =
         CASE WHEN LENGTH(likp.vehicle_code) < 10
              THEN LPAD(likp.vehicle_code, 10, '0')
              ELSE likp.vehicle_code
          END
     AND tvdlv.language_code = 'R'
),

        x.bill_of_lading_code,
        b.bill_of_lading_type_code,
        b.bill_of_lading_number,
        tv.vessel_name
    FROM (
        SELECT DISTINCT
            perw.plant_code,
            perw.transportation_service_invoice_code,
            perw.bill_of_lading_code
        FROM dds.transportation_service_invoice_subposition perw

         AND perw.transportation_service_invoice_code = perh.transportation_service_invoice_code
        WHERE perh.dt_transportation_service_invoice >= DATE '2023-01-01'
          AND perw.bill_of_lading_code IS NOT NULL
    ) x
    JOIN dds.bill_of_lading b
      ON b.bill_of_lading_code = x.bill_of_lading_code
    LEFT JOIN dds.nomination nom
      ON nom.nomination_code = b.nomination_code
    LEFT JOIN dict_dds.transport_vehicle_texts tv
      ON tv.vessel_code = nom.vessel_number
     AND tv.language_code = 'R'
