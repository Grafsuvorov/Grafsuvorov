#!/usr/bin/bash
  set -euo pipefail

  unset http_proxy
  unset HTTPS_PROXY
  unset https_proxy
  unset HTTP_PROXY

  echo 'gogogogo'

  # здесь НЕТ git pull — считаем, что /root/git_dir уже обновлён на хосте

  cd /root/table-dependency-viewer/

  # meta_info (как было)
  rsync -atv --delete /root/git_dir/etl/meta_info/ /root/table-dependency-viewer/meta_info/

  # meta -> config_files/meta
  mkdir -p /root/table-dependency-viewer/config_files
  rsync -atv --delete /root/git_dir/etl/meta/ /root/table-dependency-viewer/config_files/meta/

  echo 'Закончили упражнение'
  echo 'Как же ты хорош!'

  Дальше:

  chmod +x /root/table-dependency-viewer/scripts/ci_cd.sh
  docker compose up -d --force-recreate

  Теперь кнопка будет работать без git в контейнере.
  Если хочешь — добавлю отдельную кнопку для host git pull (но это уже нужно делать вне контейнера, через systemd/cron/ssh).
