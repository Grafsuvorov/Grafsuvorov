SELECT
    n.nspname AS schema_name,
    c.relname AS table_name,
    pg_get_userbyid(c.relowner) AS owner_name,

    ROUND(
        pg_total_relation_size(c.oid) / 1024.0 / 1024 / 1024,
        2
    ) AS total_size_gb,

    ROUND(
        pg_relation_size(c.oid) / 1024.0 / 1024 / 1024,
        2
    ) AS table_size_gb,

    ROUND(
        (pg_total_relation_size(c.oid) - pg_relation_size(c.oid))
        / 1024.0 / 1024 / 1024,
        2
    ) AS additional_size_gb,

    NULL::timestamp AS created_at,
    NULL::integer   AS days_since_creation

FROM pg_class c
JOIN pg_namespace n
    ON n.oid = c.relnamespace

WHERE n.nspname = 'dds'
  AND c.relkind = 'r'

ORDER BY pg_total_relation_size(c.oid) DESC
LIMIT 30;
