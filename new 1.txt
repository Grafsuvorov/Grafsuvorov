import os, json, requests, urllib3
  urllib3.disable_warnings()

  issue = "DWH-10867"
  url = os.environ["YOUTRACK_URL"].rstrip("/")
  token = os.environ["YOUTRACK_TOKEN"]

  r = requests.get(
      f"{url}/api/issues/{issue}/activitiesPage",
      headers={"Authorization": f"Bearer {token}", "Accept": "application/json"},
      params={
          "fields":
  "activities(timestamp,author(name,login),field(name),added(name,fullName,login,text,presentation),removed(name,fullName,login,text,presentation),target(id,idReadable
  ))"
      },
      verify=False,
      timeout=60,
  )
  r.raise_for_status()
  data = r.json()

  for row in data.get("activities", []):
      field = (row.get("field") or {}).get("name")
      if field in {"Направление", "Дашборд КХД/Направление", "Дашборд SD"}:
          print(json.dumps(row, ensure_ascii=False, indent=2))
  PY
