def to_date(value):
    value = clean_text(value)

    if value is None:
        return None

    # Иногда из SAP приходит дата-время вместо даты:
    # 2023-05-26T12:00:00
    # Для date-поля берём только YYYY-MM-DD
    if "T" in value:
        value = value.split("T")[0]

    if " " in value:
        value = value.split(" ")[0]

    return datetime.date.fromisoformat(value)
