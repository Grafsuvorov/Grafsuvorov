def get_template_context(self, context: Context = None) -> dict:
    self._ensure_params_enriched()
    ctx = super().get_template_context(context)  # <= передаём context
    merged = {**self.params, **(ctx.get("params") or {})}
    ctx["params"] = merged
    return ctx
