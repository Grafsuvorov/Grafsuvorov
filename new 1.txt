create or replace function tech_etl.backup_entity_tables(
      p_entity_id integer,
      p_table_schema text
  )
  returns table (
      source_table text,
      backup_table text,
      status text
  )
  language plpgsql
  as
  $$
  declare
      r record;
      v_backup_suffix text := to_char(current_date, 'YYYY_MM_DD');
      v_backup_table_name text;
  begin
      for r in
          select t.table_name
          from tech_etl.tables_meta t
          where t.entity_id = p_entity_id
            and t.table_schema = p_table_schema
          order by t.table_name
      loop
          v_backup_table_name := format('%s_backup_%s', r.table_name, v_backup_suffix);

          execute format(
              'drop table if exists %I.%I',
              p_table_schema,
              v_backup_table_name
          );

          execute format(
              'create table %I.%I (like %I.%I including all)',
              p_table_schema,
              v_backup_table_name,
              p_table_schema,
              r.table_name
          );

          execute format(
              'insert into %I.%I select * from %I.%I',
              p_table_schema,
              v_backup_table_name,
              p_table_schema,
              r.table_name
          );

          source_table := format('%s.%s', p_table_schema, r.table_name);
          backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
          status := 'created';
          return next;
      end loop;
  end;
  $$;

  Запуск:

  select *
  from tech_etl.backup_entity_tables(35, 'dm');
