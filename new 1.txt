DROP TABLE IF EXISTS base_as_eu;

CREATE TEMP TABLE base_as_eu
DISTRIBUTED BY (rn) AS
SELECT
    b.rn,
    b.storage_cost_calculation_type_name,

    -- поля договора
    f.purchase_contract_code,
    f.position_line_item_code,
    f.order_asia,
    f.order_euro_svh,
    f.kopos,
    f.kbetr,

    f.konws,
    f.service_code,
    f.price_uom_code,
    f.currency_dogovor,
    f.datab,
    f.datbi,
    f.netpr,
    f.tmp_1,
    f.tmp_2,
    f.zzsrvpos,

    -- поля из базовой витрины
    b.sales_bundle_gross_weight,
    b.warehouse_code,
    b.dt_storage_end,
    b.dt_storage_start,
    b.storage_cost_special_calculation_type_code,
    b.storage_duration_total_calendar_days,
    b.forwarder_code,
    b.material_shape_code,
    b.etsng_code,
    b.transport_type_code,
    b.dt_for_price_search_in_purchase_contract
FROM dm_calc.storage_sales_bundles_weight AS b
JOIN for_1079_base AS f
  ON f.bukrs  = b.unit_balance_code
 AND f.lifnr  = b.forwarder_code
 AND f.werks  = b.plant_code
 AND f.purchase_document_subtype_code = 'ZDTL'
 AND b.dt_for_price_search_in_purchase_contract BETWEEN f.dt_valid_from AND f.dt_valid_to
 AND b.dt_for_price_search_in_purchase_contract BETWEEN f.datab        AND f.datbi
 AND (f.zzforma     = b.material_shape_code OR f.zzforma     IS NULL)
 AND (f.zzet_tarif  = b.etsng_code         OR f.zzet_tarif  IS NULL)
 AND (f.sdabw       = b.transport_type_code OR f.sdabw      IS NULL)
 AND (f.zzport      = b.port_of_discharge_code OR f.zzport  IS NULL)
 AND (f.zzterminal  = b.warehouse_code OR f.zzterminal IS NULL)
 AND b.storage_cost_calculation_type_name IN ('Логика Азия', 'Логика Европа');
Здесь мы один раз делаем тяжёлый join и сразу храним всё нужное (в т.ч. order_euro_svh, tmp_1, tmp_2, zzsrvpos).

4.3. Предрасчёт для Европы (base_eu)
Тут повторяем всю логику с for_1082_1/2/3, param_sap, param_srv_1, но уже от base_as_eu, а не от исходных таблиц:

sql
Копировать код
DROP TABLE IF EXISTS base_eu;

CREATE TEMP TABLE base_eu
DISTRIBUTED BY (rn) AS
SELECT
    b.*,
    -- LIFNRA (перевозчик)
    f1082_1.range_low_value AS lifnra_range,
    -- FORMSTEU по форме материала
    f1082_2.range_low_value AS formsteu_range,
    -- S_USLST по tmp_2 (EU%)
    f1082_3.range_low_value AS uslst_range,
    -- FORMSTEU для param_sap
    psap.range_low_value    AS param_sap_range,
    -- S_USLSTC для param_srv_1
    psrv.range_low_value    AS param_srv_range
FROM base_as_eu AS b
LEFT JOIN dict_dds.settings_and_parameters_sap AS f1082_1
  ON f1082_1.abap_program_code = '/RUSAL/ZLE112M'
 AND f1082_1.case_code        = 'PERET_FRN'
 AND f1082_1.parameter_code   = 'S_LIFNRA'
 AND f1082_1.range_low_value  = b.forwarder_code

LEFT JOIN dict_dds.settings_and_parameters_sap AS f1082_2
  ON f1082_2.abap_program_code = '/RUSAL/SD4553M'
 AND f1082_2.parameter_code    = 'S_FORMSTEU'
 AND f1082_2.range_low_value   = b.material_shape_code

LEFT JOIN dict_dds.settings_and_parameters_sap AS f1082_3
  ON b.tmp_2 IS NOT NULL
 AND f1082_3.range_low_value   LIKE 'EU%'
 AND f1082_3.abap_program_code = '/RUSAL/SD4553M'
 AND f1082_3.parameter_code    = 'S_USLST'

LEFT JOIN dict_dds.settings_and_parameters_sap AS psap
  ON psap.range_low_value      = b.material_shape_code
 AND psap.abap_program_code    = '/RUSAL/SD4553M'
 AND psap.parameter_code       = 'FORMSTEU'

LEFT JOIN dict_dds.settings_and_parameters_sap AS psrv
  ON psrv.range_low_value      = f1082_3.range_low_value
 AND psrv.abap_program_code    = '/RUSAL/SD4553M'
 AND psrv.parameter_code       = 'S_USLSTC'

WHERE b.storage_cost_calculation_type_name = 'Логика Европа'
  AND (
        (f1082_1.range_low_value IS NOT NULL AND b.zzsrvpos = 'S49')
     OR (f1082_1.range_low_value IS NULL AND f1082_2.range_low_value IS NOT NULL AND b.tmp_1 IS NOT NULL)
     OR (f1082_1.range_low_value IS NULL AND f1082_2.range_low_value IS NULL    AND b.tmp_2 IS NOT NULL)
  );
Это 1:1 перенос твоей логики фильтра для Европы, просто на предрасчитанном слое.

4.4. Месячный вес по складу
Вместо тяжёлого SUM() OVER (PARTITION BY warehouse_code, month) на 18 млн строк — считаем агрегат один раз:

sql
Копировать код
DROP TABLE IF EXISTS eu_wh_month_sum;

CREATE TEMP TABLE eu_wh_month_sum
DISTRIBUTED BY (warehouse_code) AS
SELECT
    warehouse_code,
    date_trunc('month', dt_storage_end)::date AS month_end,
    SUM(sales_bundle_gross_weight) AS warehouse_weight_month_end
FROM base_eu
GROUP BY
    warehouse_code,
    date_trunc('month', dt_storage_end)::date;
4.5. Финальный for_1079_euro с выбором лучшего договора
sql
Копировать код
DROP TABLE IF EXISTS for_1079_euro;

CREATE TEMP TABLE for_1079_euro
DISTRIBUTED BY (rn) AS
SELECT *
FROM (
    SELECT
        e.rn,
        e.purchase_contract_code,
        e.position_line_item_code,
        e.konws,
        e.service_code,
        e.price_uom_code,

        w.warehouse_weight_month_end,

        CASE
          WHEN EXTRACT(DAY FROM e.dt_storage_end)
             < EXTRACT(DAY FROM e.dt_storage_start)
          THEN (EXTRACT(YEAR  FROM e.dt_storage_end) - EXTRACT(YEAR  FROM e.dt_storage_start)) * 12
             + (EXTRACT(MONTH FROM e.dt_storage_end) - EXTRACT(MONTH FROM e.dt_storage_start))
          ELSE (EXTRACT(YEAR  FROM e.dt_storage_end) - EXTRACT(YEAR  FROM e.dt_storage_start)) * 12
             + (EXTRACT(MONTH FROM e.dt_storage_end) - EXTRACT(MONTH FROM e.dt_storage_start)) + 1
        END AS month_difference_2,

        e.netpr,

        CASE
          WHEN e.storage_cost_special_calculation_type_code = '4'
            THEN CEIL(e.storage_duration_total_calendar_days / 7::numeric)
          WHEN e.storage_cost_special_calculation_type_code = '5'
            THEN CEIL(e.storage_duration_total_calendar_days / 30::numeric)
        END AS special,

        e.currency_dogovor,
        e.datab,
        e.datbi,

        ROW_NUMBER() OVER (
          PARTITION BY e.rn
          ORDER BY
            e.order_euro_svh,
            e.kopos,
            e.kbetr,
            CASE
              WHEN e.param_sap_range IS NOT NULL AND e.param_srv_range IS NOT NULL THEN 1
              WHEN e.param_sap_range IS NULL     AND e.param_srv_range IS NULL     THEN 1
              ELSE 0
            END DESC,
            e.purchase_contract_code,
            e.position_line_item_code
        ) AS row_num
    FROM base_eu e
    LEFT JOIN eu_wh_month_sum w
      ON w.warehouse_code = e.warehouse_code
     AND w.month_end      = date_trunc('month', e.dt_storage_end)::date
) AS t
WHERE row_num = 1;
