create temp   table base_as_eu as 
select 
base_2.rn,
		for_1079_base.purchase_contract_code,
		for_1079_base.position_line_item_code,
		for_1079_base.order_asia, for_1079_base.kopos, for_1079_base.kbetr,
	    storage_cost_calculation_type_name,
		for_1079_base.konws,
		for_1079_base.service_code,
		for_1079_base.price_uom_code,
		for_1079_base.currency_dogovor,
		for_1079_base.datab,
		for_1079_base.datbi,
		base_2.sales_bundle_gross_weight,
		base_2.warehouse_code,
		base_2.dt_storage_end,
		base_2.dt_storage_start,
		base_2.storage_cost_special_calculation_type_code,
		base_2.storage_duration_total_calendar_days,
		base_2.forwarder_code,
		base_2.material_shape_code,
		for_1079_base.tmp_2,
		zzsrvpos
		
from
	dm_calc.storage_sales_bundles_weight as base_2
		join for_1079_base on
			for_1079_base.bukrs = base_2.unit_balance_code 
			and for_1079_base.lifnr = base_2.forwarder_code
			and for_1079_base.purchase_document_subtype_code = 'ZDTL'
			and base_2.dt_for_price_search_in_purchase_contract between for_1079_base.dt_valid_from and for_1079_base.dt_valid_to
			and base_2.dt_for_price_search_in_purchase_contract between for_1079_base.datab and for_1079_base.datbi
			and (for_1079_base.zzforma = base_2.material_shape_code or for_1079_base.zzforma is null)
			and (for_1079_base.zzet_tarif = base_2.etsng_code or for_1079_base.zzet_tarif is null)
			and (for_1079_base.sdabw = base_2.transport_type_code or for_1079_base.sdabw is null)
			and (for_1079_base.zzport = base_2.port_of_discharge_code or for_1079_base.zzport is null) 
			and (for_1079_base.zzterminal = base_2.warehouse_code or for_1079_base.zzterminal is null)
			and for_1079_base.werks = base_2.plant_code --and for_1079_base.zzsrvpos = 'S49'
			and base_2.storage_cost_calculation_type_name in ( 'Логика Европа')
 distributed by (rn)


create temporary table for_1079_euro-- on commit drop 
as (
select
	dfdf.rn,
	dfdf.purchase_contract_code,
	dfdf.position_line_item_code,
	dfdf.konws,
	dfdf.service_code,
	dfdf.price_uom_code,
	dfdf.warehouse_weight_month_end,
	dfdf.month_difference_2,
	dfdf.netpr,
	dfdf.special,
	dfdf.currency_dogovor,
	dfdf.datab,
	dfdf.datbi
from
	(

select
		base_2.rn,
		for_1079_base.purchase_contract_code,
		for_1079_base.position_line_item_code,
		row_number() over(partition by base_2.rn order by order_euro_svh,
															for_1079_base.kopos, 
															for_1079_base.kbetr, 
		
															case
																when param_sap.range_low_value is not null and param_srv_1.range_low_value is not null then 1
																when param_sap.range_low_value is null and param_srv_1.range_low_value is null then 1
																else 0
															end desc,
															
														for_1079_base.purchase_contract_code, for_1079_base.position_line_item_code) as row_num,
		
		
		
		
		for_1079_base.konws, 
		for_1079_base.service_code,
		for_1079_base.price_uom_code,
		sum(base_2.sales_bundle_gross_weight) over(partition by base_2.warehouse_code, date_trunc('month', base_2.dt_storage_end)) as warehouse_weight_month_end,

       case when extract(day from base_2.dt_storage_end) < extract(day from base_2.dt_storage_start) then
				(extract(year from base_2.dt_storage_end) - extract(year from base_2.dt_storage_start)) * 12 +
       			(extract(month from base_2.dt_storage_end) - extract(month from base_2.dt_storage_start))
			else (extract(year from base_2.dt_storage_end) - extract(year from base_2.dt_storage_start)) * 12 +
       			(extract(month from base_2.dt_storage_end) - extract(month from base_2.dt_storage_start)) + 1 end as month_difference_2,
       
		for_1079_base.netpr,

		case
			when base_2.storage_cost_special_calculation_type_code = '4' then ceil(base_2.storage_duration_total_calendar_days / 7 :: numeric)
			when base_2.storage_cost_special_calculation_type_code = '5' then ceil(base_2.storage_duration_total_calendar_days / 30 :: numeric)
		end as special,
		for_1079_base.currency_dogovor,
		for_1079_base.datab,
		for_1079_base.datbi
	from dm_calc.storage_sales_bundles_weight as base_2
	left join dict_dds.settings_and_parameters_sap as for_1082_1 on
		for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
		and for_1082_1.case_code = 'PERET_FRN'
		and for_1082_1.parameter_code = 'S_LIFNRA'
		and for_1082_1.range_low_value = base_2.forwarder_code
	left join dict_dds.settings_and_parameters_sap as for_1082_2 on
		for_1082_2.abap_program_code = '/RUSAL/SD4553M'
		and for_1082_2.parameter_code = 'S_FORMSTEU'
		and for_1082_2.range_low_value = base_2.material_shape_code
	
		
	join for_1079_base on
		base_2.storage_cost_calculation_type_name = 'Логика Европа'
		and for_1079_base.purchase_document_subtype_code = 'ZDTL'
		and ((for_1082_1.range_low_value is not null and for_1079_base.zzsrvpos = 'S49')
		or (for_1082_1.range_low_value is null and for_1082_2.range_low_value is not null and for_1079_base.tmp_1 is not null)
		or (for_1082_1.range_low_value is null and for_1082_2.range_low_value is null and for_1079_base.tmp_2 is not null))
			and for_1079_base.bukrs = base_2.unit_balance_code --'1520'
			and for_1079_base.lifnr = base_2.forwarder_code
			and for_1079_base.werks = base_2.plant_code --'1521'
			and (for_1079_base.zzport = base_2.port_of_discharge_code or for_1079_base.zzport is null)
			and (for_1079_base.zzforma = base_2.material_shape_code or for_1079_base.zzforma is null)
			and (for_1079_base.zzet_tarif = base_2.etsng_code or for_1079_base.zzet_tarif is null)
			and (for_1079_base.sdabw = base_2.transport_type_code or for_1079_base.sdabw is null)
			and base_2.dt_for_price_search_in_purchase_contract between for_1079_base.dt_valid_from and for_1079_base.dt_valid_to
			and base_2.dt_for_price_search_in_purchase_contract between for_1079_base.datab and for_1079_base.datbi
			and (for_1079_base.zzterminal = base_2.warehouse_code or for_1079_base.zzterminal is null)
			
		left join dict_dds.settings_and_parameters_sap as for_1082_3 on
			for_1079_base.tmp_2 is not null
			and for_1082_3.range_low_value like 'EU%'
			and for_1082_3.abap_program_code = '/RUSAL/SD4553M'
			and for_1082_3.parameter_code = 'S_USLST'

		left join dict_dds.settings_and_parameters_sap as param_sap on
			param_sap.range_low_value = base_2.material_shape_code
			and param_sap.abap_program_code = '/RUSAL/SD4553M'
			and param_sap.parameter_code = 'FORMSTEU'
			   
		left join dict_dds.settings_and_parameters_sap as param_srv_1 on
			param_srv_1.range_low_value = for_1082_3.range_low_value
			and param_srv_1.abap_program_code = '/RUSAL/SD4553M'
			and param_srv_1.parameter_code = 'S_USLSTC'
		) as dfdf where dfdf.row_num = 1
)
distributed by (rn);
