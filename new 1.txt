 drop table if exists dq_sd_0005_tmp;
 
create temporary table dq_sd_0005_tmp (
    change_type_code text,
    total_weight_net numeric,
    total_including_smelters_wh_weight_net numeric,
    total_shipped_metal_in_inventory_weight_net numeric,
    total_not_yet_delivered_metal_weight_net numeric,
    dt_report date,
    delivery_number_sales text,
    batch text,
    uni text,
    weight_nk numeric,
    weight_net numeric,
    location text,
    business_location_name text,
    plant_producer_code text,
    warehouse_shipment_type_name text,
    delivery_number_of_producer_plant text,
    dttm_inserted timestamp
)
distributed by (dttm_inserted, dt_report, delivery_number_sales, batch);
 
insert into dq_sd_0005_tmp (
    change_type_code,
    total_weight_net,
    total_including_smelters_wh_weight_net,
    total_shipped_metal_in_inventory_weight_net,
    total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant,
    dttm_inserted
)
select
    '3-total_weight' as change_type_code,
    sum(weight_net)  as total_weight_net, --без фильтров с дашборда
    sum(case
        when location ilike 'Scheduled%'
            then 0
        else weight_net
    end)             as total_including_smelters_wh_weight_net, -- фильтры с дашборда
    sum(case
        when
            business_location_name in (
                'Status Smelter WH',
                'At station'
            )
            then 0
        when location ilike 'Scheduled%'
            then 0
        else weight_net
    end)             as total_shipped_metal_in_inventory_weight_net, -- фильтры с дашборда
    sum(case
        when
            business_location_name in (
                'Status Smelter WH',
                'At station',
                'Delivered'
            )
            then 0
        when location ilike 'Scheduled%'
            then 0
        else weight_net
    end)             as total_not_yet_delivered_metal_weight_net,
    dt_report,
    null             as delivery_number_sales,
    null             as batch,
    null             as uni,
    null             as weight_nk,
    null             as weight_net,
    null             as location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    null             as delivery_number_of_producer_plant,
    now()            as dttm_inserted
from
    dm.sb_wuc        as wuc
group by dt_report, business_location_name, plant_producer_code, warehouse_shipment_type_name
union all
select
    '1-add'          as change_type_code,
    0                as total_weight_net,
    0                as total_including_smelters_wh_weight_net,
    0                as total_shipped_metal_in_inventory_weight_net,
    0                as total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant,
    now()            as dttm_inserted
from
    dm.sb_wuc
where
    location not ilike 'Scheduled%'
    or location is null
except
select
    '1-add'          as change_type_code,
    0                as total_weight_net,
    0                as total_including_smelters_wh_weight_net,
    0                as total_shipped_metal_in_inventory_weight_net,
    0                as total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant,
    now()            as dttm_inserted
from
    dm_calc.sb_wuc_backup
where
    location not ilike 'Scheduled%'
    or location is null
union all
select
    '2-delete'       as change_type_code,
    0                as total_weight_net,
    0                as total_including_smelters_wh_weight_net,
    0                as total_shipped_metal_in_inventory_weight_net,
    0                as total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant,
    now()            as dttm_inserted
from
    dm_calc.sb_wuc_backup
where
    location not ilike 'Scheduled%'
    or location is null
except
select
    '2-delete'       as change_type_code,
    0                as total_weight_net,
    0                as total_including_smelters_wh_weight_net,
    0                as total_shipped_metal_in_inventory_weight_net,
    0                as total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant,
    now()            as dttm_inserted
from
    dm.sb_wuc
where
    location not ilike 'Scheduled%'
    or location is null;
 
with dt as (
    select max(dttm_inserted::date) as table_last_load
    from dm.dq_sd0005
),
 
de as (
    select
        error_code,
        error_short_name,
        error_full_name,
        business_area_code,
        error_type_code,
        severity_type_code,
        table_source_code,
        error_description_text,
        error_algorithm_text
    from
        dict_dds.dq_error
    where
        error_code = 'dq_sd0005'
)
 
insert into dm.dq_sd0005 (
    error_code,
    error_short_name,
    error_full_name,
    business_area_code,
    error_type_code,
    severity_type_code,
    table_source_code,
    error_description_text,
    error_algorithm_text,
    change_type_code,
    total_weight_net,
    total_including_smelters_wh_weight_net,
    total_shipped_metal_in_inventory_weight_net,
    total_not_yet_delivered_metal_weight_net,
    dt_report,
    delivery_number_sales,
    batch,
    uni,
    weight_nk,
    weight_net,
    location,
    business_location_name,
    plant_producer_code,
    warehouse_shipment_type_name,
    delivery_number_of_producer_plant
)
select
    de.error_code,
    de.error_short_name,
    de.error_full_name,
    de.business_area_code,
    de.error_type_code,
    de.severity_type_code,
    de.table_source_code,
    de.error_description_text,
    de.error_algorithm_text,
    dq.change_type_code,
    dq.total_weight_net,
    dq.total_including_smelters_wh_weight_net,
    dq.total_shipped_metal_in_inventory_weight_net,
    dq.total_not_yet_delivered_metal_weight_net,
    dq.dt_report,
    dq.delivery_number_sales,
    dq.batch,
    dq.uni,
    dq.weight_nk,
    dq.weight_net,
    dq.location,
    dq.business_location_name,
    dq.plant_producer_code,
    dq.warehouse_shipment_type_name,
    dq.delivery_number_of_producer_plant
from
    dq_sd_0005_tmp as dq
cross join de
where dq.dttm_inserted::date > (select table_last_load from dt);
 
drop table if exists dq_sd_0005_tmp;
