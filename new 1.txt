  
  SELECT *
FROM dm."toro_material_cost_analysis"
WHERE CONCAT_WS('~', "material_document_code",  "dt_material_document",  "operation_code",  "material_code",  "toro_order_code" )
IN (
    SELECT CONCAT_WS('~', "material_document_code",  "dt_material_document",  "operation_code",  "material_code",  "toro_order_code" )
    FROM dm."toro_material_cost_analysis"
    WHERE deleted_flag IS FALSE
    GROUP BY "material_document_code",  "dt_material_document",  "operation_code",  "material_code",  "toro_order_code" 
    HAVING COUNT(1)>1
)
