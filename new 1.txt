-- Основной интервал хранения
intervals AS (
    SELECT
        b.*,
        -- начало интервала
        b.dt_transportation_stage_start_p AS dt_start,
        -- конец интервала хранения
        COALESCE(b.dt_transportation_stage_start_r, current_date) AS dt_end
    FROM base b
    CROSS JOIN bounds
    WHERE b.dt_transportation_stage_start_p <= bounds.dt_max
      AND COALESCE(b.dt_transportation_stage_start_r, current_date) >= bounds.dt_min
),

-- Пересечение интервала bundle с периодом витрины
expanded AS (
    SELECT 
        i.sales_bundle_code,
        i.plant_producer_delivery_code,
        i.dt_start,
        i.dt_end,
        d.dt_balance AS dt_calc,
        i.sales_bundle_gross_weight,
        i.sales_bundle_net_weight,
        i.storage_cost_calculation_type_name,
        i.material_code,
        i.batch_code,
        i.transport_type_code,

        i.dt_arrival_to_russian_port,
        i.id_for_unique
    FROM intervals i
    JOIN date_for_vitr_tmp d
      ON d.dt_balance BETWEEN i.dt_start AND i.dt_end
),

-- Справочник бесплатных дней по договору
free_days AS (
    SELECT 
        f1087.purchase_contract_code,
        f1087.purchase_contract_position_code,
        f1087.dt_price_condition_valid_from,
        f1087.dt_price_condition_valid_to,
        GREATEST(f1087.price_scale_limit_from_quantity - 1, 0) AS free_days
    FROM for_1087 f1087
),

-- Спец-логика LE.001089–001125
calc_days AS (
    SELECT 
        e.*,
        fd.free_days,
        -- фактический номер дня в интервале хранения
        (e.dt_calc - e.dt_start) + 1 AS storage_day_number,
        CASE WHEN (e.dt_calc - e.dt_start) + 1 <= fd.free_days THEN 1 ELSE 0 END AS is_free_day,
        CASE WHEN (e.dt_calc - e.dt_start) + 1  > fd.free_days THEN 1 ELSE 0 END AS is_pay_day
    FROM expanded e
    LEFT JOIN free_days fd
      ON fd.purchase_contract_code          = e.purchase_contract_code
     AND fd.purchase_contract_position_code = e.purchase_contract_position_code
     AND fd.dt_price_condition_valid_from   = e.dt_price_condition_valid_from
     AND fd.dt_price_condition_valid_to     = e.dt_price_condition_valid_to
)
