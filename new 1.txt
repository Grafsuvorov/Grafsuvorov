SELECT
    event_time,
    type,
    query_id,
    query_duration_ms,
    exception_code,
    exception,
    query
FROM system.query_log
WHERE event_time >= now() - INTERVAL 3 HOUR
  AND (
        query ILIKE '%EXCHANGE%'
        OR query ILIKE '%имя_таблицы%'
      )
ORDER BY event_time;

SELECT
    database,
    name,
    uuid,
    engine,
    total_rows,
    metadata_modification_time
FROM system.tables
WHERE database = 'db'
  AND name IN ('table_main', 'table_tmp');

SELECT
    elapsed,
    query_id,
    user,
    query
FROM system.processes
WHERE query ILIKE '%EXCHANGE%'
   OR query ILIKE '%имя_таблицы%';
