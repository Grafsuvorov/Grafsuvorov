select distinct
       nv.nspname as view_schema,
       v.relname  as view_name
from pg_depend d
join pg_rewrite r
  on r.oid = d.objid
join pg_class v
  on v.oid = r.ev_class
join pg_namespace nv
  on nv.oid = v.relnamespace
join pg_class t
  on t.oid = d.refobjid
join pg_namespace nt
  on nt.oid = t.relnamespace
where nt.nspname = 'dm'
  and t.relname = 'my_table'
  and v.relkind = 'v'
order by 1, 2;

Результат будет примерно такой:

view_schema | view_name
------------+-------------------
dm          | my_view
report      | account_report_v
2. Получить SQL для пересоздания VIEW
select distinct
       nv.nspname as view_schema,
       v.relname  as view_name,
       'create or replace view '
       || quote_ident(nv.nspname)
       || '.'
       || quote_ident(v.relname)
       || ' as '
       || pg_get_viewdef(v.oid, true)
       || ';' as create_view_sql
from pg_depend d
join pg_rewrite r
  on r.oid = d.objid
join pg_class v
  on v.oid = r.ev_class
join pg_namespace nv
  on nv.oid = v.relnamespace
join pg_class t
  on t.oid = d.refobjid
join pg_namespace nt
  on nt.oid = t.relnamespace
where nt.nspname = 'dm'
  and t.relname = 'my_table'
  and v.relkind = 'v'
order by 1, 2;
