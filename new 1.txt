    -- dt_storage_end

        WHEN base_1_5.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')

    ON base_1_5.storage_cost_calculation_type_name = 'Логика Европа'
   AND for_1082_3.rn = 1

    ON for_1083_4.abap_program_code = '/RUSAL/ZLE112M'
   AND for_1083_4.case_code = 'PERET_FRN'
   AND for_1083_4.parameter_code = 'S_LIFNRA'
   AND for_1083_4.range_low_value = base_1_5.forwarder_code

    ON trwsa.plant_code = base_1_5.receiving_plant_code
   AND trwsa.transport_hub_code = base_1_5.warehouse_code
   AND trwsa.transportation_stage_code = base_1_5.transportation_stage_code

       OR (base_1_5.wwgsg_01 IS NOT NULL
