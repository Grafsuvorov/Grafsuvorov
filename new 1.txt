#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Lineage extractor for proto_*.sql
- Поиск файлов: рекурсивно, case-insensitive, маска proto_*.sql
- Цели: INSERT INTO / CREATE [MATERIALIZED] VIEW|TABLE / MERGE INTO
- Источники: FROM / JOIN / USING / UPDATE ... FROM / DELETE FROM
- Игнор временных таблиц: если объект без схемы (нет точки) — пропускаем
- Исключаем CTE: имена из WITH ... AS (...) не считаем источниками
- Сохраняет итог в lineage.json и lineage.csv
"""

import argparse
import json
import os
import re
from collections import defaultdict
from typing import Dict, List, Set, Tuple

# ---------- утилиты нормализации ----------

def _strip_quotes(name: str) -> str:
    # убираем двойные кавычки вокруг частей идентификатора
    return name.replace('"', '').replace("`", "").strip()

def _norm_ident(raw: str) -> str:
    """
    Нормализуем идентификатор:
    - убираем кавычки
    - схлопываем пробелы вокруг точки
    - приводим к нижнему регистру
    """
    s = _strip_quotes(raw)
    s = re.sub(r'\s*\.\s*', '.', s)
    s = re.sub(r'\s+', ' ', s).strip()
    return s.lower()

def _has_schema(obj: str) -> bool:
    """Считаем объект «реальным», если есть схема (schema.table)."""
    return '.' in obj

# ---------- регэкспы ----------

# Цели
RE_INSERT_INTO = re.compile(r'(?is)\binsert\s+into\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_CREATE_TBL  = re.compile(r'(?is)\bcreate\s+table\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_CREATE_VIEW = re.compile(r'(?is)\bcreate\s+(?:or\s+replace\s+)?(?:materialized\s+)?view\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_MERGE_INTO  = re.compile(r'(?is)\bmerge\s+into\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')

# Источники
RE_FROM_JOIN   = re.compile(r'(?is)\b(?:from|join)\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_USING       = re.compile(r'(?is)\busing\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_UPDATE_FROM = re.compile(r'(?is)\bupdate\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)\s+set\b.*?\bfrom\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)', re.S)
RE_DELETE_FROM = re.compile(r'(?is)\bdelete\s+from\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')

# CTE
RE_WITH_CTE    = re.compile(r'(?is)\bwith\s+(.+?)\bselect\b', re.S)
RE_CTE_NAME    = re.compile(r'(?is)\b([a-z0-9_."`$]+)\s+as\s*\(')

# Комментарии
RE_LINE_COMMENT = re.compile(r'--.*?$', re.M)
RE_BLOCK_COMMENT = re.compile(r'/\*.*?\*/', re.S)

def _remove_comments(sql: str) -> str:
    sql = RE_BLOCK_COMMENT.sub(' ', sql)
    sql = RE_LINE_COMMENT.sub(' ', sql)
    return sql

# ---------- парсер одного SQL ----------

def parse_sql(sql_text: str) -> Tuple[Set[str], Set[str]]:
    """
    Возвращает (targets, sources) — множество целевых и исходных объектов со схемой.
    """
    sql = _remove_comments(sql_text)

    # соберём CTE-имена, чтобы потом исключить из источников
    cte_names: Set[str] = set()
    for with_block in RE_WITH_CTE.findall(sql):
        for m in RE_CTE_NAME.finditer(with_block):
            nm = _norm_ident(m.group(1))
            # CTE — всегда без схемы, но исключим по имени
            cte_names.add(_strip_quotes(nm).lower())

    targets: Set[str] = set()
    sources: Set[str] = set()

    # цели
    for rx in (RE_INSERT_INTO, RE_CREATE_TBL, RE_CREATE_VIEW, RE_MERGE_INTO):
        for m in rx.finditer(sql):
            obj = _norm_ident(m.group(1))
            if _has_schema(obj):
                targets.add(obj)

    # источники: FROM/JOIN/USING
    for m in RE_FROM_JOIN.finditer(sql):
        obj = _norm_ident(m.group(1))
        if '(' in obj:  # защита от вызовов функций/подзапросов
            continue
        if not _has_schema(obj):
            continue
        # исключим, если совпадает с CTE (бывает, когда пишут schema.cte? редко, но на всякий случай)
        base = obj.split('.')[-1]
        if base in cte_names:
            continue
        sources.add(obj)

    for m in RE_USING.finditer(sql):
        obj = _norm_ident(m.group(1))
        if '(' in obj:
            continue
        if not _has_schema(obj):
            continue
        base = obj.split('.')[-1]
        if base in cte_names:
            continue
        sources.add(obj)

    # UPDATE ... FROM
    for m in RE_UPDATE_FROM.finditer(sql):
        tgt = _norm_ident(m.group(1))
        src = _norm_ident(m.group(2))
        if _has_schema(tgt):
            targets.add(tgt)
        if _has_schema(src):
            base = src.split('.')[-1]
            if base not in cte_names:
                sources.add(src)

    # DELETE FROM (цель — таблица, но источников может не быть)
    for m in RE_DELETE_FROM.finditer(sql):
        tgt = _norm_ident(m.group(1))
        if _has_schema(tgt):
            targets.add(tgt)

    # Убираем случаи, где цель и источник совпали (самообновления не информативны для «из чего собирается»)
    sources -= targets

    return targets, sources

# ---------- обход файлов ----------

def iter_proto_sql_files(root: str) -> List[str]:
    out = []
    for dirpath, dirnames, filenames in os.walk(root):
        for fn in filenames:
            fl = fn.lower()
            if fl.startswith('proto_') and fl.endswith('.sql'):
                out.append(os.path.join(dirpath, fn))
    return sorted(out)

# ---------- основной сбор lineage ----------

def build_lineage(root: str) -> Dict[str, Dict]:
    """
    Возвращает словарь:
    {
      "schema.table": {
        "file": "путь/к/файлу.sql",
        "sources": ["schema.src1", "schema.src2", ...]
      },
      ...
    }
    """
    lineage: Dict[str, Dict] = {}
    files = iter_proto_sql_files(root)

    for path in files:
        try:
            with open(path, 'r', encoding='utf-8', errors='ignore') as f:
                text = f.read()
        except Exception as e:
            print(f"[WARN] Не удалось прочитать {path}: {e}")
            continue

        targets, sources = parse_sql(text)

        # Для каждого таргета — привязываем источники, агрегируем
        for tgt in targets:
            entry = lineage.setdefault(tgt, {"file": path, "sources": set()})
            entry["file"] = entry.get("file") or path
            entry["sources"].update(sources)

    # Приводим множества к спискам и сортируем
    for tgt, data in list(lineage.items()):
        srcs = sorted(data["sources"])
        data["sources"] = srcs

    return lineage

# ---------- вывод ----------

def save_outputs(lineage: Dict[str, Dict], out_prefix: str) -> None:
    json_path = f"{out_prefix}.json"
    csv_path  = f"{out_prefix}.csv"

    with open(json_path, 'w', encoding='utf-8') as jf:
        json.dump(lineage, jf, ensure_ascii=False, indent=2)

    # CSV: target,source (многострочно, по одной строке на зависимость; если источников нет — одиночная строка)
    with open(csv_path, 'w', encoding='utf-8') as cf:
        cf.write("target,source,file\n")
        for tgt, data in sorted(lineage.items()):
            if data["sources"]:
                for src in data["sources"]:
                    cf.write(f"{tgt},{src},{data['file']}\n")
            else:
                cf.write(f"{tgt},,{data['file']}\n")

    print(f"[OK] Сохранено:\n  - {json_path}\n  - {csv_path}")

def print_summary(lineage: Dict[str, Dict]) -> None:
    print("\n=== Lineage summary ===")
    print(f"Целей (таблиц/представлений): {len(lineage)}")
    for tgt, data in sorted(lineage.items()):
        srcs = ", ".join(data["sources"]) if data["sources"] else "—"
        print(f"- {tgt}  <=  {srcs}")

# ---------- CLI ----------

def main():
    ap = argparse.ArgumentParser(
        description="Сбор lineage для proto_*.sql (игнор временных таблиц: без схемы)."
    )
    ap.add_argument(
        "root", nargs="?", default=os.getcwd(),
        help="Корневая папка проекта (по умолчанию — текущая директория)"
    )
    ap.add_argument(
        "--out", default="lineage",
        help="Префикс для выходных файлов (JSON/CSV). По умолчанию: lineage"
    )
    args = ap.parse_args()

    root = os.path.abspath(args.root)
    print(f"[INFO] Корневая папка: {root}")

    lineage = build_lineage(root)
    save_outputs(lineage, args.out)
    print_summary(lineage)

if __name__ == "__main__":
    main()
