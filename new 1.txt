------------------------------------------------------------
-- 0. База: только "Логика Европа" и нужные поля
------------------------------------------------------------
DROP TABLE IF EXISTS tmp_storage_eu;

CREATE TEMP TABLE tmp_storage_eu
-- можно подобрать другой ключ, если rn плохой, но rn даёт локальный window по rn
DISTRIBUTED BY (rn) AS
SELECT
    rn,
    warehouse_code,
    unit_balance_code,
    forwarder_code,
    plant_code,
    port_of_discharge_code,
    material_shape_code,
    etsng_code,
    transport_type_code,
    dt_for_price_search_in_purchase_contract,
    dt_storage_start,
    dt_storage_end,
    storage_duration_total_calendar_days,
    storage_cost_special_calculation_type_code,
    sales_bundle_gross_weight
FROM dm_calc.storage_sales_bundles_weight
WHERE storage_cost_calculation_type_name = 'Логика Европа';


------------------------------------------------------------
-- 1. Помесячный вес по складу
------------------------------------------------------------
DROP TABLE IF EXISTS tmp_storage_eu_month;

CREATE TEMP TABLE tmp_storage_eu_month
DISTRIBUTED BY (warehouse_code) AS
SELECT
    warehouse_code,
    date_trunc('month', dt_storage_end)::date AS month_end,
    SUM(sales_bundle_gross_weight) AS warehouse_weight_month_end
FROM tmp_storage_eu
GROUP BY 1,2;


------------------------------------------------------------
-- 2. Джоин с for_1079_base и всеми SAP-таблицами (без окон)
------------------------------------------------------------
DROP TABLE IF EXISTS tmp_storage_eu_with_price;

CREATE TEMP TABLE tmp_storage_eu_with_price
DISTRIBUTED BY (rn) AS
SELECT
    base.rn,
    base.warehouse_code,
    base.dt_storage_start,
    base.dt_storage_end,
    base.storage_duration_total_calendar_days,
    base.storage_cost_special_calculation_type_code,
    base.unit_balance_code,
    base.forwarder_code,
    base.plant_code,
    base.port_of_discharge_code,
    base.material_shape_code,
    base.etsng_code,
    base.transport_type_code,
    base.dt_for_price_search_in_purchase_contract,

    -- поля из for_1079_base
    for_1079_base.purchase_contract_code,
    for_1079_base.position_line_item_code,
    for_1079_base.konws,
    for_1079_base.service_code,
    for_1079_base.price_uom_code,
    for_1079_base.netpr,
    for_1079_base.currency_dogovor,
    for_1079_base.datab,
    for_1079_base.datbi,
    for_1079_base.order_euro_svh,
    for_1079_base.kopos,
    for_1079_base.kbetr,

    -- эквивалент твоего CASE с param_sap / param_srv_1 в ORDER BY
    CASE
        WHEN param_sap.range_low_value IS NOT NULL
             AND param_srv_1.range_low_value IS NOT NULL THEN 1
        WHEN param_sap.range_low_value IS NULL
             AND param_srv_1.range_low_value IS NULL THEN 1
        ELSE 0
    END AS param_rank
FROM tmp_storage_eu AS base
LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_1
  ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
 AND for_1082_1.case_code        = 'PERET_FRN'
 AND for_1082_1.parameter_code   = 'S_LIFNRA'
 AND for_1082_1.range_low_value  = base.forwarder_code
LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_2
  ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
 AND for_1082_2.parameter_code    = 'S_FORMSTEU'
 AND for_1082_2.range_low_value   = base.material_shape_code
JOIN for_1079_base
  ON for_1079_base.purchase_document_subtype_code = 'ZDTL'
 -- основные ключи
 AND for_1079_base.bukrs = base.unit_balance_code
 AND for_1079_base.lifnr = base.forwarder_code
 AND for_1079_base.werks = base.plant_code
 -- опциональные условия
 AND (for_1079_base.zzport    = base.port_of_discharge_code OR for_1079_base.zzport    IS NULL)
 AND (for_1079_base.zzforma   = base.material_shape_code     OR for_1079_base.zzforma   IS NULL)
 AND (for_1079_base.zzet_tarif= base.etsng_code              OR for_1079_base.zzet_tarif IS NULL)
 AND (for_1079_base.sdabw     = base.transport_type_code     OR for_1079_base.sdabw     IS NULL)
 AND base.dt_for_price_search_in_purchase_contract
        BETWEEN for_1079_base.dt_valid_from AND for_1079_base.dt_valid_to
 AND base.dt_for_price_search_in_purchase_contract
        BETWEEN for_1079_base.datab        AND for_1079_base.datbi
 AND (for_1079_base.zzterminal = base.warehouse_code OR for_1079_base.zzterminal IS NULL)
 -- логика включения строки как и была
 AND (
        (for_1082_1.range_low_value IS NOT NULL AND for_1079_base.zzsrvpos = 'S49')
     OR (for_1082_1.range_low_value IS NULL
         AND for_1082_2.range_low_value IS NOT NULL
         AND for_1079_base.tmp_1 IS NOT NULL)
     OR (for_1082_1.range_low_value IS NULL
         AND for_1082_2.range_low_value IS NULL
         AND for_1079_base.tmp_2 IS NOT NULL)
     )
LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_3
  ON for_1079_base.tmp_2 IS NOT NULL
 AND for_1082_3.range_low_value LIKE 'EU%'
 AND for_1082_3.abap_program_code = '/RUSAL/SD4553M'
 AND for_1082_3.parameter_code    = 'S_USLST'
LEFT JOIN dict_dds.settings_and_parameters_sap AS param_sap
  ON param_sap.range_low_value   = base.material_shape_code
 AND param_sap.abap_program_code = '/RUSAL/SD4553M'
 AND param_sap.parameter_code    = 'FORMSTEU'
LEFT JOIN dict_dds.settings_and_parameters_sap AS param_srv_1
  ON param_srv_1.range_low_value   = for_1082_3.range_low_value
 AND param_srv_1.abap_program_code = '/RUSAL/SD4553M'
 AND param_srv_1.parameter_code    = 'S_USLSTC';


------------------------------------------------------------
-- 3. Финальный запрос = row_number + join помесячного веса
------------------------------------------------------------
EXPLAIN (VERBOSE, FORMAT JSON)
SELECT
    t.rn,
    t.purchase_contract_code,
    t.position_line_item_code,
    t.konws,
    t.service_code,
    t.price_uom_code,
    m.warehouse_weight_month_end,
    CASE
        WHEN EXTRACT(DAY FROM t.dt_storage_end)
             < EXTRACT(DAY FROM t.dt_storage_start)
            THEN (EXTRACT(YEAR  FROM t.dt_storage_end) -
                  EXTRACT(YEAR  FROM t.dt_storage_start)) * 12 +
                 (EXTRACT(MONTH FROM t.dt_storage_end) -
                  EXTRACT(MONTH FROM t.dt_storage_start))
        ELSE (EXTRACT(YEAR  FROM t.dt_storage_end) -
              EXTRACT(YEAR  FROM t.dt_storage_start)) * 12 +
             (EXTRACT(MONTH FROM t.dt_storage_end) -
              EXTRACT(MONTH FROM t.dt_storage_start)) + 1
    END AS month_difference_2,
    t.netpr,
    CASE
        WHEN t.storage_cost_special_calculation_type_code = '4'
            THEN ceil(t.storage_duration_total_calendar_days / 7::numeric)
        WHEN t.storage_cost_special_calculation_type_code = '5'
            THEN ceil(t.storage_duration_total_calendar_days / 30::numeric)
    END AS special,
    t.currency_dogovor,
    t.datab,
    t.datbi
FROM (
    SELECT
        tj.*,
        ROW_NUMBER() OVER (
            PARTITION BY tj.rn
            ORDER BY
                tj.order_euro_svh,
                tj.kopos,
                tj.kbetr,
                tj.param_rank DESC,
                tj.purchase_contract_code,
                tj.position_line_item_code
        ) AS row_num
    FROM tmp_storage_eu_with_price tj
) AS t
JOIN tmp_storage_eu_month m
  ON m.warehouse_code = t.warehouse_code
 AND m.month_end = date_trunc('month', t.dt_storage_end)::date
WHERE t.row_num = 1;
