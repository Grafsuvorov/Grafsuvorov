insert into dm.sales_metal_stock_at_plant_for_transportation_reporting (
	batch,
	weight_net_with_wirerod,
	business_location_for_cargo_reporting_code,
	business_location_for_cargo_reporting_name,
	business_location_group_for_cargo_reporting_name,
	transport_departure_hub_code,
	transport_destination_hub_code,
	dt_business_location_status_acquired
)
select 
	sgp.batch as batch,
	sum(sgp.weight_net) as weight_net_with_wirerod,
	'M01' as business_location_for_cargo_reporting_code,
	tst.transportation_status_name as business_location_for_cargo_reporting_name,
	ts.transportation_status_group_code as business_location_group_for_cargo_reporting_name,
	sgp.plant_manufact as transport_departure_hub_code,
	sgp.plant_manufact as transport_destination_hub_code,
	sgp.dt_report as dt_business_location_status_acquired
from dm_calc.finish_goods_warehouse_stock_plant_by_date as sgp
left join dict_dds.transportation_status_texts as tst
	on tst.transportation_status_code = 'M01'
	and tst.language_code = 'R'
left join dict_dds.transportation_status as ts
	on ts.transportation_status_code = 'M01'
where sgp.dt_report = current_date - 1
group by
	sgp.batch,
	tst.transportation_status_name,
	ts.transportation_status_group_code,
	sgp.plant_manufact,
	sgp.dt_report
;

		base.foreign_port_vessel_loading_normative_days,
		base.normative_marine_transit2_duration_days_quantity,
		base.foreign_port_delivery_to_customer_normative_days,
    	base.foreign_port_storage_before_vessel_loading_norm_days,
    	base.route_trip_railway_plant_to_border_normative_days,
		case 
			when stock.delivery_number_sales is not null
				or svh.delivery_number_sales is not null
				or realized.delivery_number_sales is not null
				then 'X'
			else null
		end as transportation_reporting_indicator
	from delivery_schema as base
	left join dm_calc.sd_sales_stock_by_date as stock
		on stock.dt_report = current_date - 1
		and stock.delivery_number_sales = base.delivery_number_sales
		and stock.batch = base.batch
		and stock.dt_realization is null
		and ((stock.warehouse_shipment_type_name is not null and stock.shipment_market_code in ('1', '4'))
	        or stock.warehouse_shipment_type_name is null)
	left join dm_calc.sd_sales_svh_stock_by_date as svh
		on svh.dt_report = current_date - 1
		and svh.dt_realization is null
		and svh.delivery_number_sales = base.delivery_number_sales
		and svh.batch = base.batch
	left join realized_delivery as realized
		on realized.delivery_number_sales = base.delivery_number_sales
		and realized.batch = base.batch
),
unique_routes as (
	select
	    transport_route_technical_code,						-- SD.001670 ID маршрута в цепочке поставок
		business_location_transport_departure_hub_code,  	-- SD.001673 Код реперной точки отправления
	    business_location_transport_destination_hub_code, 	-- SD.001674 Код реперной точки назначения
	    max(dt_delivery_actual)	as 	dt_delivery_actual		-- SD.001672 Фактическая дата в цепочке доставки
	from delivery_schema
	where dt_delivery_actual is not null
	group by
	    transport_route_technical_code,
		business_location_transport_departure_hub_code,
	    business_location_transport_destination_hub_code
),
preliminary_routes as (
	select
		base.transport_route_technical_code, 										-- SD.001670 ID маршрута в цепочке поставок
		base.business_location_transport_departure_hub_code,  						-- SD.001673 Код реперной точки отправления
		base.business_location_transport_destination_hub_code, 						-- SD.001674 Код реперной точки назначения
		-- SD.001675 Кол.дней на этапе маршрута (сумма)
		sum(base.business_location_stay_duration_days_quantity) as sum_business_location_stay_duration_days_quantity,
		-- количество уникальных строк для каждой найденной группы
		count(*) as count_rows_unique_group,
		-- сумма "Кол.дней на этапе маршрута"
		sum(base.business_location_stay_duration_days_quantity * ssms.weight_net) as sum_w_business_location_stay_duration_days_quantity,
		-- сумма "Вес нетто"
		sum(ssms.weight_net) as sum_weight_net
	from precalc as base
	inner join unique_routes as ur
		on ur.transport_route_technical_code = base.transport_route_technical_code
		and ur.business_location_transport_departure_hub_code = base.business_location_transport_departure_hub_code
		and ur.business_location_transport_destination_hub_code = base.business_location_transport_destination_hub_code
		and ur.dt_delivery_actual >= (base.dt_delivery_actual - 90)
	left join dm_calc.sd_sales_main_scm as ssms
		on ssms.delivery_number_sales = base.delivery_number_sales
		and ssms.batch = base.batch
	group by base.transport_route_technical_code,
			 base.business_location_transport_departure_hub_code,
			 base.business_location_transport_destination_hub_code
),
avg_metrics as (
	select
		base.transport_route_technical_code, 										-- SD.001670 ID маршрута в цепочке поставок
		base.business_location_transport_departure_hub_code,  						-- SD.001673 Код реперной точки отправления
		base.business_location_transport_destination_hub_code, 						-- SD.001674 Код реперной точки назначения
		-- SD.001778 Средневзвешенное количество дней в локации
		round(base.sum_w_business_location_stay_duration_days_quantity / nullif(base.sum_weight_net, 0)) as location_stay_duration_average_weighted_days,
		-- SD.001779 Среднее количество дней в локации
		round(base.sum_business_location_stay_duration_days_quantity / nullif(base.count_rows_unique_group, 0)) as location_stay_duration_average_days
	from preliminary_routes as base
)
select
    base.transport_route_type_name,
    base.delivery_number_sales,
    base.batch,
    base.weight_net_with_wirerod,
    base.transport_route_technical_code,
    base.dt_delivery_plan,
    base.dt_delivery_actual,
    base.business_location_transport_departure_hub_code,
    base.business_location_transport_destination_hub_code,
    base.business_location_stay_duration_days_quantity,
    base.dt_delivery_estimated,
    base.is_delivery_plan_date_overdue,
    (base.dt_realization_estimated
    +
	coalesce(
	        sum(coalesce(m.location_stay_duration_average_weighted_days, 0)) over (
	            partition by base.delivery_number_sales, base.batch
	            order by base.transport_route_technical_code
	            rows between 1 following and unbounded following
        	),
       		0
    )::int) as dt_realization_estimated,
    base.dt_realization_basis_document,
	base.destination_warehouse_code,
	base.destination_transport_hub_code,
    (base.dt_delivery_to_customer_estimated
    +
	coalesce(
	        sum(coalesce(m.location_stay_duration_average_weighted_days, 0)) over (
	            partition by base.delivery_number_sales, base.batch
	            order by base.transport_route_technical_code
	            rows between 1 following and unbounded following
	        ),
	        0
    )::int) as dt_delivery_to_customer_estimated,	
    base.business_location_at_report_date_code,
    base.business_location_for_cargo_reporting_code,
    base.business_location_for_cargo_reporting_name,
    base.business_location_group_for_cargo_reporting_name,
    base.transport_departure_hub_code,
    base.transport_destination_hub_code,
	base.cargo_document_code,
    base.transport_type_name,
    base.route_plant_code,
    base.transport_hub_enroute_code,
    base.border_cross_to_final_station_normative_days,
    base.normative_route_trip_duration_days_quantity,
    base.normative_railway_trip_duration_days_quantity,
    base.storage_before_repacking_normative_days,
    base.arrival_to_port_after_repacking_normative_days,
    base.route_delivery_after_repacking_normative_days,
    base.acceptance_normative_russian_port_days_amount,
    base.port_of_loading_storage_normative_days,
    base.normative_marine_transit1_duration_days_quantity,
    base.acceptance_normative_foreign_port_days_amount,
    base.foreign_port_storage_normative_days,
    base.acceptance_normative_second_foreign_port_days_amount,
    base.second_foreign_port_storage_normative_days,
    base.foreign_port_vessel_loading_normative_days,
    base.normative_marine_transit2_duration_days_quantity,
    base.foreign_port_delivery_to_customer_normative_days,
    base.foreign_port_storage_before_vessel_loading_norm_days,
    base.route_trip_railway_plant_to_border_normative_days,
    m.location_stay_duration_average_weighted_days,
    m.location_stay_duration_average_days,
    base.transportation_reporting_indicator
from precalc as base
left join avg_metrics as m
	on m.transport_route_technical_code = base.transport_route_technical_code
	and m.business_location_transport_departure_hub_code = base.business_location_transport_departure_hub_code
	and m.business_location_transport_destination_hub_code = base.business_location_transport_destination_hub_code

		-- SD.001688 Норма хранения в ин.порту
		case
			when rt.route_code = 'ROUTE070' and di.is_shipped_via_overseas_warehouse is not null then di.foreign_port_storage_normative_days
			else 0
		end as foreign_port_storage_normative_days,
		-- SD.000710 Норма приемки терминал/склад ин. порта 2
		case
			when rt.route_code = 'ROUTE100' then di.acceptance_normative_second_foreign_port_days_amount
			else 0
		end as acceptance_normative_second_foreign_port_days_amount,
		-- SD.001689 Норма хранения в ин.порту 2
		case
			when rt.route_code = 'ROUTE110' and di.is_shipped_via_overseas_second_foreign_warehouse is not null then di.second_foreign_port_storage_normative_days
			else 0
		end as second_foreign_port_storage_normative_days,
		-- SD.001690 Норма погрузки на судно в ин. порту
		case
			when rt.route_code = 'ROUTE080' and di.is_shipped_via_overseas_warehouse is not null then di.foreign_port_vessel_loading_normative_days
			else 0
		end as foreign_port_vessel_loading_normative_days,
		-- SD.000777 Норма морского транзита 2
		case
			when rt.route_code = 'ROUTE100' then di.normative_marine_transit2_duration_days_quantity::int
			else 0
		end as normative_marine_transit2_duration_days_quantity,
		-- SD.001691 Норма доставки по маршруту со склада ин. порта
		case
			when rt.route_code = 'ROUTE120' and di.is_shipped_via_overseas_second_foreign_warehouse is not null then di.foreign_port_delivery_to_customer_normative_days
			else 0
		end as foreign_port_delivery_to_customer_normative_days,
		-- SD.001780 Норма хранения до погрузки на судно в ин.порту
		case
			when rt.route_code = 'ROUTE070' and di.is_shipped_via_overseas_warehouse is not null then di.foreign_port_storage_before_vessel_loading_norm_days
			else 0
		end as foreign_port_storage_before_vessel_loading_norm_days,		
		-- SD.001668 Тип ТС в дислокации
		case
			when rt.route_code in ('ROUTE130', 'ROUTE120', 'ROUTE110', 'ROUTE100', 'ROUTE090')
				then di.transport_railcar_type_code
			when rt.route_code in ('ROUTE080', 'ROUTE070', 'ROUTE060', 'ROUTE050') and di.dt_repacked is not null
				then di.container_after_repacking
			when rt.route_code in ('ROUTE080', 'ROUTE070', 'ROUTE060', 'ROUTE050', 
                         		   'ROUTE040', 'ROUTE030', 'ROUTE020', 'ROUTE010')
                then di.transport_type_at_plant_code
		end as transport_type_name,
		di.dt_realization_basis_document,				-- SD.001679 Дата основания для реализации
		di.destination_warehouse_code,					-- SD.001680 Склад грузополучателя по маршруту завода (код)
		-- SD.001681 Конечный пункт назначения в маршруте завода (код)	
		coalesce(trs_con.transport_route_stage_destination_hub_code, trs_dest.transport_route_stage_destination_hub_code) as destination_transport_hub_code,
		di.delivery_basis,								-- SD.000067 Базис поставки
		di.final_release_code,							-- SD.000588 Номер Финальный релиз
		di.delivery_notice_number,						-- SD.000133 Номер нотиса о доставке
		di.bill_of_lading_in_foreign_port,				-- SD.000048 Коносамент в ин.порту
		di.bill_of_lading_number,						-- SD.000041 Номер коносамента
		di.transport_bill,								-- SD.000014 Накладная
		-- SD.001781 Норма движения по ЖД (дни) от Завода до Пограничного перехода в Китай
		0 as route_trip_railway_plant_to_border_normative_days
	from dm_calc.delivery_schema_general_info as di
	left join route_types as rt
		on rt.type_code = '1'
	left join  (
				select distinct range_low_value
				from dict_dds.settings_and_parameters_sap
				where abap_program_code = '/RUSAL/MK_TRACK_ROUTE'
					and parameter_code in ('INCO1CIF', 'INCO1CIP')
					and range_low_value is not null
			  ) as inco1_params
		on inco1_params.range_low_value = di.delivery_basis
	left join ( select distinct on (transport_route_code)
					transport_route_code,
					transport_route_stage_departure_hub_code
				from dict_dds.transport_route_stage
				order by 
						transport_route_code,
						transport_route_stage_number
			) as trs_dep
		on trs_dep.transport_route_code = di.route_plant_code
	left join ( select distinct on (transport_route_code)
					transport_route_code,
					transport_route_stage_destination_hub_code
				from dict_dds.transport_route_stage
				order by 
						transport_route_code,
						transport_route_stage_number desc
			) as trs_dest
		on trs_dest.transport_route_code = di.route_plant_code
	-- SD.001681 Конечный пункт назначения в маршруте завода (код)
	left join ( select distinct on (transport_route_code)
					transport_route_code,
					transport_route_stage_destination_hub_code
				from dict_dds.transport_route_stage
				order by 
						transport_route_code,
						transport_route_stage_number desc
			) as trs_con
		on trs_con.transport_route_code = di.bill_of_lading_route
	--	Определение реперных точек
	left join dict_dds.transport_hub as th
		on th.transport_hub_code = di.port_of_discharge_code
	left join dict_dds.transport_hub_texts as tht_1
		on tht_1.language_code = 'R'
		and tht_1.transport_hub_code like 'G%'
		and tht_1.transport_hub_name = di.foreigh_port_terminal_name
	left join dict_dds.transport_hub_texts as tht_2
		on tht_2.language_code = 'R'
		and tht_2.transport_hub_code like 'G%'
		and tht_2.transport_hub_name = di.fwrd_info_second_foreign_warehouse_location_name
	left join dict_dds.location_sales as ls
		on ls.location_code = di.tsw_location_code
	left join dict_dds.incoterms_location as il
		on il.incoterms_location_code = di.delivery_point_name
	where di.transport_route_type_name = 'X'
	order by
			delivery_number_sales,
			batch,
			transport_route_technical_code
),
delivery_routes_adv as (
	select
		drb.transport_route_type_name,  							-- SD.001669 Тип маршрута в цепочке поставок Морской/Сухопутный/Внутренний рынок
		drb.delivery_number_sales,  								-- SD.000002 Продажная поставка
		drb.batch,													-- SD.000004 Партия
		drb.weight_net_with_wirerod,								-- SD.000033 Вес Н&K
		drb.transport_route_technical_code,  						-- SD.001670 ID маршрута в цепочке поставок
		drb.dt_delivery_plan,										-- SD.001671 Плановая дата в цепочке доставки
		drb.is_relevant_route,										-- признак релевантности маршрута
		-- предыдущее значение SD.001671 относительно текущего маршрута SD.001670
		lag(drb.dt_delivery_plan) over (partition by 
										drb.delivery_number_sales,
										drb.batch
									order by drb.transport_route_technical_code) as dt_delivery_plan_prev, 
		-- последующее значение SD.001671 относительно текущего маршрута SD.001670
		lead(drb.dt_delivery_plan) over (partition by 
													drb.delivery_number_sales,
													drb.batch
									order by drb.transport_route_technical_code) as dt_delivery_plan_next, 
		drb.dt_delivery_actual,										-- SD.001672 Фактическая дата в цепочке доставки
		-- предыдущее значение SD.001672 относительно текущего маршрута SD.001670
		lag(drb.dt_delivery_actual) over (partition by 
													drb.delivery_number_sales,
													drb.batch

		lead(drb.dt_delivery_actual) over (partition by 
													drb.delivery_number_sales,
													drb.batch

		case

							then drb.dt_delivery_actual + drb.storage_before_repacking_normative_days + drb.arrival_to_port_after_repacking_normative_days + drb.route_delivery_after_repacking_normative_days + drb.port_of_loading_storage_normative_days 
						else drb.dt_delivery_plan + drb.storage_before_repacking_normative_days + drb.arrival_to_port_after_repacking_normative_days + drb.route_delivery_after_repacking_normative_days + drb.port_of_loading_storage_normative_days

						else drb.dt_delivery_plan + drb.port_of_loading_storage_normative_days

						else drb.dt_delivery_plan + drb.normative_marine_transit1_duration_days_quantity + drb.acceptance_normative_foreign_port_days_amount
					 end
			when drb.transport_route_technical_code = 'ROUTE070'
				then case
						when drb.dt_delivery_actual is not null then drb.dt_delivery_actual + drb.foreign_port_storage_normative_days + drb.foreign_port_vessel_loading_normative_days + drb.foreign_port_storage_before_vessel_loading_norm_days
						else drb.dt_delivery_plan + drb.foreign_port_storage_normative_days + drb.foreign_port_vessel_loading_normative_days + drb.foreign_port_storage_before_vessel_loading_norm_days
					 end
			when drb.transport_route_technical_code = 'ROUTE080'
				then case
						when drb.dt_delivery_actual is not null then drb.dt_delivery_actual + drb.foreign_port_vessel_loading_normative_days
						else drb.dt_delivery_plan + drb.foreign_port_vessel_loading_normative_days

						when drb.dt_delivery_actual is not null then drb.dt_delivery_actual + drb.normative_marine_transit2_duration_days_quantity + drb.acceptance_normative_second_foreign_port_days_amount
						else drb.dt_delivery_plan + drb.normative_marine_transit2_duration_days_quantity + drb.acceptance_normative_second_foreign_port_days_amount
					 end
			when drb.transport_route_technical_code = 'ROUTE110'
				then case
						when drb.dt_delivery_actual is not null then drb.dt_delivery_actual + drb.second_foreign_port_storage_normative_days
						else drb.dt_delivery_plan + drb.second_foreign_port_storage_normative_days
					 end
			when drb.transport_route_technical_code = 'ROUTE120'
				then case
						when drb.dt_delivery_actual is not null then drb.dt_delivery_actual + drb.foreign_port_delivery_to_customer_normative_days
						else drb.dt_delivery_plan + drb.foreign_port_delivery_to_customer_normative_days

			else null
			end as dt_delivery_estimated,
		sde.business_location_code as business_location_at_report_date_code,								-- SD.001272 ID Статуса Среза
		bl.transportation_status_code as business_location_for_cargo_reporting_code,						-- SD.001661 ID статуса - отчётность по грузоперевозкам
		tstext.transportation_status_name as business_location_for_cargo_reporting_name,					-- SD.001662 Имя статуса - отчётность по грузоперевозкам
		ts.transportation_status_group_code as business_location_group_for_cargo_reporting_name,			-- SD.001663 Группа статуса - отчётность по грузоперевозкам
		coalesce(dep_en.transport_hub_name, dep_ru.transport_hub_name) as transport_departure_hub_code, 	-- SD.001664 Пункт отправления
		coalesce(dest_en.transport_hub_name, dest_ru.transport_hub_name) as transport_destination_hub_code, -- SD.001665 Пункт назначения
		drb.dt_realization_basis_document,																	-- SD.001679 Дата основания для реализации
		drb.destination_warehouse_code,																		-- SD.001680 Склад грузополучателя по маршруту завода (код)
		drb.destination_transport_hub_code,																	-- SD.001681 Конечный пункт назначения в маршруте завода (код)
		drb.transport_type_name,																			-- SD.001668 Тип ТС в дислокации
		drb.route_plant_code,  																				-- SD.000118 Маршрут завода
		drb.border_cross_to_final_station_normative_days,  													-- SD.001683 Норматив Пограничный переход-Конечная станция
		drb.normative_route_trip_duration_days_quantity,  													-- SD.000775 Норма доставки по маршруту завода
		drb.normative_railway_trip_duration_days_quantity,  												-- SD.000774 Норма движения по жд (дни)
		drb.storage_before_repacking_normative_days,  														-- SD.001684 Норма хранения до перетарки
		drb.arrival_to_port_after_repacking_normative_days,  												-- SD.001685 Норма прибытия в порт после перетарки
		drb.route_delivery_after_repacking_normative_days,  												-- SD.001686 Норма доставки по маршруту перетарки
		drb.acceptance_normative_russian_port_days_amount,  												-- SD.000708 Норма приемки контейнера в порту
		drb.port_of_loading_storage_normative_days,  														-- SD.001687 Норма хранения в порту погрузки
		drb.normative_marine_transit1_duration_days_quantity,  												-- SD.000776 Норма морского транзита
		drb.acceptance_normative_foreign_port_days_amount,  												-- SD.000709 Норма приемки терминал/склад ин. порта

		drb.acceptance_normative_second_foreign_port_days_amount,  											-- SD.000710 Норма приемки терминал/склад ин. порта 2

		drb.normative_marine_transit2_duration_days_quantity,  												-- SD.000777 Норма морского транзита 2
		drb.foreign_port_delivery_to_customer_normative_days,  												-- SD.001691 Норма доставки по маршруту со склада ин. порта
		drb.foreign_port_storage_before_vessel_loading_norm_days,											-- SD.001780 Норма хранения до погрузки на судно в ин.порту

					when ildr.logistics_document_type_code = 'Д' then drb.delivery_notice_number
					when drb.bill_of_lading_in_foreign_port is not null then drb.bill_of_lading_in_foreign_port

					else drb.transport_bill
				end
			when drb.transport_route_technical_code in ('ROUTE120', 'ROUTE110', 'ROUTE100', 'ROUTE090') then
				case
					when drb.bill_of_lading_in_foreign_port is not null then drb.bill_of_lading_in_foreign_port
					when drb.bill_of_lading_number is not null then drb.bill_of_lading_number
					else drb.transport_bill
				end
			when drb.transport_route_technical_code in ('ROUTE080', 'ROUTE070', 'ROUTE060', 'ROUTE050') then
				case
					when drb.bill_of_lading_number is not null then drb.bill_of_lading_number
					else drb.transport_bill
				end			
			when drb.transport_route_technical_code in ('ROUTE040', 'ROUTE030', 'ROUTE020', 'ROUTE010') then drb.transport_bill
		end as cargo_document_code,
		-- расчет кумулятивных нормативов относительно текущего маршрута
		case 
			when drb.transport_route_technical_code <> 'ROUTE130' then
				( 
				-- сумма SD.001683 Норматив Пограничный переход-Конечная станция, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.border_cross_to_final_station_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000775 Норма доставки по маршруту завода, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.normative_route_trip_duration_days_quantity) over ( partition by

																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000774 Норма движения по жд (дни), начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.normative_railway_trip_duration_days_quantity) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001684 Норма хранения до перетарки, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.storage_before_repacking_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001685 Норма прибытия в порт после перетарки, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.arrival_to_port_after_repacking_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001686 Норма доставки по маршруту перетарки, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.route_delivery_after_repacking_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000708 Норма приемки контейнера в порту, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.acceptance_normative_russian_port_days_amount) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001687 Норма хранения в порту погрузки, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.port_of_loading_storage_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000776 Норма морского транзита, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.normative_marine_transit1_duration_days_quantity) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000709 Норма приемки терминал/склад ин. порта, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.acceptance_normative_foreign_port_days_amount) over ( partition by

				-- сумма SD.000710 Норма приемки терминал/склад ин. порта 2, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.acceptance_normative_second_foreign_port_days_amount) over ( partition by

																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )

				-- сумма SD.001689 Норма хранения в ин.порту 2, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.second_foreign_port_storage_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001690 Норма погрузки на судно в ин. порту, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.foreign_port_vessel_loading_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.000777 Норма морского транзита 2, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.normative_marine_transit2_duration_days_quantity) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001691 Норма доставки по маршруту со склада ин. порта, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.foreign_port_delivery_to_customer_normative_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
				+
				-- сумма SD.001780 Норма хранения до погрузки на судно в ин.порту, начиная с последующего (относительно текущего) маршрута в рамках поставки и партии
				sum(drb.foreign_port_storage_before_vessel_loading_norm_days) over ( partition by 
																			 		drb.delivery_number_sales,
																			 		drb.batch
																		    order by drb.transport_route_technical_code
																		    rows between 1 following and unbounded following
																		  )
			)
			else 0
		end as cumulative_normatives

	left join dict_dds.transport_hub_texts as dep_en

	    and sde.sales_delivery_code = drb.delivery_number_sales

		and tstext.language_code = 'R'

	left join dict_dds.incoterms_and_logistics_document_relation as ildr
		on ildr.incoterms_code = drb.delivery_basis
		and ildr.ownership_transfer_code = '001'
