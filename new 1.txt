-- Блок вместо твоего dfdf

WITH base_eu AS (
    -- Базовый набор: только нужная логика + все джойны
    SELECT
        b.rn,
        b.warehouse_code,
        b.storage_cost_special_calculation_type_code,
        b.dt_storage_start,
        b.dt_storage_end,
        b.storage_duration_total_calendar_days,
        b.sales_bundle_gross_weight,
        b.unit_balance_code,
        b.forwarder_code,
        b.plant_code,
        b.port_of_discharge_code,
        b.material_shape_code,
        b.etsng_code,
        b.transport_type_code,
        b.dt_for_price_search_in_purchase_contract,

        f.purchase_contract_code,
        f.position_line_item_code,
        f.konws,
        f.service_code,
        f.price_uom_code,
        f.netpr,
        f.currency_dogovor,
        f.datab,
        f.datbi,
        f.order_euro_svh,
        f.kopos,
        f.kbetr,

        param_sap.range_low_value   AS param_sap_low_value,
        param_srv_1.range_low_value AS param_srv_low_value
    FROM dm_calc.storage_sales_bundles_weight AS b
    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_1
           ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
          AND for_1082_1.case_code        = 'PERET_FRN'
          AND for_1082_1.parameter_code   = 'S_LIFNRA'
          AND for_1082_1.range_low_value  = b.forwarder_code
    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_2
           ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
          AND for_1082_2.parameter_code    = 'S_FORMSTEU'
          AND for_1082_2.range_low_value   = b.material_shape_code
    JOIN for_1079_base AS f
      ON b.storage_cost_calculation_type_name = 'Логика Европа'
     AND f.purchase_document_subtype_code = 'ZDTL'
     AND (
            (for_1082_1.range_low_value IS NOT NULL AND f.zzsrvpos = 'S49')
         OR (for_1082_1.range_low_value IS NULL
             AND for_1082_2.range_low_value IS NOT NULL
             AND f.tmp_1 IS NOT NULL)
         OR (for_1082_1.range_low_value IS NULL
             AND for_1082_2.range_low_value IS NULL
             AND f.tmp_2 IS NOT NULL)
         )
     AND f.bukrs = b.unit_balance_code
     AND f.lifnr = b.forwarder_code
     AND f.werks = b.plant_code
     AND (f.zzport      = b.port_of_discharge_code OR f.zzport IS NULL)
     AND (f.zzforma     = b.material_shape_code     OR f.zzforma IS NULL)
     AND (f.zzet_tarif  = b.etsng_code             OR f.zzet_tarif IS NULL)
     AND (f.sdabw       = b.transport_type_code     OR f.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract
           BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract
           BETWEEN f.datab        AND f.datbi
     AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_3
           ON f.tmp_2 IS NOT NULL
          AND for_1082_3.range_low_value LIKE 'EU%'
          AND for_1082_3.abap_program_code = '/RUSAL/SD4553M'
          AND for_1082_3.parameter_code    = 'S_USLST'
    LEFT JOIN dict_dds.settings_and_parameters_sap AS param_sap
           ON param_sap.range_low_value   = b.material_shape_code
          AND param_sap.abap_program_code = '/RUSAL/SD4553M'
          AND param_sap.parameter_code    = 'FORMSTEU'
    LEFT JOIN dict_dds.settings_and_parameters_sap AS param_srv_1
           ON param_srv_1.range_low_value   = for_1082_3.range_low_value
          AND param_srv_1.abap_program_code = '/RUSAL/SD4553M'
          AND param_srv_1.parameter_code    = 'S_USLSTC'
    WHERE b.storage_cost_calculation_type_name = 'Логика Европа'
),

ranked AS (
    -- Считаем row_number по rn и сразу помечаем лучшую строку
    SELECT
        base_eu.*,
        ROW_NUMBER() OVER (
            PARTITION BY base_eu.rn
            ORDER BY
                base_eu.order_euro_svh,
                base_eu.kopos,
                base_eu.kbetr,
                CASE
                    WHEN base_eu.param_sap_low_value IS NOT NULL
                         AND base_eu.param_srv_low_value IS NOT NULL THEN 1
                    WHEN base_eu.param_sap_low_value IS NULL
                         AND base_eu.param_srv_low_value IS NULL THEN 1
                    ELSE 0
                END DESC,
                base_eu.purchase_contract_code,
                base_eu.position_line_item_code
        ) AS row_num
    FROM base_eu
),

wh_month AS (
    -- Предагрегация веса за месяц по складу
    SELECT
        warehouse_code,
        date_trunc('month', dt_storage_end)::date AS month_key,
        SUM(sales_bundle_gross_weight) AS warehouse_weight_month_end
    FROM dm_calc.storage_sales_bundles_weight
    WHERE storage_cost_calculation_type_name = 'Логика Европа'
    GROUP BY
        warehouse_code,
        date_trunc('month', dt_storage_end)::date
)

SELECT
    r.rn,
    r.purchase_contract_code,
    r.position_line_item_code,
    r.konws,
    r.service_code,
    r.price_uom_code,
    w.warehouse_weight_month_end,
    CASE
        WHEN EXTRACT(DAY FROM r.dt_storage_end)
             < EXTRACT(DAY FROM r.dt_storage_start)
            THEN (EXTRACT(YEAR  FROM r.dt_storage_end) -
                  EXTRACT(YEAR  FROM r.dt_storage_start)) * 12 +
                 (EXTRACT(MONTH FROM r.dt_storage_end) -
                  EXTRACT(MONTH FROM r.dt_storage_start))
        ELSE (EXTRACT(YEAR  FROM r.dt_storage_end) -
              EXTRACT(YEAR  FROM r.dt_storage_start)) * 12 +
             (EXTRACT(MONTH FROM r.dt_storage_end) -
              EXTRACT(MONTH FROM r.dt_storage_start)) + 1
    END AS month_difference_2,
    r.netpr,
    CASE
        WHEN r.storage_cost_special_calculation_type_code = '4'
            THEN ceil(r.storage_duration_total_calendar_days / 7::numeric)
        WHEN r.storage_cost_special_calculation_type_code = '5'
            THEN ceil(r.storage_duration_total_calendar_days / 30::numeric)
    END AS special,
    r.currency_dogovor,
    r.datab,
    r.datbi
FROM ranked r
JOIN wh_month w
  ON r.warehouse_code = w.warehouse_code
 AND date_trunc('month', r.dt_storage_end)::date = w.month_key
WHERE r.row_num = 1;
