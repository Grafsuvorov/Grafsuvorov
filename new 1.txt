sql
Копировать код
SELECT 'warehouse' AS hub_type, COUNT(*)
FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN tmp_hub_ext hub
    ON hub.transport_hub_code = base.warehouse_code

UNION ALL
SELECT 'port', COUNT(*)
FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN tmp_hub_ext hub
    ON hub.transport_hub_code = base.port_of_discharge_code

UNION ALL
SELECT 'departure', COUNT(*)
FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN tmp_hub_ext hub
    ON hub.transport_hub_code = base.transport_departure_hub_code

UNION ALL
SELECT 'destination', COUNT(*)
FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN tmp_hub_ext hub
    ON hub.transport_hub_code = base.transport_destination_hub_code

UNION ALL
