import psycopg2
import gzip
import csv
import time
from lxml import etree

# ---------------------------
# параметры подключения
# ---------------------------

conn = psycopg2.connect(
    host="HOST",
    dbname="DB",
    user="USER",
    password="PASSWORD",
    port=5432
)

# ---------------------------
# узнаем сколько строк всего
# ---------------------------

cur = conn.cursor()
cur.execute("""
select count(*)
from stg.RAL_IS2_TORO2_ORD_HDR_OPR_items_attr
""")

total_rows = cur.fetchone()[0]

print(f"Всего XML строк: {total_rows}")

# ---------------------------
# курсор потоковый
# ---------------------------

cur = conn.cursor(name="xml_cursor")

cur.itersize = 5000

cur.execute("""
select uuid, xml_item
from stg.RAL_IS2_TORO2_ORD_HDR_OPR_items_attr
""")

# ---------------------------
# файл
# ---------------------------

outfile = "parsed_init.csv.gz"

f = gzip.open(outfile, "wt", newline="", encoding="utf8")
writer = csv.writer(f)

writer.writerow([
    "uuid",
    "VORNR",
    "KTSCH",
    "ARBPL",
    "WERKS",
    "NTANF_DT"
])

# ---------------------------
# обработка
# ---------------------------

start_time = time.time()

processed = 0
written_rows = 0

for uuid, xml in cur:

    processed += 1

    root = etree.fromstring(xml)

    for opr in root.xpath("//ORDER_OPR"):

        writer.writerow([
            uuid,
            opr.findtext("VORNR"),
            opr.findtext("KTSCH"),
            opr.findtext("ARBPL"),
            opr.findtext("WERKS"),
            opr.findtext("NTANF_DT")
        ])

        written_rows += 1

    # каждые 5000 строк выводим прогресс
    if processed % 5000 == 0:

        elapsed = time.time() - start_time
        speed = processed / elapsed

        remaining = total_rows - processed
        eta = remaining / speed if speed > 0 else 0

        percent = processed / total_rows * 100

        print(
            f"""
-----------------------------------------
Обработано XML: {processed}/{total_rows} ({percent:.2f}%)
Скорость: {speed:.1f} строк/сек
Записано строк: {written_rows}
Осталось: {remaining}
ETA: {eta/60:.1f} минут
-----------------------------------------
"""
        )

f.close()

total_time = time.time() - start_time

print("\nINIT завершен")
print(f"Всего XML: {processed}")
print(f"Всего записано строк: {written_rows}")
print(f"Файл: {outfile}")
print(f"Время: {total_time/60:.2f} минут")
