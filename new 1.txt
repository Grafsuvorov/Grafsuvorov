drop table if exists tmp_dates;

create temp table tmp_dates as
select generate_series(
    date '2026-01-01',
    current_date,
    interval '1 day'
)::date as dt_report
distributed randomly;

analyze tmp_dates;


-- =========================================================
-- 1. Заводы
-- =========================================================

drop table if exists tmp_plant_id;

create temp table tmp_plant_id as
select
    pas.asuzdc_plant_code,
    pas.plant_code,
    pas.plant_short_name,
    case
        when pas.plant_code = '5201' then 1
        when pas.plant_code = '5101' then 2
        when pas.plant_code = '5401' then 3
        when pas.plant_code = '5301' then 4
        when pas.plant_code = '6100' then 5
        when pas.plant_code = '5601' then 6
        when pas.plant_code = '6300' then 7
        when pas.plant_code = '6800' then 8
        when pas.plant_code = '5701' then 9
        else 100
    end as subsequent_code
from dict_dds.plant_and_subsidiary pas
join dict_dds.plant_and_subsidiary_texts past
    on past.plant_code = pas.plant_code
   and past.language_code = 'R'
where pas.asuzdc_plant_code is not null
  and pas.deleted_flag = false
  and past.deleted_flag = false
distributed by (plant_code);

analyze tmp_plant_id;


-- =========================================================
-- 2. Даты вагонов с учетом timezone
-- =========================================================

drop table if exists tmp_dwz;

create temp table tmp_dwz as
select
    rdc.rw_part,
    rdc.kpr_home,

    date(((rdc.dt_in_plant at time zone 'UTC') at time zone past.time_zone_iana_code) - interval '3 hours') as dt_in_plant,
    date(((rdc.dt_out_plant at time zone 'UTC') at time zone past.time_zone_iana_code) - interval '3 hours') as dt_out_plant,
    date(((rdc.dt_in_gdc at time zone 'UTC') at time zone past.time_zone_iana_code) - interval '3 hours') as dt_in_gdc,
    date(((rdc.dt_out_gdc at time zone 'UTC') at time zone past.time_zone_iana_code) - interval '3 hours') as dt_out_gdc,
    date(((rdc.dt_end_unload at time zone 'UTC') at time zone past.time_zone_iana_code) - interval '3 hours') as dt_end_unload

from ods.asuzdc_rwcar_date_control rdc
join dict_dds.plant_and_subsidiary_td past
    on past.asuzdc_plant_code =
        case
            when rdc.kpr_home = '15' then '2'
            else rdc.kpr_home
        end
where past.asuzdc_plant_code is not null
  and (
        rdc.kpr_home <> '15'
        or (rdc.kpr_home = '15' and rdc.dt_in_plant is null)
      )
distributed by (rw_part);

analyze tmp_dwz;


-- =========================================================
-- 3. Базовая таблица вагонов
-- =========================================================

drop table if exists tmp_cars_base;

create temp table tmp_cars_base as
select
    dwz.rw_part,

    coalesce(docs.id_mdm_p, docs1.id_mdm_p) as id_mdm,

    rwp.car_type,

    case
        when coalesce(docs.rw_st_p, docs1.rw_st_p) = '924906'
            then 1
        else 0
    end as pr_avto,

    case
        when cert.railcar_type_asuzdc_code in ('23', '24', '25', '94', '9 ')
             and ugm.cargo_group_asuzdc_code = '1' then '3'
        when coalesce(dc1.cargo_places, dc2.cargo_places) = 5
             and ugm.cargo_group_asuzdc_code = '1' then '1'
        when coalesce(dc1.cargo_places, dc2.cargo_places) = 1
             and ugm.cargo_group_asuzdc_code = '1' then '2'
        when coalesce(dc1.cargo_places, dc2.cargo_places) > 5
             and ugm.cargo_group_asuzdc_code = '1' then '4'
        when coalesce(dc1.cargo_places, dc2.cargo_places) = 0
             and cert.railcar_type_asuzdc_code in ('6 ', '68')
             and ugm.cargo_group_asuzdc_code = '1' then '2'
        when ugm.cargo_group_asuzdc_code in ('2', '3', '4') then '0'
        else '6'
    end as pack_id,

    dwz.dt_in_plant,
    dwz.dt_out_plant,
    dwz.dt_in_gdc,
    dwz.dt_out_gdc,
    dwz.dt_end_unload,

    coalesce(dc1.cargo_weight, dc2.cargo_weight) as cargo_weight

from tmp_dwz dwz

left join ods.asuzdc_rwcar_parts rwp
    on dwz.rw_part = rwp.rw_part
   and rwp.car_type in ('2', '3')

left join ods.asuzdc_docs_cars dc0
    on dc0.rw_part = rwp.rw_part
   and dc0.doc_type = '1'

left join ods.asuzdc_docs_cars dc1
    on dc1.rw_part = rwp.rw_part
   and dc1.doc_type = '2'

left join ods.asuzdc_docs docs
    on docs.rn_doc::varchar = dc1.rn_doc

left join ods.asuzdc_docs_cars dc2
    on dc2.rw_part = dwz.rw_part
   and dc2.doc_type = '10'

left join ods.asuzdc_docs docs1
    on docs1.rn_doc::varchar = dc2.rn_doc

left join dict_dds.railcar_certificate cert
    on cert.railcar_code = rwp.car_num

left join dict_dds.etsng ugm
    on coalesce(coalesce(dc1.rw_kg, dc2.rw_kg), dc0.rw_kg) = ugm.etsng_code

where ugm.cargo_group_asuzdc_code = '1'
  and coalesce(docs.id_mdm_p, docs1.id_mdm_p) is not null
distributed by (id_mdm);

analyze tmp_cars_base;


-- =========================================================
-- 4. Дневные события: приход на завод и разгрузка
-- =========================================================

drop table if exists tmp_day_fact;

create temp table tmp_day_fact as
select
    dt_report,
    id_mdm,
    car_type,
    pr_avto,
    pack_id,

    sum(railcar_arrived_to_plant_actual_quantity) as railcar_arrived_to_plant_actual_quantity,
    sum(railcar_arrived_to_plant_actual_weight) as railcar_arrived_to_plant_actual_weight,

    sum(railcar_at_plant_unloaded_actual_quantity) as railcar_at_plant_unloaded_actual_quantity,
    sum(railcar_at_plant_unloaded_actual_weight) as railcar_at_plant_unloaded_actual_weight

from (
    select
        dt_in_plant as dt_report,
        id_mdm,
        car_type,
        pr_avto,
        pack_id,
        1 as railcar_arrived_to_plant_actual_quantity,
        cargo_weight / 1000 as railcar_arrived_to_plant_actual_weight,
        0 as railcar_at_plant_unloaded_actual_quantity,
        0::numeric as railcar_at_plant_unloaded_actual_weight
    from tmp_cars_base
    where dt_in_plant is not null

    union all

    select
        dt_end_unload as dt_report,
        id_mdm,
        car_type,
        pr_avto,
        pack_id,
        0,
        0::numeric,
        1,
        cargo_weight / 1000
    from tmp_cars_base
    where dt_end_unload is not null
) x
group by
    dt_report,
    id_mdm,
    car_type,
    pr_avto,
    pack_id
distributed by (id_mdm);

analyze tmp_day_fact;


-- =========================================================
-- 5. Остатки / наличие на заводе и станции
-- =========================================================

drop table if exists tmp_balance_day;

create temp table tmp_balance_day as
select
    d.dt_report,
    c.id_mdm,
    c.car_type,
    c.pr_avto,
    c.pack_id,

    sum(
        case
            when c.dt_in_plant is not null
             and c.dt_in_plant <= d.dt_report - 1
             and (c.dt_out_plant is null or c.dt_out_plant > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then 1 else 0
        end
    ) as railcar_at_plant_actual_quantity,

    sum(
        case
            when c.dt_in_gdc is not null
             and c.dt_in_gdc <= d.dt_report - 1
             and (c.dt_out_gdc is null or c.dt_out_gdc > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then 1 else 0
        end
    )
    -
    sum(
        case
            when c.dt_in_plant is not null
             and c.dt_in_plant <= d.dt_report - 1
             and (c.dt_out_plant is null or c.dt_out_plant > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then 1 else 0
        end
    ) as railcar_at_station_actual_quantity,

    sum(
        case
            when c.dt_in_plant is not null
             and c.dt_in_plant <= d.dt_report - 1
             and (c.dt_out_plant is null or c.dt_out_plant > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then c.cargo_weight / 1000 else 0
        end
    ) as railcar_at_plant_actual_weight,

    sum(
        case
            when c.dt_in_gdc is not null
             and c.dt_in_gdc <= d.dt_report - 1
             and (c.dt_out_gdc is null or c.dt_out_gdc > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then c.cargo_weight / 1000 else 0
        end
    )
    -
    sum(
        case
            when c.dt_in_plant is not null
             and c.dt_in_plant <= d.dt_report - 1
             and (c.dt_out_plant is null or c.dt_out_plant > d.dt_report - 1)
             and (c.dt_end_unload is null or c.dt_end_unload > d.dt_report - 1)
            then c.cargo_weight / 1000 else 0
        end
    ) as railcar_at_station_actual_weight

from tmp_dates d
join tmp_cars_base c
    on d.dt_report between least(
                            coalesce(c.dt_in_gdc, c.dt_in_plant, c.dt_end_unload),
                            coalesce(c.dt_in_plant, c.dt_in_gdc, c.dt_end_unload)
                         )
                       and coalesce(c.dt_out_plant, c.dt_out_gdc, c.dt_end_unload, current_date)
group by
    d.dt_report,
    c.id_mdm,
    c.car_type,
    c.pr_avto,
    c.pack_id
distributed by (id_mdm);

analyze tmp_balance_day;


-- =========================================================
-- 6. Объединение факта
-- =========================================================

drop table if exists tmp_fact_day;

create temp table tmp_fact_day as
select
    coalesce(b.dt_report, f.dt_report) as dt_report,
    coalesce(b.id_mdm, f.id_mdm) as id_mdm,
    coalesce(b.car_type, f.car_type) as car_type,
    coalesce(b.pr_avto, f.pr_avto) as pr_avto,
    coalesce(b.pack_id, f.pack_id) as pack_id,

    coalesce(b.railcar_at_plant_actual_quantity, 0) as railcar_at_plant_actual_quantity,
    coalesce(b.railcar_at_station_actual_quantity, 0) as railcar_at_station_actual_quantity,
    coalesce(b.railcar_at_plant_actual_weight, 0) as railcar_at_plant_actual_weight,
    coalesce(b.railcar_at_station_actual_weight, 0) as railcar_at_station_actual_weight,

    coalesce(f.railcar_arrived_to_plant_actual_quantity, 0) as railcar_arrived_to_plant_actual_quantity,
    coalesce(f.railcar_arrived_to_plant_actual_weight, 0) as railcar_arrived_to_plant_actual_weight,
    coalesce(f.railcar_at_plant_unloaded_actual_quantity, 0) as railcar_at_plant_unloaded_actual_quantity,
    coalesce(f.railcar_at_plant_unloaded_actual_weight, 0) as railcar_at_plant_unloaded_actual_weight

from tmp_balance_day b
full join tmp_day_fact f
    on b.dt_report = f.dt_report
   and b.id_mdm = f.id_mdm
   and b.car_type = f.car_type
   and b.pr_avto = f.pr_avto
   and b.pack_id = f.pack_id
distributed by (id_mdm);

analyze tmp_fact_day;

        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('month', dt_report)

        rows between unbounded preceding and current row

    sum(railcar_arrived_to_plant_actual_weight) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('month', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_arrived_to_plant_actual_mtd_weight,

    sum(railcar_at_plant_unloaded_actual_weight) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('month', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_at_plant_unloaded_actual_mtd_weight,

    sum(railcar_arrived_to_plant_actual_quantity) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('year', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_arrived_to_plant_actual_ytd_quantity,

    sum(railcar_at_plant_unloaded_actual_quantity) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('year', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_at_plant_unloaded_actual_ytd_quantity,

    sum(railcar_arrived_to_plant_actual_weight) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('year', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_arrived_to_plant_actual_ytd_weight,

    sum(railcar_at_plant_unloaded_actual_weight) over (
        partition by id_mdm, car_type, pr_avto, pack_id, date_trunc('year', dt_report)
        order by dt_report
        rows between unbounded preceding and current row
    ) as railcar_at_plant_unloaded_actual_ytd_weight

from tmp_fact_day f
distributed by (id_mdm);

analyze tmp_fact_cum;


-- =========================================================
-- 8. Нормативы на день
-- =========================================================

drop table if exists tmp_norm_day;

create temp table tmp_norm_day as
select
    d.dt_report,
    aurp.consignee_mdm_code as id_mdm,
    aurp.railcar_type_asuzdc_code as car_type,
    aurp.package_type_asuzdc_code as pack_id,
    aurp.is_unloaded_outside_plant_code as pr_avto,
    past.asuzdc_plant_code as kpr_home,
    aurp.plant_code,
    aurp.package_type_sap_code,

    coalesce(aurp.unload_daily_normative_quantity, 0) as railcar_at_plant_unload_normative_quantity,
    coalesce(aurp.unload_daily_normative_weight, 0) as railcar_at_plant_unload_normative_weight

from tmp_dates d
join dict_dds.raw_material_unload_at_plant_normative_td aurp
    on aurp.dt_valid_from < d.dt_report
   and (aurp.dt_valid_to >= d.dt_report or aurp.dt_valid_to is null)
   and aurp.cargo_group_asuzdc_code = '1'
left join dict_dds.plant_and_subsidiary_td past
    on past.plant_code = aurp.plant_code
distributed by (id_mdm);

analyze tmp_norm_day;

    sum(q.railcar_at_station_actual_weight) as railcar_at_station_actual_weight,
    sum(q.railcar_total_actual_weight) as railcar_total_actual_weight,
    sum(q.railcar_arrived_to_plant_actual_weight) as railcar_arrived_to_plant_actual_weight,
    sum(q.railcar_at_plant_ready_for_unload_actual_weight) as railcar_at_plant_ready_for_unload_actual_weight,
    sum(q.railcar_at_plant_unload_normative_weight) as railcar_at_plant_unload_normative_weight,
    sum(q.railcar_at_plant_unloaded_actual_weight) as railcar_at_plant_unloaded_actual_weight,
    sum(q.railcar_at_plant_unload_normative_vs_actual_weight) as railcar_at_plant_unload_normative_vs_actual_weight,
    sum(q.railcar_at_plant_left_after_unload_actual_weight) as railcar_at_plant_left_after_unload_actual_weight,

    sum(q.railcar_arrived_to_plant_actual_mtd_quantity) as railcar_arrived_to_plant_actual_mtd_quantity,
    sum(q.railcar_at_plant_unload_normative_mtd_quantity) as railcar_at_plant_unload_normative_mtd_quantity,
    sum(q.railcar_at_plant_unloaded_actual_mtd_quantity) as railcar_at_plant_unloaded_actual_mtd_quantity,
    sum(q.railcar_at_plant_unload_normative_vs_actual_mtd_quantity) as railcar_at_plant_unload_normative_vs_actual_mtd_quantity,
    sum(q.railcar_arrived_to_plant_actual_mtd_weight) as railcar_arrived_to_plant_actual_mtd_weight,
    sum(q.railcar_at_plant_unload_normative_mtd_weight) as railcar_at_plant_unload_normative_mtd_weight,
    sum(q.railcar_at_plant_unloaded_actual_mtd_weight) as railcar_at_plant_unloaded_actual_mtd_weight,
    sum(q.railcar_at_plant_unload_normative_vs_actual_mtd_weight) as railcar_at_plant_unload_normative_vs_actual_mtd_weight,

    sum(q.railcar_arrived_to_plant_actual_ytd_quantity) as railcar_arrived_to_plant_actual_ytd_quantity,
    sum(q.railcar_at_plant_unload_normative_ytd_quantity) as railcar_at_plant_unload_normative_ytd_quantity,
    sum(q.railcar_at_plant_unloaded_actual_ytd_quantity) as railcar_at_plant_unloaded_actual_ytd_quantity,
    sum(q.railcar_at_plant_unload_normative_vs_actual_ytd_quantity) as railcar_at_plant_unload_normative_vs_actual_ytd_quantity,
    sum(q.railcar_arrived_to_plant_actual_ytd_weight) as railcar_arrived_to_plant_actual_ytd_weight,
    sum(q.railcar_at_plant_unload_normative_ytd_weight) as railcar_at_plant_unload_normative_ytd_weight,
    sum(q.railcar_at_plant_unloaded_actual_ytd_weight) as railcar_at_plant_unloaded_actual_ytd_weight,
    sum(q.railcar_at_plant_unload_normative_vs_actual_ytd_weight) as railcar_at_plant_unload_normative_vs_actual_ytd_weight

from (
    select
        k.dt_report,

        p.asuzdc_plant_code as plant_code,
        p.plant_short_name as plant_name,
        p.asuzdc_plant_code || '-' || p.plant_short_name as plant_search_name,
        p.subsequent_code,

        k.pack_id as package_type_code,
        ptt.package_type_name,
        k.pack_id || '-' || ptt.package_type_name as package_type_search,

        k.car_type as transport_type_code,
        case
            when k.car_type = '2' then 'Вагон'
            when k.car_type = '3' then 'Контейнер'
        end as transport_type_name,
        k.car_type || '-' ||
        case
            when k.car_type = '2' then 'Вагон'
            when k.car_type = '3' then 'Контейнер'
        end as transport_type_search,

        coalesce(f.railcar_at_plant_actual_quantity, 0) as railcar_at_plant_actual_quantity,
        coalesce(f.railcar_at_station_actual_quantity, 0) as railcar_at_station_actual_quantity,
        coalesce(f.railcar_at_plant_actual_quantity, 0)
            + coalesce(f.railcar_at_station_actual_quantity, 0) as railcar_total_actual_quantity,

        coalesce(f.railcar_arrived_to_plant_actual_quantity, 0) as railcar_arrived_to_plant_actual_quantity,

        coalesce(f.railcar_at_plant_actual_quantity, 0)
            + coalesce(f.railcar_arrived_to_plant_actual_quantity, 0)
            + case
                when k.pr_avto = 1 then coalesce(f.railcar_at_station_actual_quantity, 0)
                else 0
              end as railcar_at_plant_ready_for_unload_actual_quantity,

        coalesce(n.railcar_at_plant_unload_normative_quantity, 0) as railcar_at_plant_unload_normative_quantity,

        coalesce(f.railcar_at_plant_unloaded_actual_quantity, 0) as railcar_at_plant_unloaded_actual_quantity,

        coalesce(f.railcar_at_plant_unloaded_actual_quantity, 0)
            -
        (
            coalesce(f.railcar_at_plant_actual_quantity, 0)
            + coalesce(f.railcar_arrived_to_plant_actual_quantity, 0)
            + case
                when k.pr_avto = 1 then coalesce(f.railcar_at_station_actual_quantity, 0)
                else 0
              end
        ) as railcar_at_plant_left_after_unload_actual_quantity,

        coalesce(f.railcar_at_plant_unloaded_actual_quantity, 0)
            - coalesce(n.railcar_at_plant_unload_normative_quantity, 0)
            as railcar_at_plant_unload_normative_vs_actual_quantity,

        ur.reason_text as comment,

        coalesce(f.railcar_at_plant_actual_weight, 0) as railcar_at_plant_actual_weight,
        coalesce(f.railcar_at_station_actual_weight, 0) as railcar_at_station_actual_weight,
        coalesce(f.railcar_at_plant_actual_weight, 0)
            + coalesce(f.railcar_at_station_actual_weight, 0) as railcar_total_actual_weight,

        coalesce(f.railcar_arrived_to_plant_actual_weight, 0) as railcar_arrived_to_plant_actual_weight,

        coalesce(f.railcar_at_plant_actual_weight, 0)
            + coalesce(f.railcar_arrived_to_plant_actual_weight, 0)
            + case
                when k.pr_avto = 1 then coalesce(f.railcar_at_station_actual_weight, 0)
                else 0
              end as railcar_at_plant_ready_for_unload_actual_weight,

        coalesce(n.railcar_at_plant_unload_normative_weight, 0) as railcar_at_plant_unload_normative_weight,

        coalesce(f.railcar_at_plant_unloaded_actual_weight, 0) as railcar_at_plant_unloaded_actual_weight,

        coalesce(f.railcar_at_plant_unloaded_actual_weight, 0)
            - coalesce(n.railcar_at_plant_unload_normative_weight, 0)
            as railcar_at_plant_unload_normative_vs_actual_weight,

        coalesce(f.railcar_at_plant_unloaded_actual_weight, 0)
            -
        (
            coalesce(f.railcar_at_plant_actual_weight, 0)
            + coalesce(f.railcar_arrived_to_plant_actual_weight, 0)
            + case
                when k.pr_avto = 1 then coalesce(f.railcar_at_station_actual_weight, 0)
                else 0
              end
        ) as railcar_at_plant_left_after_unload_actual_weight,

        coalesce(f.railcar_arrived_to_plant_actual_mtd_quantity, 0) as railcar_arrived_to_plant_actual_mtd_quantity,
        coalesce(n.railcar_at_plant_unload_normative_mtd_quantity, 0) as railcar_at_plant_unload_normative_mtd_quantity,
        coalesce(f.railcar_at_plant_unloaded_actual_mtd_quantity, 0) as railcar_at_plant_unloaded_actual_mtd_quantity,

        coalesce(f.railcar_at_plant_unloaded_actual_mtd_quantity, 0)
            - coalesce(n.railcar_at_plant_unload_normative_mtd_quantity, 0)
            as railcar_at_plant_unload_normative_vs_actual_mtd_quantity,

        coalesce(f.railcar_arrived_to_plant_actual_mtd_weight, 0) as railcar_arrived_to_plant_actual_mtd_weight,
        coalesce(n.railcar_at_plant_unload_normative_mtd_weight, 0) as railcar_at_plant_unload_normative_mtd_weight,
        coalesce(f.railcar_at_plant_unloaded_actual_mtd_weight, 0) as railcar_at_plant_unloaded_actual_mtd_weight,

        coalesce(f.railcar_arrived_to_plant_actual_mtd_weight, 0)
            - coalesce(n.railcar_at_plant_unload_normative_mtd_weight, 0)
            as railcar_at_plant_unload_normative_vs_actual_mtd_weight,

        coalesce(f.railcar_arrived_to_plant_actual_ytd_quantity, 0) as railcar_arrived_to_plant_actual_ytd_quantity,
        coalesce(n.railcar_at_plant_unload_normative_ytd_quantity, 0) as railcar_at_plant_unload_normative_ytd_quantity,
        coalesce(f.railcar_at_plant_unloaded_actual_ytd_quantity, 0) as railcar_at_plant_unloaded_actual_ytd_quantity,

        coalesce(f.railcar_at_plant_unloaded_actual_ytd_quantity, 0)
            - coalesce(n.railcar_at_plant_unload_normative_ytd_quantity, 0)
            as railcar_at_plant_unload_normative_vs_actual_ytd_quantity,

        coalesce(f.railcar_arrived_to_plant_actual_ytd_weight, 0) as railcar_arrived_to_plant_actual_ytd_weight,
        coalesce(n.railcar_at_plant_unload_normative_ytd_weight, 0) as railcar_at_plant_unload_normative_ytd_weight,
        coalesce(f.railcar_at_plant_unloaded_actual_ytd_weight, 0) as railcar_at_plant_unloaded_actual_ytd_weight,

        coalesce(f.railcar_arrived_to_plant_actual_ytd_weight, 0)
            - coalesce(n.railcar_at_plant_unload_normative_ytd_weight, 0)
            as railcar_at_plant_unload_normative_vs_actual_ytd_weight

    from tmp_final_keys k

    left join tmp_fact_cum f
        on f.dt_report = k.dt_report
       and f.id_mdm = k.id_mdm
       and f.car_type = k.car_type
       and f.pr_avto = k.pr_avto
       and f.pack_id = k.pack_id

    left join tmp_norm_cum n
        on n.dt_report = k.dt_report
       and n.id_mdm = k.id_mdm
       and n.car_type = k.car_type
       and n.pr_avto::int = k.pr_avto
       and n.pack_id = k.pack_id

    left join tmp_plant_id p
        on p.plant_code = n.plant_code

    left join dict_dds.package_type_texts ptt
        on ptt.package_type_code = n.package_type_sap_code::varchar
       and ptt.language_code = 'R'

    left join ods.asuzdc_unload_reasons ur
        on ur.id_group = '1'
       and ur.id_mdm = k.id_mdm
       and ur.kp = '-1'
       and ur.dt = k.dt_report

    where p.plant_short_name is not null
) q

where
       (q.railcar_at_plant_actual_quantity > 0 and q.railcar_at_plant_actual_quantity is not null)
    or (q.railcar_at_station_actual_quantity > 0 and q.railcar_at_station_actual_quantity is not null)
    or (q.railcar_arrived_to_plant_actual_quantity > 0 and q.railcar_arrived_to_plant_actual_quantity is not null)
    or (q.railcar_at_plant_unload_normative_quantity > 0 and q.railcar_at_plant_unload_normative_quantity is not null)

group by
    q.dt_report,
    q.plant_code,
    q.plant_name,
    q.plant_search_name,
    q.subsequent_code,
    q.package_type_code,
    q.package_type_name,
    q.package_type_search,
    q.transport_type_code,
    q.transport_type_name,
    q.transport_type_search,
    q.comment;
