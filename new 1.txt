----------------------------------------------------------------------
-- DM: storage_sales_bundles_amount_enriched
-- Полная загрузка из dm_calc.storage_sales_bundles_amount
-- с обогащением справочниками через CTE.
----------------------------------------------------------------------

DROP TABLE IF EXISTS dm.storage_sales_bundles_amount_enriched;

CREATE TABLE dm.storage_sales_bundles_amount_enriched
WITH (
    appendonly  = true,
    orientation = column,
    compresstype = zstd,
    compresslevel = 3
)
DISTRIBUTED BY (sales_bundle_code)  -- при необходимости поменять на ваш ключ
AS
WITH
----------------------------------------------------------------------
-- 1. Базовый факт
----------------------------------------------------------------------
fact AS (
    SELECT
        s.*
    FROM dm_calc.storage_sales_bundles_amount s
),

----------------------------------------------------------------------
-- 2. Справочник материалов
----------------------------------------------------------------------
material_texts AS (
    SELECT DISTINCT
        m.material_code,
        m.material_name
    FROM dict_dds.material_texts m
),

----------------------------------------------------------------------
-- 3. Формы материала
----------------------------------------------------------------------
material_shape AS (
    SELECT DISTINCT
        ms.material_shape_code,
        ms.material_shape_full_name
    FROM dict_dds.material_shape ms
),

----------------------------------------------------------------------
-- 4. Услуги (перевозка/хранение)
----------------------------------------------------------------------
service_texts AS (
    SELECT DISTINCT
        t.service_code,
        t.transportation_service_name
    FROM dict_dds.transportation_service_texts t
),

----------------------------------------------------------------------
-- 5. ЕТСНГ
----------------------------------------------------------------------
etsng_texts AS (
    SELECT DISTINCT
        e.etsng_code,
        e.etsng_name
    FROM dict_dds.etsng_texts e
),

----------------------------------------------------------------------
-- 6. Типы транспорта / транспортировки
----------------------------------------------------------------------
transport_transfer_type_texts AS (
    SELECT DISTINCT
        tt.transport_transfer_type_code,
        tt.transport_transfer_type_name
    FROM dict_dds.transport_transfer_type_texts tt
),

----------------------------------------------------------------------
-- 7. Локации / страны / регионы рынка
--    (упорядоченная “широкая” таблица hub_ext)
----------------------------------------------------------------------
hub_ext AS (
    SELECT
        h.transport_hub_code,
        h.transport_hub_name_rus,
        h.transport_hub_name_eng,
        l.location_code,
        c.country_code,
        c.country_full_name,
        mr.market_region1_name
    FROM dict_dds.transport_hub h
    LEFT JOIN dict_dds.location l
        ON l.location_code = h.location_code
    LEFT JOIN dict_dds.country c
        ON c.country_code = l.country_code
    LEFT JOIN dict_dds.market_region1 mr
        ON mr.market_region1_code = c.market_region1_code
),

----------------------------------------------------------------------
-- 8. Заводы
----------------------------------------------------------------------
plant AS (
    SELECT DISTINCT
        p.plant_code,
        p.plant_short_name,
        p.plant_full_name
    FROM dict_dds.plant p
),

----------------------------------------------------------------------
-- 9. Контрагенты
----------------------------------------------------------------------
counterparty AS (
    SELECT DISTINCT
        c.counterparty_code,
        c.counterparty_short_name,
        c.address_code
    FROM dict_dds.counterparty c
),

----------------------------------------------------------------------
-- 10. Адреса контрагентов
----------------------------------------------------------------------
address_full AS (
    SELECT DISTINCT
        a.address_code,
        a.address_full_name
    FROM dict_dds.address a
),

----------------------------------------------------------------------
-- 11. Этапы транспортировки
----------------------------------------------------------------------
transportation_stage AS (
    SELECT DISTINCT
        ts.transportation_stage_code,
        ts.transportation_stage_name
    FROM dict_dds.transportation_stage ts
),

----------------------------------------------------------------------
-- 12. Связка "доставка -> завод-производитель"
--     (если в модели есть отдельная таблица доставок)
----------------------------------------------------------------------
delivery_plant AS (
    SELECT DISTINCT
        d.delivery_code,
        d.plant_producer_code
    FROM dds.delivery d
)

----------------------------------------------------------------------
-- 13. Итоговый SELECT
----------------------------------------------------------------------
SELECT
    -- базовые ключи и измерения
    f.sales_bundle_code,
    f.plant_producer_delivery_code,
    f.plant_producer_delivery_position_code,
    f.initial_delivery_code,
    f.material_code,
    mt.material_name,
    (f.material_code || '-' || COALESCE(mt.material_name, '')) AS material_key,

    f.dt_shipment,
    f.railcar_code,
    f.transport_bill_code,
    f.uni,

    f.material_shape_code,
    msh.material_shape_full_name,
    (f.material_shape_code || '-' || COALESCE(msh.material_shape_full_name, '')) AS material_shape_key,

    ------------------------------------------------------------------
    -- Порты / склады / хабы
    ------------------------------------------------------------------
    f.port_of_discharge_code,
    h_port.transport_hub_name_eng           AS port_of_discharge_name_eng,
    f.dt_arrival_to_russian_port,
    f.dt_storage_end_in_release,
    (f.port_of_discharge_code || '-' || COALESCE(h_port.transport_hub_name_eng, '')) AS port_of_discharge_key,

    -- Склад хранения
    f.warehouse_code,
    CASE
        WHEN country_warehouse.country_full_name = 'Россия'
            THEN COALESCE(warehouse_hub.transport_hub_name_rus, warehouse_hub.transport_hub_name_eng)
        ELSE COALESCE(warehouse_hub.transport_hub_name_eng, warehouse_hub.transport_hub_name_rus)
    END AS warehouse_name,
    (f.warehouse_code || '-' ||
        CASE
            WHEN country_warehouse.country_full_name = 'Россия'
                 THEN COALESCE(warehouse_hub.transport_hub_name_rus, warehouse_hub.transport_hub_name_eng)
            ELSE COALESCE(warehouse_hub.transport_hub_name_eng, warehouse_hub.transport_hub_name_rus)
        END
    ) AS warehouse_key,

    -- Удалённый склад (remote_warehouse_code)
    f.remote_warehouse_code,
    CASE
        WHEN country_remote.country_full_name = 'Россия'
            THEN COALESCE(remote_hub.transport_hub_name_rus, remote_hub.transport_hub_name_eng)
        ELSE COALESCE(remote_hub.transport_hub_name_eng, remote_hub.transport_hub_name_rus)
    END AS remote_warehouse_name,
    country_remote.country_code            AS remote_country_code,
    country_remote.country_full_name       AS remote_country_name,

    ------------------------------------------------------------------
    -- Регион/страна назначения (по основному хабу)
    ------------------------------------------------------------------
    CASE
        WHEN hub_main.market_region1_name = 'Турция'                  THEN 'Западная Азия'
        WHEN hub_main.market_region1_name = 'Средний Восток и Афр'    THEN 'Ближний Восток'
        WHEN hub_main.market_region1_name IN ('Азия', 'Китай')        THEN 'Восточная Азия'
        ELSE hub_main.market_region1_name
    END AS market_region1_group,

    CASE
        WHEN hub_main.country_full_name = 'Корея, республика' THEN 'Южная Корея'
        WHEN hub_main.country_full_name = 'Тайвань (Китай)'   THEN 'Тайвань'
        ELSE hub_main.country_full_name
    END AS country_full_name_norm,

    ------------------------------------------------------------------
    -- Заводы
    ------------------------------------------------------------------
    f.receiving_plant_code,
    p_recv.plant_short_name  AS receiving_plant_short_name,
    p_recv.plant_full_name   AS receiving_plant_full_name,
    (f.receiving_plant_code || '-' ||
        p_recv.plant_short_name || ' ' || p_recv.plant_full_name
    ) AS receiving_plant_key,

    f.plant_code,
    p_base.plant_short_name  AS base_plant_short_name,
    p_base.plant_full_name   AS base_plant_full_name,
    (p_base.plant_short_name || ' ' || p_base.plant_full_name) AS base_plant_name_full,

    -- Завод-производитель по доставке (если есть)
    p_prod.plant_short_name  AS producer_plant_short_name,
    p_prod.plant_full_name   AS producer_plant_full_name,
    (p_prod.plant_short_name || ' ' || p_prod.plant_full_name) AS producer_plant_name_full,

    ------------------------------------------------------------------
    -- Типы транспорта / этапы
    ------------------------------------------------------------------
    f.transport_type_code,
    ttt.transport_transfer_type_name,
    (f.transport_type_code || '-' || COALESCE(ttt.transport_transfer_type_name,'')) AS transport_type_key,

    f.transport_departure_hub_code,
    h_dep.transport_hub_name_rus AS departure_hub_name_rus,

    f.transport_destination_hub_code,
    h_dest.transport_hub_name_rus AS destination_hub_name_rus,

    f.transportation_outbound_delivery_code,
    f.transportation_outbound_delivery_position_code,
    f.transportation_inbound_delivery_code,
    f.transportation_inbound_delivery_position_code,

    f.is_final_transportation_stage_code,
    f.transportation_stage_code,
    ts.transportation_stage_name,
    CASE
        WHEN f.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
            THEN f.transportation_stage_code || '-' || COALESCE(ts.transportation_stage_name,'')
        ELSE NULL
    END AS transportation_stage_key,

    ------------------------------------------------------------------
    -- Услуги / контрагенты
    ------------------------------------------------------------------
    f.service_code,
    st.transportation_service_name,
    (f.service_code || '-' || COALESCE(st.transportation_service_name,'')) AS service_key,

    f.forwarder_code,
    COALESCE(cp_forwarder.counterparty_short_name, 'Не определено') AS forwarder_short_name,
    COALESCE(
        f.forwarder_code || '-' || cp_forwarder.counterparty_short_name,
        'Не определено'
    ) AS forwarder_key,

    f.creditor_in_shipment_instruction_code,
    addr.address_full_name AS creditor_address_full_name,
    (f.creditor_in_shipment_instruction_code || '-' ||
        COALESCE(addr.address_full_name,'')
    ) AS creditor_key,

    ------------------------------------------------------------------
    -- Стоимость/дни хранения
    ------------------------------------------------------------------
    f.storage_cost_calculation_type_name,
    f.storage_calculation_bundle_quantity,
    f.storage_duration_total_calendar_days,
    f.storage_duration_free_by_contract_calendar_days,
    f.storage_duration_free_by_delivery_calendar_days,
    f.storage_duration_payable_calendar_days,

    f.storage_cost_calculated_amount,
    f.storage_cost_calculated_amount_local,

    f.storage_payable_001_030_days_gross_weight,
    f.storage_payable_031_060_days_gross_weight,
    f.storage_payable_061_090_days_gross_weight,
    f.storage_payable_091_180_days_gross_weight,
    f.storage_payable_181_365_days_gross_weight,
    f.storage_payable_over_365_days_gross_weight,

    f.storage_calculated_cost_001_030_amount,
    f.storage_calculated_cost_031_060_amount,
    f.storage_calculated_cost_061_090_amount,
    f.storage_calculated_cost_091_180_amount,
    f.storage_calculated_cost_181_365_amount,
    f.storage_calculated_cost_over_365_amount,

    f.is_stored_in_russian_port_code,

    -- Тип объекта хранения
    CASE
        WHEN f.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN 'Терминал порта'
        WHEN f.storage_cost_calculation_type_name = 'Логика СВХ'
            THEN 'Склад СВХ'
        ELSE 'Не определено'
    END AS storage_object_type,

    -- Тип груза (контейнер/балк)
    CASE
        WHEN f.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'
            THEN 'Контейнеры'
        WHEN f.storage_cost_calculation_type_name = 'Логика РФ-балк'
            THEN 'Балк'
        ELSE 'Не определено'
    END AS storage_cargo_type,

    ------------------------------------------------------------------
    -- Прочие поля факта, которые тебе нужны как есть
    ------------------------------------------------------------------
    f.dt_report,
    f.sales_bundle_net_weight,
    f.sales_bundle_gross_weight,
    f.batch_code,
    f.dt_arrival,
    f.sales_delivery_code,
    f.delivery_for_storage_calculation_code,
    f.delivery_in_final_release_code,
    f.dt_final_release,
    f.bill_of_lading_code,
    f.bill_of_lading_number,
    f.dt_bill_of_lading,
    f.incoterms_code,
    f.storage_cost_special_calculation_type_code,
    f.unit_balance_code,
    f.dt_storage_start,
    f.dt_shipped_from_warehouse,
    f.dt_storage_end,
    f.service_number,
    f.shipment_instruction_number,
    f.dt_for_price_search_in_purchase_contract,
    f.purchase_contract_code,
    f.purchase_contract_position_code,
    f.purchase_contract_position_currency_code,
    f.purchase_contract_position_uom_code,
    f.dt_transportation_stage_start_p,
    f.dt_transportation_stage_start_r,
    f.dt_shipment_from_foreign_warehouse,
    f.metal_in_china_stock_balance_closing_weight

FROM fact f
    -- материал
    LEFT JOIN material_texts mt
        ON mt.material_code = f.material_code

    -- форма материала
    LEFT JOIN material_shape msh
        ON msh.material_shape_code = f.material_shape_code

    -- услуги
    LEFT JOIN service_texts st
        ON st.service_code = f.service_code

    -- ЕТСНГ
    LEFT JOIN etsng_texts et
        ON et.etsng_code = f.etsng_code

    -- тип транспорта
    LEFT JOIN transport_transfer_type_texts ttt
        ON ttt.transport_transfer_type_code = f.transport_type_code

    -- основной хаб (например, склад/порт назначения, зависит от твоей логики)
    LEFT JOIN hub_ext hub_main
        ON hub_main.transport_hub_code = f.warehouse_code

    -- порт выгрузки
    LEFT JOIN hub_ext h_port
        ON h_port.transport_hub_code = f.port_of_discharge_code

    -- склад хранения
    LEFT JOIN hub_ext warehouse_hub
        ON warehouse_hub.transport_hub_code = f.warehouse_code
    LEFT JOIN hub_ext country_warehouse
        ON country_warehouse.transport_hub_code = f.warehouse_code

    -- удалённый склад
    LEFT JOIN hub_ext remote_hub
        ON remote_hub.transport_hub_code = f.remote_warehouse_code
    LEFT JOIN hub_ext country_remote
        ON country_remote.transport_hub_code = f.remote_warehouse_code

    -- хабы отправления/назначения по перевозкам
    LEFT JOIN hub_ext h_dep
        ON h_dep.transport_hub_code = f.transport_departure_hub_code
    LEFT JOIN hub_ext h_dest
        ON h_dest.transport_hub_code = f.transport_destination_hub_code

    -- заводы
    LEFT JOIN plant p_recv
        ON p_recv.plant_code = f.receiving_plant_code
    LEFT JOIN plant p_base
        ON p_base.plant_code = f.plant_code

    -- завод-производитель (по доставке)
    LEFT JOIN delivery_plant dp
        ON dp.delivery_code = f.delivery_for_storage_calculation_code
    LEFT JOIN plant p_prod
        ON p_prod.plant_code = dp.plant_producer_code

    -- этапы транспортировки
    LEFT JOIN transportation_stage ts
        ON ts.transportation_stage_code = f.transportation_stage_code

    -- контрагенты
    LEFT JOIN counterparty cp_forwarder
        ON cp_forwarder.counterparty_code = f.forwarder_code
    LEFT JOIN counterparty cp_creditor
        ON cp_creditor.counterparty_code = f.creditor_in_shipment_instruction_code
    LEFT JOIN address_full addr
        ON addr.address_code = cp_creditor.address_code
;
