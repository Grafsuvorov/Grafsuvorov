 WITH ai_field AS (
      SELECT DISTINCT ON (issue_id)
          issue_id,
          field_value AS ai_saving_text,
          dttm_loaded
      FROM tech_etl.yt_issue_custom_field
      WHERE lower(trim(field_name)) = lower('Экономия за счет ИИ')
        AND nullif(trim(field_value), '') IS NOT NULL
      ORDER BY issue_id, dttm_loaded DESC
  ),

  ai_parts AS (
      SELECT
          issue_id,
          ai_saving_text,

          replace(
              substring(
                  lower(ai_saving_text)
                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(дн|день|дня|дней)'
              ),
              ',',
              '.'
          )::numeric AS days_value,

          replace(
              substring(
                  lower(ai_saving_text)
                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(ч|час|часа|часов)'
              ),
              ',',
              '.'
          )::numeric AS hours_value,

          replace(
              substring(
                  lower(ai_saving_text)
                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(мин|минута|минуты|минут|м)'
              ),
              ',',
              '.'
          )::numeric AS minutes_value

      FROM ai_field
  ),

  ai AS (
      SELECT
          issue_id,
          ai_saving_text,

          CASE
              -- Если указано просто число, считаем его часами,
              -- как это сейчас делает приложение.
              WHEN trim(ai_saving_text) ~ '^[0-9]+([.,][0-9]+)?$'
                  THEN round(replace(trim(ai_saving_text), ',', '.')::numeric * 60)::int

              ELSE round(
                  coalesce(days_value, 0) * 24 * 60
                  + coalesce(hours_value, 0) * 60
                  + coalesce(minutes_value, 0)
              )::int
          END AS ai_saving_minutes

      FROM ai_parts
  )

  SELECT
      ai.issue_id                                      AS task_id,
      'https://yt.rusal.ru/issue/' || ai.issue_id      AS task_url,
      issue.summary                                    AS task_summary,
      issue.assignee                                   AS task_assignee,

      ai.ai_saving_text,
      ai.ai_saving_minutes,
      round(ai.ai_saving_minutes / 60.0, 2)            AS ai_saving_hours,

      worklog.author                                   AS worklog_author,
      worklog.creator                                  AS worklog_creator,
      worklog.work_date,
      worklog.work_date::date                          AS work_date,
      worklog.minutes                                  AS worklog_minutes,
      round(worklog.minutes / 60.0, 2)                 AS worklog_hours,
      worklog.work_type,
      worklog.work_text

  FROM ai

  LEFT JOIN tech_etl.yt_issue_snapshot issue
      ON issue.issue_id = ai.issue_id

  LEFT JOIN tech_etl.yt_issue_worklog worklog
      ON worklog.issue_id = ai.issue_id

  WHERE ai.ai_saving_minutes > 0

  ORDER BY
      worklog.work_date DESC NULLS LAST,
      ai.issue_id,
      worklog.author;

  ## Отчёт на уровне задачи

  Для управленческой отчётности безопаснее сначала агрегировать списания по задаче:

  WITH ai_field AS (
      SELECT DISTINCT ON (issue_id)
          issue_id,
          field_value AS ai_saving_text,
          dttm_loaded
      FROM tech_etl.yt_issue_custom_field
      WHERE lower(trim(field_name)) = lower('Экономия за счет ИИ')
        AND nullif(trim(field_value), '') IS NOT NULL
      ORDER BY issue_id, dttm_loaded DESC
  ),

  ai AS (
      SELECT
          issue_id,
          ai_saving_text,
          CASE
              WHEN trim(ai_saving_text) ~ '^[0-9]+([.,][0-9]+)?$'
                  THEN round(replace(trim(ai_saving_text), ',', '.')::numeric * 60)::int
              ELSE
                  round(
                      coalesce(
                          replace(
                              substring(
                                  lower(ai_saving_text)
                                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(дн|день|дня|дней)'
                              ),
                              ',',
                              '.'
                          )::numeric,
                          0
                      ) * 1440
                      +
                      coalesce(
                          replace(
                              substring(
                                  lower(ai_saving_text)
                                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(ч|час|часа|часов)'
                              ),
                              ',',
                              '.'
                          )::numeric,
                          0
                      ) * 60
                      +
                      coalesce(
                          replace(
                              substring(
                                  lower(ai_saving_text)
                                  FROM '([0-9]+([.,][0-9]+)?)[[:space:]]*(мин|минута|минуты|минут|м)'
                              ),
                              ',',
                              '.'
                          )::numeric,
                          0
                      )
                  )::int
          END AS ai_saving_minutes
      FROM ai_field
  ),

  worklog AS (
      SELECT
          issue_id,
          string_agg(DISTINCT author, ', ' ORDER BY author) AS worklog_authors,
          count(DISTINCT author)                           AS authors_count,
          sum(coalesce(minutes, 0))                        AS worklog_minutes,
          min(work_date)                                   AS first_work_date,
          max(work_date)                                   AS last_work_date
      FROM tech_etl.yt_issue_worklog
      GROUP BY issue_id
  )

  SELECT
      ai.issue_id                                         AS task_id,
      'https://yt.rusal.ru/issue/' || ai.issue_id         AS task_url,
      issue.summary                                       AS task_summary,
      issue.assignee,
      worklog.worklog_authors,
      worklog.authors_count,
      worklog.first_work_date,
      worklog.last_work_date,
      coalesce(worklog.worklog_minutes, 0)                AS worklog_minutes,
      round(coalesce(worklog.worklog_minutes, 0) / 60.0, 2)
                                                            AS worklog_hours,
      ai.ai_saving_text,
      ai.ai_saving_minutes,
      round(ai.ai_saving_minutes / 60.0, 2)               AS ai_saving_hours

  FROM ai

  LEFT JOIN tech_etl.yt_issue_snapshot issue
      ON issue.issue_id = ai.issue_id

  LEFT JOIN worklog
      ON worklog.issue_id = ai.issue_id

  WHERE ai.ai_saving_minutes > 0

  ORDER BY worklog.last_work_date DESC NULLS LAST;

  ## Важное ограничение данных

  Сейчас yt_issue_custom_field содержит только:

  issue_id
  field_name
  field_value
  dttm_loaded

  Поэтому из неё невозможно узнать:

  - кто заполнил поле «Экономия за счет ИИ»;
  - когда именно поле было изменено;
  - как менялось значение;
  - как распределить экономию между несколькими исполнителями.

  dttm_loaded — это дата загрузки данных в Greenplum, а не дата изменения поля в YouTrack.

  Чтобы получить точную историю, загрузчик YouTrack нужно расширить: сохранять изменение поля «Экономия за счет ИИ» в yt_issue_timeline с полями:

  issue_id
  ts
  author
  field_name
  value_from
  value_to
