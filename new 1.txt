DROP TABLE IF EXISTS t_contract_match;

CREATE TEMP TABLE t_contract_match ON COMMIT DROP AS
WITH

-- ===========================
-- 1. РФ-контейнеры
-- ===========================
rf_cont AS (
    SELECT
        b.rn,
        t.purchase_contract_code,
        t.position_line_item_code,
        t.konws,
        t.price_uom_code,
        t.service_code,
        t.currency_dogovor,
        t.datab,
        t.datbi,
        row_number() OVER (
            PARTITION BY b.rn
            ORDER BY
                t.order_rf,
                t.kopos,
                t.kbetr,
                t.purchase_contract_code,
                t.position_line_item_code
        ) AS rn_choice
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN t_1079_base t
      ON t.bukrs = '1120'
     AND t.lifnr = b.creditor_in_shipment_instruction_code
     AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
     AND t.werks = '1121'
     AND t.zzsrvpos = 'S49'
     AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
     AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
     AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab       AND t.datbi
     AND b.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'
),
rf_cont_final AS (
    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        price_uom_code,
        service_code,
        currency_dogovor,
        datab,
        datbi
    FROM rf_cont
    WHERE rn_choice = 1
),

-- ===========================
-- 2. РФ-балк (вариант 1: zzsrvpos = service_number)
-- ===========================
rf_balk_1 AS (
    SELECT
        b.rn,
        t.purchase_contract_code,
        t.position_line_item_code,
        t.konws,
        t.price_uom_code,
        t.service_code,
        t.currency_dogovor,
        t.datab,
        t.datbi,
        row_number() OVER (
            PARTITION BY b.rn
            ORDER BY
                t.order_rf,
                t.kopos,
                t.kbetr,
                t.purchase_contract_code,
                t.position_line_item_code
        ) AS rn_choice
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN t_1079_base t
      ON t.bukrs = '1120'
     AND t.lifnr = b.creditor_in_shipment_instruction_code
     AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
     AND t.werks = '1121'
     AND t.zzsrvpos = b.service_number
     AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
     AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
     AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab       AND t.datbi
     AND b.storage_cost_calculation_type_name = 'Логика РФ-балк'
),
rf_balk_1_final AS (
    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        price_uom_code,
        service_code,
        currency_dogovor,
        datab,
        datbi
    FROM rf_balk_1
    WHERE rn_choice = 1
),

-- ===========================
-- 3. РФ-балк (вариант 2: zzsrvpos = 'S49')
-- ===========================
rf_balk_2 AS (
    SELECT
        b.rn,
        t.purchase_contract_code,
        t.position_line_item_code,
        t.konws,
        t.price_uom_code,
        t.service_code,
        t.currency_dogovor,
        t.datab,
        t.datbi,
        row_number() OVER (
            PARTITION BY b.rn
            ORDER BY
                t.order_rf,
                t.kopos,
                t.kbetr,
                t.purchase_contract_code,
                t.position_line_item_code
        ) AS rn_choice
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN t_1079_base t
      ON t.bukrs = '1120'
     AND t.lifnr = b.creditor_in_shipment_instruction_code
     AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
     AND t.werks = '1121'
     AND t.zzsrvpos = 'S49'
     AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
     AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
     AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab       AND t.datbi
     AND b.storage_cost_calculation_type_name = 'Логика РФ-балк'
),
rf_balk_2_final AS (
    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        price_uom_code,
        service_code,
        currency_dogovor,
        datab,
        datbi
    FROM rf_balk_2
    WHERE rn_choice = 1
),

-- ===========================
-- 4. Логика Азия
-- ===========================
asia AS (
    SELECT
        b.rn,
        t.purchase_contract_code,
        t.position_line_item_code,
        t.konws,
        t.price_uom_code,
        t.service_code,
        t.currency_dogovor,
        t.datab,
        t.datbi,
        row_number() OVER (
            PARTITION BY b.rn
            ORDER BY
                t.order_asia,
                t.kopos,
                t.kbetr,
                t.purchase_contract_code,
                t.position_line_item_code
        ) AS rn_choice
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN t_1079_base t
      ON t.bukrs = b.unit_balance_code
     AND t.lifnr = b.forwarder_code
     AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
     AND t.werks = b.plant_code
     AND t.zzsrvpos = 'S49'
     AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
     AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
     AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
     AND (t.zzterminal = b.warehouse_code OR t.zzterminal IS NULL)
     AND (t.vzzzone   = b.warehouse_storage_area_code OR t.vzzzone IS NULL)
     AND t.purchase_document_subtype_code = 'ZDTL'
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab       AND t.datbi
     AND b.storage_cost_calculation_type_name = 'Логика Азия'

    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        price_uom_code,
        service_code,
        currency_dogovor,
        datab,
        datbi
    FROM asia
    WHERE rn_choice = 1
),

-- ===========================
-- 5. Логика СВХ
-- ===========================

        b.rn,
        t.purchase_contract_code,
        t.position_line_item_code,
        t.konws,
        t.price_uom_code,
        t.service_code,
        t.currency_dogovor,
        t.datab,
        t.datbi,
        row_number() OVER (
            PARTITION BY b.rn
            ORDER BY
                t.order_euro_svh,
                t.kopos,
                t.kbetr,
                t.purchase_contract_code,
                t.position_line_item_code
        ) AS rn_choice
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN t_1079_base t
      ON t.bukrs = '1120'
     AND t.lifnr = b.forwarder_code
     AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
     AND t.werks = '1121'
     AND t.zzsrvpos = 'S49'
     AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
     AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
     AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
     AND (t.zzterminal = b.warehouse_code OR t.zzterminal IS NULL)
     AND t.purchase_document_subtype_code = 'ZDTL'
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab       AND t.datbi
     AND b.storage_cost_calculation_type_name = 'Логика СВХ'
),
svh_final AS (
    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        price_uom_code,
        service_code,
        currency_dogovor,
        datab,
        datbi
    FROM svh
    WHERE rn_choice = 1
)

SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    price_uom_code,
    service_code,
    currency_dogovor,
    datab,
    datbi
FROM rf_cont_final

UNION ALL
SELECT * FROM rf_balk_1_final
UNION ALL
SELECT * FROM rf_balk_2_final
UNION ALL
SELECT * FROM asia_final
UNION ALL
SELECT * FROM svh_final
;
