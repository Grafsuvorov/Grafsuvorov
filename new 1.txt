-- =========================================================
-- 3) CR на 1-е следующего месяца БЕЗ GROUP BY (дублей нет)
--    Ограничиваемся только тройками из PP через EXISTS (дешево)
-- =========================================================
CREATE TEMP TABLE cr_next ON COMMIT DROP 
AS
SELECT
    c.kalnr,
    (make_date(c.bdatj::int, substring(c.poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    c.salk3
FROM cr_curr_small c
WHERE EXISTS (
    SELECT 1
    FROM pp_base p
    WHERE p.kalnr = c.kalnr
      AND p.bdatj = c.bdatj
      AND p.poper = c.poper
)
DISTRIBUTED BY (kalnr);
ANALYZE cr_next; --36

-- 5) DATES через LEAD() (один проход по pp_dates)

CREATE TEMP TABLE dates ON COMMIT drop
AS
SELECT
    kalnr,
    dt_valid_from,
    COALESCE(

    )::date AS dt_valid_to

DISTRIBUTED BY (kalnr);

    -- rule3
    CASE
      WHEN (t.umkumo + t.abkumo + t.zukumo) = 0 THEN 0
      ELSE TRUNC(
             CAST(ROUND(
                   (t.stprs / NULLIF(t.peinh,0)) * (t.umkumo + t.abkumo)

------------- цена куммулютивного запаса для равной даты
case when (ckmlpp.abkumo + ckmlpp.umkumo + ckmlpp.zukumo) <> 0 
	then trunc(cast(ckmlcr_2.salk3 + coalesce(mlcd.salk3 + mlcd.estprd + mlcd.mstprd, 0) as numeric(40, 20)) / (ckmlpp.abkumo + ckmlpp.umkumo + ckmlpp.zukumo), 15)
	else 0 end as average_weighted_stock_price_rule2_prev_date_amount, -- price_588_2_ravno
------------- цена куммулютивного запаса для не равной даты
case when (ckmlpp.lbkum) <> 0 then trunc(cast(ckmlcr.salk3 as numeric(40, 20)) / (ckmlpp.lbkum), 15)
	else 0 end as average_weighted_stock_price_rule2_next_date_amount, -- price_588_2_ne_ravno
ckmlhd.price_rule_code as price_rule_code
from ods.ckmlpp_ral as ckmlpp
left join ods.ckmlcr_ral as ckmlcr 
on ckmlcr.kalnr = ckmlpp.kalnr and ckmlcr.bdatj = ckmlpp.bdatj and ckmlcr.poper = ckmlpp.poper and ckmlcr.untper = '000' and ckmlcr.curtp = '10'
left join ckmlcr_2 on ckmlcr_2.kalnr = ckmlpp.kalnr and ckmlcr_2.dt_posting = (ckmlpp.bdatj || substring(ckmlpp.poper, 2, 2) || '01') ::date
left join mlcd on mlcd.kalnr = ckmlpp.kalnr and mlcd.bdatj = ckmlpp.bdatj and mlcd.poper = ckmlpp.poper
left join dds.material_ledger_header as ckmlhd on ckmlhd.calculation_code = ckmlpp.kalnr
where ckmlpp.untper = '000';

  create temporary table ckmlcr_2 ON COMMIT DROP as (
	select
		kalnr,
		(ckmlcr_2.bdatj || substring(ckmlcr_2.poper, 2, 2) || '01')::date + interval '1 month' as dt_posting,
		salk3,
		poper,
		bdatj,
		stprs,
		vprsv,
		peinh,
		abprd_o,
		abprd_mo,
		pvprs 
	from
		ods.ckmlcr_ral as ckmlcr_2
	where
		ckmlcr_2.untper = '000'
		and ckmlcr_2.curtp = '10'
)
distributed by (kalnr);


create temporary table mlcd ON COMMIT DROP as (
	select
		kalnr,
		bdatj,
		poper,
		sum(salk3) as salk3,
		sum(estprd) as estprd,
		sum(mstprd) as mstprd 
	from
		ods.mlcd_ral
	where
		categ in ('ZU', 'VP', 'PC')
		and curtp = '10' 
	group by kalnr, bdatj, poper
)
distributed by (kalnr);


create temporary table ckmlpp ON COMMIT DROP as (
	select
	    ckmlpp.kalnr,
	    ckmlpp.bdatj,
		ckmlpp.poper,
		(ckmlpp.bdatj || substring(ckmlpp.poper, 2, 2) || '01')::date as dt_valid_from,
		ckmlpp.umkumo,
		ckmlpp.abkumo,
		ckmlpp.zukumo,
		ckmlpp.lbkum,
		ckmlpp.untper
	from
		ods.ckmlpp_ral as ckmlpp
	where
		ckmlpp.untper = '000'
)
distributed by (kalnr, bdatj, poper);

--insert into dm.material_price_calculation
--insert into  userdata.saa_material_price_calculation
select
	ckmlpp.kalnr as calculation_code,
	ckmlpp.dt_valid_from,
	case
		when lead(ckmlpp.dt_valid_from) over (partition by ckmlpp.kalnr order by ckmlpp.dt_valid_from) is null
		then '22991231'::date
		else lead(ckmlpp.dt_valid_from) over (partition by ckmlpp.kalnr order by ckmlpp.dt_valid_from - interval '1 day' )
	end as dt_valid_to,
	ckmlcr.vprsv as price_valuation_type_code, 
	ckmlcr.stprs as standard_price_amount, 
	ckmlcr.peinh as price_unit_code, 
	ckmlcr.pvprs as moving_average_price_amount, 
	case 
		when ckmlpp.umkumo + ckmlpp.abkumo + ckmlpp.zukumo = 0
		then 0
		else
			trunc(
				cast(
					round(ckmlcr.stprs / ckmlcr.peinh * (ckmlpp.umkumo + ckmlpp.abkumo) + ckmlcr.abprd_o + ckmlcr.abprd_mo /*стоймость остатков на начало*/
						+ coalesce(mlcd.salk3 + mlcd.estprd + mlcd.mstprd, 0), 2
					) as numeric(40, 20)
				) /*стоймость куммулютивного запаса*/ / (ckmlpp.umkumo + ckmlpp.abkumo + ckmlpp.zukumo), 15
			)
	end as average_weighted_stock_price_rule3_amount, -- Цена кумулятивного запаса в ветке = 3 price_588_3
	------------- цена куммулютивного запаса для равной даты
	case
		when (ckmlpp.abkumo + ckmlpp.umkumo + ckmlpp.zukumo) <> 0 
		then
			trunc(
				cast(
					ckmlcr_2.salk3 + coalesce(mlcd.salk3 + mlcd.estprd + mlcd.mstprd, 0) as numeric(40, 20)
				) / (ckmlpp.abkumo + ckmlpp.umkumo + ckmlpp.zukumo), 15
			)
		else 0
	end as average_weighted_stock_price_rule2_prev_date_amount, -- price_588_2_ravno
	------------- цена куммулютивного запаса для не равной даты
	case
		when ckmlpp.lbkum <> 0
		then trunc(cast(ckmlcr.salk3 as numeric(40, 20)) / (ckmlpp.lbkum), 15)
		else 0
	end as average_weighted_stock_price_rule2_next_date_amount, -- price_588_2_ne_ravno
	ckmlhd.price_rule_code as price_rule_code
from
	ckmlpp
	left join ckmlcr_2 ckmlcr
		on ckmlcr.kalnr = ckmlpp.kalnr
		and ckmlcr.bdatj = ckmlpp.bdatj
		and ckmlcr.poper = ckmlpp.poper
	left join ckmlcr_2
		on ckmlcr_2.kalnr = ckmlpp.kalnr
		and ckmlcr_2.dt_posting = ckmlpp.dt_valid_from
	left join mlcd
		on mlcd.kalnr = ckmlpp.kalnr
		and mlcd.bdatj = ckmlpp.bdatj
		and mlcd.poper = ckmlpp.poper
	left join dds.material_ledger_header as ckmlhd
		on ckmlhd.calculation_code = ckmlpp.kalnr
;
