CREATE TEMP TABLE for_1079_asia AS
SELECT
    b2.rn,
    fb.purchase_contract_code,
    fb.position_line_item_code,
    fb.konws,
    fb.service_code,
    fb.price_uom_code,
    fb.currency_dogovor,
    fb.datab,
    fb.datbi
FROM dm_calc.storage_sales_bundles_weight b2
LEFT JOIN LATERAL (
    SELECT
        f.*
    FROM for_1079_base f
    WHERE f.bukrs = b2.unit_balance_code
      AND f.lifnr = b2.forwarder_code
      AND (f.zzport = b2.port_of_discharge_code OR f.zzport IS NULL)
      AND f.werks = b2.plant_code
      AND f.zzsrvpos = 'S49'
      AND (f.zzforma = b2.material_shape_code OR f.zzforma IS NULL)
      AND (f.zzet_tarif = b2.etsng_code OR f.zzet_tarif IS NULL)
      AND (f.sdabw = b2.transport_type_code OR f.sdabw IS NULL)
      AND b2.dt_for_price_search_in_purchase_contract BETWEEN f.dt_valid_from AND f.dt_valid_to
      AND b2.dt_for_price_search_in_purchase_contract BETWEEN f.datab AND f.datbi
      AND b2.storage_cost_calculation_type_name = 'Логика Азия'
      AND (f.zzterminal = b2.warehouse_code OR f.zzterminal IS NULL)
      AND (f.vzzzone = b2.warehouse_storage_area_code OR f.vzzzone IS NULL)
      AND f.purchase_document_subtype_code = 'ZDTL'
    ORDER BY
        f.order_asia,
        f.kopos,
        f.kbetr,
        f.purchase_contract_code,
        f.position_line_item_code
    LIMIT 1
) fb ON TRUE
DISTRIBUTED BY (rn);
