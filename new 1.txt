select
    tv.airtable_vessel_type_name as vessel_type_name,																-- Тип судна LE_TR2.000004
    fca.dt_report as dt_operation,																					-- Конечная дата месяца LE_TR2.000025
    lpad(extract(day from fca.dt_report)::text, 2, '0') as dt_operation_dd,														
    lpad(extract(month  from fca.dt_report)::text, 2, '0') as dt_operation_mm,													
    extract(year from fca.dt_report) as dt_operation_yyyy,													
    fca.heavy_fuel_oil_consumption_plan_weight,																		-- Объем топлива Heavy Fuel Oil в месяце по БП LE_TR2.000026
    fca.heavy_fuel_oil_consumption_actual_weight,																	-- Объем топлива Heavy Fuel Oil в месяце по факту LE_TR2.000027
    fca.marine_gas_oil_consumption_plan_weight,																		-- Объем топлива Marine Gas Oil в месяце по БП LE_TR2.000028
    fca.marine_gas_oil_consumption_actual_weight,																	-- Объем топлива Marine Gas Oil в месяце по факту LE_TR2.000029
    fca.vessel_name as feeder_vessel_name,																			-- Наименование судна LE_TR2.000030
    fca.port_name as port_of_loading_name,																			-- Наименование порта LE_TR2.000031
    date(fca.dt_created) as dt_created,																				-- Дата создания записи LE_TR2.000032
    date(fca.dt_updated) as dt_updated																				-- Дата изменения записи LE_TR2.000033	
from 
    {{ ref('dds_s_int_transportation_transshipment_fuel_bp_vs_actual') }} as fca
left join {{ ref('dict_dds_s_transport_vehicle') }} as tv
    on tv.vessel_name = fca.vessel_name
where 
    fca.dt_report is not null
    and fca.source_system_code = 'airtable'
