select
    purchase_document_code,
    purchase_document_position_code,
    material_document_code,
    fiscal_year,
    material_document_position_code,
    assignment_code,
    count(*) as cnt
from dds.purchase_document_history
group by
    purchase_document_code,
    purchase_document_position_code,
    material_document_code,
    fiscal_year,
    material_document_position_code,
    assignment_code
having count(*) > 1
order by cnt desc
limit 100;
