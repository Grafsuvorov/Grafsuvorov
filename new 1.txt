       AND cotex.language_code = 'R'

        ON COALESCE(trht.transport_hub_code, '##NULL##') = COALESCE(base.transport_departure_hub_code, '##NULL##')

        ON COALESCE(trht_2.transport_hub_code, '##NULL##') = COALESCE(base.transport_destination_hub_code, '##NULL##')

                                      WHEN cotex.country_full_name = 'Россия' THEN 'R'
                                      ELSE 'E'
                                  END

        ON trht_4.language_code = 'R'
       AND COALESCE(trht_4.transport_hub_code, '##NULL##') = COALESCE(base.warehouse_storage_area_code, '##NULL##')

        ON COALESCE(deldp.delivery_code, '##NULL##') = COALESCE(base.delivery_for_storage_calculation_code, '##NULL##')
       AND deldp.delivery_position_line_item_code = '000010'

        ON COALESCE(pas_1124.plant_code, '##NULL##') = COALESCE(deldp.plant_producer_code, '##NULL##')

        ON COALESCE(trst.transportation_stage_code, '##NULL##') = COALESCE(base.transportation_stage_code, '##NULL##')
       AND trst.language_code = 'R'
    LEFT JOIN dict_dds.counterparty AS counter
        ON COALESCE(counter.counterparty_code, '##NULL##') = COALESCE(base.forwarder_code, '##NULL##')
    LEFT JOIN dict_dds.transport_hub AS tr_hub
        ON COALESCE(tr_hub.transport_hub_code, '##NULL##') = COALESCE(base.port_of_discharge_code, '##NULL##')
    LEFT JOIN dict_dds.transport_hub AS trhu_2
        ON COALESCE(trhu_2.transport_hub_code, '##NULL##') = COALESCE(base.remote_warehouse_code, '##NULL##')
    LEFT JOIN adrc AS adrc_3
        ON COALESCE(adrc_3.address_code, '##NULL##') = COALESCE(trhu_2.address_code, '##NULL##')
    LEFT JOIN dict_dds.country_texts AS cotex_2
        ON COALESCE(cotex_2.country_code, '##NULL##') = COALESCE(adrc_3.country_code, '##NULL##')
       AND cotex_2.language_code = 'R'
    LEFT JOIN dict_dds.transport_hub_texts AS trht_5
        ON COALESCE(trht_5.transport_hub_code, '##NULL##') = COALESCE(base.remote_warehouse_code, '##NULL##')
       AND trht_5.language_code = CASE
                                      WHEN cotex_2.country_full_name = 'Россия' THEN 'R'
                                      ELSE 'E'
                                  END
    LEFT JOIN dict_dds.material_specification AS dd_ms_s_n
        ON COALESCE(dd_ms_s_n.material_code, '##NULL##') = COALESCE(base.material_code, '##NULL##')
    LEFT JOIN dict_dds.material_shape_texts AS mf
        ON COALESCE(mf.shape_code, '##NULL##') = COALESCE(dd_ms_s_n.shape_code, '##NULL##')
       AND mf.language_code = 'R'
    LEFT JOIN dict_dds.transport_transfer_type_texts AS ipd
        ON COALESCE(ipd.transport_transfer_type_code, '##NULL##') = COALESCE(base.transport_type_code, '##NULL##')
       AND ipd.language_code = 'R'
    LEFT JOIN dict_dds.counterparty AS counter_2
        ON COALESCE(counter_2.counterparty_code, '##NULL##') = COALESCE(base.creditor_in_shipment_instruction_code, '##NULL##')
    LEFT JOIN dict_dds.address AS addr_8
        ON COALESCE(addr_8.address_code, '##NULL##') = COALESCE(counter_2.address_code, '##NULL##')
       AND addr_8.international_display_format_code IS NULL;
