import os
import yaml
from datetime import datetime, timedelta
from pathlib import Path
from airflow import DAG
from airflow.utils.task_group import TaskGroup
from plugins.custom_operators.dbtBashOperator import dbtBashOperator
from plugins.custom_operators.ExecutionGuidSensor import ExecutionGuidSensor
from plugins.custom_operators.DbtModelReadySensor import DbtModelReadySensor
from plugins.custom_operators.DbtRunWatcherSensor import DbtRunWatcherSensor

AIRFLOW_HOME_DIR = Path(os.environ.get('AIRFLOW_HOME', '/home/airflow'))
CONFIGS_DIR = AIRFLOW_HOME_DIR / 'config_files' / 'dbt_dags'

def create_default_args():
    return {
        'owner': 'airflow',
        'depends_on_past': False,
        'start_date': datetime(2026, 8, 1),
        'retries': 0,
        'retry_delay': timedelta(minutes=5),
    }

CLUSTER = os.environ.get("CLUSTER", "DEV")

if CONFIGS_DIR.is_dir():
    try:
        yaml_files = [f.name for f in CONFIGS_DIR.iterdir() if f.suffix == '.yml']
    except FileNotFoundError:
        yaml_files = []

    for yaml_file in yaml_files:
        entity_name = yaml_file.replace('.yml', '')
        config_path = CONFIGS_DIR / yaml_file

        try:
            with open(config_path, 'r') as f:
                config = yaml.safe_load(f)
        except Exception as e:
            print(f"Error while parsing:{yaml_file}: {e}")
            continue

        dag_id = f'{entity_name}_dbt'
        default_args = create_default_args()
        raw_schedule = config.get('schedule', None)
        if isinstance(raw_schedule, dict):
            schedule_val = raw_schedule.get(CLUSTER, raw_schedule.get('DEV'))
        else:
            schedule_val = raw_schedule

        clickhouse_config = config.get('clickhouse', {})
        tables_list = clickhouse_config.get('tables', [])
        clickhouse_enabled = clickhouse_config.get('enabled', False)
        if isinstance(clickhouse_enabled, dict):
            clickhouse_enabled_flag = clickhouse_enabled.get(CLUSTER, clickhouse_enabled.get('DEV'))
        else:
            clickhouse_enabled_flag = clickhouse_enabled

        watcher_poke_interval = clickhouse_config.get('poke_interval_watcher', 120)

        TEMPLATE_DIR_DBT = AIRFLOW_HOME_DIR / 'template_dir' / 'dbt_operator_templates'
        TEMPLATE_DIR_SENSOR = AIRFLOW_HOME_DIR / 'template_dir' / 'dbt_dags_generator'

        with DAG(
            dag_id=dag_id,
            default_args=default_args,
            schedule_interval=schedule_val,
            catchup=False,
            max_active_tasks=6,
            max_active_runs=1,
            tags=['dbt-clickhouse', 'dbt'],
            on_failure_callback=None,
            template_searchpath=[str(TEMPLATE_DIR_DBT), str(TEMPLATE_DIR_SENSOR)],
            is_paused_upon_creation=True
        ) as dag:
            with TaskGroup(group_id='dbt') as dbt_group:
                run_dbt = dbtBashOperator(
                    task_id=f'run_dbt_{entity_name}',
                    dbt_command=config['dbt']['dbt_command'],
                    additional_vars=config['dbt']['additional_vars']
                )

            if tables_list and clickhouse_enabled_flag:
                get_execution_guid_dbt = ExecutionGuidSensor(
                    task_id='get_execution_guid_dbt',
                    sql='dbt_get_exec_guid.sql',
                    poke_interval=30,
                    timeout=1200,
                    mode='reschedule',
                    soft_fail=False
                )
                watch_dbt_run = DbtRunWatcherSensor(
                    task_id='watch_dbt_run',
                    models=tables_list,
                    sql='dbt_get_models_list.sql',
                    poke_interval=watcher_poke_interval,
                    timeout=7200,
                    mode='reschedule',
                    soft_fail=False
                )
                with TaskGroup(group_id='clickhouse') as ch_group:
                    for table_info in tables_list:
                        model_name = table_info['dbt_model']
                        trigger_target = table_info['dag_name']
                        unique_suffix = model_name.replace(".", "_")
                        check_trigger_task = DbtModelReadySensor(
                            task_id=f'check_trigger_{unique_suffix}',
                            model_name=model_name,
                            trigger_dag_id=trigger_target,
                            poke_interval=watcher_poke_interval,
                            timeout=7200,
                            mode='reschedule',
                            soft_fail=False
                        )

                get_execution_guid_dbt >> watch_dbt_run
                get_execution_guid_dbt >> ch_group
