/* =====================================================================
   НОВАЯ АРХИТЕКТУРА:
   
   LANDING
      ↓ новые XML по watermark
   tmp_new_docs
      ↓ XML parsing
   tmp_delta
      ├──────────────→ STG HISTORY
      │
      └→ latest snapshot
             ↓
         ODS CURRENT

   Дополнительно:
   meta.s1c_debt_watermark       - что уже обработал ETL
   meta.s1c_debt_source_control  - когда каждая 1С/БЕ приезжала последний раз
   ===================================================================== */


/* =====================================================================
   БЛОК 0.
   СОЗДАЁМ META-СХЕМУ
   Выполняется ОДИН РАЗ.
   ===================================================================== */

CREATE SCHEMA IF NOT EXISTS meta;


/* =====================================================================
   БЛОК 1.
   WATERMARK

   Зачем:
   Хранит до какого dt_insert LANDING уже успешно обработан.

   Вместо текущего:
       select max(dt_insert) from ods...

   ODS больше не управляет состоянием ETL.
   ===================================================================== */

CREATE TABLE meta.s1c_debt_watermark
(
    process_name    varchar(100) NOT NULL,
    last_dt_insert  timestamp    NOT NULL,
    updated_dttm    timestamp    NOT NULL
)
DISTRIBUTED REPLICATED;


/* ---------------------------------------------------------------------
   INITIAL LOAD WATERMARK

   ВЫПОЛНЯЕТСЯ ТОЛЬКО ОДИН РАЗ при переходе на новый алгоритм.

   Берём max(dt_insert), который уже попал в STG.
   --------------------------------------------------------------------- */

INSERT INTO meta.s1c_debt_watermark
(
    process_name,
    last_dt_insert,
    updated_dttm
)
SELECT
    's1c_debt_for_working_capital',
    COALESCE(
        MAX(dt_insert),
        timestamp '1900-01-01 00:00:00'
    ),
    clock_timestamp()
FROM stg.s1c_debt_for_working_capital;


/* =====================================================================
   БЛОК 2.
   SOURCE CONTROL

   Зачем:
   Контроль того, когда конкретная 1С + БЕ
   последний раз присылала данные.

   Например:

   ERP01 + BE01 -> сегодня
   ERP02 + BE02 -> сегодня
   ERP03 + BE03 -> три дня назад -> ALERT
   ===================================================================== */

CREATE TABLE meta.s1c_debt_source_control
(
    database_code_1c          varchar(255) NOT NULL,
    unit_balance_mdm_code_1c  varchar(255) NOT NULL,

    /* участвует ли источник в мониторинге */
    is_active                 boolean NOT NULL DEFAULT true,

    /*
       Сколько источник может молчать.
       Например 36 часов.
    */
    stale_after_hours         integer NOT NULL DEFAULT 36,

    last_dt_report            date,
    last_dt_insert            timestamp,
    last_create_date_time     timestamp,

    updated_dttm              timestamp NOT NULL
)
DISTRIBUTED BY
(
    database_code_1c,
    unit_balance_mdm_code_1c
);


/* ---------------------------------------------------------------------
   INITIAL LOAD SOURCE CONTROL

   ОДИН РАЗ при внедрении.

   Здесь можно использовать ODS, потому что нам нужно только
   текущее состояние каждого источника.
   --------------------------------------------------------------------- */

INSERT INTO meta.s1c_debt_source_control
(
    database_code_1c,
    unit_balance_mdm_code_1c,
    is_active,
    stale_after_hours,
    last_dt_report,
    last_dt_insert,
    last_create_date_time,
    updated_dttm
)
SELECT
    database_code_1c,
    unit_balance_mdm_code_1c,
    true,
    36,
    MAX(dt_report),
    MAX(dt_insert),
    MAX(create_date_time),
    clock_timestamp()

FROM ods.account_debt_for_working_capital_1c

GROUP BY
    database_code_1c,
    unit_balance_mdm_code_1c;


/* =====================================================================
   =====================================================================
   
                       ЕЖЕДНЕВНЫЙ ПРОЦЕСС
   
   Всё ниже запускается DAG-ом каждый день.
   
   =====================================================================
   ===================================================================== */


DO
$$

DECLARE

    v_watermark      timestamp;
    v_new_watermark  timestamp;

    v_row_cnt        bigint;
    v_start_dttm     timestamp;

BEGIN

    v_start_dttm := clock_timestamp();

    RAISE NOTICE 'START %', v_start_dttm;


    /* ================================================================
       STEP 1.
       ПОЛУЧАЕМ WATERMARK
       ================================================================ */

    SELECT last_dt_insert

      INTO v_watermark

      FROM meta.s1c_debt_watermark

     WHERE process_name =
           's1c_debt_for_working_capital';


    RAISE NOTICE 'Current watermark = %', v_watermark;


    /* ================================================================
       STEP 2.
       БЕРЁМ ИЗ LANDING ТОЛЬКО НОВЫЕ XML

       Например:

       LANDING = много миллионов XML

       watermark = вчера 08:00

       Берём только:
           dt_insert > вчера 08:00

       Эта таблица TEMP.
       После COMMIT исчезнет.
       ================================================================ */

    CREATE TEMP TABLE tmp_new_docs

    ON COMMIT DROP

    AS

    SELECT
        t.uuid,
        t.dt_insert,
        t.document_xml

    FROM landing."INPUT_DATA_FROM_SAPXI_IN" t

    WHERE t.flow_id = 'SI_WorkingCapital_AI'

      AND t.dt_insert > v_watermark

    DISTRIBUTED BY (uuid);


    GET DIAGNOSTICS v_row_cnt = ROW_COUNT;

    RAISE NOTICE 'New XML documents = %', v_row_cnt;


    /*
       Если новых документов нет —
       делать дальше нечего.
    */

    IF v_row_cnt = 0 THEN

        RAISE NOTICE 'No new documents';

        RETURN;

    END IF;


    /* ================================================================
       Определяем будущий watermark.

       Пока его НЕ записываем.

       Запишем только после успешного ODS.
       ================================================================ */

    SELECT MAX(dt_insert)

      INTO v_new_watermark

      FROM tmp_new_docs;


    /* ================================================================
       STEP 3.
       TMP_DELTA

       Вот та таблица, про которую ты спрашивал.

       ЗАЧЕМ ОНА:

       tmp_new_docs содержит XML.

       tmp_delta содержит УЖЕ РАСПАРСЕННЫЕ строки.

       XML разбираем один раз.

       Затем tmp_delta используем:
           1. для STG
           2. для ODS
           3. для мониторинга источников

       После COMMIT таблица исчезает.
       ================================================================ */


    CREATE TEMP TABLE tmp_delta
    (
        LIKE stg.s1c_debt_for_working_capital
    )

    ON COMMIT DROP

    DISTRIBUTED BY
    (
        unit_balance_mdm_code_1c
    );


    /* ================================================================
       STEP 4.
       XML PARSING

       Это практически твой существующий STG-script.

       Главное изменение:

       БЫЛО:
           landing
           +
           not exists к STG 350 млн

       СТАЛО:
           tmp_new_docs

       То есть парсим только новые XML.
       ================================================================ */


    INSERT INTO tmp_delta
    (
        sender,
        receiver,
        create_date_time,
        dt_report,
        database_code_1c,
        database_name_1c,
        unit_balance_mdm_code_1c,
        posting_uid_code_1c,
        invoice_registration_number,
        counterparty_role_name,
        debit_or_credit_name,
        accounting_document_descriprion_text,
        document_1c_code,
        responsibility_center_hfm_code,
        region_of_end_user_code,
        country_of_end_user_code,
        counterparty_mdm_code,
        counterparty_annual_turnover_amount,
        finish_goods_group_name,
        contract_number,
        paydox_document_url,
        contract_type_name,
        contract_amount,
        contract_currency_code,
        dt_contract_start,
        dt_contract_end,
        terms_of_payment_name,
        dt_overdue,
        debt_coverage_type_name,
        coverage_amount,
        coverage_currency_code,
        dt_debt,
        document_currency_code,
        debt_balance_document_currency_amount,
        debt_balance_rub_currency_amount,
        debt_balance_usd_currency_amount,
        debt_overdue_document_currency_amount,
        debt_overdue_rub_currency_amount,
        debt_overdue_usd_currency_amount,
        bank_receiver_name,
        receivable_claim_paydox_url,
        dt_receivable_claim,
        dt_claim_send_to_law_court,
        contract_supervisor_name,
        contract_supervisor_ad_login_code,
        general_ledger_account_code,
        general_ledger_account_name,
        dt_contract_registration,
        document_currency_amount,
        contract_supervisor_employee_1c_number,
        contract_supervisor_employee_sap_number,
        sales_market_code,
        sales_market_name,
        bad_debt_provision_amount,
        bad_debt_provision_currency_code,
        credit_limit_rub_currency_amount,
        dt_credit_limit_valid_to,
        paydox_credit_limit_url,
        insured_amount,
        insurance_currency_code,
        dt_insurance_valid_to,
        insurance_company_mdm_code,
        bank_guarantee_amount,
        bank_guarantee_currency_code,
        dt_bank_guarantee_valid_to,
        bank_guarantee_mdm_code,
        third_party_guarantee_amount,
        third_party_guarantee_currency_code,
        dt_third_party_guarantee_valid_to,
        third_party_guarantee_mdm_code,
        counterparty_out_of_mdm_text,
        uuid,
        dt_insert
    )

    WITH src AS
    (
        SELECT
            t.uuid,
            t.dt_insert,
            t.document_xml,

            (
                xpath(
                    'Header/Sender/text()',
                    document_xml
                )
            )[1]::varchar AS sender,

            (
                xpath(
                    'Header/Receiver/text()',
                    document_xml
                )
            )[1]::varchar AS receiver,

            (
                (
                    xpath(
                        'Header/CreatedDateTime/text()',
                        document_xml
                    )
                )[1]::varchar
            )::timestamp AS create_date_time

        FROM tmp_new_docs t
    ),

    items AS
    (
        SELECT
            uuid,
            dt_insert,
            sender,
            receiver,
            create_date_time,
            item_xml,

            (
                xpath(
                    'dtRp/text()',
                    item_xml
                )
            )[1]::varchar::date AS dt_report,

            (
                xpath(
                    'dbCd/text()',
                    item_xml
                )
            )[1]::varchar AS database_code_1c,

            (
                xpath(
                    'dbName/text()',
                    item_xml
                )
            )[1]::varchar AS database_name_1c,

            (
                xpath(
                    'unitMd/text()',
                    item_xml
                )
            )[1]::varchar AS unit_balance_mdm_code_1c

        FROM
        (
            SELECT
                uuid,
                dt_insert,
                sender,
                receiver,
                create_date_time,

                unnest(
                    xpath(
                        'Body/Item',
                        document_xml
                    )
                ) AS item_xml

            FROM src

        ) t
    ),

    entrys AS
    (
        SELECT
            uuid,
            dt_insert,
            sender,
            receiver,
            create_date_time,
            dt_report,
            database_code_1c,
            database_name_1c,
            unit_balance_mdm_code_1c,
            entry_xml

        FROM
        (
            SELECT
                *,

                unnest(
                    xpath(
                        'entry',
                        item_xml
                    )
                ) AS entry_xml

            FROM items

        ) t
    )

    SELECT
        sender,
        receiver,
        create_date_time,
        dt_report,
        database_code_1c,
        database_name_1c,
        unit_balance_mdm_code_1c,

        (xpath('id/text()', entry_xml))[1]::varchar,
        (xpath('invNum/text()', entry_xml))[1]::varchar,
        (xpath('ctrpRole/text()', entry_xml))[1]::varchar,
        (xpath('dcName/text()', entry_xml))[1]::varchar,
        (xpath('docDesc/text()', entry_xml))[1]::varchar,
        (xpath('docNum/text()', entry_xml))[1]::varchar,
        (xpath('respCntr/text()', entry_xml))[1]::varchar,
        (xpath('regEndUsr/text()', entry_xml))[1]::varchar,
        (xpath('cntEndUsr/text()', entry_xml))[1]::varchar,
        (xpath('ctrpMd/text()', entry_xml))[1]::varchar,

        (xpath('ctrpTurnov/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('fgGrName/text()', entry_xml))[1]::varchar,
        (xpath('ctNum/text()', entry_xml))[1]::varchar,
        (xpath('pdUrl/text()', entry_xml))[1]::varchar,
        (xpath('ctType/text()', entry_xml))[1]::varchar,

        (xpath('ctAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('ctCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dtCtStart/text()', entry_xml))[1]::varchar::date,
        (xpath('dtCtEnd/text()', entry_xml))[1]::varchar::date,

        (xpath('payTerm/text()', entry_xml))[1]::varchar,

        (xpath('dtOverdue/text()', entry_xml))[1]::varchar::date,

        (xpath('covTypeName/text()', entry_xml))[1]::varchar,

        (xpath('covAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('covCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dtDebt/text()', entry_xml))[1]::varchar::date,

        (xpath('docCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dbDocCurAmt/text()', entry_xml))[1]::varchar::decimal(20,6),
        (xpath('dbRubAmt/text()', entry_xml))[1]::varchar::decimal(20,6),
        (xpath('dbUsdAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('odDocCurAmt/text()', entry_xml))[1]::varchar::decimal(20,6),
        (xpath('odRubAmt/text()', entry_xml))[1]::varchar::decimal(20,6),
        (xpath('odUsdAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('bankRecName/text()', entry_xml))[1]::varchar,
        (xpath('recClmUrl/text()', entry_xml))[1]::varchar,

        (xpath('dtRecClm/text()', entry_xml))[1]::varchar::date,
        (xpath('dtLawCrt/text()', entry_xml))[1]::varchar::date,

        (xpath('ctrSupName/text()', entry_xml))[1]::varchar,
        (xpath('ctrSupId/text()', entry_xml))[1]::varchar,

        (xpath('glAccCd/text()', entry_xml))[1]::varchar,
        (xpath('glAccName/text()', entry_xml))[1]::varchar,

        (xpath('dtCtReg/text()', entry_xml))[1]::varchar::date,

        (xpath('docCurAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('ctrSup1c/text()', entry_xml))[1]::varchar,
        (xpath('ctrSupSap/text()', entry_xml))[1]::varchar,

        (xpath('smCode/text()', entry_xml))[1]::varchar,
        (xpath('smName/text()', entry_xml))[1]::varchar,

        (xpath('badDbAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('badDbCurCd/text()', entry_xml))[1]::varchar,

        (xpath('crLimRubAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('dtCrLimEnd/text()', entry_xml))[1]::varchar::date,

        (xpath('crUrlPd/text()', entry_xml))[1]::varchar,

        (xpath('insAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('insCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dtInsEnd/text()', entry_xml))[1]::varchar::date,

        (xpath('insCoMd/text()', entry_xml))[1]::varchar,

        (xpath('bgAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('bgCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dtBgEnd/text()', entry_xml))[1]::varchar::date,

        (xpath('bgMd/text()', entry_xml))[1]::varchar,

        (xpath('thpgAmt/text()', entry_xml))[1]::varchar::decimal(20,6),

        (xpath('thpgCurCd/text()', entry_xml))[1]::varchar,

        (xpath('dtThpgEnd/text()', entry_xml))[1]::varchar::date,

        (xpath('thpgMd/text()', entry_xml))[1]::varchar,

        (xpath('ctrpName/text()', entry_xml))[1]::varchar,

        uuid,
        dt_insert

    FROM entrys;


    GET DIAGNOSTICS v_row_cnt = ROW_COUNT;

    RAISE NOTICE 'Parsed rows tmp_delta = %', v_row_cnt;


    /* ================================================================
       STEP 5.
       ДОПИСЫВАЕМ ИСТОРИЮ В STG

       STG остаётся полной историей.

       Но дальше для расчёта ODS мы её НЕ ЧИТАЕМ.
       ================================================================ */

    INSERT INTO stg.s1c_debt_for_working_capital
    SELECT *
    FROM tmp_delta;


    GET DIAGNOSTICS v_row_cnt = ROW_COUNT;

    RAISE NOTICE 'Inserted into STG = %', v_row_cnt;


    /* ================================================================
       STEP 6.
       ОПРЕДЕЛЯЕМ, КАКИЕ SNAPSHOT ИЗМЕНИЛИСЬ

       Сейчас у тебя min(dt_report) / max(dt_report).

       Теперь конкретный список:

       BE01 + 07.08
       BE02 + 07.08
       BE15 + 05.03
       ================================================================ */


    CREATE TEMP TABLE tmp_changed_snapshots

    ON COMMIT DROP

    AS

    SELECT DISTINCT
        unit_balance_mdm_code_1c,
        dt_report

    FROM tmp_delta

    DISTRIBUTED BY
    (
        unit_balance_mdm_code_1c
    );


    /* ================================================================
       STEP 7.
       ВЫБИРАЕМ ПОСЛЕДНЮЮ ПОСТАВКУ SNAPSHOT

       Сохраняем твою текущую логику:

       dense_rank(
           partition by BE, dt_report
           order by dt_insert desc
       )

       Но выполняем её только по сегодняшним ~2 млн.
       ================================================================ */


    CREATE TEMP TABLE tmp_latest_snapshot

    ON COMMIT DROP

    AS

    SELECT *

    FROM
    (
        SELECT
            d.*,

            DENSE_RANK() OVER
            (
                PARTITION BY
                    d.unit_balance_mdm_code_1c,
                    d.dt_report

                ORDER BY
                    d.dt_insert DESC

            ) AS snapshot_rn

        FROM tmp_delta d

    ) s

    WHERE s.snapshot_rn = 1

    DISTRIBUTED BY
    (
        unit_balance_mdm_code_1c
    );


    /* ================================================================
       STEP 8.
       ГОТОВИМ ДАННЫЕ ДЛЯ ODS

       Здесь сохраняются текущие преобразования:
       - даты 1900..2299
       - Paydox URL
       - dedup posting
       ================================================================ */


    CREATE TEMP TABLE tmp_ods_delta

    ON COMMIT DROP

    AS

    WITH prepared AS
    (
        SELECT DISTINCT

            c.sender,
            c.receiver,
            c.create_date_time,

            CASE
                WHEN c.dt_report < DATE '1900-01-01'
                    THEN DATE '1900-01-01'

                WHEN c.dt_report > DATE '2299-12-31'
                    THEN DATE '2299-12-31'

                ELSE c.dt_report
            END AS dt_report,

            c.database_code_1c,
            c.database_name_1c,
            c.unit_balance_mdm_code_1c,
            c.posting_uid_code_1c,
            c.invoice_registration_number,
            c.counterparty_role_name,
            c.debit_or_credit_name,
            c.accounting_document_descriprion_text,
            c.document_1c_code,
            c.responsibility_center_hfm_code,
            c.region_of_end_user_code,
            c.country_of_end_user_code,
            c.counterparty_mdm_code,
            c.counterparty_annual_turnover_amount,
            c.finish_goods_group_name,
            c.contract_number,

            CASE
                WHEN c.paydox_document_url LIKE '%https://paydox%'
                  OR c.paydox_document_url LIKE '%http://paydox%'
                THEN c.paydox_document_url
                ELSE NULL
            END AS paydox_document_url,

            c.contract_type_name,
            c.contract_amount,
            c.contract_currency_code,

            CASE
                WHEN c.dt_contract_start < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_contract_start > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_contract_start
            END AS dt_contract_start,

            CASE
                WHEN c.dt_contract_end < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_contract_end > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_contract_end
            END AS dt_contract_end,

            c.terms_of_payment_name,

            CASE
                WHEN c.dt_overdue < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_overdue > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_overdue
            END AS dt_overdue,

            c.debt_coverage_type_name,
            c.coverage_amount,
            c.coverage_currency_code,

            CASE
                WHEN c.dt_debt < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_debt > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_debt
            END AS dt_debt,

            c.document_currency_code,
            c.debt_balance_document_currency_amount,
            c.debt_balance_rub_currency_amount,
            c.debt_balance_usd_currency_amount,
            c.debt_overdue_document_currency_amount,
            c.debt_overdue_rub_currency_amount,
            c.debt_overdue_usd_currency_amount,
            c.bank_receiver_name,
            c.receivable_claim_paydox_url,

            CASE
                WHEN c.dt_receivable_claim < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_receivable_claim > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_receivable_claim
            END AS dt_receivable_claim,

            CASE
                WHEN c.dt_claim_send_to_law_court < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_claim_send_to_law_court > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_claim_send_to_law_court
            END AS dt_claim_send_to_law_court,

            c.contract_supervisor_name,
            c.contract_supervisor_ad_login_code,
            c.general_ledger_account_code,
            c.general_ledger_account_name,

            CASE
                WHEN c.dt_contract_registration < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_contract_registration > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_contract_registration
            END AS dt_contract_registration,

            c.document_currency_amount,
            c.contract_supervisor_employee_1c_number,
            c.contract_supervisor_employee_sap_number,
            c.sales_market_code,
            c.sales_market_name,
            c.bad_debt_provision_amount,
            c.bad_debt_provision_currency_code,
            c.credit_limit_rub_currency_amount,

            CASE
                WHEN c.dt_credit_limit_valid_to < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_credit_limit_valid_to > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_credit_limit_valid_to
            END AS dt_credit_limit_valid_to,

            c.paydox_credit_limit_url,
            c.insured_amount,
            c.insurance_currency_code,

            CASE
                WHEN c.dt_insurance_valid_to < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_insurance_valid_to > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_insurance_valid_to
            END AS dt_insurance_valid_to,

            c.insurance_company_mdm_code,
            c.bank_guarantee_amount,
            c.bank_guarantee_currency_code,

            CASE
                WHEN c.dt_bank_guarantee_valid_to < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_bank_guarantee_valid_to > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_bank_guarantee_valid_to
            END AS dt_bank_guarantee_valid_to,

            c.bank_guarantee_mdm_code,
            c.third_party_guarantee_amount,
            c.third_party_guarantee_currency_code,

            CASE
                WHEN c.dt_third_party_guarantee_valid_to < DATE '1900-01-01'
                    THEN DATE '1900-01-01'
                WHEN c.dt_third_party_guarantee_valid_to > DATE '2299-12-31'
                    THEN DATE '2299-12-31'
                ELSE c.dt_third_party_guarantee_valid_to
            END AS dt_third_party_guarantee_valid_to,

            c.third_party_guarantee_mdm_code,
            c.counterparty_out_of_mdm_text,
            c.dt_insert

        FROM tmp_latest_snapshot c

        WHERE c.dt_report <> DATE '1900-01-01'
    ),

    dedup AS
    (
        SELECT
            p.*,

            ROW_NUMBER() OVER
            (
                PARTITION BY
                    p.unit_balance_mdm_code_1c,
                    p.posting_uid_code_1c,
                    p.dt_report

                ORDER BY
                    p.create_date_time DESC
            ) AS rn

        FROM prepared p
    )

    SELECT
        sender,
        receiver,
        create_date_time,
        dt_report,
        database_code_1c,
        database_name_1c,
        unit_balance_mdm_code_1c,
        posting_uid_code_1c,
        invoice_registration_number,
        counterparty_role_name,
        debit_or_credit_name,
        accounting_document_descriprion_text,
        document_1c_code,
        responsibility_center_hfm_code,
        region_of_end_user_code,
        country_of_end_user_code,
        counterparty_mdm_code,
        counterparty_annual_turnover_amount,
        finish_goods_group_name,
        contract_number,
        paydox_document_url,
        contract_type_name,
        contract_amount,
        contract_currency_code,
        dt_contract_start,
        dt_contract_end,
        terms_of_payment_name,
        dt_overdue,
        debt_coverage_type_name,
        coverage_amount,
        coverage_currency_code,
        dt_debt,
        document_currency_code,
        debt_balance_document_currency_amount,
        debt_balance_rub_currency_amount,
        debt_balance_usd_currency_amount,
        debt_overdue_document_currency_amount,
        debt_overdue_rub_currency_amount,
        debt_overdue_usd_currency_amount,
        bank_receiver_name,
        receivable_claim_paydox_url,
        dt_receivable_claim,
        dt_claim_send_to_law_court,
        contract_supervisor_name,
        contract_supervisor_ad_login_code,
        general_ledger_account_code,
        general_ledger_account_name,
        dt_contract_registration,
        document_currency_amount,
        contract_supervisor_employee_1c_number,
        contract_supervisor_employee_sap_number,
        sales_market_code,
        sales_market_name,
        bad_debt_provision_amount,
        bad_debt_provision_currency_code,
        credit_limit_rub_currency_amount,
        dt_credit_limit_valid_to,
        paydox_credit_limit_url,
        insured_amount,
        insurance_currency_code,
        dt_insurance_valid_to,
        insurance_company_mdm_code,
        bank_guarantee_amount,
        bank_guarantee_currency_code,
        dt_bank_guarantee_valid_to,
        bank_guarantee_mdm_code,
        third_party_guarantee_amount,
        third_party_guarantee_currency_code,
        dt_third_party_guarantee_valid_to,
        third_party_guarantee_mdm_code,
        counterparty_out_of_mdm_text,
        dt_insert

    FROM dedup

    WHERE rn = 1

    DISTRIBUTED BY
    (
        unit_balance_mdm_code_1c
    );


    /* ================================================================
       STEP 9.
       УДАЛЯЕМ ИЗ ODS ТОЛЬКО ИЗМЕНИВШИЕСЯ SNAPSHOT

       КЛЮЧЕВОЕ ОТЛИЧИЕ.

       Больше НЕТ:

           dt_report between min and max

       Есть конкретные:
           BE + dt_report
       ================================================================ */


    DELETE FROM ods.account_debt_for_working_capital_1c o

    USING tmp_changed_snapshots c

    WHERE
        o.unit_balance_mdm_code_1c =
        c.unit_balance_mdm_code_1c

    AND o.dt_report =
        c.dt_report;


    GET DIAGNOSTICS v_row_cnt = ROW_COUNT;

    RAISE NOTICE 'Deleted ODS rows = %', v_row_cnt;


    /* ================================================================
       STEP 10.
       ВСТАВЛЯЕМ НОВЫЙ SNAPSHOT
       ================================================================ */


    INSERT INTO ods.account_debt_for_working_capital_1c
    (
        sender,
        receiver,
        create_date_time,
        dt_report,
        database_code_1c,
        database_name_1c,
        unit_balance_mdm_code_1c,
        posting_uid_code_1c,
        invoice_registration_number,
        counterparty_role_name,
        debit_or_credit_name,
        accounting_document_descriprion_text,
        document_1c_code,
        responsibility_center_hfm_code,
        region_of_end_user_code,
        country_of_end_user_code,
        counterparty_mdm_code,
        counterparty_annual_turnover_amount,
        finish_goods_group_name,
        contract_number,
        paydox_document_url,
        contract_type_name,
        contract_amount,
        contract_currency_code,
        dt_contract_start,
        dt_contract_end,
        terms_of_payment_name,
        dt_overdue,
        debt_coverage_type_name,
        coverage_amount,
        coverage_currency_code,
        dt_debt,
        document_currency_code,
        debt_balance_document_currency_amount,
        debt_balance_rub_currency_amount,
        debt_balance_usd_currency_amount,
        debt_overdue_document_currency_amount,
        debt_overdue_rub_currency_amount,
        debt_overdue_usd_currency_amount,
        bank_receiver_name,
        receivable_claim_paydox_url,
        dt_receivable_claim,
        dt_claim_send_to_law_court,
        contract_supervisor_name,
        contract_supervisor_ad_login_code,
        general_ledger_account_code,
        general_ledger_account_name,
        dt_contract_registration,
        document_currency_amount,
        contract_supervisor_employee_1c_number,
        contract_supervisor_employee_sap_number,
        sales_market_code,
        sales_market_name,
        bad_debt_provision_amount,
        bad_debt_provision_currency_code,
        credit_limit_rub_currency_amount,
        dt_credit_limit_valid_to,
        paydox_credit_limit_url,
        insured_amount,
        insurance_currency_code,
        dt_insurance_valid_to,
        insurance_company_mdm_code,
        bank_guarantee_amount,
        bank_guarantee_currency_code,
        dt_bank_guarantee_valid_to,
        bank_guarantee_mdm_code,
        third_party_guarantee_amount,
        third_party_guarantee_currency_code,
        dt_third_party_guarantee_valid_to,
        third_party_guarantee_mdm_code,
        counterparty_out_of_mdm_text,
        dt_insert
    )

    SELECT *
    FROM tmp_ods_delta;


    GET DIAGNOSTICS v_row_cnt = ROW_COUNT;

    RAISE NOTICE 'Inserted ODS rows = %', v_row_cnt;


    /* ================================================================
       STEP 11.
       SOURCE MONITORING

       Определяем последнее состояние источников,
       которые реально приехали в этом запуске.
       ================================================================ */


    CREATE TEMP TABLE tmp_source_state

    ON COMMIT DROP

    AS

    SELECT DISTINCT ON
    (
        database_code_1c,
        unit_balance_mdm_code_1c
    )

        database_code_1c,
        unit_balance_mdm_code_1c,
        dt_report,
        dt_insert,
        create_date_time

    FROM tmp_delta

    ORDER BY
        database_code_1c,
        unit_balance_mdm_code_1c,
        dt_insert DESC,
        create_date_time DESC

    DISTRIBUTED BY
    (
        database_code_1c,
        unit_balance_mdm_code_1c
    );


    /* ================================================================
       Сначала UPDATE существующих источников
       ================================================================ */

    UPDATE meta.s1c_debt_source_control t

       SET last_dt_report =
               s.dt_report,

           last_dt_insert =
               s.dt_insert,

           last_create_date_time =
               s.create_date_time,

           updated_dttm =
               clock_timestamp()

      FROM tmp_source_state s

     WHERE t.database_code_1c =
           s.database_code_1c

       AND t.unit_balance_mdm_code_1c =
           s.unit_balance_mdm_code_1c;


    /* ================================================================
       Добавляем совершенно новые 1С/БЕ
       ================================================================ */

    INSERT INTO meta.s1c_debt_source_control
    (
        database_code_1c,
        unit_balance_mdm_code_1c,
        is_active,
        stale_after_hours,
        last_dt_report,
        last_dt_insert,
        last_create_date_time,
        updated_dttm
    )

    SELECT
        s.database_code_1c,
        s.unit_balance_mdm_code_1c,

        true,

        36,

        s.dt_report,
        s.dt_insert,
        s.create_date_time,

        clock_timestamp()

    FROM tmp_source_state s

    WHERE NOT EXISTS
    (
        SELECT 1

        FROM meta.s1c_debt_source_control t

        WHERE t.database_code_1c =
              s.database_code_1c

          AND t.unit_balance_mdm_code_1c =
              s.unit_balance_mdm_code_1c
    );


    /* ================================================================
       STEP 12.
       WATERMARK ОБНОВЛЯЕМ ТОЛЬКО В САМОМ КОНЦЕ

       Если выше произошла ошибка —
       транзакция откатится,
       watermark останется прежним.

       Следующий запуск снова возьмёт эту поставку.
       ================================================================ */


    UPDATE meta.s1c_debt_watermark

       SET last_dt_insert =
               v_new_watermark,

           updated_dttm =
               clock_timestamp()

     WHERE process_name =
           's1c_debt_for_working_capital';


    RAISE NOTICE
        'New watermark = %',
        v_new_watermark;


    /* ================================================================
       ANALYZE

       ODS имеет смысл обновить.

       STG 350+ млн я бы НЕ ANALYZE-ил после каждого ежедневного append.
       ================================================================ */

    ANALYZE ods.account_debt_for_working_capital_1c;


    RAISE NOTICE
        'FINISH. Runtime = %',
        clock_timestamp() - v_start_dttm;

END;

$$;


/* =====================================================================
   =====================================================================
   ОТДЕЛЬНЫЙ ЗАПРОС МОНИТОРИНГА

   Можно запускать после DAG или отдельным мониторингом.

   Покажет 1С/БЕ, которые давно ничего не отправляли.
   =====================================================================
   ===================================================================== */


SELECT
    database_code_1c,
    unit_balance_mdm_code_1c,

    last_dt_report,
    last_dt_insert,

    stale_after_hours,

    clock_timestamp() - last_dt_insert
        AS current_delay

FROM meta.s1c_debt_source_control

WHERE is_active = true

  AND last_dt_insert <
      clock_timestamp()
      - stale_after_hours * interval '1 hour'

ORDER BY last_dt_insert;


/* =====================================================================
   ОДНОРАЗОВАЯ ПРОВЕРКА КЛЮЧА SNAPSHOT

   ОБЯЗАТЕЛЬНО ВЫПОЛНИТЬ ПЕРЕД ПЕРЕХОДОМ.

   Если запрос вернул 0 строк:
       текущий ключ BE + dt_report нормальный.

   Если вернул строки:
       надо отдельно разбирать ключ,
       возможно добавить database_code_1c.
   ===================================================================== */


SELECT
    unit_balance_mdm_code_1c,
    dt_report,

    COUNT(
        DISTINCT database_code_1c
    ) AS cnt_database

FROM stg.s1c_debt_for_working_capital

GROUP BY
    unit_balance_mdm_code_1c,
    dt_report

HAVING COUNT(
    DISTINCT database_code_1c
) > 1

LIMIT 100;
