DROP TABLE IF EXISTS for_1079_all;

CREATE TEMP TABLE for_1079_all
DISTRIBUTED BY (rn)
AS
-- РФ-контейнеры
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    NULL::numeric        AS month_difference_2,
    NULL::varchar(50)    AS special
FROM for_1079_rf_cont

UNION ALL
-- РФ-балка 1
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    NULL::numeric        AS month_difference_2,
    NULL::varchar(50)    AS special
FROM for_1079_rf_balk_1

UNION ALL
-- РФ-балка 2
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    NULL::numeric        AS month_difference_2,
    NULL::varchar(50)    AS special
FROM for_1079_rf_balk_2

UNION ALL
-- EURO (тут уже есть month_difference_2 и special)
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    month_difference_2,
    special
FROM for_1079_euro

UNION ALL
-- СВХ
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    NULL::numeric        AS month_difference_2,
    NULL::varchar(50)    AS special
FROM for_1079_svh

UNION ALL
-- ASIA
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    NULL::numeric        AS month_difference_2,
    NULL::varchar(50)    AS special
FROM for_1079_asia;



SELECT
    base_2.sales_bundle_code,
    base_2.plant_producer_delivery_code,
    base_2.plant_producer_delivery_position_code,
    base_2.initial_delivery_code,
    base_2.material_code,
    base_2.dt_shipment,
    base_2.railcar_code,
    base_2.transport_bill_code,
    base_2.uni,
    base_2.material_shape_code,
    base_2.port_of_discharge_code,
    base_2.dt_arrival_to_russian_port,
    base_2.dt_storage_end_in_release,
    base_2.transport_type_code,
    base_2.etsng_code,
    base_2.dt_report,
    base_2.sales_bundle_net_weight,
    base_2.sales_bundle_gross_weight,
    base_2.batch_code,
    base_2.dt_arrival,
    base_2.receiving_plant_code,
    base_2.sales_delivery_code,
    base_2.transportation_inbound_delivery_code,
    base_2.delivery_for_storage_calculation_code,
    base_2.transportation_inbound_delivery_position_code,
    base_2.delivery_position_for_storage_calculation_code,
    base_2.forwarder_code,
    base_2.storage_cost_calculation_type_name,
    base_2.transport_departure_hub_code,
    base_2.transport_destination_hub_code,
    base_2.transportation_outbound_delivery_code,
    base_2.transportation_outbound_delivery_position_code,
    base_2.is_final_transportation_stage_code,
    base_2.transportation_stage_code,
    base_2.storage_calculation_bundle_quantity,
    base_2.warehouse_code,
    base_2.delivery_in_final_release_code,
    base_2.dt_final_release,
    base_2.bill_of_lading_code,
    base_2.bill_of_lading_number,
    base_2.dt_bill_of_lading,
    base_2.incoterms_code,
    base_2.storage_cost_special_calculation_type_code,
    base_2.warehouse_storage_area_code,
    base_2.plant_code,
    base_2.unit_balance_code,
    base_2.creditor_in_shipment_instruction_code,
    base_2.dt_storage_start,
    base_2.dt_shipped_from_warehouse,
    base_2.dt_storage_end,
    base_2.service_number,
    base_2.storage_duration_total_calendar_days,
    base_2.shipment_instruction_number,
    base_2.dt_for_price_search_in_purchase_contract,
    base_2.is_stored_in_russian_port_code,
    base_2.remote_warehouse_code,
    base_2.dt_transportation_stage_start_p,
    base_2.dt_transportation_stage_start_r,

    -- вместо длинного COALESCE по 6 таблицам:
    f1079.purchase_contract_code                                    AS purchase_contract_code,
    f1079.position_line_item_code                                   AS purchase_contract_position_code,
    f1079.konws                                                     AS purchase_contract_position_currency_code,
    f1079.service_code                                              AS service_code,
    f1079.price_uom_code                                            AS purchase_contract_position_uom_code,

    GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)       AS storage_duration_free_by_contract_calendar_days,

    CASE
        WHEN COALESCE(base_2.storage_cost_special_calculation_type_code,'fdgdfg') <> '6'
            THEN LEAST(
                    base_2.storage_duration_total_calendar_days,
                    GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
                 )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND calendar_start.calendar_year = calendar_end.calendar_year
         AND calendar_start.calendar_month = calendar_end.calendar_month
         AND base_2.storage_duration_total_calendar_days < 15
            THEN 0

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND calendar_start.calendar_year = calendar_end.calendar_year
         AND calendar_start.calendar_month = calendar_end.calendar_month
         AND base_2.storage_duration_total_calendar_days < 30
            THEN 1

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND calendar_start.calendar_year = calendar_end.calendar_year
         AND calendar_start.calendar_month = calendar_end.calendar_month
         AND base_2.storage_duration_total_calendar_days > 29
            THEN 3

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) < 15
         AND calendar_end.calendar_day <> calendar_end.days_in_month
            THEN 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month) - 1
            )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) < 30
         AND calendar_end.calendar_day <> calendar_end.days_in_month
            THEN 1 + 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month) - 1
            )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) > 29
         AND calendar_end.calendar_day <> calendar_end.days_in_month
            THEN 3 + 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month) - 1
            )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) < 15
         AND calendar_end.calendar_day = calendar_end.days_in_month
            THEN 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month)
            )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) < 30
         AND calendar_end.calendar_day = calendar_end.days_in_month
            THEN 1 + 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month)
            )

        WHEN base_2.storage_cost_special_calculation_type_code = '6'
         AND (calendar_start.days_in_month - calendar_start.calendar_day + 1) > 29
         AND calendar_end.calendar_day = calendar_end.days_in_month
            THEN 3 + 3 * (
                (calendar_end.calendar_year - calendar_start.calendar_year) * 12 +
                (calendar_end.calendar_month - calendar_start.calendar_month)
            )
    END AS storage_duration_free_by_delivery_calendar_days,

    -- euro-специфика (будет NULL для остальных типов)
    f1079.month_difference_2                                         AS for_1079_euro_month_difference_2,
    f1079.special                                                    AS for_1079_euro_special,

    f1079.currency_dogovor                                           AS currency_dogovor,
    for_1087.price_scale_limit_from_quantity                         AS for_1087_price_scale_limit_from_quantity,
    base_2.weight_for_1090_razdel,
    base_2.weight_for_1090_umno,
    f1079.datbi                                                      AS datbi,
    f1079.datab                                                      AS datab,
    base_2.rn
FROM dm_calc.storage_sales_bundles_weight AS base_2
LEFT JOIN dict_dds.calendar AS calendar_start
  ON base_2.storage_cost_special_calculation_type_code = '6'
 AND calendar_start.dt = base_2.dt_storage_start
LEFT JOIN dict_dds.calendar AS calendar_end
  ON base_2.storage_cost_special_calculation_type_code = '6'
 AND calendar_end.dt = base_2.dt_storage_end
LEFT JOIN for_1079_all AS f1079
  ON f1079.rn = base_2.rn
LEFT JOIN for_1087
  ON  for_1087.purchase_contract_code          = f1079.purchase_contract_code
 AND for_1087.purchase_contract_position_code  = f1079.position_line_item_code
 AND for_1087.dt_price_condition_valid_to      = f1079.datbi
 AND for_1087.dt_price_condition_valid_from    = f1079.datab;
