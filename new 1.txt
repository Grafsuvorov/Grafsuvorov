CREATE TEMP TABLE storage_europe_prep AS
SELECT 
    sw.rn,
    sw.warehouse_code,
    sw.warehouse_storage_area_code,
    sw.unit_balance_code,
    sw.forwarder_code,
    sw.plant_code,
    sw.transport_type_code,
    sw.etsng_code,
    sw.port_of_discharge_code,
    sw.material_shape_code,
    sw.dt_for_price_search_in_purchase_contract,
    sw.storage_duration_total_calendar_days,
    sw.storage_cost_special_calculation_type_code,
    sum(sw.sales_bundle_gross_weight)
        OVER (PARTITION BY sw.warehouse_code, date_trunc('month', sw.dt_storage_end)) 
            AS warehouse_weight_month_end,
    CASE 
        WHEN extract(day FROM sw.dt_storage_end) < extract(day FROM sw.dt_storage_start)
            THEN (extract(year FROM sw.dt_storage_end) - extract(year FROM sw.dt_storage_start)) * 12 +
                 (extract(month FROM sw.dt_storage_end) - extract(month FROM sw.dt_storage_start))
        ELSE
             (extract(year FROM sw.dt_storage_end) - extract(year FROM sw.dt_storage_start)) * 12 +
             (extract(month FROM sw.dt_storage_end) - extract(month FROM sw.dt_storage_start)) + 1
    END AS month_difference_2,
    CASE
        WHEN sw.storage_cost_special_calculation_type_code = '4'
             THEN ceil(sw.storage_duration_total_calendar_days / 7.0)
        WHEN sw.storage_cost_special_calculation_type_code = '5'
             THEN ceil(sw.storage_duration_total_calendar_days / 30.0)
    END AS special
FROM dm_calc.storage_sales_bundles_weight sw
WHERE sw.storage_cost_calculation_type_name = 'Логика Европа'
DISTRIBUTED BY (rn);

CREATE TEMP TABLE for_1079_base_norm AS
WITH base AS (
    SELECT 
        purchase_contract_code,
        position_line_item_code,
        bukrs,
        lifnr,
        zzport,
        werks,
        zzsrvpos,
        zzforma,
        zzet_tarif,
        sdabw,
        zzterminal,
        vzzzone,
        konws,
        service_code,
        price_uom_code,
        netpr,
        kbetr,
        order_euro_svh,
        currency_dogovor,
        datab,
        datbi,
        purchase_document_subtype_code,
        tmp_1,
        tmp_2,
        LEAD(datab) OVER (
            PARTITION BY purchase_contract_code, position_line_item_code
            ORDER BY datab
        ) AS next_datab
    FROM for_1079_base
)
SELECT *,
       COALESCE(next_datab - INTERVAL '1 day', datbi) AS effective_datbi
FROM base
WHERE purchase_document_subtype_code = 'ZDTL'
DISTRIBUTED REPLICATED;

CREATE TEMP TABLE europe_contract_choice AS
SELECT DISTINCT ON (p.rn)
    p.rn,
    b.purchase_contract_code,
    b.position_line_item_code,
    b.konws,
    b.service_code,
    b.price_uom_code,
    p.warehouse_weight_month_end,
    p.month_difference_2,
    b.netpr,
    p.special,
    b.currency_dogovor,
    b.datab,
    b.effective_datbi
FROM storage_europe_prep p
JOIN for_1079_base_norm b
     ON b.bukrs = p.unit_balance_code
    AND b.lifnr = p.forwarder_code
    AND (b.zzport = p.port_of_discharge_code OR b.zzport IS NULL)
    AND b.werks = p.plant_code
    AND (b.zzforma = p.material_shape_code OR b.zzforma IS NULL)
    AND (b.zzet_tarif = p.etsng_code OR b.zzet_tarif IS NULL)
    AND (b.sdabw = p.transport_type_code OR b.sdabw IS NULL)
    AND (b.zzterminal = p.warehouse_code OR b.zzterminal IS NULL)
    AND (b.vzzzone = p.warehouse_storage_area_code OR b.vzzzone IS NULL)
    AND p.dt_for_price_search_in_purchase_contract BETWEEN b.datab AND b.effective_datbi
    AND (
         (b.zzsrvpos = 'S49' AND EXISTS (SELECT 1 FROM dict_dds.settings_and_parameters_sap s1 
                                         WHERE s1.abap_program_code='/RUSAL/ZLE112M'
                                           AND s1.parameter_code='S_LIFNRA'
                                           AND s1.range_low_value = p.forwarder_code))
         OR
         (b.tmp_1 IS NOT NULL)
         OR
         (b.tmp_2 IS NOT NULL)
        )
ORDER BY p.rn,
         b.order_euro_svh,
         b.kopos,
         b.kbetr;
