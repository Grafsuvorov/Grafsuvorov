select format(
      'drop table if exists %I.%I;',
      table_schema,
      table_name || '_backup_' || to_char(current_date, 'YYYY_MM_DD')
  ) as drop_sql
  from tech_etl.tables_meta
  where entity_id = 35
    and table_schema = 'dm'
  order by table_name;

  Если нужна не текущая дата, а конкретная:

  select format(
      'drop table if exists %I.%I;',
      table_schema,
      table_name || '_backup_2026_06_02'
  ) as drop_sql
  from tech_etl.tables_meta
  where entity_id = 35
    and table_schema = 'dm'
  order by table_name;

  Если хотите сразу одной строкой в один большой скрипт:

  select string_agg(
      format(
          'drop table if exists %I.%I;',
          table_schema,
          table_name || '_backup_2026_06_02'
      ),
      E'\n'
      order by table_name
  ) as drop_script
  from tech_etl.tables_meta
  where entity_id = 35
    and table_schema = 'dm';
