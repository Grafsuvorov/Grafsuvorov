/* ---------------------------------------------------------
   ПЕРИОД — последние 12 месяцев
--------------------------------------------------------- */

DROP TABLE IF EXISTS date_for_vitr_tmp;

CREATE TEMP TABLE date_for_vitr_tmp
ON COMMIT DROP AS
SELECT 
    dt::date AS dt_balance
FROM generate_series(
        current_date - interval '1 year',
        current_date,
        interval '1 day'
     ) AS gs(dt)
DISTRIBUTED REPLICATED;



/* ---------------------------------------------------------
   СПРАВОЧНИКИ — оптимизированные для GP
--------------------------------------------------------- */


-- ADRC – убираем дубли
DROP TABLE IF EXISTS adrc;
CREATE TEMP TABLE adrc
ON COMMIT DROP AS
SELECT
    address_code,
    MIN(country_code) AS country_code
FROM dict_dds.address
GROUP BY address_code
DISTRIBUTED REPLICATED;



-- Азия — контрагенты
DROP TABLE IF EXISTS asia;
CREATE TEMP TABLE asia
ON COMMIT DROP AS
SELECT DISTINCT counterparty_code
FROM dict_dds.transportation_storage_calculation_method
DISTRIBUTED REPLICATED;



-- Европа 1 — сервисы EU%
DROP TABLE IF EXISTS euro_1;
CREATE TEMP TABLE euro_1
ON COMMIT DROP AS
SELECT DISTINCT counterparty_code
FROM dict_dds.transportation_storage_calculation_method
WHERE service_code LIKE 'EU%'
DISTRIBUTED REPLICATED;



-- Европа 2 — все контрагенты
DROP TABLE IF EXISTS euro_2;
CREATE TEMP TABLE euro_2
ON COMMIT DROP AS
SELECT DISTINCT counterparty_code
FROM dict_dds.transportation_storage_calculation_method
DISTRIBUTED REPLICATED;



-- Europa — таблица логики
DROP TABLE IF EXISTS euro_log;
CREATE TEMP TABLE euro_log
ON COMMIT DROP AS
SELECT DISTINCT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code ILIKE '/RUSAL/ZLE112m'
  AND case_code = 'PERET_FRN'
  AND parameter_code = 'S_LIFNRA'
  AND range_low_value IS NOT NULL
DISTRIBUTED REPLICATED;



-- СВХ
DROP TABLE IF EXISTS svh;
CREATE TEMP TABLE svh
ON COMMIT DROP AS
SELECT DISTINCT counterparty_code
FROM dict_dds.transportation_general_calculation_method
WHERE general_calculation_method_code IN ('16', '04')
DISTRIBUTED REPLICATED;



-- Азия — логика параметров
DROP TABLE IF EXISTS asia_log;
CREATE TEMP TABLE asia_log
ON COMMIT DROP AS
SELECT DISTINCT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code ILIKE '/RUSAL/ZLE112m'
  AND case_code = 'PERET_FRN'
  AND parameter_code = 'LIFNR_FR'
  AND range_low_value IS NOT NULL
DISTRIBUTED REPLICATED;



-- LE.001035 — количество пакетов
DROP TABLE IF EXISTS for_LE_001035;
CREATE TEMP TABLE for_LE_001035
ON COMMIT DROP AS
SELECT
    delivery_code,
    batch_code,
    COUNT(*) AS cnt
FROM dds.sales_bundle_and_delivery_relationship
WHERE is_deleted_code IS NULL
GROUP BY delivery_code, batch_code
DISTRIBUTED BY (delivery_code);



-- DEDOPO_2 — уникальная связь delivery → sales_document
DROP TABLE IF EXISTS dedopo_2;
CREATE TEMP TABLE dedopo_2
ON COMMIT DROP AS
SELECT DISTINCT delivery_code, sales_document_code
FROM dds.delivery_document_position
WHERE sales_document_code IS NOT NULL
DISTRIBUTED BY (delivery_code);



-- FOR_1055 — дата окончания бесплатного хранения
DROP TABLE IF EXISTS for_1055;
CREATE TEMP TABLE for_1055
ON COMMIT DROP AS
SELECT
    vbss.delivery_code AS vbeln,
    MIN(vbsk.dt_end_of_free_storage) AS dt_end_of_free_storage
FROM dds.logistics_document_and_delivery_relationship vbss
LEFT JOIN dds.release vbsk
       ON vbsk.release_code = vbss.logistics_document_code
      AND vbsk.release_type_code = 'X'
LEFT JOIN dict_dds.settings_and_parameters_sap frel_kod
       ON vbsk.release_template_code = frel_kod.range_low_value
WHERE frel_kod.abap_program_code = '/RUSAL/SD2921M'
  AND frel_kod.parameter_code = 'FREL_KOD'
  AND frel_kod.range_sign_code = 'I'
  AND frel_kod.range_option_code = 'EQ'
  AND vbsk.release_code IS NOT NULL
  AND vbsk.dt_end_of_free_storage IS NOT NULL
GROUP BY vbss.delivery_code
DISTRIBUTED REPLICATED;

/* ==========================================================
   LAKE — оптимизированный под Greenplum
   Выбираем уникальные sd_sales_main_scm по sales_bundle_code
========================================================== */

DROP TABLE IF EXISTS lake;

CREATE TEMP TABLE lake
ON COMMIT DROP AS
SELECT DISTINCT ON (sd4798.sales_bundle_code)
    ssms.delivery_number_sales,
    ssms.delivery_number_of_producer_plant      AS SD_000003,
    ssms.delivery_item_of_plant                AS SD_000110,
    ssms.delivery_number_initial               AS SD_000001,
    ssms.material_code                         AS SD_000143,
    ssms.dt_shipment                           AS SD_000010,
    ssms.railcar                               AS SD_000013,
    ssms.transport_bill                        AS SD_000014,
    ssms.uni                                   AS SD_000151,
    ssms.port_of_discharge_in_foreign_port_code AS SD_000054,
    ssms.port_of_loading_code                  AS SD_000649,
    ssms.port_of_discharge_code                AS SD_000044,
    ssms.transport_railcar_type_code           AS SD_000028,
    ssms.bill_of_lading_group_code             AS SD_000040,
    ssms.bill_of_lading_number                 AS SD_000041,
    ssms.dt_bill_of_lading                     AS SD_000042,
    ssms.dt_final_release                      AS SD_000482,
    ssms.batch                                 AS SD_000004,
    ssms.delivery_basis                        AS SD_000067,
    ssms.bill_of_lading_group_code_in_foreign_port AS SD_000047,
    ssms.bill_of_lading_in_foreign_port        AS SD_000048,
    ssms.dt_bill_of_lading_in_foreign_port     AS SD_000049,
    ssms.dt_warehouse                          AS SD_000024,
    sd4798.sales_bundle_code
FROM dm_calc.sd_sales_main_scm ssms
JOIN dds.sales_bundle_and_delivery_relationship sd4798
  ON ssms.delivery_number_sales = sd4798.delivery_code
ORDER BY sd4798.sales_bundle_code, ssms.delivery_number_initial DESC
DISTRIBUTED BY (sales_bundle_code);



/* ==========================================================
   BASE — основная сборка логики до base_1_5
========================================================== */

DROP TABLE IF EXISTS base;

CREATE TEMP TABLE base
ON COMMIT DROP AS
WITH forwapd_1 AS (
    SELECT 
        transport_hub_code,
        MIN(market_region1_code) AS market_region1_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE market_region1_code IN ('03','04')
      AND (is_terminal_code IS NOT NULL 
        OR is_foreign_warehouse_code IS NOT NULL 
        OR is_temporary_warehouse_code IS NOT NULL)
    GROUP BY transport_hub_code
),
forwapd_2 AS (
    SELECT DISTINCT transport_hub_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE is_russian_port_code IS NOT NULL
)

SELECT
    t4.sales_bundle_code,
    lake.SD_000003  AS plant_producer_delivery_code,
    lake.SD_000110  AS plant_producer_delivery_position_code,
    lake.SD_000001  AS initial_delivery_code,
    lake.SD_000143  AS material_code,
    lake.SD_000010  AS dt_shipment,
    lake.SD_000013  AS railcar_code,
    lake.SD_000014  AS transport_bill_code,
    lake.SD_000151  AS uni,
    maspe.shape_code AS material_shape_code,
    lake.SD_000028  AS transport_type_code,
    mater.etsng_code,
    sabu.sales_bundle_net_weight / 1000 AS sales_bundle_net_weight,
    sabu.sales_bundle_gross_weight / 1000 AS sales_bundle_gross_weight,
    lake.SD_000004 AS batch_code,

    /* Дата прибытия */
    t4.dt_transportation_stage_start_r AS dt_arrival,

    /* Завод продажной поставки */
    dedopo_1.plant_producer_code AS receiving_plant_code,

    lake.delivery_number_sales AS sales_delivery_code,

    /* Входящая тех. поставка по складской логике */
    CASE WHEN forwapd_1.transport_hub_code IS NOT NULL
         THEN t4.delivery_code_le_p END AS transportation_inbound_delivery_code,

    /* Поставка расчёта хранения (LE.001016) */
    CASE
        WHEN forwapd_1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN sabadr_03.delivery_code
        WHEN forwapd_1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN sabadr_04.delivery_code
        WHEN forwapd_2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
    END AS delivery_for_storage_calculation_code,

    /* Позиция тех. поставки входа */
    t4.transportation_inbound_delivery_position_code,

    /* Позиция поставки расчета хранения */
    CASE
        WHEN forwapd_1.market_region1_code IN ('03','04')
          AND (sabadr_03.delivery_code IS NOT NULL OR sabadr_04.delivery_code IS NOT NULL)
          THEN t4.transportation_inbound_delivery_position_code
        ELSE '10'
    END AS delivery_position_for_storage_calculation_code,

    /* Экспедитор */
    COALESCE(ascr_1.supplier_code, ascr_2.supplier_code) AS forwarder_code,

    /* Тип логики хранения */
    CASE
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 = 'TL04' THEN 'Логика РФ-контейнеры'
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 <> 'TL04' THEN 'Логика РФ-балк'
        WHEN likp.delivery_type_code = 'ZLE' AND asia_log.range_low_value IS NOT NULL AND asia.counterparty_code IS NULL THEN 'Логика Азия'
        WHEN likp.delivery_type_code = 'ZLE' AND euro_1.counterparty_code IS NOT NULL THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE' AND euro_2.counterparty_code IS NOT NULL AND euro_log.range_low_value IS NOT NULL THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE' AND svh.counterparty_code IS NOT NULL THEN 'Логика СВХ'
        ELSE 'Логика не определена'
    END AS storage_cost_calculation_type_name,

    /* Узлы */
    sbudr_base.transport_route_departure_hub_code,
    sbudr_base.transport_route_destination_hub_code,
    sbudr_base.is_final_transportation_stage_code,
    sbudr_base.transportation_stage_code,

    for_LE_001035.cnt AS storage_calculation_bundle_quantity,

    /* Разное */
    lake.SD_000482 AS dt_final_release,
    lake.SD_000041 AS bill_of_lading_number,
    lake.SD_000042 AS dt_bill_of_lading,
    lake.SD_000047 AS SD_000047,
    lake.SD_000048 AS SD_000048,
    lake.SD_000054 AS SD_000054,
    lake.SD_000049 AS SD_000049,
    lake.SD_000067 AS incoterms_code,
    lake.SD_000649,
    lake.SD_000044,
    lake.SD_000040,
    t4.delivery_code_le_r AS SD_000516,
    t4.transportation_outbound_delivery_position_code AS SD_000897,

    /* NET / BRUTTO */
    likp.weight_net AS ntgew,
    likp.weight_net_with_wirerod AS btgew,

    /* Удалённый склад */
    t4.knote AS remote_warehouse_code,

    /* Дата прибытия в РФ порт */
    CASE
        WHEN forwapd_1.market_region1_code IN ('03','04')
         AND (sabadr_03.delivery_code IS NOT NULL OR sabadr_04.delivery_code IS NOT NULL)
         THEN NULL
        ELSE lake.SD_000024
    END AS dt_arrival_to_russian_port,

    /* Интервалы */
    t4.dt_transportation_stage_start_p,
    COALESCE(t4.dt_transportation_stage_start_r, current_date) AS dt_transportation_stage_start_r,

    /* ID для уникальности */
    CONCAT(t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code) AS id_for_unique

FROM dm_calc.sales_bundle_transport_hub_turnover_sdt0004 t4
LEFT JOIN lake ON t4.sales_bundle_code = lake.sales_bundle_code
LEFT JOIN dds.bill_of_lading vbsk
       ON vbsk.bill_of_lading_code = lake.SD_000040
      AND vbsk.bill_of_lading_type_code = 'Y'
LEFT JOIN for_1055 ON for_1055.vbeln = lake.delivery_number_sales
LEFT JOIN dict_dds.material mater ON mater.material_code = lake.SD_000143
LEFT JOIN dds.sales_bundle sabu ON sabu.sales_bundle_code = t4.sales_bundle_code
LEFT JOIN forwapd_1 ON forwapd_1.transport_hub_code = t4.knote
LEFT JOIN forwapd_2 ON forwapd_2.transport_hub_code = t4.knote

/* Азия / Европа / СВХ логика */
LEFT JOIN dds.sales_bundle_and_delivery_relationship sabadr_03 ON
    sabadr_03.sales_bundle_code = t4.sales_bundle_code
    AND sabadr_03.transportation_stage_code IN (
        SELECT range_low_value
        FROM dict_dds.settings_and_parameters_sap
        WHERE abap_program_code = '/RUSAL/SD2973M_2'
          AND parameter_code = 'STAGE_EU'
          AND range_low_value IS NOT NULL
    )
    AND sabadr_03.transport_route_destination_hub_code = t4.knote
    AND forwapd_1.market_region1_code = '03'
    AND sabadr_03.is_deleted_code IS NULL

LEFT JOIN dds.sales_bundle_and_delivery_relationship sabadr_04 ON
    sabadr_04.sales_bundle_code = t4.sales_bundle_code
    AND sabadr_04.transportation_stage_code IN (
        SELECT range_low_value
        FROM dict_dds.settings_and_parameters_sap
        WHERE abap_program_code = '/RUSAL/SD2973M_2'
          AND parameter_code = 'STAGE_A'
          AND range_low_value IS NOT NULL
    )
    AND sabadr_04.transport_route_destination_hub_code = t4.knote
    AND forwapd_1.market_region1_code = '04'
    AND sabadr_04.is_deleted_code IS NULL

/* Основной sbudr запись */
LEFT JOIN dds.sales_bundle_and_delivery_relationship sbudr_base
    ON sbudr_base.sales_bundle_code = t4.sales_bundle_code
   AND sbudr_base.delivery_code = t4.delivery_code_le_p
   AND sbudr_base.delivery_position_code = t4.transportation_inbound_delivery_position_code
   AND sbudr_base.is_deleted_code IS NULL

/* Завод */
LEFT JOIN dds.delivery_document_position dedopo_1
    ON dedopo_1.delivery_code = lake.delivery_number_sales

LEFT JOIN dds.sales_document_counterparty_role ascr_1
    ON ascr_1.sales_document_code = t4.delivery_code_le_p
   AND ascr_1.counterparty_role_code = 'XR'

LEFT JOIN dds.sales_document_counterparty_role ascr_2
    ON ascr_2.sales_document_code = t4.delivery_code_le_p
   AND ascr_2.counterparty_role_code = 'ZU'

LEFT JOIN dds.delivery_document_header likp
    ON likp.delivery_code =
        CASE
            WHEN forwapd_1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN sabadr_03.delivery_code
            WHEN forwapd_1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN sabadr_04.delivery_code
            WHEN forwapd_2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
        END

LEFT JOIN asia ON asia.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN euro_1 ON euro_1.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN euro_2 ON euro_2.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN svh    ON svh.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN asia_log ON asia_log.range_low_value = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN euro_log ON euro_log.range_low_value = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

LEFT JOIN dict_dds.material_specification maspe
    ON maspe.material_code = lake.SD_000143

LEFT JOIN for_LE_001035
    ON for_LE_001035.delivery_code = t4.delivery_code_le_p
   AND for_LE_001035.batch_code = lake.SD_000004

WHERE
    t4.dt_transportation_stage_start_p <= (SELECT MAX(dt_balance) FROM date_for_vitr_tmp)
    AND COALESCE(t4.dt_transportation_stage_start_r, current_date) >= (SELECT MIN(dt_balance) FROM date_for_vitr_tmp)
    AND t4.dt_transportation_stage_start_p <> COALESCE(t4.dt_transportation_stage_start_r, DATE '2299-12-31')
    AND t4.transport_route_destination_hub_code LIKE 'G%'
    AND (
            CASE
                WHEN forwapd_1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN sabadr_03.delivery_code
                WHEN forwapd_1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN sabadr_04.delivery_code
                WHEN forwapd_2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
            END
        ) IS NOT NULL
DISTRIBUTED BY (sales_bundle_code, plant_producer_delivery_code);
