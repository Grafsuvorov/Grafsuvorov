Копировать код
CREATE TEMP TABLE t_contract_match ON COMMIT DROP AS
SELECT
    b.rn,
    x.purchase_contract_code,
    x.position_line_item_code,
    x.konws,
    x.price_uom_code,
    x.service_code,
    x.currency_dogovor,
    x.datab,
    x.datbi
FROM dm_calc.storage_sales_bundles_weight b
JOIN LATERAL
(
    SELECT *
    FROM t_1079_base t
    WHERE t.lifnr = b.forwarder_code
      AND (t.zzport = b.port_of_discharge_code OR t.zzport IS NULL)
      AND (t.zzforma = b.material_shape_code OR t.zzforma IS NULL)
      AND (t.zzet_tarif = b.etsng_code OR t.zzet_tarif IS NULL)
      AND (t.sdabw = b.transport_type_code OR t.sdabw IS NULL)
      AND b.dt_for_price_search_in_purchase_contract BETWEEN t.datab AND t.datbi
      AND b.dt_for_price_search_in_purchase_contract BETWEEN t.dt_valid_from AND t.dt_valid_to
    ORDER BY
        CASE b.storage_cost_calculation_type_name
            WHEN 'Логика РФ-контейнеры' THEN order_rf
            WHEN 'Логика РФ-балк'       THEN order_rf
            WHEN 'Логика Азия'          THEN order_asia
            WHEN 'Логика Европа'        THEN order_euro_svh
            WHEN 'Логика СВХ'           THEN order_euro_svh
        END,
        kopos,
        kbetr
    LIMIT 1
) x ON TRUE
DISTRIBUTED BY (rn);

CREATE TEMP TABLE t_1087 ON COMMIT DROP AS
SELECT
    purchase_contract_code,
    purchase_contract_position_code,
    dt_price_condition_valid_to,
    dt_price_condition_valid_from,
    MIN(price_scale_limit_from_quantity) AS min_free_days
FROM dds.purchase_contract_price_conditions
WHERE price_scale_limit_from_quantity IS NOT NULL
GROUP BY 1,2,3,4
DISTRIBUTED BY (purchase_contract_code);

             AND b.dt_storage_end_in_release IS NOT NULL

    CASE WHEN storage_duration_payable_calendar_days BETWEEN 61 AND 90 THEN b.sales_bundle_gross_weight ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 91 AND 180 THEN b.sales_bundle_gross_weight ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 181 AND 365 THEN b.sales_bundle_gross_weight ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days > 365 THEN b.sales_bundle_gross_weight ELSE 0 END,

    b.is_stored_in_russian_port_code,

    /* Стоимость по диапазонам */
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 1 AND 30 THEN k1.konp_kbetr ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 31 AND 60 THEN k1.konp_kbetr ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 61 AND 90 THEN k1.konp_kbetr ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 91 AND 180 THEN k1.konp_kbetr ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days BETWEEN 181 AND 365 THEN k1.konp_kbetr ELSE 0 END,
    CASE WHEN storage_duration_payable_calendar_days > 365 THEN k1.konp_kbetr ELSE 0 END,

    NULL AS metal_in_china_stock_balance_closing_weight

FROM dm_calc.storage_sales_bundles_weight b
LEFT JOIN t_contract_match cm
    ON cm.rn = b.rn
LEFT JOIN t_1087 t87
    ON t87.purchase_contract_code = cm.purchase_contract_code
   AND t87.purchase_contract_position_code = cm.position_line_item_code
   AND t87.dt_price_condition_valid_to = cm.datbi
   AND t87.dt_price_condition_valid_from = cm.datab
LEFT JOIN t_1090 k1
    ON k1.evrtn = cm.purchase_contract_code
   AND k1.evrtp = cm.position_line_item_code
   AND k1.dt_price_condition_valid_to = cm.datbi
   AND k1.dt_price_condition_valid_from = cm.datab
   AND k1.price_scale_basis_code IS NULL
LEFT JOIN dict_dds.currency_rates cr
    ON cr.currency_rate_type_code = 'M'
   AND cr.currency_from_code = cm.konws
   AND cr.currency_to_code = 'USD'
   AND cr.dt_currency_rate = b.dt_report
   AND cr.deleted_flag = false;
