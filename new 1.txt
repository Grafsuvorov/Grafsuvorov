CREATE TEMP TABLE for_1079_asia AS

WITH
-- ============================================================
-- 1) Основной быстрый JOIN (строгое совпадение по всем полям)
-- ============================================================
cte_main AS (
    SELECT
        b.rn,
        f.purchase_contract_code,
        f.position_line_item_code,
        f.konws,
        f.service_code,
        f.price_uom_code,
        f.currency_dogovor,
        f.datab,
        f.datbi,
        f.order_asia,
        f.kopos,
        f.kbetr
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN for_1079_base f
      ON f.bukrs = b.unit_balance_code
     AND f.lifnr = b.forwarder_code
     AND f.zzport = b.port_of_discharge_code
     AND f.werks = b.plant_code
     AND f.zzsrvpos = 'S49'
     AND f.zzforma = b.material_shape_code
     AND f.zzet_tarif = b.etsng_code
     AND f.sdabw = b.transport_type_code
     AND f.zzterminal = b.warehouse_code
     AND f.vzzzone = b.warehouse_storage_area_code
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.datab AND f.datbi
     AND f.purchase_document_subtype_code = 'ZDTL'
     AND b.storage_cost_calculation_type_name IN ('Логика Азия')
),

-- ============================================================
-- 2) NULL-варианты (очень редкие) — отдельный блок
--    Здесь f.zzforma / zzet_tarif / sdabw / zzterminal / vzzzone могут быть NULL.
--    Выполняется быстро, т.к. данных мало.
-- ============================================================
cte_nulls AS (
    SELECT
        b.rn,
        f.purchase_contract_code,
        f.position_line_item_code,
        f.konws,
        f.service_code,
        f.price_uom_code,
        f.currency_dogovor,
        f.datab,
        f.datbi,
        f.order_asia,
        f.kopos,
        f.kbetr
    FROM dm_calc.storage_sales_bundles_weight b
    JOIN for_1079_base f
      ON f.bukrs = b.unit_balance_code
     AND f.lifnr = b.forwarder_code
     AND f.zzport = b.port_of_discharge_code
     AND f.werks = b.plant_code
     AND f.zzsrvpos = 'S49'
     AND (f.zzforma IS NULL OR f.zzforma = b.material_shape_code)
     AND (f.zzet_tarif IS NULL OR f.zzet_tarif = b.etsng_code)
     AND (f.sdabw IS NULL OR f.sdabw = b.transport_type_code)
     AND (f.zzterminal IS NULL OR f.zzterminal = b.warehouse_code)
     AND (f.vzzzone IS NULL OR f.vzzzone = b.warehouse_storage_area_code)
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract
         BETWEEN f.datab AND f.datbi
     AND f.purchase_document_subtype_code = 'ZDTL'
     AND b.storage_cost_calculation_type_name IN ('Логика Азия')
),

-- ============================================================
-- 3) Объединяем два варианта
-- ============================================================
cte AS (
    SELECT
        rn,
        purchase_contract_code,
        position_line_item_code,
        konws,
        service_code,
        price_uom_code,
        currency_dogovor,
        datab,
        datbi,
        ROW_NUMBER() OVER (
            PARTITION BY rn
            ORDER BY order_asia, kopos, kbetr, purchase_contract_code, position_line_item_code
        ) AS row_num
    FROM (
        SELECT * FROM cte_main
        UNION ALL
        SELECT * FROM cte_nulls
    ) x
)

-- ============================================================
-- 4) Финальный выбор
-- ============================================================
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi
FROM cte
WHERE row_num = 1
DISTRIBUTED BY (rn);
