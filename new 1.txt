SELECT
    -- Базовые поля из dm_calc.storage_sales_bundles_weight
    b.sales_bundle_code,
    b.plant_producer_delivery_code,
    b.plant_producer_delivery_position_code,
    b.initial_delivery_code,
    b.material_code,
    b.dt_shipment,
    b.railcar_code,
    b.transport_bill_code,
    b.uni,
    b.material_shape_code,
    b.port_of_discharge_code,
    b.dt_arrival_to_russian_port,
    b.dt_storage_end_in_release,
    b.transport_type_code,
    b.etsng_code,
    b.dt_report,
    b.sales_bundle_net_weight,
    b.sales_bundle_gross_weight,
    b.batch_code,
    b.dt_arrival,
    b.receiving_plant_code,
    b.sales_delivery_code,
    b.transportation_inbound_delivery_code,
    b.delivery_for_storage_calculation_code,
    b.transportation_inbound_delivery_position_code,
    b.delivery_position_for_storage_calculation_code,
    b.forwarder_code,
    b.storage_cost_calculation_type_name,
    b.transport_departure_hub_code,
    b.transport_destination_hub_code,
    b.transportation_outbound_delivery_code,
    b.transportation_outbound_delivery_position_code,
    b.is_final_transportation_stage_code,
    b.transportation_stage_code,
    b.storage_calculation_bundle_quantity,
    b.warehouse_code,
    b.delivery_in_final_release_code,
    b.dt_final_release,
    b.bill_of_lading_code,
    b.bill_of_lading_number,
    b.dt_bill_of_lading,
    b.incoterms_code,
    b.storage_cost_special_calculation_type_code,
    b.warehouse_storage_area_code,
    b.plant_code,
    b.unit_balance_code,
    b.creditor_in_shipment_instruction_code,
    b.dt_storage_start,
    b.dt_shipped_from_warehouse,
    b.dt_storage_end,
    b.service_number,
    b.storage_duration_total_calendar_days,
    b.shipment_instruction_number,
    b.dt_for_price_search_in_purchase_contract,
    b.is_stored_in_russian_port_code,
    b.remote_warehouse_code,
    b.dt_transportation_stage_start_p,
    b.dt_transportation_stage_start_r,

    /* --- LE.001079 — договор хранения --- */
    main_contract.contract_code         AS purchase_contract_code,

    /* --- LE.001080 — позиция договора --- */
    main_contract.contract_position     AS purchase_contract_position_code,

    /* --- LE.001081 — валюта позиции договора --- */
    main_contract.contract_konws        AS purchase_contract_position_currency_code,

    /* --- LE.001083 — услуга договора --- */
    main_contract.contract_service_code AS service_code,

    /* --- LE.001091 — ед. измерения позиции договора --- */
    main_contract.contract_price_uom    AS purchase_contract_position_uom_code,

    /* --- LE.001087 — бесплатные дни по договору --- */
    GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0) AS storage_duration_free_by_contract_calendar_days,

    /* --- LE.001088 — бесплатные дни по поставке --- */
    LEAST(
        b.storage_duration_total_calendar_days,
        GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
    ) AS storage_duration_free_by_delivery_calendar_days,

    /* --- LE.001089 — платные дни --- */
    b.storage_duration_total_calendar_days
      - LEAST(
            b.storage_duration_total_calendar_days,
            GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
        ) AS storage_duration_payable_calendar_days,

    /* --- LE.001125 — стоимость хранения в валюте договора --- */
    CASE
        WHEN b.storage_cost_special_calculation_type_code = '2'
        THEN
            COALESCE(for_1090_1.konp_kbetr, for_1090_2.konp_kbetr)
            * euro.month_difference_2
            * b.sales_bundle_gross_weight

        WHEN for_prices.range_low_value IS NOT NULL
             AND for_1090_1.evrtn IS NOT NULL
        THEN
            for_1090_1.konp_kbetr
            * (
                  COALESCE(euro.special, b.storage_duration_total_calendar_days)
                  - LEAST(
                        COALESCE(euro.special, b.storage_duration_total_calendar_days),
                        GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
                    )
              )
            / b.weight_for_1090_razdel
            * b.weight_for_1090_umno
            * main_contract.contract_currency_dogovor

        WHEN for_prices.range_low_value IS NOT NULL
             AND for_1090_2.evrtn IS NOT NULL
        THEN
            COALESCE(for_1090_2.presum, 0)
            + GREATEST(
                COALESCE(euro.special, b.storage_duration_total_calendar_days)
                - for_1090_2.kstbm + 1,
                0
              )
              * for_1090_2.konm_kbetr
              / b.weight_for_1090_razdel
              * b.weight_for_1090_umno
              * main_contract.contract_currency_dogovor

        WHEN for_1090_1.evrtn IS NOT NULL
        THEN
            for_1090_1.konp_kbetr
            * (
                  COALESCE(euro.special, b.storage_duration_total_calendar_days)
                  - LEAST(
                        COALESCE(euro.special, b.storage_duration_total_calendar_days),
                        GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
                    )
              )
            * b.weight_for_1090_umno
            * main_contract.contract_currency_dogovor

        WHEN for_1090_2.evrtn IS NOT NULL
        THEN
            (COALESCE(for_1090_2.presum, 0)
             + GREATEST(
                    COALESCE(euro.special, b.storage_duration_total_calendar_days)
                    - for_1090_2.kstbm + 1,
                    0
               ) * for_1090_2.konm_kbetr
            )
            * b.weight_for_1090_umno
            * main_contract.contract_currency_dogovor
    END AS storage_cost_calculated_amount_local,

    /* --- LE.001095–LE.001100 — распределение веса по интервалам платных дней --- */
    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            )
            BETWEEN 1 AND 30
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_001_030_days_gross_weight,

    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            )
            BETWEEN 31 AND 60
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_031_060_days_gross_weight,

    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            )
            BETWEEN 61 AND 90
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_061_090_days_gross_weight,

    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            )
            BETWEEN 91 AND 180
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_091_180_days_gross_weight,

    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            )
            BETWEEN 181 AND 365
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_181_365_days_gross_weight,

    CASE
        WHEN
            COALESCE(euro.special, b.storage_duration_total_calendar_days)
            - LEAST(
                b.storage_duration_total_calendar_days,
                GREATEST(for_1087.price_scale_limit_from_quantity - 1, 0)
            ) > 365
        THEN b.sales_bundle_gross_weight
        ELSE 0
    END AS storage_payable_over_365_days_gross_weight,

    b.rn
FROM dm_calc.storage_sales_bundles_weight b
    LEFT JOIN for_1079_rf_cont   rf_cont  ON rf_cont.rn  = b.rn
    LEFT JOIN for_1079_rf_balk_1 rf_balk1 ON rf_balk1.rn = b.rn
    LEFT JOIN for_1079_rf_balk_2 rf_balk2 ON rf_balk2.rn = b.rn
    LEFT JOIN euro_base          euro     ON euro.rn     = b.rn
    LEFT JOIN for_1079_svh       svh      ON svh.rn      = b.rn
    LEFT JOIN for_1079_asia      asia     ON asia.rn     = b.rn

    -- Единый выбор "главного" договора/позиции/валюты/дат
    CROSS JOIN LATERAL (
        SELECT
            COALESCE(
                rf_cont.purchase_contract_code,
                rf_balk1.purchase_contract_code,
                rf_balk2.purchase_contract_code,
                euro.purchase_contract_code,
                svh.purchase_contract_code,
                asia.purchase_contract_code
            ) AS contract_code,

            COALESCE(
                rf_cont.position_line_item_code,
                rf_balk1.position_line_item_code,
                rf_balk2.position_line_item_code,
                euro.position_line_item_code,
                svh.position_line_item_code,
                asia.position_line_item_code
            ) AS contract_position,

            COALESCE(
                rf_cont.konws,
                rf_balk1.konws,
                rf_balk2.konws,
                euro.konws,
                svh.konws,
                asia.konws
            ) AS contract_konws,

            COALESCE(
                rf_cont.service_code,
                rf_balk1.service_code,
                rf_balk2.service_code,
                euro.service_code,
                svh.service_code,
                asia.service_code
            ) AS contract_service_code,

            COALESCE(
                rf_cont.price_uom_code,
                rf_balk1.price_uom_code,
                rf_balk2.price_uom_code,
                euro.price_uom_code,
                svh.price_uom_code,
                asia.price_uom_code
            ) AS contract_price_uom,

            COALESCE(
                rf_cont.currency_dogovor,
                rf_balk1.currency_dogovor,
                rf_balk2.currency_dogovor,
                euro.currency_dogovor,
                svh.currency_dogovor,
                asia.currency_dogovor
            ) AS contract_currency_dogovor,

            COALESCE(
                rf_cont.datab,
                rf_balk1.datab,
                rf_balk2.datab,
                euro.datab,
                svh.datab,
                asia.datab
            ) AS contract_valid_from,

            COALESCE(
                rf_cont.datbi,
                rf_balk1.datbi,
                rf_balk2.datbi,
                euro.datbi,
                svh.datbi,
                asia.datbi
            ) AS contract_valid_to
    ) AS main_contract

    LEFT JOIN for_1087
           ON for_1087.purchase_contract_code         = main_contract.contract_code
          AND for_1087.purchase_contract_position_code = main_contract.contract_position
          AND for_1087.dt_price_condition_valid_to     = main_contract.contract_valid_to
          AND for_1087.dt_price_condition_valid_from   = main_contract.contract_valid_from

    LEFT JOIN for_1090 AS for_1090_1
           ON for_1090_1.evrtn                        = main_contract.contract_code
          AND for_1090_1.evrtp                        = main_contract.contract_position
          AND for_1090_1.dt_price_condition_valid_to  = main_contract.contract_valid_to
          AND for_1090_1.dt_price_condition_valid_from= main_contract.contract_valid_from
          AND for_1090_1.price_scale_basis_code IS NULL

    LEFT JOIN for_1090 AS for_1090_2
           ON for_1090_2.evrtn                        = main_contract.contract_code
          AND for_1090_2.evrtp                        = main_contract.contract_position
          AND for_1090_2.dt_price_condition_valid_to  = main_contract.contract_valid_to
          AND for_1090_2.dt_price_condition_valid_from= main_contract.contract_valid_from
          AND for_1090_2.price_scale_basis_code       = 'C'
          AND COALESCE(euro.special, b.storage_duration_total_calendar_days)
              BETWEEN for_1090_2.kstbm AND for_1090_2.kstbm_next

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_prices
           ON for_prices.range_low_value   = main_contract.contract_price_uom
          AND for_prices.abap_program_code = '/RUSAL/ZLE112M'
          AND for_prices.parameter_code    = 'S_BPRME';
