def clean_text(value):
    if value is None:
        return None

    value = str(value).strip()

    if value == "":
        return None

    return value


def text_or_none(node, path):
    if node is None:
        return None

    found = node.find(path)

    if found is None:
        return None

    return clean_text(found.text)


def to_decimal(value):
    value = clean_text(value)

    if value is None:
        return None

    value = value.replace(" ", "").replace(",", ".")

    if value == "":
        return None

    return decimal.Decimal(value)


def to_date(value):
    value = clean_text(value)

    if value is None:
        return None


    if "T" in value:
        value = value.split("T")[0]

    if " " in value:
        value = value.split(" ")[0]

    return datetime.date.fromisoformat(value)


def to_timestamp(value):
    value = clean_text(value)

    if value is None:
        return None

    value = value.replace("T", " ")
    value = value.replace("Z", "")

    return datetime.datetime.fromisoformat(value)

def value_for_copy(value):
    """
    Для COPY CSV.
    None грузим как \\N.
    Остальное приводим к строке.
    """
    if value is None:
        return r"\N"

    if isinstance(value, decimal.Decimal):
        return str(value)

    if isinstance(value, datetime.datetime):
        return value.strftime("%Y-%m-%d %H:%M:%S")

    if isinstance(value, datetime.date):
        return value.strftime("%Y-%m-%d")

    return str(value)

# COPY в таблицу загруженных uuid
# =========================

def copy_loaded_uuid(conn, uuid_rows):
    if not uuid_rows:
        return 0

    buffer = io.StringIO()

    writer = csv.writer(
        buffer,
        delimiter="\t",
        quotechar='"',
        quoting=csv.QUOTE_MINIMAL,
        lineterminator="\n",
        escapechar="\\"
    )

    for uuid, dt_insert in uuid_rows:
        writer.writerow([
            value_for_copy(uuid),
            value_for_copy(dt_insert),
        ])

    buffer.seek(0)

    copy_sql = """
        COPY stg.s1c_debt_for_working_capital_loaded_uuid
        (
            uuid,
            dt_insert
        )
        FROM STDIN
        WITH (
            FORMAT csv,
            DELIMITER E'\\t',
            NULL '\\N',
            QUOTE '"',
            ESCAPE '\\'
        )
    """

    with conn.cursor() as cur:
        cur.copy_expert(copy_sql, buffer)

    return len(uuid_rows)

def main():
    read_conn = psycopg2.connect(**GP_CONN)
    write_conn = psycopg2.connect(**GP_CONN)

    read_conn.autocommit = False
    write_conn.autocommit = False

    total_xml = 0
    total_rows = 0
    total_uuid_loaded = 0
    error_count = 0

    batch_rows = []
    batch_uuid_rows = []
    seen_uuid = set()

    select_sql = """
        SELECT
            t.uuid::varchar AS uuid,
            t.dt_insert,
            t.document_xml::text AS document_xml
        FROM landing."INPUT_DATA_FROM_SAPXI_IN" t
        WHERE t.flow_id = %s
          AND NOT EXISTS (
              SELECT 1
              FROM stg.s1c_debt_for_working_capital_loaded_uuid u
              WHERE u.uuid = t.uuid
          )
        ORDER BY t.dt_insert
    """

    try:
        with read_conn.cursor(name="s1c_wc_xml_cursor") as cur:
            cur.itersize = FETCH_BATCH_SIZE
            cur.execute(select_sql, (FLOW_ID,))

            for uuid, dt_insert, document_xml in cur:
                if uuid in seen_uuid:
                    continue

                seen_uuid.add(uuid)
                total_xml += 1

                try:
                    parsed_rows = parse_xml_row(uuid, dt_insert, document_xml)

                    batch_rows.extend(parsed_rows)
                    batch_uuid_rows.append((uuid, dt_insert))

                except Exception as e:
                    error_count += 1
                    print("=" * 100)
                    print(f"ERROR while parsing uuid={uuid}, dt_insert={dt_insert}")
                    print(str(e))
                    print(traceback.format_exc())
                    print("=" * 100)

                if len(batch_rows) >= COPY_ROWS_LIMIT:
                    inserted_rows = copy_rows_to_stg(write_conn, batch_rows)
                    inserted_uuids = copy_loaded_uuid(write_conn, batch_uuid_rows)

                    write_conn.commit()

                    total_rows += inserted_rows
                    total_uuid_loaded += inserted_uuids

                    print(
                        f"XML processed: {total_xml}, "
                        f"rows inserted: {total_rows}, "
                        f"uuid loaded: {total_uuid_loaded}, "
                        f"errors: {error_count}"
                    )

                    batch_rows = []
                    batch_uuid_rows = []

            if batch_rows or batch_uuid_rows:
                inserted_rows = copy_rows_to_stg(write_conn, batch_rows)
                inserted_uuids = copy_loaded_uuid(write_conn, batch_uuid_rows)

                write_conn.commit()

                total_rows += inserted_rows
                total_uuid_loaded += inserted_uuids

        print("=" * 100)
        print("DONE")
        print(f"Total XML processed: {total_xml}")
        print(f"Total rows inserted: {total_rows}")
        print(f"Total uuid loaded: {total_uuid_loaded}")
        print(f"Errors: {error_count}")
        print("=" * 100)

    except Exception:
        write_conn.rollback()
        print("FATAL ERROR")
        print(traceback.format_exc())
        sys.exit(1)

    finally:
        read_conn.close()
        write_conn.close()


if __name__ == "__main__":
    main()
