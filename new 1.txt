	
	

CREATE TEMP TABLE pp --ON COMMIT drop
AS
SELECT
    kalnr,
    bdatj,
    poper,
    make_date(bdatj::int, substring(poper, 2, 2)::int, 1) AS dt_valid_from,
    umkumo, abkumo, zukumo, lbkum
FROM ods.ckmlpp_ral
WHERE untper = '000'
DISTRIBUTED BY (kalnr);
ANALYZE pp;

--------------------------
-- 2) CKMLCR (текущий период): гарантируем 1 строку на (kalnr,bdatj,poper)
--    Если в источнике встречаются дубляжи, выберите бизнес-правило:
--    SUM(...) / MAX(...) / LAST_VALUE по нужному полю (пример ниже: MAX для “параметров”, SUM для потоков).
--------------------------
CREATE TEMP TABLE cr_curr --ON COMMIT DROP 
AS
SELECT
    kalnr, bdatj, poper,
    -- при необходимости заменить на агрегаты, согласованные с бизнесом
    MAX(stprs)    AS stprs,
    MAX(peinh)    AS peinh,
    MAX(pvprs)    AS pvprs,
    MAX(abprd_o)  AS abprd_o,
    MAX(abprd_mo) AS abprd_mo,
    MAX(vprsv)    AS vprsv,
    MAX(salk3)    AS salk3   -- или SUM(...) если это поток/движение, а не срез
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);
ANALYZE cr_curr;

--------------------------
-- 3) CKMLCR (следующий месяц): salk3 на dt_posting = 1-е число следующего месяца (DATE!)
--------------------------
CREATE TEMP TABLE cr_next --ON COMMIT DROP 
AS
SELECT
    kalnr,
    (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    -- если бывают дубли на один и тот же dt_posting — также агрегируйте
    MAX(salk3) AS salk3
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);
ANALYZE cr_next;

--------------------------
-- 4) MLCD агрегаты
--------------------------
CREATE TEMP TABLE mlcd_agg --ON COMMIT drop
AS
SELECT
    kalnr, bdatj, poper,
    SUM(salk3)  AS salk3,
    SUM(estprd) AS estprd,
    SUM(mstprd) AS mstprd
FROM ods.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr);
ANALYZE mlcd_agg;

--------------------------
-- 5) Один WindowAgg для границ действия
--------------------------
--explain(verbose, format json)
--insert into    userdata.material_price_calculation
explain(verbose, format json)
WITH ranges AS (
    SELECT
        kalnr,
        dt_valid_from,
        COALESCE(
            LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
            DATE '2299-12-31'
        )::date AS dt_valid_to
    FROM pp
)
SELECT
    r.kalnr AS calculation_code,
    r.dt_valid_from,
    r.dt_valid_to,
    cr_curr.vprsv AS price_valuation_type_code,
    cr_curr.stprs  AS standard_price_amount,
    cr_curr.peinh  AS price_unit_code,
    cr_curr.pvprs  AS moving_average_price_amount,

    /* rule3 */
    CASE WHEN (pp.umkumo + pp.abkumo + pp.zukumo) = 0 THEN 0
         ELSE TRUNC(
                CAST(ROUND(
                      (cr_curr.stprs / cr_curr.peinh) * (pp.umkumo + pp.abkumo)
                    + cr_curr.abprd_o + cr_curr.abprd_mo
                    + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                , 2) AS NUMERIC(40,20))
                / (pp.umkumo + pp.abkumo + pp.zukumo), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE WHEN (pp.abkumo + pp.umkumo + pp.zukumo) <> 0
         THEN TRUNC(
                CAST(
                    cr_next.salk3 + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                AS NUMERIC(40,20))
                / (pp.abkumo + pp.umkumo + pp.zukumo), 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE WHEN pp.lbkum <> 0
         THEN TRUNC(CAST(cr_curr.salk3 AS NUMERIC(40,20)) / pp.lbkum, 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    ckmlhd.price_rule_code
FROM ranges r
JOIN pp
  ON pp.kalnr = r.kalnr
 AND pp.dt_valid_from = r.dt_valid_from
LEFT JOIN cr_curr
  ON cr_curr.kalnr = pp.kalnr
 AND cr_curr.bdatj = pp.bdatj
 AND cr_curr.poper = pp.poper
LEFT JOIN cr_next
  ON cr_next.kalnr = pp.kalnr
 AND cr_next.dt_posting = pp.dt_valid_from
LEFT JOIN mlcd_agg
  ON mlcd_agg.kalnr = pp.kalnr
 AND mlcd_agg.bdatj = pp.bdatj
 AND mlcd_agg.poper = pp.poper
LEFT JOIN dds.material_ledger_header ckmlhd
  ON ckmlhd.calculation_code = pp.kalnr
;
