insert into pg_temp.cost_plan (cal_day, month_day, plant_code, cost_plan_day, cum_sum_cost_plan)
select
    ct.cal_day,
    cp.month_day,
    ct.plant_code,
    cp.cost_plan_day,
    case
        when sum(cost_plan_day) over (partition by ct.plant_code, cp.month_day order by ct.cal_day asc) = cp.cost_plan_day
            then 0 -- Для первого числа месяца не учитываем расходы, они уже учтены в начальном балансе balance_opening
        else sum(cost_plan_day) over (
            partition by ct.plant_code, cp.month_day
            order by ct.cal_day asc
            rows between unbounded preceding and 1 preceding
        )
    end                                 as cum_sum_cost_plan
from (
    select
        make_date(year::int, month::int, 1) as month_day,
        plant_code,
        round(
            sum(pcp.cost_plan)
            / extract(days from make_date(year::int, month::int, 1) + interval '1 month - 1 day')::numeric(15, 3), 3
        ) as cost_plan_day
    from dm.production_cost_plan as pcp
    group by make_date(year::int, month::int, 1), plant_code
)                                       as cp
inner join pg_temp.calendar_plant_table as ct
    on
        date_trunc('month', cal_day) = cp.month_day
        and cp.plant_code = ct.plant_code;

insert into pg_temp.arrival (cal_day, plant_code, fact_bal_inc_arrival_day, cum_sum_fact_bal_inc_arrival_closed, cum_sum_fact_bal_inc_arrival_open)
select
    cp.cal_day,
    cp.plant_code,
    coalesce(ar.fact_bal_inc_arrival_day, 0) as fact_bal_inc_arrival_day,
    case
        when
            sum(coalesce(ar.fact_bal_inc_arrival_day, 0))
                over (partition by cp.plant_code, date_trunc('month', cp.cal_day) order by cp.cal_day asc)
            = coalesce(ar.fact_bal_inc_arrival_day, 0)
            then 0 --  Для первого числа месяца не учитываем поступления, они уже учтены в начальном балансе balance_opening
        else sum(coalesce(ar.fact_bal_inc_arrival_day, 0))
            over (partition by cp.plant_code, date_trunc('month', cp.cal_day) order by cp.cal_day asc)
    end                                      as cum_sum_fact_bal_inc_arrival_closed,
    case
        when date_trunc('month', cp.cal_day) = cp.cal_day
            then 0 --  Для первого числа месяца не учитываем поступления, они уже учтены в начальном балансе balance_opening
        else sum(coalesce(ar.fact_bal_inc_arrival_day, 0))
            over (
                partition by cp.plant_code, date_trunc('month', cp.cal_day)
                order by cp.cal_day asc
                rows between unbounded preceding and 1 preceding
            )
    end                                      as cum_sum_fact_bal_inc_arrival_open
from pg_temp.calendar_plant_table            as cp
left join (
    select
        dt_arrived,
        date_trunc('month', dt_arrived) as month_arrived,
        plant,
        sum(volum_value) as fact_bal_inc_arrival_day
    from dm.asugdc_arrived_cargo
    group by dt_arrived, plant
)                                            as ar
    on
        cp.cal_day = ar.dt_arrived
        and cp.plant_code = ar.plant;

insert into dm.material_alumina_balance (
    dt_report,
    plant_code,
    plant_name,
    material_code,
    material_balance_opening_weight,
    material_receipt_weight,
    material_consumption_weight,
    material_balance_closing_weight,
    material_unfactured_weight,
    material_technical_minimum_weight,
    material_normative_weight,
    fact_asu_rest,
    forecast_bal_closing,
    data_type_code,
    alumina_at_plant_on_wheels_weight --MM.001056 Объем глинозема "на колесах"
)
--Итоговая сборка
select
    dt_report,
    plant_code,
    plant_name,
    material_code,
    material_balance_opening_weight,
    material_receipt_weight,
    material_consumption_weight,
    material_balance_opening_weight + material_receipt_weight - material_consumption_weight as material_balance_closing_weight, -- Баланс на конец дня
    material_unfactured_weight,
    material_technical_minimum_weight,
    material_normative_weight,
    0                                                                                       as fact_asu_rest,                                                                            -- no
    0                                                                                       as forecast_bal_closing,                                                                     -- no
    data_type_code,
    alumina_at_plant_on_wheels_weight
from (
    select
        cpt.cal_day as dt_report,
        cpt.plant_code,
        p.plant_short_name as plant_name,
        '000000000001000001'::varchar(54) as material_code,
        coalesce(cp.cost_plan_day, 0) as material_consumption_weight, --Расход плановый за день
        case
            when cpt.cal_day <= now()::date
                then coalesce(arr.fact_bal_inc_arrival_day, 0)
            else coalesce(arr_f.fact_bal_inc_arrival_day, 0)
        end as material_receipt_weight,                              -- Поступления за день
        case
            when coalesce(bo.balance_opening_amt, 0) = 0 then 0 -- Если исходного баланса нет, ничего не считаем
            else
                coalesce(bo.balance_opening_amt, 0)
                + coalesce(arr.cum_sum_fact_bal_inc_arrival_open, 0)
                + coalesce(arr_f.cum_sum_fact_bal_inc_arrival, 0)
                - coalesce(cp.cum_sum_cost_plan, 0)
        end as material_balance_opening_weight,
        coalesce(cp.cum_sum_cost_plan, 0) as cum_sum_cost_plan,
        coalesce(arr.cum_sum_fact_bal_inc_arrival_open, 0) as cum_sum_fact_bal_inc_arrival,
        coalesce(car.cum_sum_car_unfactured, 0) as cum_sum_car_unfactured,
        coalesce(car_last.last_month, 0) as last_month,
        case
            when date_trunc('month', cpt.cal_day) = cpt.cal_day then coalesce(car.last_month, 0)
            else 0
        end as material_unfactured_weight,
        coalesce(tnt.material_technical_minimum_weight, 0) as material_technical_minimum_weight,
        coalesce(tnt.material_normative_weight, 0) as material_normative_weight,
        bo.balance_opening_amt,
        case
            when cpt.cal_day < now()::date
                then 'Факт'
            else 'Прогноз'
        end as data_type_code,
        coalesce(ap.alumina_at_plant_on_wheels_weight, 0) as alumina_at_plant_on_wheels_weight --MM.001056 Объем глинозема "на колесах"
    from pg_temp.calendar_plant_table as cpt
    left join dict_dds.plant_and_subsidiary as p
        on cpt.plant_code = p.plant_code
    left join pg_temp.techmin_normativ_table as tnt
        on
            cpt.cal_day = tnt.cal_day
            and cpt.plant_code = tnt.plant_code
    left join pg_temp.balance_opening as bo -- Исходный баланс, остатки на 1 число
        on
            date_trunc('month', cpt.cal_day) = bo.month_day
            and cpt.plant_code = bo.plant_code
    left join pg_temp.cost_plan as cp -- Расходы, месячный план раскиданный по дням
        on
            cpt.cal_day = cp.cal_day
            and cpt.plant_code = cp.plant_code
    left join pg_temp.arrival as arr -- Обьем поступлений на завод по дням
        on
            cpt.cal_day = arr.cal_day
            and cpt.plant_code = arr.plant_code
    left join pg_temp.arrival_forecast as arr_f -- Прогнозные поступления
        on
            cpt.cal_day = arr_f.cal_day
            and cpt.plant_code = arr_f.plant_code
    left join pg_temp.car_unfactured_table as car -- Запас на колесах, текущий месяц
        on
            cpt.cal_day = car.cal_day
            and cpt.plant_code = car.plant_code
    left join pg_temp.car_unfactured_table as car_last -- Запас на колесах, предудыщий месяц
        on
            date_trunc('month', cpt.cal_day) = car_last.cal_day
            and cpt.plant_code = car_last.plant_code
    left join pg_temp.alumina_plow as ap --Для поля MM.001056 Объем глинозема "на колесах"
        on
            ap.dt_report = cpt.cal_day
            and ap.plant_code = cpt.plant_code
)                                                                                           as sub_q;
