"""
GP Plan Doctor — TEMP edition (v5)
Цели:
- Найти и создать ВСЕ временные таблицы (TEMP/TEMPORARY), независимо от имени.
  * CREATE TEMP (columns…)  → создаём как есть.
  * CREATE TEMP … AS SELECT → создаём "пустую" структуру через LIMIT 0 (без данных).
  * DROP TEMP              → пропускаем.
- Для каждого оператора, где участвуют временные таблицы, строим EXPLAIN (JSON) и анализ:
  * размерность (rows×width), Seq Scan без фильтра, Hash/Agg/Sort, Motion.
  * эвристика по DISTRIBUTED BY vs JOIN keys (совпадает ли дистрибьюция с условием соединения).
  * рекомендации: предфильтры, предагрегации, согласование распределения / REPLICATE для мелких.
- Запросы, которые пишут в НЕ временные таблицы, не выполняем (даже EXPLAIN можно, но мы их пропустим).
- Планируем INSERT/SELECT/CTE, но только EXPLAIN — без изменения данных.
"""

def strip_comments_sql(s: str) -> str:
    # Удаление /* */ и -- (простая, но устойчивая версия)
    out = []
    in_block = False
    i = 0
    while i < len(s):
        if not in_block and s.startswith("/*", i):
            in_block = True; i += 2; continue
        if in_block and s.startswith("*/", i):
            in_block = False; i += 2; continue
        if not in_block:
            out.append(s[i])
        i += 1
    txt = "".join(out)
    # построчно глушим --
    return re.sub(r"(^|\n)\s*--.*(?=\n|$)", r"\1", txt)

@dataclass
class Stmt:
    idx: int
    sql: str
    start: int
    end: int
    path: str

def split_sql_with_lines(sql_text: str, file_path: str) -> List[Stmt]:
    # Сплитим по ';' вне блок-комментов
    text = sql_text
    stmts, buf, start_ln, ln, in_block = [], [], 1, 1, False
    for line in text.splitlines():
        if "/*" in line and "*/" not in line: in_block = True
        if "*/" in line: in_block = False
        buf.append(line)
        if ";" in line and not in_block:
            chunk = "\n".join(buf)
            parts = chunk.split(";")
            for part in parts[:-1]:
                p = part.strip()
                if p:
                    stmts.append(Stmt(len(stmts)+1, p, start_ln, ln, file_path))
                start_ln = ln
            buf = [parts[-1]]
        ln += 1
    tail = "\n".join(buf).strip()
    if tail:
        stmts.append(Stmt(len(stmts)+1, tail, start_ln, ln-1, file_path))
    return stmts

def low(s: str) -> str:
    return strip_comments_sql(s).lower().strip()

def is_temp_create(sql: str) -> bool:
    h = low(sql)
    return h.startswith("create") and (" temp " in f" {h} " or " temporary " in f" {h} ")

def temp_name_from_create(sql: str) -> Optional[str]:
    m = re.search(r"create\s+(?:temporary|temp)\s+table\s+([^\s(]+)", strip_comments_sql(sql), re.IGNORECASE)
    return m.group(1).split(".")[-1].strip('"').lower() if m else None

def is_ctas(sql: str) -> bool:
    return bool(re.search(r"create\s+(?:temporary|temp)\s+table\s+\S+\s+as\s+select", strip_comments_sql(sql), re.IGNORECASE|re.DOTALL))

def is_drop(sql: str) -> bool:
    return low(sql).startswith("drop ")

def is_set(sql: str) -> bool:
    h = low(sql)
    return h.startswith("set ") or h.startswith("reset ")

def is_explainable(sql: str) -> bool:
    h = low(sql)
    return h.startswith("select") or h.startswith("with") or h.startswith("insert") or h.startswith("update") or h.startswith("delete") or h.startswith("merge") or h.startswith("create")  # CTAS

def insert_target_name(sql: str) -> Optional[str]:
    m = re.search(r"\binsert\s+into\s+([a-zA-Z_]\w*(?:\.[a-zA-Z_]\w*)?)", strip_comments_sql(sql), re.IGNORECASE)
    return m.group(1).split(".")[-1].lower() if m else None

def update_target_name(sql: str) -> Optional[str]:
    m = re.search(r"\bupdate\s+([a-zA-Z_]\w*(?:\.[a-zA-Z_]\w*)?)", strip_comments_sql(sql), re.IGNORECASE)
    return m.group(1).split(".")[-1].lower() if m else None

def delete_target_name(sql: str) -> Optional[str]:
    m = re.search(r"\bdelete\s+from\s+([a-zA-Z_]\w*(?:\.[a-zA-Z_]\w*)?)", strip_comments_sql(sql), re.IGNORECASE)
    return m.group(1).split(".")[-1].lower() if m else None

def mentioned_tables(sql: str) -> Set[str]:
    txt = strip_comments_sql(sql)
    names = set()
    for kw in ("from","join","update","delete\s+from","into"):
        for t in re.findall(rf"\b{kw}\s+([a-zA-Z_]\w*(?:\.[a-zA-Z_]\w*)?)", txt, re.IGNORECASE):
            names.add(t.split(".")[-1].lower())
    return names

def is_temp_name(name: str) -> bool:
    # В Greenplum temp в текущей сессии живут в schema pg_temp_NNN — имена любые.
    # Мы считаем времянкой то, что встретилось в CREATE TEMP ранее (карта temp_names)
    return False  # реальная проверка идёт по карте temp_defs

@dataclass
class TempDef:
    name: str
    ddl_stmt: Stmt
    distributed: Optional[str] = None     # RAW: 'DISTRIBUTED BY (...)' / 'DISTRIBUTED RANDOMLY' / 'DISTRIBUTED REPLICATED'
    dist_cols: Tuple[str, ...] = field(default_factory=tuple)  # имена колонок (lower) из BY (...)
    is_ctas: bool = False

def parse_distribution(sql: str) -> Tuple[Optional[str], Tuple[str,...]]:
    txt = strip_comments_sql(sql)
    m_rep = re.search(r"\bdistributed\s+replicated\b", txt, re.IGNORECASE)
    if m_rep: return ("REPLICATED", tuple())
    m_rand = re.search(r"\bdistributed\s+randomly\b", txt, re.IGNORECASE)
    if m_rand: return ("RANDOMLY", tuple())
    m_by = re.search(r"\bdistributed\s+by\s*\(([^)]+)\)", txt, re.IGNORECASE|re.DOTALL)
    if m_by:
        cols = tuple(c.strip().strip('"').lower() for c in m_by.group(1).split(","))
        return ("BY", cols)
    return (None, tuple())

def extract_join_keys(cond_text: str, table_a: str, table_b: str) -> Tuple[Tuple[str,...], Tuple[str,...]]:
    """
    Пытаемся из Hash/Merge Cond вытащить пары ключей между table_a и table_b.
    Примеры условий:
      "(a.x = b.x) AND (a.y = b.y)" / "(b.k1 = a.k1)"
    Возвращаем два одинаковой длины набора ключей в порядке a-стороны и b-стороны.
    """
    if not cond_text: return (tuple(), tuple())
    a_keys, b_keys = [], []
    # нормализуем
    t = cond_text.replace('"','').lower()
    # разбиваем на простые равенства
    parts = re.split(r"\)\s+and\s+\(", t.strip("() "))
    for p in parts:
        m = re.search(r"([a-z_]\w*)\.([a-z_]\w*)\s*=\s*([a-z_]\w*)\.([a-z_]\w*)", p)
        if not m: continue
        t1,c1,t2,c2 = m.groups()
        # кто из них a/b — решаем по совпадению имени
        if t1.endswith(f".{table_a}") or t1==table_a:
            a_keys.append(c1); b_keys.append(c2)
        elif t1.endswith(f".{table_b}") or t2==table_a:
            # не всегда встречается schema.table — мы просто сравним имена
            if t1==table_a:
                a_keys.append(c1); b_keys.append(c2)
            elif t2==table_a:
                a_keys.append(c2); b_keys.append(c1)
            else:
                # эвристика: если t1==table_b
                if t1==table_b:
                    a_keys.append(c2); b_keys.append(c1)
                elif t2==table_b:
                    a_keys.append(c1); b_keys.append(c2)
        else:
            # базовая эвристика: если одно из имён совпадает со второй таблицей
            if t1==table_b:
                a_keys.append(c2); b_keys.append(c1)
            elif t2==table_b:
                a_keys.append(c1); b_keys.append(c2)
    return (tuple(a_keys), tuple(b_keys))

def build_explain(sql: str) -> str:
    return f"EXPLAIN (VERBOSE, FORMAT JSON) {sql}"

def as_limit0(sql: str) -> str:
    # Превращаем CTAS в CTAS LIMIT 0 (сохраняем "AS <select> LIMIT 0")
    txt = strip_comments_sql(sql)
    m = re.search(r"\bas\b(.*)$", txt, re.IGNORECASE|re.DOTALL)
    if not m:
        return sql
    sel = m.group(1).strip()
    if re.search(r"\blimit\s+\d+\b", sel, re.IGNORECASE):
        return sql  # уже есть лимит
    lim = CTAS_CREATE_LIMIT if CTAS_CREATE_LIMIT is not None else 0
    return txt[:m.start(1)] + f"{sel}\n LIMIT {lim}"

def main(sql_path: str, dsn: str):
    full = open(sql_path, "r", encoding="utf-8").read()
    stmts = split_sql_with_lines(full, sql_path)

    # Собираем все TEMP DDL
    temp_defs: Dict[str, TempDef] = {}
    for st in stmts:
        if is_temp_create(st.sql):
            name = temp_name_from_create(st.sql)
            if not name: continue
            dist, cols = parse_distribution(st.sql)
            td = TempDef(name=name, ddl_stmt=st, distributed=dist, dist_cols=cols, is_ctas=is_ctas(st.sql))
            temp_defs[name] = td

    # Помощники подключения
    conn = psycopg2.connect(dsn); conn.autocommit=False
    cur  = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)

    def sp(tag): cur.execute(f"SAVEPOINT {tag}")
    def spr(tag):
        try: cur.execute(f"ROLLBACK TO SAVEPOINT {tag}")
        except Exception: pass

    created: Set[str] = set()
    errors: List[str] = []
    reports_temp: Dict[str, Dict] = {t: {
        "ddl_lines": (temp_defs[t].ddl_stmt.start, temp_defs[t].ddl_stmt.end),
        "distributed": temp_defs[t].distributed,
        "dist_cols": temp_defs[t].dist_cols,
        "uses": [],                   # список (stmt_idx, kind, summary)
        "plans": [],                  # JSON plans for statements where temp involved
        "notes": []                   # WARN/INFO strings
    } for t in temp_defs}

    # Создаём все временные таблицы (DDL / CTAS LIMIT 0)
    for tname, td in temp_defs.items():
        sql = td.ddl_stmt.sql
        try:
            sp(f"ct_{tname}")
            if td.is_ctas:
                cur.execute(as_limit0(sql))
            else:
                cur.execute(sql)
            created.add(tname)
        except Exception as e:
            spr(f"ct_{tname}")
            errors.append(f"[CREATE TEMP {tname}] {e}")
        # не коммитим, останемся в транзакции до ROLLBACK

    # Функция: относится ли оператор к времянкам и «безопасен ли»
    def is_safe_for_temp_explain(st: Stmt) -> bool:
        # Пропускаем любые DROP; SET/RESET исполним.
        if is_drop(st.sql): return False
        if is_set(st.sql):  return True
        if not is_explainable(st.sql): return False

        # Не планируем команды, которые пишут в НЕ временные таблицы
        tgt = insert_target_name(st.sql) or update_target_name(st.sql) or delete_target_name(st.sql)
        if tgt and tgt not in temp_defs:
            return False

        # Если в операторе фигурирует хотя бы одна временная таблица — да
        tabs = mentioned_tables(st.sql)
        return any(t in temp_defs for t in tabs)

    # Построение EXPLAIN JSON с автодиагностикой
    def explain_collect(st: Stmt):
        tag = f"exp_{st.idx}"
        try:
            sp(tag)
            cur.execute(build_explain(st.sql))
            rows = cur.fetchall()
            cur.execute(f"RELEASE SAVEPOINT {tag}")
        except Exception as e:
            spr(tag)
            errors.append(f"[EXPLAIN stmt#{st.idx}] {e} [{st.path} {st.start}–{st.end}]")
            return None
        if not rows: return None
        return rows[0]["QUERY PLAN"]

    # Аналитика плана (SeqScan, Heavy nodes, Motion, Join conds для эвристики dists)
    def analyze_plan(plan_json, stmt: Stmt):
        out_notes = []
        involved_temps: Set[str] = set()

        try:
            root = plan_json[0]["Plan"]
        except Exception:
            return out_notes, involved_temps, {}

        heavy_motions = []
        table_estimates = {}  # name -> (rows,width)

        def walk(n):
            if not isinstance(n, dict): return
            ntype = n.get("Node Type","")
            rows  = float(n.get("Plan Rows", 0) or 0)
            width = float(n.get("Plan Width", 0) or 0)
            est_mb = rows*max(width,1)/1024/1024

            rel  = n.get("Relation Name")
            alias= n.get("Alias")

            if rel and rel.lower() in temp_defs:
                involved_temps.add(rel.lower())
                table_estimates[rel.lower()] = (rows, width)

            if ntype == "Seq Scan" and not n.get("Filter"):
                lvl = "WARN" if est_mb >= SIZE_MB_SEQSCAN_WARN else "INFO"
                out_notes.append(f"[{lvl}] Seq Scan без фильтра: {rel or alias or '?'} — ~{rows:,.0f} × {int(width)} B/row ({est_mb:.2f} MB).".replace(","," "))

            if ntype in ("Hash","Aggregate","Sort"):
                lvl = "WARN" if est_mb >= SIZE_MB_HEAVY_WARN else "INFO"
                out_notes.append(f"[{lvl}] Тяжёлый узел {ntype} — ~{est_mb:.1f} MB.")

            if "Motion Type" in n:
                lvl = "WARN" if est_mb >= SIZE_MB_MOTION_WARN else "INFO"
                mt = n.get("Motion Type")
                heavy_motions.append((mt, est_mb))
                out_notes.append(f"[{lvl}] {mt} — ~{est_mb:.1f} MB.")

            # JOIN эвристика — пробуем разобрать cond
            if ntype in ("Hash Join","Merge Join","Nested Loop"):
                cond = n.get("Hash Cond") or n.get("Merge Cond") or n.get("Join Filter")
                # попробуем для каждой пары времянка↔другая таблица вытащить ключи
                for t in list(temp_defs.keys()):
                    if cond and (f".{t}." in (cond.lower()) or cond.lower().find(f"{t}.")>=0):
                        # в JSON нет точных имён обеих таблиц; эвристически проверим присутствие t
                        pass

            for k in ("Plans","Inner","Outer"):
                ch = n.get(k)
                if isinstance(ch, list):
                    for c in ch: walk(c)
                elif isinstance(ch, dict):
                    walk(ch)

        walk(root)
        # Сводные метрики по Motion
        if heavy_motions:
            worst = max(heavy_motions, key=lambda x: x[1])
            out_notes.append(f"[INFO] Motion суммарно: {len(heavy_motions)} узлов, худший {worst[0]} ~{worst[1]:.1f} MB.")

        return out_notes, involved_temps, table_estimates

    # Основной проход: EXPLAIN только для операторов, где задействованы временные таблицы
    for st in stmts:
        if is_set(st.sql):
            try:
                sp(f"set_{st.idx}"); cur.execute(st.sql)
            except Exception:
                spr(f"set_{st.idx}")
            continue
        if not is_safe_for_temp_explain(st):
            continue

        plan = explain_collect(st)
        if not plan: continue
        notes, temps, ests = analyze_plan(plan, st)
        for t in temps:
            rep = reports_temp[t]
            rep["uses"].append((st.idx, low(st.sql).split()[0], f"{st.path} {st.start}–{st.end}"))
            rep["plans"].append(plan)
            rep["notes"].extend(notes)
            # сохраним оценку размера
            if "est" not in rep:
                rep["est"] = {}
            rep["est"][st.idx] = ests.get(t, (0,0))

    # Оценка DISTRIBUTED BY vs JOIN ключи (эвристика на основе найденных условий в планах)
    def evaluate_distribution_heuristic(rep: Dict, tname: str):
        dist = rep.get("distributed")
        dcols= rep.get("dist_cols", tuple())
        notes= rep["notes"]
        # если REPLICATED — ок, но предупредим про габариты
        est_rows, est_width = 0, 0
        if "est" in rep and rep["est"]:
            # возьмём худший по объёму
            mx = 0
            for (_sid,(r,w)) in rep["est"].items():
                vol = r*max(w,1)
                if vol>mx: mx, est_rows, est_width = vol, r, w
        vol_mb = est_rows*max(est_width,1)/1024/1024

        if dist == "REPLICATED":
            if vol_mb > SMALL_REPLICATE_CEILMB:
                notes.append(f"[WARN] {tname}: REPLICATED и объём ~{vol_mb:.1f} MB — рассмотрите DISTRIBUTED BY по ключам JOIN.")
            else:
                notes.append(f"[INFO] {tname}: REPLICATED — малый объём (~{vol_mb:.1f} MB) допустим.")
            return
        if dist == "RANDOMLY":
            notes.append(f"[WARN] {tname}: DISTRIBUTED RANDOMLY — вероятны Redistribute Motion на JOIN; лучше BY по ключу(ам) JOIN.")
            return
        if dist == "BY":
            if not dcols:
                notes.append(f"[WARN] {tname}: DISTRIBUTED BY без колонок? Проверьте DDL.")
                return
            # эвристика: если в планах часто встречаются Motion WARN — возможно, BY не совпадает с ключами JOIN
            has_motion_warn = any(n.startswith("[WARN] Redistribute") or ("Motion" in n and n.startswith("[WARN]")) for n in notes)
            if has_motion_warn:
                notes.append(f"[WARN] {tname}: есть тяжёлые Motion; проверьте, совпадают ли DISTRIBUTED BY {list(dcols)} с равенственными ключами JOIN.")
            else:
                notes.append(f"[INFO] {tname}: DISTRIBUTED BY {list(dcols)} — выглядит разумно (явных тяжёлых Motion не найдено).")

    for tname, rep in reports_temp.items():
        evaluate_distribution_heuristic(rep, tname)

    # Итог: печать отчёта
    def print_section(title: str):
        print("\n" + title)
        print("="*len(title))

    print_section("GREENPLUM PLAN DOCTOR — TEMP SUMMARY")
    print(f"Файл: {SQL_FILE_PATH}")
    print(f"Найдено временных таблиц: {len(temp_defs)}")
    if errors:
        print(f"\nОшибки ({len(errors)}):")
        for e in errors:
            print("  " + e)

    for tname, rep in reports_temp.items():
        print_section(f"TEMP {tname}")
        dl = rep["ddl_lines"]
        print(f"DDL строки: {dl[0]}–{dl[1]}")
        dist = rep.get("distributed") or "—"
        dcols= list(rep.get("dist_cols") or [])
        print(f"DISTRIBUTION: {dist} {(' ' + str(dcols)) if dcols else ''}")

        uses = rep["uses"]
        if uses:
            print(f"Использования ({len(uses)}):")
            for (sid, kind, where) in uses[:20]:
                print(f"  stmt#{sid:>3}  {kind:6}  @ {where}")
            if len(uses) > 20:
                print(f"  … ещё {len(uses)-20} операторов")

        # Сводные подсказки
        notes = rep["notes"]
        if notes:
            print("Подсказки:")
            # Сгруппируем WARN впереди
            for n in sorted(notes, key=lambda x: 0 if x.startswith("[WARN]") else 1):
                print("  " + n)
        else:
            print("Подсказки: нет явных проблем (по планам).")

    # откатываемся на всякий
    try: conn.rollback()
    finally:
        cur.close(); conn.close()

if __name__ == "__main__":
    main(SQL_FILE_PATH, DEFAULT_DSN)
