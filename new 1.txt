#!/usr/bin/bash
  set -euo pipefail

  unset http_proxy
  unset https_proxy
  unset HTTP_PROXY
  unset HTTPS_PROXY
  export GIT_SSL_NO_VERIFY=1

  REPO_ETL="/root/git_dir/etl"
  REPO_META="/root/git_dir/meta"
  APP_ROOT="/app"
  BRANCH="main"

  echo "[ci_cd] start $(date -u +"%Y-%m-%dT%H:%M:%SZ")"

  # Обновляем git-репозиторий etl до чистого origin/main
  git -C "${REPO_ETL}" fetch origin
  git -C "${REPO_ETL}" checkout "${BRANCH}"
  git -C "${REPO_ETL}" reset --hard "origin/${BRANCH}"
  git -C "${REPO_ETL}" clean -fd

  # Готовим целевые каталоги приложения
  mkdir -p "${APP_ROOT}/meta_info"
  mkdir -p "${APP_ROOT}/config_files/dm"
  mkdir -p "${APP_ROOT}/config_files/dm_view"

  # Полностью пересобираем target из source
  rm -rf "${APP_ROOT}/meta_info"/*
  rm -rf "${APP_ROOT}/config_files/dm"/*
  rm -rf "${APP_ROOT}/config_files/dm_view"/*

  # Копируем актуальные данные
  cp -a "${REPO_ETL}/meta_info/." "${APP_ROOT}/meta_info/"
  cp -a "${REPO_META}/dm/." "${APP_ROOT}/config_files/dm/"
  cp -a "${REPO_META}/dm_view/." "${APP_ROOT}/config_files/dm_view/"

  echo "[ci_cd] synced meta_info, dm, dm_view"
  echo "[ci_cd] done"
