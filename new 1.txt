class SqlExecuteOperator(BaseOperator):
    """
    Performs checks against a DB. Expects a SQL returning one row.
    get_db_hook() must return a DB hook.
    """
    template_fields = ('sql',)
    template_ext = ('.hql', '.sql',)
    ui_color = '#fff7e6'

    @apply_defaults
    def __init__(
            self,
            sql: str,
            conn_id: Optional[str] = None,
            *args, **kwargs
    ):
        super(SqlExecuteOperator, self).__init__(*args, **kwargs)
        self.conn_id = conn_id
        self.sql = sql

    def get_db_hook(self):
        return BaseHook.get_hook(conn_id=self.conn_id)

    def execute(self, context: Any = None) -> int:
        logging.info('Executing SQL statement: %s', self.sql)
        records = self.get_db_hook().get_first(self.sql)
        logging.info("Record: %s", records)
        return int(records[0]) if records and records[0] is not None else 0

class SqlMetaMinioOperator(SQLExecuteQueryOperator):
    template_fields: Sequence[str] = ("sql", *SQLExecuteQueryOperator.template_fields)
    template_ext: Sequence[str] = (".sql",)
    template_fields_renderers = {"columns": "sql",}
    ui_color = "#89d3f5"
    
    def __init__(
            self,
            sql: str,
            autocommit: bool,
            conn_id: str,
            minio_conn: str,
            secure=bool,     # оставил как у вас; (хотя лучше secure: bool = False)
            **kwargs
    ):
        super().__init__(sql=sql, **kwargs)
        self.sql = sql
        self.autocommit = autocommit
        self.conn_id = conn_id
        self.minio_conn = minio_conn
        self.secure = secure
        self.params.update()

    # === FIX: готовим params ДО рендера Jinja, чтобы {{ params.columns }} существовал ===
    def prepare_template_context(self, context: Context) -> dict:
        ctx = super().prepare_template_context(context)

        # берём базовые params из self и из ctx, объединяем (ctx перекрывает self)
        base_params: Dict[str, Any] = (getattr(self, "params", {}) or {}).copy()
        ctx.setdefault("params", {})
        p_in_ctx: Dict[str, Any] = ctx["params"] or {}
        p: Dict[str, Any] = {**base_params, **p_in_ctx}

        # дефолты (шаблон не упадёт, даже если YAML пустой)
        p.setdefault("columns", "")
        p.setdefault("settings_external_table", "")

        # из чего строим путь к YAML
        schema = p.get("table_schema")
        table = p.get("table_name")
        if not schema or not table:
            # ключей пока нет — отдаём пустые поля; Jinja не упадёт
            ctx["params"] = p
            return ctx

        meta_path = os.path.join(
            os.environ.get("AIRFLOW_HOME"),
            'meta_info',
            'database',
            'clickhouse',
            'schema_name',
            schema,
            'table' if schema != 'dm_view' else 'view',
            table,
            'meta_data_file.yaml'
        )

        yaml_data = {}
        if os.path.exists(meta_path):
            with open(meta_path, 'r') as file:
                yaml_data = yaml.safe_load(file) or {}

            # собираем columns в вашем формате (с одинарными кавычками и переносами)
            columns = ""
            for att in yaml_data.get('attributes', []):
                columns += "'" + att['column_name_click'] + "',\n"
            p['columns'] = columns[:-2] if columns else ""

            # опциональные settings
            if yaml_data.get('settings_external_table'):
                _set = yaml_data.get('settings_external_table')
                p['settings_external_table'] = 'settings ' + ', '.join(
                    [f"{key} = {value}" for key, value in _set.items()]
                )

        ctx["params"] = p
        return ctx

    def remove_file(self, client, load_dir):
        for removed_object in client.list_objects(
                bucket_name=self.params.get('minio_bucket'),
                prefix=f"{load_dir}",
                recursive=True
        ):
            self.log.info(f'Delete {removed_object.object_name}')
            client.remove_object(
                bucket_name=self.params.get('minio_bucket'),
                object_name=removed_object.object_name
            )

            f'{hook.host}:{hook.port}',
            access_key=hook.login,
            secret_key=hook.password,
