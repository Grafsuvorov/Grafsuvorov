python3 - <<'PY'
  import os, requests
  issue_id="2-56109"
  url=f"https://yt.rusal.ru/api/issues/{issue_id}/activities"
  headers={"Authorization":f"Bearer {os.environ['YOUTRACK_TOKEN']}", "Accept":"application/json"}
  params={
    "fields":"author(name,login),timestamp,field(name),added(name),removed(name),to(name),target(customFields(name,value(name)))",
    "categories":"IssueCustomFieldHistoryCategory,IssueHistoryCategory,CommentsCategory,WorkItemCategory,ProjectCategory"
  }
  r=requests.get(url,headers=headers,params=params,verify=False)
  print("status", r.status_code)
  print(r.text[:2000])
  PY

  python3 - <<'PY'
  import os, requests
  issue_id="2-56109"
  url=f"https://yt.rusal.ru/api/issues/{issue_id}/activitiesPage"
  headers={"Authorization":f"Bearer {os.environ['YOUTRACK_TOKEN']}", "Accept":"application/json"}
  params={
    "fields":"activities(author(name,login),timestamp,field(name),added(name),removed(name),to(name),target(customFields(name,value(name))))",
    "categories":"IssueCustomFieldHistoryCategory,IssueHistoryCategory,CommentsCategory,WorkItemCategory",
    "reverse":"true"
  }
  r=requests.get(url,headers=headers,params=params,verify=False)
  print("status", r.status_code)
  print(r.text[:2000])
  PY

  ———

  Если после этого нет State/Assignee — значит API не отдаёт историю. Тогда можно только:

  - current state/assignee из customFields
  - комментарии
  - work items
