with cte as (
    select
        temp1.unit_balance_code,
        temp1.asset_main_code,
        temp1.asset_sub_code,
        temp1.dt_from,
        row_number() over (
            partition by temp1.unit_balance_code, temp1.asset_main_code, temp1.asset_sub_code
            order by temp1.dt_from
        ) as rn
    from (
        select
            t.unit_balance_code,
            t.asset_main_code,
            t.asset_sub_code,
            t.dt_from
        from (
            select
                quot.unit_balance_code,
                quot.asset_main_code,
                quot.asset_sub_code,
                quot.dt_assignment_valid_from as dt_from
            from tmp_fixed_asset_quotation_td as quot
            union all
            select
                conv.unit_balance_code,
                conv.asset_main_code,
                conv.asset_sub_code,
                conv.dt_conservated_from as dt_from
            from tmp_fixed_asset_conservation_td as conv
        ) as t
        group by
            t.unit_balance_code,
            t.asset_main_code,
            t.asset_sub_code,
            t.dt_from
    ) as temp1
),

sep_int as (
    select
        t1.unit_balance_code,
        t1.asset_main_code,
        t1.asset_sub_code,
        coalesce(t1.dt_from, to_date('19000101'::text, 'YYYYMMDD')) as dt_from,
        case
            when t1.dt_from is null and t2.dt_from is null then (tcf_test.dt_from - '1day'::interval)::date
            when t2.dt_from is null then to_date('99991231'::text, 'YYYYMMDD')
            else (t2.dt_from - '1day'::interval)::date
        end as dt_to,
        t1.rn
    from cte as t1
    left join cte as t2
        on
            t1.unit_balance_code = t2.unit_balance_code
            and t1.asset_main_code = t2.asset_main_code
            and t1.asset_sub_code = t2.asset_sub_code
            and t1.rn = t2.rn - 1
    left join cte as tcf_test
        on
            tcf_test.unit_balance_code = t1.unit_balance_code
            and tcf_test.asset_main_code = t1.asset_main_code
            and tcf_test.asset_sub_code = t1.asset_sub_code
            and tcf_test.rn = t1.rn - 1
    where
        t1.unit_balance_code is not null
        and t1.asset_main_code is not null
        and t1.asset_sub_code is not null
),
