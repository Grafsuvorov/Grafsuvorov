drop table if exists match_candidates;

create temporary table match_candidates
distributed by (rn)
as
-- 1) Логика РФ-контейнеры
select
    sw.rn,
    sw.storage_cost_calculation_type_name,
    fb.purchase_contract_code,
    fb.position_line_item_code,
    fb.datab,
    fb.datbi,
    fb.price_uom_code,
    fb.service_code,
    fb.konws,
    fb.currency_dogovor,

    case when fb.zzport   = sw.port_of_discharge_code or fb.zzport   is null then 1 else 0 end as m_port,
    case when fb.zzforma  = sw.material_shape_code    or fb.zzforma  is null then 1 else 0 end as m_shape,
    case when fb.zzet_tarif = sw.etsng_code           or fb.zzet_tarif is null then 1 else 0 end as m_etsng,
    case when fb.sdabw    = sw.transport_type_code    or fb.sdabw    is null then 1 else 0 end as m_transport,
    case when fb.zzterminal = sw.warehouse_code       or fb.zzterminal is null then 1 else 0 end as m_terminal,
    case when fb.vzzzone  = sw.warehouse_storage_area_code or fb.vzzzone is null then 1 else 0 end as m_area
from dm_calc.storage_sales_bundles_weight sw
join for_1079_base_clean fb
  on fb.bukrs   = '1120'
 and fb.lifnr   = sw.creditor_in_shipment_instruction_code
 and fb.werks   = '1121'
 and fb.service_code = 'S49'
 and (fb.zzport   = sw.port_of_discharge_code    or fb.zzport   is null)
 and (fb.zzforma  = sw.material_shape_code       or fb.zzforma  is null)
 and (fb.zzet_tarif = sw.etsng_code              or fb.zzet_tarif is null)
 and (fb.sdabw    = sw.transport_type_code       or fb.sdabw    is null)
 and sw.dt_for_price_search_in_purchase_contract between fb.dt_valid_from and fb.dt_valid_to
 and sw.dt_for_price_search_in_purchase_contract between fb.datab       and fb.datbi
where sw.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'

UNION ALL
