-- 1.1. Упрощённый адрес: один country_code на address_code
CREATE TEMPORARY TABLE adrc ON COMMIT DROP AS
SELECT
    address_code,
    MIN(country_code) AS country_code
FROM dict_dds.address
GROUP BY address_code
DISTRIBUTED REPLICATED;


-- 1.2. Единый справочник по hub’ам с RU/EN именами
CREATE TEMPORARY TABLE hub_texts_unified ON COMMIT DROP AS
SELECT
    th.transport_hub_code,
    MAX(CASE WHEN th.language_code = 'R' THEN th.transport_hub_name END) AS name_r,
    MAX(CASE WHEN th.language_code = 'E' THEN th.transport_hub_name END) AS name_e
FROM dict_dds.transport_hub_texts th
GROUP BY th.transport_hub_code
DISTRIBUTED REPLICATED;

        ON matet.material_code  = base.material_code
       AND matet.language_code  = 'R'

    LEFT JOIN dict_dds.transportation_service_texts AS zsrvpost
        ON zsrvpost.transportation_service_code = base.service_code
       AND zsrvpost.language_code              = 'R'

    LEFT JOIN dict_dds.etsng_texts AS ette
        ON ette.etsng_code    = base.etsng_code
       AND ette.language_code = 'R'

    -- Определение региона удалённого склада
    LEFT JOIN dict_dds.transport_hub AS tvkn
        ON tvkn.transport_hub_code = base.warehouse_code

    LEFT JOIN adrc AS adrc_2
        ON adrc_2.address_code = tvkn.address_code

    LEFT JOIN dict_dds.country AS r_sd556
        ON r_sd556.country_code = adrc_2.country_code

    LEFT JOIN dict_dds.market_region1_texts AS t25a1
        ON t25a1.market_region1_code = r_sd556.market_region1_code
       AND t25a1.language_code       = 'R'

    -- Завод поставки
    LEFT JOIN dict_dds.plant_and_subsidiary AS pas
        ON pas.plant_code = base.receiving_plant_code

    -- Страна склада (для country_of_remote_warehouse_name)
    LEFT JOIN dict_dds.transport_hub AS trhu
        ON trhu.transport_hub_code = base.warehouse_code

    LEFT JOIN adrc AS adrc
        ON adrc.address_code = trhu.address_code

    LEFT JOIN dict_dds.country_texts AS cotex
        ON cotex.country_code   = adrc.country_code
       AND cotex.language_code  = 'R'

    -- Единый справочник имён hub’ов

        ON h_dep.transport_hub_code = base.transport_departure_hub_code

    LEFT JOIN hub_texts_unified AS h_dest
        ON h_dest.transport_hub_code = base.transport_destination_hub_code

    LEFT JOIN hub_texts_unified AS h_wh
        ON h_wh.transport_hub_code = base.warehouse_code

    LEFT JOIN hub_texts_unified AS h_area
        ON h_area.transport_hub_code = base.warehouse_storage_area_code

    LEFT JOIN hub_texts_unified AS h_port
        ON h_port.transport_hub_code = base.port_of_discharge_code

    LEFT JOIN hub_texts_unified AS h_remote
        ON h_remote.transport_hub_code = base.remote_warehouse_code

    -- Собственник металла (через поставку хранения)
    LEFT JOIN dds.delivery_document_position AS deldp
        ON deldp.delivery_code                 = base.delivery_for_storage_calculation_code
       AND deldp.delivery_position_line_item_code = '000010'

    LEFT JOIN dict_dds.plant_and_subsidiary AS pas_1124
        ON pas_1124.plant_code = deldp.plant_producer_code

    -- Этап перевозки
    LEFT JOIN dict_dds.transportation_stage_texts AS trst
        ON trst.transportation_stage_code = base.transportation_stage_code
       AND trst.language_code            = 'R'

    -- Экспедитор хранения
    LEFT JOIN dict_dds.counterparty AS counter
        ON counter.counterparty_code = base.forwarder_code

    -- Удалённый склад (код страны)
    LEFT JOIN dict_dds.transport_hub AS trhu_2
        ON trhu_2.transport_hub_code = base.remote_warehouse_code

    LEFT JOIN adrc AS adrc_3
        ON adrc_3.address_code = trhu_2.address_code

    -- Форма металла через спецификацию
    LEFT JOIN dict_dds.material_specification AS dd_ms_s_n
        ON dd_ms_s_n.material_code = base.material_code

    LEFT JOIN dict_dds.material_shape_texts AS mf
        ON mf.shape_code    = dd_ms_s_n.shape_code
       AND mf.language_code = 'R'

    -- Тип подвижного состава
    LEFT JOIN dict_dds.transport_transfer_type_texts AS ipd
        ON ipd.transport_transfer_type_code = base.transport_type_code
       AND ipd.language_code               = 'R'

    -- Кредитор хранения (адрес)
    LEFT JOIN dict_dds.counterparty AS counter_2
        ON counter_2.counterparty_code = base.creditor_in_shipment_instruction_code

    LEFT JOIN dict_dds.address AS addr_8
        ON addr_8.address_code = counter_2.address_code
       AND addr_8.international_display_format_code IS NULL
;
