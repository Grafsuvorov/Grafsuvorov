import requests
import json
import urllib3

urllib3.disable_warnings()

YOUTRACK_URL = ""
YOUTRACK_TOKEN = ""
ISSUE_ID = "DWH-10152"

HEADERS = {
    "Authorization": f"Bearer {YOUTRACK_TOKEN}",
    "Accept": "application/json",
}


def api_get(path, params=None):
    r = requests.get(
        f"{YOUTRACK_URL}{path}",
        headers=HEADERS,
        params=params or {},
        verify=False,
        timeout=60,
    )
    print(f"\n>>> GET {path} | status={r.status_code}")

    if r.status_code != 200:
        print("ERROR BODY:", r.text)
        return None

    return r.json()


# -------------------------
# 1. ISSUE (основной источник)
# -------------------------
issue = api_get(
    f"/api/issues/{ISSUE_ID}",
    {
        "fields": "idReadable,summary,customFields(name,value(name))"
    },
)

print("\n=== ISSUE CUSTOM FIELDS ===")

direction_issue = None

if issue:
    for f in issue.get("customFields", []):
        print(f["name"], "=>", f.get("value"))

        if f["name"] == "Направление":
            direction_issue = f.get("value")

print("\n=== RESULT FROM ISSUE ===")
print("Direction:", direction_issue)


# -------------------------
# 2. ISSUE /customFields (альтернатива)
# -------------------------
custom_fields = api_get(
    f"/api/issues/{ISSUE_ID}/customFields",
    {
        "fields": "name,value(name)"
    },
)

print("\n=== /customFields ===")

direction_cf = None

if custom_fields:
    for f in custom_fields:
        print(f["name"], "=>", f.get("value"))

        if f["name"] == "Направление":
            direction_cf = f.get("value")

print("\n=== RESULT FROM /customFields ===")
print("Direction:", direction_cf)


# -------------------------
# 3. ACTIVITIES (история)
# -------------------------
print("\n=== ACTIVITIES ===")

direction_history = None

activities = api_get(
    f"/api/issues/{ISSUE_ID}/activitiesPage",
    {
        "categories": "CustomFieldCategory",
        "fields": "activities(field(name),added(name))"
    },
)

if activities:
    for a in activities.get("activities", []):
        field = (a.get("field") or {}).get("name")

        if field == "Направление":
            print("FOUND IN HISTORY:", a)

            if a.get("added"):
                direction_history = a["added"].get("name")

print("\n=== RESULT FROM HISTORY ===")
print("Direction:", direction_history)


# -------------------------
# 4. FINAL RESULT
# -------------------------
print("\n=== FINAL RESULT ===")

direction = (
    direction_issue
    or direction_cf
    or direction_history
)

print("FINAL Direction:", direction)

if not direction:
    print("\n❌ Поле 'Направление' отсутствует вообще (даже в истории)")
