CREATE TEMPORARY TABLE for_1079_base
WITH (appendonly = true, compresstype = zstd, compresslevel = 5)
ON COMMIT DROP AS
SELECT
    for_1079_base_calc.purchase_contract_code,
    for_1079_base_calc.position_line_item_code,
    for_1079_base_calc.bukrs,
    for_1079_base_calc.lifnr,
    for_1079_base_calc.zzport,
    for_1079_base_calc.kdatb,
    for_1079_base_calc.kdate,
    for_1079_base_calc.werks,
    for_1079_base_calc.zzsrvpos,
    tmp_1.range_low_value AS tmp_1,
    tmp_2.range_low_value AS tmp_2,
    for_1079_base_calc.zzforma,
    for_1079_base_calc.zzet_tarif,
    for_1079_base_calc.sdabw,
    for_1079_base_calc.datbi,
    for_1079_base_calc.datab,
    for_1079_base_calc.dt_valid_to,
    for_1079_base_calc.dt_valid_from,
    for_1079_base_calc.purchase_document_subtype_code,
    for_1079_base_calc.kopos,
    for_1079_base_calc.konwa,
    for_1079_base_calc.zzterminal,
    for_1079_base_calc.vzzzone,
    for_1079_base_calc.konws,
    for_1079_base_calc.service_code,
    for_1079_base_calc.price_uom_code,
    for_1079_base_calc.netpr,
    for_1079_base_calc.kbetr,
    for_1079_base_calc.order_rf,
    for_1079_base_calc.order_asia,
    for_1079_base_calc.order_euro_svh,
    for_1079_base_calc.currency_dogovor
FROM (
    SELECT
        ekpo.purchase_contract_code,
        ekpo.position_line_item_code,
        ekko.unit_balance_code                             AS bukrs,
        ekko.supplier_code                                 AS lifnr,
        ekko.port_code                                     AS zzport,
        ekko.dt_valid_from                                 AS kdatb,
        ekko.dt_valid_to                                   AS kdate,
        ekpo.plant_code                                    AS werks,
        ekpo.service_code                                  AS zzsrvpos,
        ekpo.shape_code                                    AS zzforma,
        ekpo.etsng_code                                    AS zzet_tarif,
        ekpo.transport_type_code                           AS sdabw,
        ekpo_price.dt_price_condition_valid_to             AS datbi,
        ekpo_price.dt_price_condition_valid_from           AS datab,
        ekko.dt_valid_to,
        ekko.dt_valid_from,
        ekko.purchase_document_subtype_code,
        ekpo_price.price_condition_position_code           AS kopos,
        ekpo_price.price_condition_position_currency_code  AS konwa,
        ekpo.terminal_code                                 AS zzterminal,
        ekpo.warehouse_storage_area_code                   AS vzzzone,
        COALESCE(ekpo_price.price_scale_currency_code,
                 ekpo_price.price_condition_position_currency_code) AS konws,
        ekpo.service_code,
        ekpo.price_uom_code,
        ekpo.price_vat_excluded_document_currency_amount   AS netpr,
        ekpo_price.price_rate                              AS kbetr,

        -- приоритеты для РФ/Азия/Европа-СВХ
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END) AS order_rf,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.warehouse_storage_area_code IS NULL THEN 1 ELSE 0 END) AS order_asia,

        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END +
         CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END) AS order_euro_svh,

        CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor,

        ekpo_price.is_price_condition_position_deleted_code,
        COUNT(*) OVER (
            PARTITION BY ekpo.purchase_contract_code, ekpo.position_line_item_code
        ) AS cnt,

        ROW_NUMBER() OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code,
                         ekpo_price.dt_price_condition_valid_from,
                         ekpo_price.dt_price_condition_valid_to,
                         price_scale_basis_code
            ORDER BY price_scale_basis_consecutive_code
        ) AS rwn

    FROM dds.purchase_contract_price_conditions AS ekpo_price
    JOIN dds.purchase_contract_header AS ekko
      ON ekko.purchase_contract_code = ekpo_price.purchase_contract_code
    JOIN dds.purchase_contract_position AS ekpo
      ON ekpo.purchase_contract_code = ekko.purchase_contract_code
     AND ekpo.position_line_item_code = ekpo_price.purchase_contract_position_code
    LEFT JOIN dict_dds.currency_decimal_place_ral AS curdpr
      ON curdpr.currency_code = COALESCE(ekpo_price.price_scale_currency_code,
                                         ekpo_price.price_condition_position_currency_code)
     AND curdpr.decimal_place_number = '0'
    WHERE ekko.purchase_document_subtype_code IN ('ZDTL','ZDT','ZDE','ZDP','ZDF','ZDS')
      AND ekko.paydox_contract_type_code IN ('1','6')
      AND ekpo.is_deleted IS NULL
      AND ekpo_price.is_price_condition_position_deleted_code IS NULL
      AND ekpo_price.price_condition_type_code = 'PB00'
      AND (ekpo_price.price_scale_basis_code = 'C'
           OR ekpo_price.price_scale_basis_code IS NULL)
) AS for_1079_base_calc
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_1
  ON tmp_1.abap_program_code = '/RUSAL/SD4553M'
 AND tmp_1.parameter_code = 'S_USLSTC'
 AND tmp_1.range_low_value = for_1079_base_calc.zzsrvpos
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_2
  ON tmp_2.abap_program_code = '/RUSAL/SD4553M'
 AND tmp_2.parameter_code = 'S_USLST'
 AND tmp_2.range_low_value LIKE 'EU%'
 AND tmp_2.range_low_value = for_1079_base_calc.zzsrvpos
WHERE for_1079_base_calc.rwn = 1
DISTRIBUTED REPLICATED;

ANALYZE for_1079_base;

        SELECT
            base_sw.rn,
            for_1079_base.purchase_contract_code,

            for_1079_base.price_uom_code,
            for_1079_base.currency_dogovor,
            for_1079_base.datab,

CREATE TEMPORARY TABLE for_1090
WITH (appendonly = true, compresstype = zstd, compresslevel = 5)
ON COMMIT DROP AS
WITH tmp AS (
    SELECT
        pcpc.price_condition_code                    AS knumh,
        pcpc.purchase_contract_code                  AS evrtn,
        pcpc.purchase_contract_position_code         AS evrtp,
        pcpc.price_scale_basis_consecutive_code,
        pcpc.price_condition_position_code,
        pcpc.price_rate                              AS konp_kbetr,
        pcpc.price_scale_rate                        AS konm_kbetr,
        pcpc.price_scale_limit_from_quantity         AS kstbm,
        pcpc.price_scale_limit_to_quantity           AS kstbm_next,
        pcpc.price_scale_basis_code,
        pcpc.dt_price_condition_valid_to,
        pcpc.dt_price_condition_valid_from,
        COUNT(pcpc.price_scale_limit_from_quantity)
            OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code) AS cnt,
        ROW_NUMBER() OVER (
            PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
            ORDER BY pcpc.price_scale_basis_consecutive_code
        ) AS rn,
        MIN(price_scale_limit_from_quantity) OVER (
            PARTITION BY pcpc.purchase_contract_code,
                         pcpc.purchase_contract_position_code,
                         dt_price_condition_valid_from,
                         dt_price_condition_valid_to
        ) AS min_price_scale_limit_from_quantity,
        (pcpc.price_scale_limit_from_quantity -
         LAG(pcpc.price_scale_limit_from_quantity) OVER (
             PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
             ORDER BY pcpc.price_scale_basis_consecutive_code
         ))
         * LAG(pcpc.price_scale_rate) OVER (
             PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code
             ORDER BY pcpc.price_scale_basis_consecutive_code
         ) AS lagging
    FROM dds.purchase_contract_price_conditions AS pcpc
    WHERE pcpc.price_condition_type_code = 'PB00'
      AND (pcpc.price_scale_basis_code = 'C'
           OR pcpc.price_scale_basis_code IS NULL)
)
SELECT
    tmp.knumh,
    tmp.evrtn,
    tmp.evrtp,
    tmp.dt_price_condition_valid_to,
    tmp.dt_price_condition_valid_from,
    tmp.price_scale_basis_consecutive_code,
    tmp.price_condition_position_code,
    tmp.konp_kbetr,
    tmp.konm_kbetr,
    tmp.kstbm,
    tmp.kstbm_next,
    tmp.cnt,
    tmp.rn,
    tmp.price_scale_basis_code,
    tmp.min_price_scale_limit_from_quantity,
    SUM(tmp.lagging) OVER (
        PARTITION BY tmp.evrtn, tmp.evrtp
        ORDER BY tmp.price_scale_basis_consecutive_code
    ) AS presum
FROM tmp
DISTRIBUTED REPLICATED;

ANALYZE for_1090;

                     * base_2_5.weight_for_1090_umno
                     * base_2_5.currency_dogovor

                     * base_2_5.currency_dogovor

            ELSE 0

                 - base_2_5.storage_duration_free_by_delivery_calendar_days > 365
                THEN base_2_5.sales_bundle_gross_weight
            ELSE 0
        END AS storage_payable_over_365_days_gross_weight,

             OR bl.storage_cost_calculated_amount_local IS NULL
            THEN bl.storage_cost_calculated_amount_local
        ELSE
            bl.storage_cost_calculated_amount_local
            * COALESCE(rates.currency_to_multiplier, 1)
            * COALESCE(NULLIF(rates.currency_rate, 0), 1)
            / COALESCE(rates.currency_from_multiplier, 1)
    END AS storage_cost_calculated_amount_usd
FROM base_local bl
LEFT JOIN dict_dds.currency_rates AS rates
  ON rates.currency_rate_type_code = 'M'
 AND rates.currency_from_code = bl.purchase_contract_position_currency_code
 AND rates.currency_to_code = 'USD'
 AND rates.dt_currency_rate = bl.dt_report
 AND rates.deleted_flag = FALSE
DISTRIBUTED BY (sales_bundle_code, dt_storage_start, warehouse_code);
