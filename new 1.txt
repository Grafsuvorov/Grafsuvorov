/*===============================================================
  A) РАЗВОРАЧИВАЕМ ТЕСТОВЫЕ КОПИИ В userdata  (однократно)
  ---------------------------------------------------------------
  ВАЖНО: этот блок вы запускаете отдельно для разовой загрузки.
================================================================*/
CREATE SCHEMA IF NOT EXISTS userdata;

-- CKMLPP: распределяем по (kalnr, bdatj, poper) — ключ всех тяжёлых джойнов
DROP TABLE IF EXISTS userdata.ckmlpp_ral_tst;
CREATE TABLE userdata.ckmlpp_ral_tst
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
AS
SELECT *
FROM ods.ckmlpp_ral
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE userdata.ckmlpp_ral_tst;

-- CKMLCR: то же распределение (коллокация с PP)
DROP TABLE IF EXISTS userdata.ckmlcr_ral_tst;
CREATE TABLE userdata.ckmlcr_ral_tst
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
AS
SELECT *
FROM ods.ckmlcr_ral
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE userdata.ckmlcr_ral_tst;

-- MLCD: агрегируем по тройке — тоже распределяем по (kalnr, bdatj, poper)
DROP TABLE IF EXISTS userdata.mlcd_ral_tst;
CREATE TABLE userdata.mlcd_ral_tst
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
AS
SELECT *
FROM ods.mlcd_ral
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE userdata.mlcd_ral_tst;


/*===============================================================
  B) РАБОЧАЯ СЕССИЯ: ВРЕМЕННЫЕ ТАБЛИЦЫ + ФИНАЛЬНЫЙ ЗАПРОС
  ---------------------------------------------------------------
  Минимум подготовительных шагов, все распределения выровнены.
================================================================*/

-- 1) PP базовая выборка (сужаем ширину, считаем dt_valid_from)
DROP TABLE IF EXISTS pp_base;
CREATE TEMP TABLE pp_base ON COMMIT DROP AS
SELECT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper, 2, 2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
FROM userdata.ckmlpp_ral_tst p
WHERE p.untper = '000'
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE pp_base;

-- 2) CR текущего периода (только нужные поля)
DROP TABLE IF EXISTS cr_curr_small;
CREATE TEMP TABLE cr_curr_small ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    vprsv, stprs, peinh, pvprs,
    abprd_o, abprd_mo, salk3
FROM userdata.ckmlcr_ral_tst
WHERE untper = '000' AND curtp = '10'
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE cr_curr_small;

-- 3) CR на 1-е число следующего месяца
DROP TABLE IF EXISTS cr_next;
CREATE TEMP TABLE cr_next ON COMMIT DROP AS
SELECT
    kalnr,
    (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    MAX(salk3) AS salk3               -- страховка от дублей
FROM userdata.ckmlcr_ral_tst
WHERE untper = '000' AND curtp = '10'
GROUP BY kalnr, (make_date(bdatj::int, substring(poper, 2, 2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);
ANALYZE cr_next;

-- 4) MLCD агрегаты по тройке
DROP TABLE IF EXISTS mlcd_agg;
CREATE TEMP TABLE mlcd_agg ON COMMIT DROP AS
SELECT
    kalnr, bdatj, poper,
    SUM(salk3)  AS salk3,
    SUM(estprd) AS estprd,
    SUM(mstprd) AS mstprd
FROM userdata.mlcd_ral_tst
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE mlcd_agg;

-- 5) distinct даты начала периода
DROP TABLE IF EXISTS pp_dates;
CREATE TEMP TABLE pp_dates ON COMMIT DROP AS
SELECT kalnr, dt_valid_from
FROM pp_base
GROUP BY kalnr, dt_valid_from
DISTRIBUTED BY (kalnr);
ANALYZE pp_dates;

-- 6) интервалы (LEAD по коллокированным данным)
DROP TABLE IF EXISTS dates;
CREATE TEMP TABLE dates ON COMMIT DROP AS
SELECT
    kalnr,
    dt_valid_from,
    COALESCE(
        LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
        DATE '2299-12-31'
    )::date AS dt_valid_to
FROM pp_dates
DISTRIBUTED BY (kalnr);
ANALYZE dates;

-- 7) ФИНАЛ: один SELECT (или подмените на INSERT INTO <target> SELECT ...)
SELECT
    pp.kalnr AS calculation_code,
    pp.dt_valid_from,
    d.dt_valid_to,

    cr.vprsv AS price_valuation_type_code,
    cr.stprs AS standard_price_amount,
    cr.peinh AS price_unit_code,
    cr.pvprs AS moving_average_price_amount,

    /* rule3 */
    CASE
      WHEN (pp.umkumo + pp.abkumo + pp.zukumo) = 0 THEN 0
      ELSE TRUNC(
             CAST(ROUND(
                   (cr.stprs / NULLIF(cr.peinh,0)) * (pp.umkumo + pp.abkumo)
                 + COALESCE(cr.abprd_o,0) + COALESCE(cr.abprd_mo,0)
                 + COALESCE(ma.salk3,0) + COALESCE(ma.estprd,0) + COALESCE(ma.mstprd,0)
             , 2) AS NUMERIC(40,20))
             / NULLIF((pp.umkumo + pp.abkumo + pp.zukumo),0), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE
      WHEN (pp.abkumo + pp.umkumo + pp.zukumo) <> 0
      THEN TRUNC(
             CAST(
                 COALESCE(n.salk3,0)
               + COALESCE(ma.salk3,0) + COALESCE(ma.estprd,0) + COALESCE(ma.mstprd,0)
             AS NUMERIC(40,20))
             / NULLIF((pp.abkumo + pp.umkumo + pp.zukumo),0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE
      WHEN pp.lbkum <> 0
      THEN TRUNC(CAST(COALESCE(cr.salk3,0) AS NUMERIC(40,20)) / NULLIF(pp.lbkum,0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    h.price_rule_code
FROM pp_base pp
LEFT JOIN cr_curr_small cr
  ON cr.kalnr = pp.kalnr
 AND cr.bdatj = pp.bdatj
 AND cr.poper = pp.poper
LEFT JOIN cr_next n
  ON n.kalnr = pp.kalnr
 AND n.dt_posting = pp.dt_valid_from
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr
 AND ma.bdatj = pp.bdatj
 AND ma.poper = pp.poper
LEFT JOIN dates d
  ON d.kalnr = pp.kalnr
 AND d.dt_valid_from = pp.dt_valid_from
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = pp.kalnr
;
