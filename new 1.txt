  create or replace function tech_etl.check_entity_backup_tables(
      p_entity_id integer,
      p_table_schema text,
      p_backup_date date default current_date
  )
  returns table (
      run_dttm timestamp,
      source_table text,
      backup_table text,
      status text,
      source_cnt bigint,
      backup_cnt bigint,
      diff_cnt bigint
  )
  language plpgsql
  as
  $$
  declare
      r record;
      v_run_dttm timestamp := now();
      v_backup_suffix text := to_char(p_backup_date, 'YYYY_MM_DD');
      v_backup_table_name text;
      v_cols text;
      v_source_cnt bigint;
      v_backup_cnt bigint;
      v_diff_cnt bigint;
      v_status text;
  begin
      for r in
          select t.table_name
          from tech_etl.tables_meta t
          where t.entity_id = p_entity_id
            and t.table_schema = p_table_schema
          order by t.table_name
      loop
          v_backup_table_name := format('%s_backup_%s', r.table_name, v_backup_suffix);

          if not exists (
              select 1
              from information_schema.tables
              where table_schema = p_table_schema
                and table_name = v_backup_table_name
          ) then
              v_status := 'backup_not_found';
              v_source_cnt := null;
              v_backup_cnt := null;
              v_diff_cnt := null;

              insert into tech_etl.entity_backup_check_results (
                  run_dttm,
                  backup_date,
                  entity_id,
                  table_schema,
                  source_table,
                  backup_table,
                  status,
                  source_cnt,
                  backup_cnt,
                  diff_cnt
              )
              values (
                  v_run_dttm,
                  p_backup_date,
                  p_entity_id,
                  p_table_schema,
                  format('%s.%s', p_table_schema, r.table_name),
                  format('%s.%s', p_table_schema, v_backup_table_name),
                  v_status,
                  v_source_cnt,
                  v_backup_cnt,
                  v_diff_cnt
              );

              run_dttm := v_run_dttm;
              source_table := format('%s.%s', p_table_schema, r.table_name);
              backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
              status := v_status;
              source_cnt := v_source_cnt;
              backup_cnt := v_backup_cnt;
              diff_cnt := v_diff_cnt;
              return next;
              continue;
          end if;

          select string_agg(format('%I', c.column_name), ', ' order by c.ordinal_position)
          into v_cols
          from information_schema.columns c
          where c.table_schema = p_table_schema
            and c.table_name = r.table_name
            and c.column_name not in (
                'dttm_inserted',
                'dttm_updated',
                'job_name',
                'deleted_flag'
            );

          if v_cols is null then
              v_status := 'no_business_columns';
              v_source_cnt := null;
              v_backup_cnt := null;
              v_diff_cnt := null;

              insert into tech_etl.entity_backup_check_results (
                  run_dttm,
                  backup_date,
                  entity_id,
                  table_schema,
                  source_table,
                  backup_table,
                  status,
                  source_cnt,
                  backup_cnt,
                  diff_cnt
              )
              values (
                  v_run_dttm,
                  p_backup_date,
                  p_entity_id,
                  p_table_schema,
                  format('%s.%s', p_table_schema, r.table_name),
                  format('%s.%s', p_table_schema, v_backup_table_name),
                  v_status,
                  v_source_cnt,
                  v_backup_cnt,
                  v_diff_cnt
              );

              run_dttm := v_run_dttm;
              source_table := format('%s.%s', p_table_schema, r.table_name);
              backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
              status := v_status;
              source_cnt := v_source_cnt;
              backup_cnt := v_backup_cnt;
              diff_cnt := v_diff_cnt;
              return next;
              continue;
          end if;

          execute format(
              'select count(*) from %I.%I',
              p_table_schema,
              r.table_name
          )
          into v_source_cnt;

          execute format(
              'select count(*) from %I.%I',
              p_table_schema,
              v_backup_table_name
          )
          into v_backup_cnt;

          execute format(
              $sql$
              select count(*)
              from (
                  (select %s from %I.%I
                   except all
                   select %s from %I.%I)
                  union all
                  (select %s from %I.%I
                   except all
                   select %s from %I.%I)
              ) q
              $sql$,
              v_cols, p_table_schema, r.table_name,
              v_cols, p_table_schema, v_backup_table_name,
              v_cols, p_table_schema, v_backup_table_name,
              v_cols, p_table_schema, r.table_name
          )
          into v_diff_cnt;

          if v_source_cnt = v_backup_cnt and v_diff_cnt = 0 then
              v_status := 'ok';
          else
              v_status := 'mismatch';
          end if;

          insert into tech_etl.entity_backup_check_results (
              run_dttm,
              backup_date,
              entity_id,
              table_schema,
              source_table,
              backup_table,
              status,
              source_cnt,
              backup_cnt,
              diff_cnt
          )
          values (
              v_run_dttm,
              p_backup_date,
              p_entity_id,
              p_table_schema,
              format('%s.%s', p_table_schema, r.table_name),
              format('%s.%s', p_table_schema, v_backup_table_name),
              v_status,
              v_source_cnt,
              v_backup_cnt,
              v_diff_cnt
          );

          run_dttm := v_run_dttm;
          source_table := format('%s.%s', p_table_schema, r.table_name);
          backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
          status := v_status;
          source_cnt := v_source_cnt;
          backup_cnt := v_backup_cnt;
          diff_cnt := v_diff_cnt;
          return next;
      end loop;
  end;
  $$;
