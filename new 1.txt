-- Коэффициент скью (чем ближе к 1, тем лучше)
SELECT * 
FROM gp_toolkit.gp_skew_coefficients 
WHERE schemaname='ods' AND tablename='ekbe_ral_v2';

-- Скью по диску (полезно для AO/columnstore)
SELECT * 
FROM gp_toolkit.gp_skew_disk
WHERE sdsschemaname='ods' AND sdstablename='ekbe_ral_v2';


SELECT gp_segment_id, COUNT(*) AS rows, 
       100.0 * COUNT(*) / SUM(COUNT(*)) OVER () AS pct
FROM ods.ekbe_ral_v2
GROUP BY gp_segment_id
ORDER BY gp_segment_id;


WITH k AS (
  SELECT ebeln, ebelp, belnr, gjahr, buzei  -- <== твой ключ
)
SELECT
  COALESCE(v2.ebeln, v1.ebeln) AS ebeln,
  COUNT(*) FILTER (WHERE v1 IS NULL) AS only_in_v2,
  COUNT(*) FILTER (WHERE v2 IS NULL) AS only_in_old,
  COUNT(*) FILTER (WHERE v1 IS NOT NULL AND v2 IS NOT NULL AND (v1.dmbtr IS DISTINCT FROM v2.dmbtr)) AS dmbtr_diff,
  COUNT(*) FILTER (WHERE v1 IS NOT NULL AND v2 IS NOT NULL AND (v1.wrbtr IS DISTINCT FROM v2.wrbtr)) AS wrbtr_diff,
  COUNT(*) FILTER (WHERE v1 IS NOT NULL AND v2 IS NOT NULL AND (v1.arewr IS DISTINCT FROM v2.arewr)) AS arewr_diff
FROM ods.ekbe_ral v1
FULL JOIN ods.ekbe_ral_v2 v2
  USING (ebeln, ebelp, belnr, gjahr, buzei)
GROUP BY 1
HAVING
  COUNT(*) FILTER (WHERE v1 IS NULL OR v2 IS NULL) > 0
  OR COUNT(*) FILTER (WHERE v1 IS NOT NULL AND v2 IS NOT NULL AND (v1.dmbtr IS DISTINCT FROM v2.dmbtr OR v1.wrbtr IS DISTINCT FROM v2.wrbtr OR v1.arewr IS DISTINCT FROM v2.arewr)) > 0
LIMIT 100;


-- строки, которые есть в v2, но нет в старой
SELECT COUNT(*) AS only_in_v2
FROM (
  SELECT <cols> FROM ods.ekbe_ral_v2
  EXCEPT ALL
  SELECT <cols> FROM ods.ekbe_ral
) t;

-- строки, которые есть в старой, но нет в v2
SELECT COUNT(*) AS only_in_old
FROM (
  SELECT <cols> FROM ods.ekbe_ral
  EXCEPT ALL
  SELECT <cols> FROM ods.ekbe_ral_v2
) t;
