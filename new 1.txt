-- =====================================================================
-- Debt for working capital: delta architecture for Greenplum
-- Generated from the current production logic supplied by the user.
-- IMPORTANT: scripts preserve the CURRENT snapshot key:
--   (unit_balance_mdm_code_1c, dt_report)
-- Before production rollout, validate whether database_code_1c must also
-- be part of this key.
-- =====================================================================

-- ---------------------------------------------------------------------
-- 1. CONTROL TABLES
-- ---------------------------------------------------------------------

create schema if not exists meta;

create table if not exists meta.s1c_debt_etl_watermark (
    process_name        varchar(100) not null,
    last_dt_insert      timestamp not null,
    updated_dttm        timestamp not null default clock_timestamp()
)
distributed replicated;

create table if not exists meta.s1c_debt_processed_documents (
    uuid                varchar(60) not null,
    dt_insert           timestamp,
    processed_dttm      timestamp not null default clock_timestamp()
)
distributed by (uuid);

create table if not exists meta.s1c_debt_source_state (
    database_code_1c            varchar(255),
    unit_balance_mdm_code_1c    varchar(255),
    last_dt_report              date,
    last_dt_insert              timestamp,
    last_create_date_time       timestamp,
    last_row_count              bigint,
    updated_dttm                timestamp not null default clock_timestamp()
)
distributed by (database_code_1c, unit_balance_mdm_code_1c);

create table if not exists meta.s1c_debt_expected_source (
    database_code_1c            varchar(255),
    unit_balance_mdm_code_1c    varchar(255),
    is_active                   boolean not null default true,
    expected_interval           interval not null default interval '1 day',
    grace_interval              interval not null default interval '6 hours',
    comment_text                varchar(1000)
)
distributed by (database_code_1c, unit_balance_mdm_code_1c);


-- ---------------------------------------------------------------------
-- 2. ONE-TIME BOOTSTRAP FOR EXISTING PRODUCTION
-- ---------------------------------------------------------------------
-- Run once BEFORE switching the DAG to the new algorithm.
-- We set the watermark to the max already loaded STG dt_insert.
-- The >= predicate in the daily process is protected by processed_documents.

insert into meta.s1c_debt_etl_watermark
(process_name, last_dt_insert, updated_dttm)
select
    's1c_debt_for_working_capital',
    coalesce(max(dt_insert), timestamp '1900-01-01 00:00:00'),
    clock_timestamp()
from stg.s1c_debt_for_working_capital
where not exists (
    select 1
    from meta.s1c_debt_etl_watermark w
    where w.process_name = 's1c_debt_for_working_capital'
);

-- Mark only documents exactly on the bootstrap watermark.
-- This prevents duplicate processing when the daily query uses >=.
insert into meta.s1c_debt_processed_documents (uuid, dt_insert, processed_dttm)
select distinct l.uuid, l.dt_insert, clock_timestamp()
from landing."INPUT_DATA_FROM_SAPXI_IN" l
join meta.s1c_debt_etl_watermark w
  on w.process_name = 's1c_debt_for_working_capital'
 and l.dt_insert = w.last_dt_insert
where l.flow_id = 'SI_WorkingCapital_AI'
  and not exists (
      select 1
      from meta.s1c_debt_processed_documents p
      where p.uuid = l.uuid
  );

-- Initial source state. One-time scan of STG.
with mx as (
    select
        database_code_1c,
        unit_balance_mdm_code_1c,
        max(dt_insert) as last_dt_insert
    from stg.s1c_debt_for_working_capital
    group by 1,2
)
insert into meta.s1c_debt_source_state (
    database_code_1c,
    unit_balance_mdm_code_1c,
    last_dt_report,
    last_dt_insert,
    last_create_date_time,
    last_row_count,
    updated_dttm
)
select
    s.database_code_1c,
    s.unit_balance_mdm_code_1c,
    max(s.dt_report),
    m.last_dt_insert,
    max(s.create_date_time),
    count(*),
    clock_timestamp()
from stg.s1c_debt_for_working_capital s
join mx m
  on m.database_code_1c = s.database_code_1c
 and m.unit_balance_mdm_code_1c = s.unit_balance_mdm_code_1c
 and m.last_dt_insert = s.dt_insert
group by 1,2,4;

-- Bootstrap expected sources from everything that has historically sent data.
insert into meta.s1c_debt_expected_source (
    database_code_1c,
    unit_balance_mdm_code_1c,
    is_active,
    expected_interval,
    grace_interval,
    comment_text
)
select distinct
    s.database_code_1c,
    s.unit_balance_mdm_code_1c,
    true,
    interval '1 day',
    interval '6 hours',
    'bootstrap from STG'
from stg.s1c_debt_for_working_capital s
where not exists (
    select 1
    from meta.s1c_debt_expected_source e
    where e.database_code_1c = s.database_code_1c
      and e.unit_balance_mdm_code_1c = s.unit_balance_mdm_code_1c
);

-- ---------------------------------------------------------------------
-- 4. SOURCE FRESHNESS MONITORING
-- ---------------------------------------------------------------------
-- Returns only sources that should be sending data but are stale or never sent.

select
    e.database_code_1c,
    e.unit_balance_mdm_code_1c,
    s.last_dt_report,
    s.last_dt_insert,
    clock_timestamp() - s.last_dt_insert as data_age,
    e.expected_interval,
    e.grace_interval,
    case
        when s.last_dt_insert is null then 'NEVER_RECEIVED'
        when clock_timestamp() - s.last_dt_insert
             > e.expected_interval + e.grace_interval then 'STALE'
        else 'OK'
    end as status
from meta.s1c_debt_expected_source e
left join meta.s1c_debt_source_state s
  on s.database_code_1c = e.database_code_1c
 and s.unit_balance_mdm_code_1c = e.unit_balance_mdm_code_1c
where e.is_active
  and (
      s.last_dt_insert is null
      or clock_timestamp() - s.last_dt_insert
         > e.expected_interval + e.grace_interval
  )
order by data_age desc nulls first;

-- 5. DATA-QUALITY CHECKS FOR THE NEW ARCHITECTURE
-- ---------------------------------------------------------------------

-- 5.1 Potential collision: same BE/date from multiple 1C databases.
-- If this returns rows, strongly consider adding database_code_1c to snapshot key.
select
    unit_balance_mdm_code_1c,
    dt_report,
    count(distinct database_code_1c) as database_cnt
from stg.s1c_debt_for_working_capital
group by 1,2
having count(distinct database_code_1c) > 1
order by database_cnt desc
limit 100;

-- 5.2 Duplicate logical posting in ODS.
select
    unit_balance_mdm_code_1c,
    dt_report,
    posting_uid_code_1c,
    count(*) as cnt
from ods.account_debt_for_working_capital_1c
group by 1,2,3
having count(*) > 1
order by cnt desc
limit 100;

-- 5.3 Compare last received source state with ODS.
select
    s.database_code_1c,
    s.unit_balance_mdm_code_1c,
    s.last_dt_report,
    max(o.dt_report) as ods_max_dt_report,
    s.last_dt_insert
from meta.s1c_debt_source_state s
left join ods.account_debt_for_working_capital_1c o
  on o.database_code_1c = s.database_code_1c
 and o.unit_balance_mdm_code_1c = s.unit_balance_mdm_code_1c
group by 1,2,3,5
having max(o.dt_report) is distinct from s.last_dt_report;
