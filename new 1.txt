-- 3. Выбор одного лучшего договора на rn через DISTINCT ON
DROP TABLE IF EXISTS for_1079_asia;
CREATE TEMP TABLE for_1079_asia
ON COMMIT DROP AS
SELECT DISTINCT ON (b.rn)
    b.rn,
    f.purchase_contract_code,
    f.position_line_item_code,
    f.konws,
    f.service_code,
    f.price_uom_code,
    f.currency_dogovor,
    f.datab,
    f.datbi
FROM tmp_1079_asia_bundles b
JOIN tmp_1079_base_asia f
  ON f.bukrs = b.unit_balance_code
 AND f.lifnr = b.forwarder_code
 AND (f.zzport   = b.port_of_discharge_code    OR f.zzport   IS NULL)
 AND f.werks = b.plant_code
 AND (f.zzforma  = b.material_shape_code       OR f.zzforma  IS NULL)
 AND (f.zzet_tarif = b.etsng_code              OR f.zzet_tarif IS NULL)
 AND (f.sdabw    = b.transport_type_code       OR f.sdabw    IS NULL)
 AND (f.zzterminal = b.warehouse_code          OR f.zzterminal IS NULL)
 AND (f.vzzzone    = b.warehouse_storage_area_code OR f.vzzzone IS NULL)
 AND b.dt_for_price_search_in_purchase_contract
      BETWEEN f.dt_valid_from AND f.dt_valid_to
 AND b.dt_for_price_search_in_purchase_contract
      BETWEEN f.datab AND f.datbi
ORDER BY
    b.rn,
    f.order_asia,
    f.kopos,
    f.kbetr,
    f.purchase_contract_code,
    f.position_line_item_code
DISTRIBUTED BY (rn);

Идея
В плане по Европе видно:

два тяжёлых Redistribute Motion:

один под окно sum(weight) over (warehouse, month),

второй под row_number over (rn).

много оконных функций подряд.

Что делаем:

Отделяем расчёт суммы веса по складу/месяцу в отдельную агрегацию GROUP BY, без WINDOW.

Аналогично Азии — делаем узкий tmp_euro_bundles и tmp_1079_base_euro + REPLICATED.

Выбор лучшего договора — опять DISTINCT ON (rn).

-- 1. Бандлы под «Логика Европа» (узкий набор полей)
DROP TABLE IF EXISTS tmp_1079_euro_bundles;
CREATE TEMP TABLE tmp_1079_euro_bundles
ON COMMIT DROP AS
SELECT
    rn,
    forwarder_code,
    port_of_discharge_code,
    material_shape_code,
    etsng_code,
    transport_type_code,
    dt_for_price_search_in_purchase_contract,
    storage_cost_special_calculation_type_code,
    dt_storage_start,
    dt_storage_end,
    storage_duration_total_calendar_days,
    warehouse_code,
    sales_bundle_gross_weight
FROM dm_calc.storage_sales_bundles_weight
WHERE storage_cost_calculation_type_name = 'Логика Европа'
DISTRIBUTED BY (rn);


-- 2. Вес по складу и месяцу окончания хранения (без оконной функции)
DROP TABLE IF EXISTS tmp_1079_euro_wh_month;
CREATE TEMP TABLE tmp_1079_euro_wh_month
ON COMMIT DROP AS
SELECT
    warehouse_code,
    date_trunc('month', dt_storage_end)::date AS month_end,
    SUM(sales_bundle_gross_weight) AS warehouse_weight_month_end
FROM tmp_1079_euro_bundles
GROUP BY warehouse_code, date_trunc('month', dt_storage_end)
DISTRIBUTED BY (warehouse_code, month_end);


-- 3. Базовые настройки по договорам для Европы
DROP TABLE IF EXISTS tmp_1079_base_euro;
CREATE TEMP TABLE tmp_1079_base_euro
ON COMMIT DROP AS
SELECT
    f.purchase_contract_code,
    f.position_line_item_code,
    f.konws,
    f.service_code,
    f.price_uom_code,
    f.netpr,
    f.currency_dogovor,
    f.datab,
    f.datbi,
    f.lifnr,
    f.zzport,
    f.zzforma,
    f.zzet_tarif,
    f.sdabw,
    f.dt_valid_from,
    f.dt_valid_to,
    f.kopos,
    f.zzterminal,
    f.kbetr,
    f.order_euro_svh,
    s_eu.range_low_value AS srv_range,
    param_form.range_low_value AS form_match,
    param_srv.range_low_value  AS srv_match
FROM for_1079_base f
JOIN dict_dds.settings_and_parameters_sap s_eu
  ON s_eu.range_low_value    = f.zzsrvpos
 AND s_eu.range_low_value LIKE 'EU%'
 AND s_eu.abap_program_code  = '/RUSAL/SD4553M'
 AND s_eu.parameter_code     = 'S_USLST'
LEFT JOIN dict_dds.settings_and_parameters_sap param_form
  ON param_form.range_low_value = f.zzforma
 AND param_form.abap_program_code = '/RUSAL/SD4553M'
 AND param_form.parameter_code    = 'FORMSTEU'
LEFT JOIN dict_dds.settings_and_parameters_sap param_srv
  ON param_srv.range_low_value = s_eu.range_low_value
 AND param_srv.abap_program_code = '/RUSAL/SD4553M'
 AND param_srv.parameter_code    = 'S_USLSTC'
WHERE f.bukrs  = '1520'
  AND f.werks  = '1521'
  AND f.purchase_document_subtype_code = 'ZDTL'
DISTRIBUTED REPLICATED;


-- 4. Основной выбор одного договора на rn
DROP TABLE IF EXISTS euro_base;
CREATE TEMP TABLE euro_base
ON COMMIT DROP AS
SELECT DISTINCT ON (b.rn)
    b.rn,
    f.purchase_contract_code,
    f.position_line_item_code,
    f.konws,
    f.service_code,
    f.price_uom_code,
    w.warehouse_weight_month_end,
    /* month_difference_2 */
    CASE
        WHEN EXTRACT(day FROM b.dt_storage_end)
           < EXTRACT(day FROM b.dt_storage_start)
        THEN
            (EXTRACT(year  FROM b.dt_storage_end) - EXTRACT(year  FROM b.dt_storage_start)) * 12 +
            (EXTRACT(month FROM b.dt_storage_end) - EXTRACT(month FROM b.dt_storage_start))
        ELSE
            (EXTRACT(year  FROM b.dt_storage_end) - EXTRACT(year  FROM b.dt_storage_start)) * 12 +
            (EXTRACT(month FROM b.dt_storage_end) - EXTRACT(month FROM b.dt_storage_start)) + 1
    END AS month_difference_2,
    f.netpr,
    /* special */
    CASE
        WHEN b.storage_cost_special_calculation_type_code = '4'
            THEN CEIL(b.storage_duration_total_calendar_days / 7::numeric)
        WHEN b.storage_cost_special_calculation_type_code = '5'
            THEN CEIL(b.storage_duration_total_calendar_days / 30::numeric)
    END AS special,
    f.currency_dogovor,
    f.datab,
    f.datbi
FROM tmp_1079_euro_bundles b
JOIN tmp_1079_euro_wh_month w
  ON w.warehouse_code = b.warehouse_code
 AND w.month_end      = date_trunc('month', b.dt_storage_end)::date
JOIN tmp_1079_base_euro f
  ON f.lifnr = b.forwarder_code
 AND (f.zzport   = b.port_of_discharge_code OR f.zzport IS NULL)
 AND (f.zzforma  = b.material_shape_code    OR f.zzforma IS NULL)
 AND (f.zzet_tarif = b.etsng_code          OR f.zzet_tarif IS NULL)
 AND (f.sdabw    = b.transport_type_code    OR f.sdabw   IS NULL)
 AND (f.zzterminal = b.warehouse_code       OR f.zzterminal IS NULL)
 AND b.dt_for_price_search_in_purchase_contract
      BETWEEN f.dt_valid_from AND f.dt_valid_to
 AND b.dt_for_price_search_in_purchase_contract
      BETWEEN f.datab AND f.datbi
ORDER BY
    b.rn,
    f.order_euro_svh,
    f.kopos,
    f.kbetr,
    /* приоритет по «наличию настроек» как и было в исходном ORDER BY */
    CASE
        WHEN f.form_match IS NOT NULL AND f.srv_match IS NOT NULL THEN 1
        WHEN f.form_match IS NULL     AND f.srv_match IS NULL     THEN 1
        ELSE 0
    END DESC,
    f.purchase_contract_code,
    f.position_line_item_code
DISTRIBUTED BY (rn);
