------------------------------------------------------------
-- 2. Вспомогательные временные таблицы (все REPLICATED)
------------------------------------------------------------

-- 2.1. Адрес → страна
DROP TABLE IF EXISTS tmp_address_country;
CREATE TEMP TABLE tmp_address_country AS
SELECT
    a.address_code,
    MIN(a.country_code) AS country_code
FROM dict_dds.address a
GROUP BY a.address_code
DISTRIBUTED REPLICATED;


-- 2.2. Страна (название)
DROP TABLE IF EXISTS tmp_country_texts;
CREATE TEMP TABLE tmp_country_texts AS
SELECT
    ct.country_code,
    ct.country_full_name
FROM dict_dds.country_texts ct
WHERE ct.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.3. Страна → регион рынка (market_region1)
DROP TABLE IF EXISTS tmp_country_region;
CREATE TEMP TABLE tmp_country_region AS
SELECT
    c.country_code,
    mr.market_region1_name
FROM dict_dds.country c
LEFT JOIN dict_dds.market_region1_texts mr
    ON mr.market_region1_code = c.market_region1_code
   AND mr.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.4. Транспортные узлы с уже подтянутой страной и регионом
DROP TABLE IF EXISTS tmp_hub_ext;
CREATE TEMP TABLE tmp_hub_ext AS
SELECT
    h.transport_hub_code,
    h.address_code,
    ac.country_code,
    ct.country_full_name,
    cr.market_region1_name,
    h.transport_hub_name_rus,
    h.transport_hub_name_eng
FROM dict_dds.transport_hub h
LEFT JOIN tmp_address_country ac
    ON ac.address_code = h.address_code
LEFT JOIN tmp_country_texts ct
    ON ct.country_code = ac.country_code
LEFT JOIN tmp_country_region cr
    ON cr.country_code = ac.country_code
DISTRIBUTED REPLICATED;


-- 2.5. Материалы (название)
DROP TABLE IF EXISTS tmp_material_texts;
CREATE TEMP TABLE tmp_material_texts AS
SELECT
    m.material_code,
    m.material_name
FROM dict_dds.material_texts m
WHERE m.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.6. Услуги транспортировки (название)
DROP TABLE IF EXISTS tmp_service_texts;
CREATE TEMP TABLE tmp_service_texts AS
SELECT
    t.transportation_service_code,
    t.transportation_service_name
FROM dict_dds.transportation_service_texts t
WHERE t.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.7. ЕТСНГ (название)
DROP TABLE IF EXISTS tmp_etsng_texts;
CREATE TEMP TABLE tmp_etsng_texts AS
SELECT
    e.etsng_code,
    e.etsng_name
FROM dict_dds.etsng_texts e
WHERE e.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.8. Заводы (короткое и полное имя)
DROP TABLE IF EXISTS tmp_plant;
CREATE TEMP TABLE tmp_plant AS
SELECT
    p.plant_code,
    p.plant_short_name,
    p.plant_full_name
FROM dict_dds.plant_and_subsidiary p
DISTRIBUTED REPLICATED;


-- 2.9. Этапы перевозки
DROP TABLE IF EXISTS tmp_transportation_stage;
CREATE TEMP TABLE tmp_transportation_stage AS
SELECT
    s.transportation_stage_code,
    s.transportation_stage_name
FROM dict_dds.transportation_stage_texts s
WHERE s.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.10. Контрагенты (экспедиторы, кредиторы)
DROP TABLE IF EXISTS tmp_counterparty;
CREATE TEMP TABLE tmp_counterparty AS
SELECT
    c.counterparty_code,
    c.counterparty_short_name,
    c.address_code
FROM dict_dds.counterparty c
DISTRIBUTED REPLICATED;


-- 2.11. Материал → форма (shape)
DROP TABLE IF EXISTS tmp_material_shape;
CREATE TEMP TABLE tmp_material_shape AS
SELECT
    ms.material_code,
    MIN(mst.material_shape_full_name) AS material_shape_full_name
FROM dict_dds.material_specification ms
LEFT JOIN dict_dds.material_shape_texts mst
    ON mst.shape_code = ms.shape_code
   AND mst.language_code = 'R'
GROUP BY ms.material_code
DISTRIBUTED REPLICATED;


-- 2.12. Адреса для кредитора (только нужные нам)
DROP TABLE IF EXISTS tmp_creditor_address;
CREATE TEMP TABLE tmp_creditor_address AS
SELECT
    a.address_code,
    a.address_full_name
FROM dict_dds.address a
WHERE a.international_display_format_code IS NULL
DISTRIBUTED REPLICATED;


-- 2.13. Поставка → завод производителя (для собственника металла)
DROP TABLE IF EXISTS tmp_delivery_plant;
CREATE TEMP TABLE tmp_delivery_plant AS
SELECT
    d.delivery_code,
    d.plant_producer_code
FROM dds.delivery_document_position d
WHERE d.delivery_position_line_item_code = '000010'
DISTRIBUTED REPLICATED;


-- 2.14. Тип перевозки (transport_transfer_type_texts, язык R)
DROP TABLE IF EXISTS tmp_transport_type;
CREATE TEMP TABLE tmp_transport_type AS
SELECT
    t.transport_transfer_type_code,
    t.transport_transfer_type_name
FROM dict_dds.transport_transfer_type_texts t
WHERE t.language_code = 'R'
DISTRIBUTED REPLICATED;

DROP TABLE IF EXISTS tmp_base_stage1;
CREATE TEMP TABLE tmp_base_stage1 AS
SELECT
    base.*,

    -- материалы
    matet.material_name,
    mf.material_shape_full_name,

    -- ЕТСНГ
    ette.etsng_name,

    -- услуги
    zsrvpost.transportation_service_name,

    -- завод поставки
    pas.plant_short_name  AS recv_plant_short_name,
    pas.plant_full_name   AS recv_plant_full_name,

    -- собственник металла (завод из договора)
    deldp.plant_producer_code          AS owner_plant_code,
    pas_1124.plant_short_name          AS owner_plant_short_name,
    pas_1124.plant_full_name           AS owner_plant_full_name,

    -- этап перевозки
    trst.transportation_stage_name,

    -- экспедитор
    counter.counterparty_short_name    AS forwarder_short_name,

    -- кредитор
    counter_2.counterparty_short_name  AS creditor_short_name,
    addr_8.address_full_name          AS creditor_address_full_name,

    -- тип ПС
    ipd.transport_transfer_type_name
FROM dm_calc.storage_sales_bundles_amount base
    -- материалы
    LEFT JOIN tmp_material_texts matet
        ON matet.material_code = base.material_code
    LEFT JOIN tmp_material_shape mf
        ON mf.material_code = base.material_code

    -- услуги
    LEFT JOIN tmp_service_texts zsrvpost
        ON zsrvpost.transportation_service_code = base.service_code

    -- ЕТСНГ
    LEFT JOIN tmp_etsng_texts ette
        ON ette.etsng_code = base.etsng_code

    -- заводы
    LEFT JOIN tmp_plant pas
        ON pas.plant_code = base.receiving_plant_code

    -- собственник металла (завод из договора хранения)
    LEFT JOIN tmp_delivery_plant deldp
        ON deldp.delivery_code = base.delivery_for_storage_calculation_code
    LEFT JOIN tmp_plant pas_1124
        ON pas_1124.plant_code = deldp.plant_producer_code

    -- этап перевозки
    LEFT JOIN tmp_transportation_stage trst
        ON trst.transportation_stage_code = base.transportation_stage_code

    -- экспедитор
    LEFT JOIN tmp_counterparty counter
        ON counter.counterparty_code = base.forwarder_code

    -- кредитор + адрес
    LEFT JOIN tmp_counterparty counter_2
        ON counter_2.counterparty_code = base.creditor_in_shipment_instruction_code
    LEFT JOIN tmp_creditor_address addr_8
        ON addr_8.address_code = counter_2.address_code

    -- тип ПС (через temp, а не напрямую dict_dds)
    LEFT JOIN tmp_transport_type ipd
        ON ipd.transport_transfer_type_code = base.transport_type_code
DISTRIBUTED RANDOMLY;

ANALYZE tmp_base_stage1;

-- 2.z. ВТОРОЙ СЛОЙ ОБОГАЩЕНИЯ (хабы / склады / порты)

DROP TABLE IF EXISTS tmp_base_stage2;
CREATE TEMP TABLE tmp_base_stage2 AS
SELECT
    b.*,

    -- склад (warehouse)
    wh_hub.country_full_name        AS wh_country_full_name,
    wh_hub.market_region1_name      AS wh_market_region1_name,
    wh_hub.transport_hub_name_rus   AS wh_name_rus,
    wh_hub.transport_hub_name_eng   AS wh_name_eng,

    -- порт отгрузки
    port_hub.transport_hub_name_eng AS port_name_eng,

    -- хаб отправления
    dep_hub.transport_hub_name_rus  AS dep_name_rus,

    -- хаб назначения
    dest_hub.transport_hub_name_rus AS dest_name_rus,

    -- зона хранения (area)
    area_hub.transport_hub_name_rus AS area_name_rus,

    -- удалённый склад
    remote_hub.transport_hub_name_rus  AS remote_name_rus,
    remote_hub.transport_hub_name_eng  AS remote_name_eng,
    remote_hub.country_full_name       AS remote_country_full_name,
    remote_hub.country_code            AS remote_country_code,
    remote_hub.market_region1_name     AS remote_market_region1_name
FROM tmp_base_stage1 b
    LEFT JOIN tmp_hub_ext wh_hub
        ON wh_hub.transport_hub_code = b.warehouse_code
    LEFT JOIN tmp_hub_ext port_hub
        ON port_hub.transport_hub_code = b.port_of_discharge_code
    LEFT JOIN tmp_hub_ext dep_hub
        ON dep_hub.transport_hub_code = b.transport_departure_hub_code
    LEFT JOIN tmp_hub_ext dest_hub
        ON dest_hub.transport_hub_code = b.transport_destination_hub_code
    LEFT JOIN tmp_hub_ext area_hub
        ON area_hub.transport_hub_code = b.warehouse_storage_area_code
    LEFT JOIN tmp_hub_ext remote_hub
        ON remote_hub.transport_hub_code = b.remote_warehouse_code
DISTRIBUTED RANDOMLY;

ANALYZE tmp_base_stage2;
