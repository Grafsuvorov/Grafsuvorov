-- 2. Вспомогательные временные таблицы (все REPLICATED)
------------------------------------------------------------

-- 2.1. Адрес → страна
DROP TABLE IF EXISTS tmp_address_country;
CREATE TEMP TABLE tmp_address_country AS
SELECT
    a.address_code,
    MIN(a.country_code) AS country_code
FROM dict_dds.address a
GROUP BY a.address_code
DISTRIBUTED REPLICATED;


-- 2.2. Страна (название)
DROP TABLE IF EXISTS tmp_country_texts;
CREATE TEMP TABLE tmp_country_texts AS
SELECT
    ct.country_code,
    ct.country_full_name
FROM dict_dds.country_texts ct
WHERE ct.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.3. Страна → регион рынка (market_region1)
DROP TABLE IF EXISTS tmp_country_region;
CREATE TEMP TABLE tmp_country_region AS
SELECT
    c.country_code,
    mr.market_region1_name
FROM dict_dds.country c
LEFT JOIN dict_dds.market_region1_texts mr
    ON mr.market_region1_code = c.market_region1_code
   AND mr.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.4. Транспортные узлы с уже подтянутой страной и регионом
DROP TABLE IF EXISTS tmp_hub_ext;
CREATE TEMP TABLE tmp_hub_ext AS
SELECT
    h.transport_hub_code,
    h.address_code,
    ac.country_code,
    ct.country_full_name,
    cr.market_region1_name,
    h.transport_hub_name_rus,
    h.transport_hub_name_eng
FROM dict_dds.transport_hub h
LEFT JOIN tmp_address_country ac
    ON ac.address_code = h.address_code
LEFT JOIN tmp_country_texts ct
    ON ct.country_code = ac.country_code
LEFT JOIN tmp_country_region cr
    ON cr.country_code = ac.country_code
DISTRIBUTED REPLICATED;


-- 2.5. Материалы (название)
DROP TABLE IF EXISTS tmp_material_texts;
CREATE TEMP TABLE tmp_material_texts AS
SELECT
    m.material_code,
    m.material_name
FROM dict_dds.material_texts m
WHERE m.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.6. Услуги транспортировки (название)
DROP TABLE IF EXISTS tmp_service_texts;
CREATE TEMP TABLE tmp_service_texts AS
SELECT
    t.transportation_service_code,
    t.transportation_service_name
FROM dict_dds.transportation_service_texts t
WHERE t.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.7. ЕТСНГ (название)
DROP TABLE IF EXISTS tmp_etsng_texts;
CREATE TEMP TABLE tmp_etsng_texts AS
SELECT
    e.etsng_code,
    e.etsng_name
FROM dict_dds.etsng_texts e
WHERE e.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.8. Заводы (короткое и полное имя)
DROP TABLE IF EXISTS tmp_plant;
CREATE TEMP TABLE tmp_plant AS
SELECT
    p.plant_code,
    p.plant_short_name,
    p.plant_full_name
FROM dict_dds.plant_and_subsidiary p
DISTRIBUTED REPLICATED;


-- 2.9. Этапы перевозки
DROP TABLE IF EXISTS tmp_transportation_stage;
CREATE TEMP TABLE tmp_transportation_stage AS
SELECT
    s.transportation_stage_code,
    s.transportation_stage_name
FROM dict_dds.transportation_stage_texts s
WHERE s.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.10. Контрагенты (экспедиторы, кредиторы)
DROP TABLE IF EXISTS tmp_counterparty;
CREATE TEMP TABLE tmp_counterparty AS
SELECT
    c.counterparty_code,
    c.counterparty_short_name,
    c.address_code
FROM dict_dds.counterparty c
DISTRIBUTED REPLICATED;


-- 2.11. Материал → форма (shape)
DROP TABLE IF EXISTS tmp_material_shape;
CREATE TEMP TABLE tmp_material_shape AS
SELECT
    ms.material_code,
    mst.shape_code,
    mst.material_shape_full_name
FROM dict_dds.material_specification ms
LEFT JOIN dict_dds.material_shape_texts mst
    ON mst.shape_code = ms.shape_code
   AND mst.language_code = 'R'
DISTRIBUTED REPLICATED;


-- 2.12. Адреса для кредитора (только нужные нам)
DROP TABLE IF EXISTS tmp_creditor_address;
CREATE TEMP TABLE tmp_creditor_address AS
SELECT
    a.address_code,
    a.address_full_name
FROM dict_dds.address a
WHERE a.international_display_format_code IS NULL
DISTRIBUTED REPLICATED;


-- 2.13. Поставка → завод производителя (для собственника металла)
DROP TABLE IF EXISTS tmp_delivery_plant;
CREATE TEMP TABLE tmp_delivery_plant AS
SELECT
    d.delivery_code,
    d.plant_producer_code
FROM dds.delivery_document_position d
WHERE d.delivery_position_line_item_code = '000010'
DISTRIBUTED REPLICATED;


-- 2.14. Тип перевозки (transport_transfer_type_texts, язык R)
DROP TABLE IF EXISTS tmp_transport_type;
CREATE TEMP TABLE tmp_transport_type AS
SELECT
    t.transport_transfer_type_code,
    t.transport_transfer_type_name
FROM dict_dds.transport_transfer_type_texts t
WHERE t.language_code = 'R'
DISTRIBUTED REPLICATED;
