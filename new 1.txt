CREATE OR REPLACE FUNCTION tech_etl.util_text_to_null_validation(v_text_value text)
	RETURNS varchar
	LANGUAGE plpgsql
	IMMUTABLE
AS $$
	
	
	
	
	
	
	
declare 
l_result date;


begin
	if trim(v_text_value) = '' then 
		return null; 
	elseif trim(v_text_value) !='' and trim(v_text_value) is not null then 
		return v_text_value;
	else 
		return null;
	end if;
end;
--if trim(v_text_value) != '' then return v_text_value; end if;
 --if null then null






$$
EXECUTE ON ANY;



CREATE OR REPLACE FUNCTION tech_etl.util_text_to_date_validation(v_text_value text)
	RETURNS date
	LANGUAGE plpgsql
	IMMUTABLE
AS $$
	
	
	
	
	
declare 
l_result date;


begin
if v_text_value = '00000000' then return null; end if; --if null then null
if v_text_value = ' ' then return null; end if;
if v_text_value !~ '^\d{8}$' then return to_date('19000101', 'YYYYMMDD'); --if not date then min date
end if;
if v_text_value < '19000101' then return to_date('19000101', 'YYYYMMDD'); --uf loweer than min return min date
elsif v_text_value >= '22991231' then return to_date('22991231', 'YYYYMMDD');  --if higher than max return max date
end if;
return v_text_value::date; --conversation

exception
when others 
then
return to_date('19000101', 'YYYYMMDD'); -- if error return min date
end;



$$
EXECUTE ON ANY;
