------------------------------------------------------------------------
-- 0. ПОДГОТОВКА ВСЕХ СПРАВОЧНИКОВ (TEMP TABLES)
------------------------------------------------------------------------

BEGIN;

-- 0.1 Хабы (самый тяжёлый справочник)
CREATE TEMP TABLE tmp_hub_ext AS
SELECT
    h.transport_hub_code,
    h.address_code,
    a.country_code,
    ct.country_full_name,
    mr.market_region1_name,
    h.transport_hub_name_rus,
    h.transport_hub_name_eng
FROM dict_dds.transport_hub h
LEFT JOIN dict_dds.address a ON a.address_code = h.address_code
LEFT JOIN dict_dds.country c ON c.country_code = a.country_code
LEFT JOIN dict_dds.country_texts ct
    ON ct.country_code = c.country_code AND ct.language_code='R'
LEFT JOIN dict_dds.market_region1_texts mr
    ON mr.market_region1_code = c.market_region1_code AND mr.language_code='R';
ANALYZE tmp_hub_ext;


-- 0.2 Материалы
CREATE TEMP TABLE tmp_material_texts AS
SELECT material_code, material_name
FROM dict_dds.material_texts
WHERE language_code='R';
ANALYZE tmp_material_texts;


-- 0.3 Формы материалов
CREATE TEMP TABLE tmp_material_shape AS
SELECT ms.material_code, mst.material_shape_full_name
FROM dict_dds.material_specification ms
LEFT JOIN dict_dds.material_shape_texts mst
    ON mst.shape_code = ms.shape_code AND mst.language_code='R';
ANALYZE tmp_material_shape;


-- 0.4 Услуги транспорта
CREATE TEMP TABLE tmp_service_texts AS
SELECT transportation_service_code, transportation_service_name
FROM dict_dds.transportation_service_texts
WHERE language_code='R';
ANALYZE tmp_service_texts;


-- 0.5 ЕТСНГ
CREATE TEMP TABLE tmp_etsng_texts AS
SELECT etsng_code, etsng_name
FROM dict_dds.etsng_texts
WHERE language_code='R';
ANALYZE tmp_etsng_texts;


-- 0.6 Заводы
CREATE TEMP TABLE tmp_plant AS
SELECT plant_code, plant_short_name, plant_full_name
FROM dict_dds.plant_and_subsidiary;
ANALYZE tmp_plant;


-- 0.7 Контрагенты
CREATE TEMP TABLE tmp_counterparty AS
SELECT counterparty_code, counterparty_short_name, address_code
FROM dict_dds.counterparty;
ANALYZE tmp_counterparty;


-- 0.8 Адреса кредиторов
CREATE TEMP TABLE tmp_creditor_address AS
SELECT address_code, address_full_name
FROM dict_dds.address
WHERE international_display_format_code IS NULL;
ANALYZE tmp_creditor_address;


-- 0.9 Типы транспорта
CREATE TEMP TABLE tmp_transport_type AS
SELECT transport_transfer_type_code, transport_transfer_type_name
FROM dict_dds.transport_transfer_type_texts
WHERE language_code='R';
ANALYZE tmp_transport_type;

COMMIT;


------------------------------------------------------------------------
-- 1. ЭТАП 1 — ОБОГАЩЕНИЕ МАТЕРИАЛАМИ, УСЛУГАМИ, ФОРМАМИ, ЕТСНГ
------------------------------------------------------------------------

CREATE TEMP TABLE stage1 AS
SELECT
    base.*,

    mat.material_name,
    shp.material_shape_full_name,
    srv.transportation_service_name,
    ets.etsng_name

FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN tmp_material_texts mat USING (material_code)
LEFT JOIN tmp_material_shape shp USING (material_code)
LEFT JOIN tmp_service_texts srv USING (service_code)
LEFT JOIN tmp_etsng_texts ets USING (etsng_code)
DISTRIBUTED RANDOMLY;

ANALYZE stage1;


------------------------------------------------------------------------
-- 2. ЭТАП 2 — ДОБАВЛЕНИЕ ХАБОВ, ЗАВОДОВ, КОНТРАГЕНТОВ
------------------------------------------------------------------------

CREATE TEMP TABLE stage2 AS
SELECT
    s1.*,

    -- Хабы
    wh.transport_hub_name_rus  AS wh_name_rus,
    wh.transport_hub_name_eng  AS wh_name_eng,
    wh.country_full_name       AS wh_country,
    wh.market_region1_name     AS wh_region,

    dep.transport_hub_name_rus  AS dep_name,
    dest.transport_hub_name_rus AS dest_name,

    rem.transport_hub_name_rus  AS remote_name_rus,
    rem.transport_hub_name_eng  AS remote_name_eng,
    rem.country_full_name       AS remote_country,

    area.transport_hub_name_rus AS area_name,

    -- Заводы
    pl.plant_short_name,
    pl.plant_full_name,

    -- Собственник металла (если есть)
    pl2.plant_short_name AS owner_short_name,
    pl2.plant_full_name  AS owner_full_name,

    -- Конрагент / кредитор
    ctr.counterparty_short_name AS forwarder_short_name,
    addr.address_full_name AS creditor_address_name

FROM stage1 s1
LEFT JOIN tmp_hub_ext wh   ON wh.transport_hub_code = s1.warehouse_code
LEFT JOIN tmp_hub_ext dep  ON dep.transport_hub_code = s1.transport_departure_hub_code
LEFT JOIN tmp_hub_ext dest ON dest.transport_hub_code = s1.transport_destination_hub_code
LEFT JOIN tmp_hub_ext rem  ON rem.transport_hub_code = s1.remote_warehouse_code
LEFT JOIN tmp_hub_ext area ON area.transport_hub_code = s1.warehouse_storage_area_code

LEFT JOIN tmp_plant pl     ON pl.plant_code = s1.receiving_plant_code
LEFT JOIN tmp_plant pl2    ON pl2.plant_code = s1.plant_code

LEFT JOIN tmp_counterparty ctr ON ctr.counterparty_code = s1.forwarder_code
LEFT JOIN tmp_creditor_address addr ON addr.address_code = s1.creditor_in_shipment_instruction_code

DISTRIBUTED RANDOMLY;

ANALYZE stage2;
