select
    rl.release_id,
    rl.started_at::date as release_date,
    rl.release_type,
    count(distinct ro.task_id) as tasks_cnt
from tech_etl.release_log rl
join tech_etl.release_objects ro
    on rl.release_id = ro.release_id
group by
    rl.release_id,
    rl.started_at::date,
    rl.release_type
order by
    rl.started_at;
