
/* Optimized rewrite for Greenplum 6+
   Key changes:
   - Build intervals with LEAD() (no self-joins on rn).
   - Deduplicate early before window.
   - Defer dictionary/text joins until after interval + fact join.
*/

WITH base AS (
    /* 1) Replicate FA by valuation areas (from parameters), FA own starts, and non-liquid tech state linkage */
    SELECT
        fix.unit_balance_code,
        fix.asset_main_code,
        fix.asset_sub_code,
        q.depreciation_area_code,
        fix.dt_from
    FROM dict_dds.fixed_asset_td_v2 AS fix
    JOIN dict_dds.fixed_asset_depreciation_parameters_td_v2 AS q
      ON q.unit_balance_code = fix.unit_balance_code
     AND q.asset_main_code   = fix.asset_main_code
     AND q.asset_sub_code    = fix.asset_sub_code

    UNION ALL
    SELECT
        q.unit_balance_code,
        q.asset_main_code,
        q.asset_sub_code,
        q.depreciation_area_code,
        q.dt_valid_from AS dt_from
    FROM dict_dds.fixed_asset_depreciation_parameters_td_v2 AS q

    UNION ALL
    SELECT
        COALESCE(liq.unit_balance_code, q.unit_balance_code) AS unit_balance_code,
        COALESCE(liq.asset_main_code, q.asset_main_code)     AS asset_main_code,
        COALESCE(liq.asset_sub_code, q.asset_sub_code)       AS asset_sub_code,
        q.depreciation_area_code,
        COALESCE(liq.dt_valid_from, q.dt_valid_from)         AS dt_from
    FROM dict_dds.fixed_asset_depreciation_parameters_td_v2 AS q
    JOIN (
        SELECT
            t1.unit_balance_code,
            t1.asset_main_code,
            t1.asset_sub_code,
            t1.asset_inventory_number,
            t2.dt_valid_from,
            t2.dt_valid_to
        FROM dict_dds.fixed_asset_td_v2 AS t1
        JOIN dict_dds.fixed_asset_non_liquid_technical_state_td AS t2
          ON t2.unit_balance_code    = t1.unit_balance_code
         AND t2.asset_inventory_code = t1.asset_inventory_number
    ) AS liq
      ON q.unit_balance_code = liq.unit_balance_code
     AND q.asset_main_code   = liq.asset_main_code
     AND q.asset_sub_code    = liq.asset_sub_code
),
dedup AS (
    /* 2) Deduplicate BEFORE window to reduce sort+window volume */
    SELECT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        dt_from
    FROM base
    GROUP BY 1,2,3,4,5
),
rng AS (
    /* 3) Intervals via LEAD() in one pass */
    SELECT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        COALESCE(dt_from, DATE '1900-01-01') AS dt_valid_from,
        COALESCE(
            (LEAD(dt_from) OVER (
                PARTITION BY unit_balance_code, asset_main_code, asset_sub_code, depreciation_area_code
                ORDER BY dt_from
            ) - INTERVAL '1 day')::date,
            DATE '2299-12-31'
        ) AS dt_valid_to
    FROM dedup
),
sep_int AS (
    SELECT * FROM rng
)

SELECT
    si.unit_balance_code,
    ub.unit_balance_name,
    si.asset_main_code,
    si.asset_sub_code,
    si.dt_valid_from,
    si.dt_valid_to,
    si.depreciation_area_code AS valuation_area_code,
    favat.fixed_asset_valuation_area_full_name AS valuation_area_name,
    tb.valuation_area_currency_code,
    ct.currency_full_name AS valuation_area_currency_name,
    ub.asset_depreciation_rule_code,
    favpt.fixed_asset_valuation_plan_name AS asset_depreciation_rule_name,
    fa.asset_class_code,
    fact.fixed_asset_class_full_name AS asset_class_name,
    fa.asset_inventory_number,
    fa.asset_name,
    fa.created_by,
    fa.dt_created,
    fa.changed_by,
    fa.dt_changed,
    fa.is_marked_for_delete,
    fa.is_locked_for_operations,
    fa.general_ledger_reference_rule_code,
    t095.asset_acquisition_cost_account_code,
    t095.asset_depreciation_account_code,
    fa.dt_asset_acquisition_yyyy,
    fa.dt_asset_acquisition,
    fa.dt_asset_acquisition_reference,
    fa.dt_asset_put_into_operation AS dt_asset_recognized,
    fa.dt_reference_asset_write_off,
    fa.dt_asset_write_off,
    fa.industry_code,
    fait.fixed_asset_industry_name AS industry_name,
    fa.asset_type_for_stat_reporting_code,
    fatfsrt.fixed_asset_type_for_stat_reporting_name AS asset_type_for_stat_reporting_name,
    fa.accelerated_depreciation_reason_code,
    faadrt.accelerated_depreciation_reason_name AS accelerated_depreciation_reason_name,
    fa.depreciation_group_code,
    fadgt.depreciation_group_name,
    fa.base_uom_code,
    uom_text.uom_full_name AS base_uom_name,
    fa.asset_quantity,
    fa.asset_initial_number,
    fa.asset_initial_subnumber,
    fa.okof_code,
    fa.asset_passport_number,
    fa.dt_asset_manufacture_yyyy,
    fa.asset_serial_number,
    fa.asset_put_into_operation_code AS asset_recognition_code,
    fa.investment_activity_profile_code,
    fa.dt_assignment_valid_from,
    fa.dt_assignment_valid_to,
    fa.cost_center_code,
    cct.cost_center_full_name AS cost_center_name,
    fa.plant_code,
    pas.plant_short_name AS plant_name,
    fa.multiple_shift_coefficient_amount,
    fa.is_asset_temporary_shutdown,
    fa.asset_location_code,
    ts.location_name AS asset_location_name,
    fa.internal_order_code,
    ord.order_short_name AS internal_order_name,
    fa.personnel_code,
    pmd.employee_full_name AS personnel_name,
    fa.is_asset_conservated,
    fa.dt_conservated_from,
    fa.dt_conservated_to,
    fa.dt_conservated_actual,
    fa.document_type_code,
    facdtt.conservation_document_type_name AS document_type_name,
    fa.special_order_for_conservation_number,
    fa.dt_special_order_for_conservation,
    fa.special_order_for_cancelling_conservation_number,
    fa.dt_special_order_for_cancelling_conservation,
    fa.dt_conservation_cancelled_actual,
    rfm.dt_valid_from AS dt_approved_of_techical_state,
    rfm.non_liquid_asset_technical_state_code,
    rfm02.non_liquid_asset_technical_state_name,
    rfm.is_non_liquid_asset_record_created_manually,
    rfm.dt_valid_to AS dt_asset_status_reverse_from_non_liquid,
    fadp.dt_valid_from AS dt_depreciation_parameters_valid_from,
    fadp.dt_valid_to   AS dt_depreciation_parameters_valid_to,
    fadp.dt_depreciation_calculation_start,
    fadp.depreciation_method_code,
    fadmt.depreciation_method_name,
    fadp.planned_useful_life_in_years,
    fadp.planned_useful_life_in_periods
FROM sep_int AS si
/* Join the fact table only now; date-band on dt_valid_from */
LEFT JOIN dict_dds.fixed_asset_td_V2 AS fa
  ON si.unit_balance_code = fa.unit_balance_code
 AND si.asset_main_code   = fa.asset_main_code
 AND si.asset_sub_code    = fa.asset_sub_code
 AND si.dt_valid_from BETWEEN fa.dt_from AND fa.dt_to

LEFT JOIN dict_dds.fixed_asset_depreciation_parameters_td_v2 AS fadp
  ON si.unit_balance_code       = fadp.unit_balance_code
 AND si.asset_main_code         = fadp.asset_main_code
 And si.asset_sub_code          = fadp.asset_sub_code
 AND si.depreciation_area_code  = fadp.depreciation_area_code
 AND si.dt_valid_from BETWEEN fadp.dt_valid_from AND fadp.dt_valid_to

LEFT JOIN dict_dds.fixed_asset_non_liquid_technical_state_td AS rfm
  ON rfm.unit_balance_code   = si.unit_balance_code
 AND rfm.asset_inventory_code = fa.asset_inventory_number
 AND si.dt_valid_from BETWEEN rfm.dt_valid_from AND rfm.dt_valid_to

/* Dictionaries/texts — REPLICATED recommended */
LEFT JOIN dict_dds.unit_balance ub
  ON ub.unit_balance_code = si.unit_balance_code
 AND ub.language_code = 'R'

LEFT JOIN dict_dds.unit_balance_valuation_area AS tb
  ON tb.unit_balance_code = si.unit_balance_code
 AND tb.valuation_area_code = si.depreciation_area_code

LEFT JOIN dict_dds.currency_texts AS ct
  ON ct.currency_code = tb.valuation_area_currency_code
 AND ct.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_valuation_area_texts AS favat
  ON favat.fixed_asset_valuation_area_code = si.depreciation_area_code
 AND favat.language_code = 'R'
 AND ub.asset_depreciation_rule_code = favat.fixed_asset_valuation_plan_code

LEFT JOIN dict_dds.fixed_asset_valuation_plan_texts AS favpt
  ON favpt.fixed_asset_valuation_plan_code = ub.asset_depreciation_rule_code
 AND favpt.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_class_texts AS fact
  ON fact.fixed_asset_class_code = fa.asset_class_code
 AND fact.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_general_ledger_account_link AS t095
  ON t095.general_ledger_reference_rule_code = fa.general_ledger_reference_rule_code
 AND t095.valuation_area_code = si.depreciation_area_code
 AND t095.account_chart_code  = ub.account_chart_code

LEFT JOIN dict_dds.fixed_asset_industry_texts AS fait
  ON fait.fixed_asset_industry_code = fa.industry_code
 AND fait.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_type_for_stat_reporting_texts AS fatfsrt
  ON fatfsrt.fixed_asset_type_for_stat_reporting_code = fa.asset_type_for_stat_reporting_code
 AND fatfsrt.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_accelerated_depreciation_reason_texts AS faadrt
  ON faadrt.accelerated_depreciation_reason_code = fa.accelerated_depreciation_reason_code
 AND faadrt.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_depreciation_group_texts AS fadgt
  ON fadgt.depreciation_group_code = fa.depreciation_group_code
 AND fadgt.language_code = 'R'

LEFT JOIN dict_dds.uom_external_format_texts AS uom_text
  ON uom_text.uom_code = fa.base_uom_code
 AND uom_text.language_code = 'R'

LEFT JOIN dict_dds.plant_and_subsidiary AS pas
  ON pas.plant_code = fa.plant_code

LEFT JOIN dict_dds.fixed_asset_location AS ts
  ON ts.plant_code = fa.plant_code
 AND ts.location_code = fa.asset_location_code

LEFT JOIN dict_dds.personnel_main_data AS pmd
  ON pmd.employee_code = fa.personnel_code
 AND CURRENT_DATE BETWEEN pmd.dt_valid_from AND pmd.dt_valid_to

LEFT JOIN dict_dds.fixed_asset_non_liquid_technical_state_texts AS rfm02
  ON rfm02.non_liquid_asset_technical_state_code = rfm.non_liquid_asset_technical_state_code
 AND rfm02.language_code = 'R'

LEFT JOIN dict_dds.fixed_asset_depreciation_method_texts AS fadmt
  ON fadmt.depreciation_method_code = fadp.depreciation_method_code
 AND fadmt.fixed_asset_valuation_plan_code = ub.asset_depreciation_rule_code
 AND fadmt.language_code = 'R'

LEFT JOIN dict_dds.map_controlling_area_to_unit_balance cab
  ON cab.unit_balance_code = ub.unit_balance_code

LEFT JOIN dict_dds.cost_center_texts AS cct
  ON cct.cost_center_code = fa.cost_center_code
 AND cct.controlling_area_code = cab.controlling_area_code
 AND cct.language_code = 'R'
 AND cct.dt_valid_to >= CURRENT_DATE

LEFT JOIN dict_dds.order_controlling AS ord
  ON ord.unit_balance_code = si.unit_balance_code
 AND ord.order_code = fa.internal_order_code;
