from dm_calc.storage_sales_bundles_amount as base
	left join dict_dds.material_texts as matet on
		matet.material_code = base.material_code
		and matet.language_code = 'R'
	left join dict_dds.transportation_service_texts as zsrvpost on
		zsrvpost.transportation_service_code = base.service_code
		and zsrvpost.language_code = 'R'
	left join dict_dds.etsng_texts as ette on
		ette.etsng_code = base.etsng_code
		and ette.language_code = 'R'
	left join dict_dds.transport_hub tvkn on
		tvkn.transport_hub_code = base.warehouse_code
	left join adrc as adrc_2 on
		adrc_2.address_code = tvkn.address_code
	left join dict_dds.country as r_sd556 on
		r_sd556.country_code = adrc_2.country_code
	left join dict_dds.market_region1_texts as t25a1 on
		t25a1.market_region1_code = r_sd556.market_region1_code
		and t25a1.language_code = 'R'
		
	left join dict_dds.plant_and_subsidiary as pas on
		pas.plant_code = base.receiving_plant_code
	left join dict_dds.transport_hub as trhu on
		trhu.transport_hub_code = base.warehouse_code
	left join adrc as adrc on
		adrc.address_code = trhu.address_code
	left join dict_dds.country_texts as cotex on
		cotex.country_code = adrc.country_code
		and cotex.language_code = 'R'
	left join dict_dds.transport_hub_texts as trht on
		trht.transport_hub_code = base.transport_departure_hub_code
		and trht.language_code = 'R'
	left join dict_dds.transport_hub_texts as trht_2 on
		trht_2.transport_hub_code = base.transport_destination_hub_code
		and trht_2.language_code = 'R'
	left join dict_dds.transport_hub_texts as trht_3 on
		trht_3.language_code = case when cotex.country_full_name = 'Россия' then 'R' else 'E' end
		and trht_3.transport_hub_code = base.warehouse_code
	left join dict_dds.transport_hub_texts as trht_4 on
		trht_4.language_code = 'R'
		and trht_4.transport_hub_code = base.warehouse_storage_area_code
	left join dds.delivery_document_position as deldp on
		deldp.delivery_code = base.delivery_for_storage_calculation_code
		and deldp.delivery_position_line_item_code = '000010'
	left join dict_dds.plant_and_subsidiary as pas_1124 on
		pas_1124.plant_code = deldp.plant_producer_code
	left join dict_dds.transportation_stage_texts as trst on
		trst.transportation_stage_code = base.transportation_stage_code
		and trst.language_code = 'R'
	left join dict_dds.counterparty as counter on
		counter.counterparty_code = base.forwarder_code
	left join dict_dds.transport_hub as tr_hub on
		tr_hub.transport_hub_code = base.port_of_discharge_code
	left join dict_dds.transport_hub as trhu_2 on
		trhu_2.transport_hub_code = base.remote_warehouse_code
	left join adrc as adrc_3 on
		adrc_3.address_code = trhu_2.address_code
	left join dict_dds.country_texts as cotex_2 on
		cotex_2.country_code = adrc_3.country_code
		and cotex_2.language_code = 'R'
	left join dict_dds.transport_hub_texts as trht_5 on
		trht_5.transport_hub_code = base.remote_warehouse_code
		and trht_5.language_code = case when cotex_2.country_full_name = 'Россия' then 'R' else 'E' end
	left join dict_dds.material_specification as dd_ms_s_n on
		dd_ms_s_n.material_code = base.material_code
	left join dict_dds.material_shape_texts as mf on
		mf.shape_code = dd_ms_s_n.shape_code
		and mf.language_code = 'R'
	left join dict_dds.transport_transfer_type_texts as ipd on
		ipd.transport_transfer_type_code = base.transport_type_code
		and ipd.language_code = 'R'
	left join dict_dds.counterparty as counter_2 on
		counter_2.counterparty_code = base.creditor_in_shipment_instruction_code
	left join dict_dds.address as addr_8 on
		addr_8.address_code = counter_2.address_code
		and addr_8.international_display_format_code is null

;
