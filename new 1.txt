CREATE TEMP TABLE for_1079_base ON COMMIT DROP AS
SELECT
    f.purchase_contract_code,
    f.position_line_item_code,
    f.bukrs,
    f.lifnr,
    f.zzport,
    f.kdatb,
    f.kdate,
    f.werks,
    f.zzsrvpos,
    tmp_1.range_low_value AS tmp_1,
    tmp_2.range_low_value AS tmp_2,
    f.zzforma,
    f.zzet_tarif,
    f.sdabw,
    f.datbi,
    f.datab,
    f.dt_valid_to,
    f.dt_valid_from,
    f.purchase_document_subtype_code,
    f.kopos,
    f.konwa,
    f.zzterminal,
    f.vzzzone,
    f.konws,
    f.service_code,
    f.price_uom_code,
    f.netpr,
    f.kbetr,
    f.order_rf,
    f.order_asia,
    f.order_euro_svh,
    f.currency_dogovor
FROM (
    SELECT
        ekpo.purchase_contract_code,
        ekpo.position_line_item_code,
        ekko.unit_balance_code AS bukrs,
        ekko.supplier_code AS lifnr,
        ekko.port_code AS zzport,
        ekko.dt_valid_from AS kdatb,
        ekko.dt_valid_to AS kdate,
        ekpo.plant_code AS werks,
        ekpo.service_code AS zzsrvpos,
        ekpo.shape_code AS zzforma,
        ekpo.etsng_code AS zzet_tarif,
        ekpo.transport_type_code AS sdabw,
        ekpo_price.dt_price_condition_valid_to AS datbi,
        ekpo_price.dt_price_condition_valid_from AS datab,
        ekko.dt_valid_to,
        ekko.dt_valid_from,
        ekko.purchase_document_subtype_code,
        ekpo_price.price_condition_position_code AS kopos,
        ekpo_price.price_condition_position_currency_code AS konwa,
        ekpo.terminal_code AS zzterminal,
        ekpo.warehouse_storage_area_code AS vzzzone,
        COALESCE(ekpo_price.price_scale_currency_code, ekpo_price.price_condition_position_currency_code) AS konws,
        ekpo.service_code,
        ekpo.price_uom_code,
        ekpo.price_vat_excluded_document_currency_amount AS netpr,
        ekpo_price.price_rate AS kbetr,

        -- order_rf
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END) AS order_rf,

        -- order_asia
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.warehouse_storage_area_code IS NULL THEN 1 ELSE 0 END) AS order_asia,

        -- order_euro_svh
        (CASE WHEN ekko.port_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.shape_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.etsng_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.transport_type_code IS NULL THEN 1 ELSE 0 END)
        + (CASE WHEN ekpo.terminal_code IS NULL THEN 1 ELSE 0 END) AS order_euro_svh,

        CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor,
        ekpo_price.is_price_condition_position_deleted_code,
        ROW_NUMBER() OVER (
            PARTITION BY ekpo.purchase_contract_code,
                         ekpo.position_line_item_code,
                         ekpo_price.dt_price_condition_valid_from,
                         ekpo_price.dt_price_condition_valid_to,
                         ekpo_price.price_scale_basis_code
            ORDER BY ekpo_price.price_scale_basis_consecutive_code
        ) AS rwn
    FROM dds.purchase_contract_price_conditions AS ekpo_price
    JOIN dds.purchase_contract_header AS ekko
        ON ekko.purchase_contract_code = ekpo_price.purchase_contract_code
    JOIN dds.purchase_contract_position AS ekpo
        ON ekpo.purchase_contract_code = ekko.purchase_contract_code
       AND ekpo.position_line_item_code = ekpo_price.purchase_contract_position_code
    LEFT JOIN dict_dds.currency_decimal_place_ral AS curdpr
        ON curdpr.currency_code = COALESCE(ekpo_price.price_scale_currency_code, ekpo_price.price_condition_position_currency_code)
       AND curdpr.decimal_place_number = '0'
    WHERE ekko.purchase_document_subtype_code IN ('ZDTL', 'ZDT', 'ZDE', 'ZDP', 'ZDF', 'ZDS')
      AND ekko.paydox_contract_type_code IN ('1', '6')
      AND ekpo.is_deleted IS NULL
      AND ekpo_price.is_price_condition_position_deleted_code IS NULL
      AND ekpo_price.price_condition_type_code = 'PB00'
      AND (ekpo_price.price_scale_basis_code = 'C' OR ekpo_price.price_scale_basis_code IS NULL)
) AS f
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_1
    ON tmp_1.abap_program_code = '/RUSAL/SD4553M'
   AND tmp_1.parameter_code = 'S_USLSTC'
   AND tmp_1.range_low_value = f.zzsrvpos
LEFT JOIN dict_dds.settings_and_parameters_sap AS tmp_2
    ON tmp_2.abap_program_code = '/RUSAL/SD4553M'
   AND tmp_2.parameter_code = 'S_USLST'
   AND tmp_2.range_low_value LIKE 'EU%'
   AND tmp_2.range_low_value = f.zzsrvpos
WHERE f.rwn = 1
DISTRIBUTED REPLICATED;

ANALYZE for_1079_base;

CREATE TEMP TABLE for_1079_rf_cont ON COMMIT DROP AS
SELECT
    x.rn,
    x.purchase_contract_code,
    x.position_line_item_code,
    x.konws,
    x.service_code,
    x.price_uom_code,
    x.currency_dogovor,
    x.datab,
    x.datbi
FROM (
    SELECT
        b.rn,
        f.purchase_contract_code,
        f.position_line_item_code,
        ROW_NUMBER() OVER (
            PARTITION BY b.rn
            ORDER BY f.order_rf, f.kopos, f.kbetr, f.purchase_contract_code, f.position_line_item_code

        f.currency_dogovor,
        f.datab,
        f.datbi

    JOIN for_1079_base AS f
      ON f.bukrs = '1120'
     AND f.lifnr = b.creditor_in_shipment_instruction_code
     AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
     AND f.werks = '1121'
     AND f.zzsrvpos = 'S49'
     AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
     AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
     AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN f.datab AND f.datbi
    WHERE b.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры')

WHERE x.row_num = 1

-- Европа
CREATE TEMP TABLE for_1079_euro ON COMMIT DROP AS
SELECT
    z.rn,
    z.purchase_contract_code,
    z.position_line_item_code,
    z.konws,
    z.service_code,
    z.price_uom_code,
    z.warehouse_weight_month_end,
    z.month_difference_2,
    z.netpr,
    z.special,
    z.currency_dogovor,
    z.datab,
    z.datbi
FROM (
    SELECT
        b.rn,
        f.purchase_contract_code,
        f.position_line_item_code,

        ROW_NUMBER() OVER (
            PARTITION BY b.rn
            ORDER BY
                f.order_euro_svh,
                f.kopos,
                f.kbetr,
                CASE
                    WHEN param_sap.range_low_value IS NOT NULL AND param_srv_1.range_low_value IS NOT NULL THEN 1
                    WHEN param_sap.range_low_value IS NULL AND param_srv_1.range_low_value IS NULL THEN 1
                    ELSE 0
                END DESC,
                f.purchase_contract_code,
                f.position_line_item_code
        ) AS row_num,

        f.konws,
        f.service_code,
        f.price_uom_code,

        SUM(b.sales_bundle_gross_weight) OVER (PARTITION BY b.warehouse_code, date_trunc('month', b.dt_storage_end)) AS warehouse_weight_month_end,

        CASE
            WHEN extract(day from b.dt_storage_end) < extract(day from b.dt_storage_start) THEN
                (extract(year from b.dt_storage_end) - extract(year from b.dt_storage_start)) * 12
              + (extract(month from b.dt_storage_end) - extract(month from b.dt_storage_start))
            ELSE
                (extract(year from b.dt_storage_end) - extract(year from b.dt_storage_start)) * 12
              + (extract(month from b.dt_storage_end) - extract(month from b.dt_storage_start)) + 1
        END AS month_difference_2,

        f.netpr,

        CASE
            WHEN b.storage_cost_special_calculation_type_code = '4' THEN ceil(b.storage_duration_total_calendar_days / 7::numeric)
            WHEN b.storage_cost_special_calculation_type_code = '5' THEN ceil(b.storage_duration_total_calendar_days / 30::numeric)
        END AS special,

        f.currency_dogovor,
        f.datab,
        f.datbi
    FROM dm_calc.storage_sales_bundles_weight AS b

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_1
      ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
     AND for_1082_1.case_code = 'PERET_FRN'
     AND for_1082_1.parameter_code = 'S_LIFNRA'
     AND for_1082_1.range_low_value = b.forwarder_code

    LEFT JOIN dict_dds.settings_and_parameters_sap AS for_1082_2
      ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
     AND for_1082_2.parameter_code = 'S_FORMSTEU'
     AND for_1082_2.range_low_value = b.material_shape_code

    JOIN for_1079_base AS f
      ON b.storage_cost_calculation_type_name = 'Логика Европа'
     AND f.purchase_document_subtype_code = 'ZDTL'
     AND (
            (for_1082_1.range_low_value IS NOT NULL AND f.zzsrvpos = 'S49')
         OR (for_1082_1.range_low_value IS NULL AND for_1082_2.range_low_value IS NOT NULL AND f.tmp_1 IS NOT NULL)
         OR (for_1082_1.range_low_value IS NULL AND for_1082_2.range_low_value IS NULL AND f.tmp_2 IS NOT NULL)
         )
     AND f.bukrs = b.unit_balance_code
     AND f.lifnr = b.forwarder_code
     AND f.werks = b.plant_code
     AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
     AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
     AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
     AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
     AND b.dt_for_price_search_in_purchase_contract BETWEEN f.dt_valid_from AND f.dt_valid_to
     AND b.dt_for_price_search_in_purchase_contract BETWEEN f.datab AND f.datbi
     AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)

-- 7) for_1090 (цена хранения)
-- ------------------------------------------------------------
CREATE TEMP TABLE for_1090 ON COMMIT DROP AS
WITH tmp AS (
    SELECT
        pcpc.price_condition_code AS knumh,
        pcpc.purchase_contract_code AS evrtn,
        pcpc.purchase_contract_position_code AS evrtp,
        pcpc.price_scale_basis_consecutive_code,
        pcpc.price_condition_position_code,
        pcpc.price_rate AS konp_kbetr,
        pcpc.price_scale_rate AS konm_kbetr,
        pcpc.price_scale_limit_from_quantity AS kstbm,
        pcpc.price_scale_limit_to_quantity AS kstbm_next,
        pcpc.price_scale_basis_code,
        pcpc.dt_price_condition_valid_to,
        pcpc.dt_price_condition_valid_from,
        COUNT(pcpc.price_scale_limit_from_quantity) OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code) AS cnt,
        ROW_NUMBER() OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code ORDER BY pcpc.price_scale_basis_consecutive_code) AS rn,
        MIN(price_scale_limit_from_quantity) OVER (
            PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code, dt_price_condition_valid_from, dt_price_condition_valid_to
        ) AS min_price_scale_limit_from_quantity,

        (pcpc.price_scale_limit_from_quantity
            - LAG(pcpc.price_scale_limit_from_quantity) OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code ORDER BY pcpc.price_scale_basis_consecutive_code)
        ) * LAG(pcpc.price_scale_rate) OVER (PARTITION BY pcpc.purchase_contract_code, pcpc.purchase_contract_position_code ORDER BY pcpc.price_scale_basis_consecutive_code) AS lagging
    FROM dds.purchase_contract_price_conditions AS pcpc
    WHERE pcpc.price_condition_type_code = 'PB00'
      AND (pcpc.price_scale_basis_code = 'C' OR pcpc.price_scale_basis_code IS NULL)
)
SELECT
    tmp.knumh,
    tmp.evrtn,
    tmp.evrtp,
    tmp.dt_price_condition_valid_to,
    tmp.dt_price_condition_valid_from,
    tmp.price_scale_basis_consecutive_code,
    tmp.price_condition_position_code,
    tmp.konp_kbetr,
    tmp.konm_kbetr,
    tmp.kstbm,
    tmp.kstbm_next,
    tmp.cnt,
    tmp.rn,
    tmp.price_scale_basis_code,
    tmp.min_price_scale_limit_from_quantity,
    SUM(tmp.lagging) OVER (PARTITION BY tmp.evrtn, tmp.evrtp ORDER BY tmp.price_scale_basis_consecutive_code) AS presum
FROM tmp
DISTRIBUTED REPLICATED;

ANALYZE for_1090;

-- 9) base_3 (расчет сумм хранения)
-- ------------------------------------------------------------
CREATE TEMP TABLE base_3 ON COMMIT DROP AS
SELECT
    b.*,

    (b.storage_duration_total_calendar_days - b.storage_duration_free_by_delivery_calendar_days) AS storage_duration_payable_calendar_days, -- LE.001089

    CASE
        WHEN b.storage_cost_special_calculation_type_code = '1' THEN NULL

        WHEN b.storage_cost_special_calculation_type_code = '2'
            THEN COALESCE(v1.konp_kbetr, v2.konp_kbetr) * b.for_1079_euro_month_difference_2 * b.sales_bundle_gross_weight

        WHEN p.range_low_value IS NOT NULL AND v1.evrtn IS NOT NULL
            THEN v1.konp_kbetr
                 * (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days)
                    - LEAST(COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days), b.storage_duration_free_by_contract_calendar_days))
                 / b.weight_for_1090_razdel
                 * b.weight_for_1090_umno
                 * b.currency_dogovor

        WHEN p.range_low_value IS NOT NULL AND v2.evrtn IS NOT NULL
            THEN COALESCE(v2.presum, 0)
                 + GREATEST(COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - v2.kstbm + 1, 0)
                   * v2.konm_kbetr
                   / b.weight_for_1090_razdel
                   * b.weight_for_1090_umno
                   * b.currency_dogovor

        WHEN v1.evrtn IS NOT NULL
            THEN v1.konp_kbetr
                 * (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days)
                    - LEAST(COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days), b.storage_duration_free_by_contract_calendar_days))
                 * b.weight_for_1090_umno
                 * b.currency_dogovor

        WHEN v2.evrtn IS NOT NULL
            THEN (COALESCE(v2.presum, 0)
                 + GREATEST(COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - v2.kstbm + 1, 0)
                   * v2.konm_kbetr)
                 * b.weight_for_1090_umno
                 * b.currency_dogovor
    END AS storage_cost_calculated_amount_local, -- LE.001125

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) BETWEEN 1 AND 30
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_001_030_days_gross_weight, -- LE.001095

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) BETWEEN 31 AND 60
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_031_060_days_gross_weight, -- LE.001096

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) BETWEEN 61 AND 90
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_061_090_days_gross_weight, -- LE.001097

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) BETWEEN 91 AND 180
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_091_180_days_gross_weight, -- LE.001098

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) BETWEEN 181 AND 365
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_181_365_days_gross_weight, -- LE.001099

    CASE WHEN (COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) - b.storage_duration_free_by_delivery_calendar_days) > 365
        THEN b.sales_bundle_gross_weight ELSE 0 END AS storage_payable_over_365_days_gross_weight -- LE.001100
FROM base_2_5 AS b
LEFT JOIN for_1090 AS v1
  ON v1.evrtn = b.purchase_contract_code
 AND v1.evrtp = b.purchase_contract_position_code
 AND v1.dt_price_condition_valid_to = b.datbi
 AND v1.dt_price_condition_valid_from = b.datab
 AND v1.price_scale_basis_code IS NULL
LEFT JOIN for_1090 AS v2
  ON v2.evrtn = b.purchase_contract_code
 AND v2.evrtp = b.purchase_contract_position_code
 AND v2.dt_price_condition_valid_to = b.datbi
 AND v2.dt_price_condition_valid_from = b.datab
 AND v2.price_scale_basis_code = 'C'
 AND COALESCE(b.for_1079_euro_special, b.storage_duration_total_calendar_days) BETWEEN v2.kstbm AND v2.kstbm_next
LEFT JOIN dict_dds.settings_and_parameters_sap AS p
  ON p.range_low_value = b.purchase_contract_position_uom_code
 AND p.abap_program_code = '/RUSAL/ZLE112M'
 AND p.parameter_code = 'S_BPRME'
DISTRIBUTED BY (sales_bundle_code);

            WHEN b.purchase_contract_position_currency_code = 'USD' OR b.storage_cost_calculated_amount_local IS NULL

            ELSE

                * CASE WHEN COALESCE(r.currency_rate, 1) > 0 THEN COALESCE(r.currency_rate, 1) ELSE 1 END
                / (CASE WHEN COALESCE(r.currency_rate, 1) < 0 THEN abs(COALESCE(r.currency_rate, 1)) ELSE 1 END * COALESCE(r.currency_from_multiplier, 1))
        END, 0) ELSE 0 END AS storage_calculated_cost_061_090_amount,

    CASE WHEN b.storage_duration_payable_calendar_days BETWEEN 91 AND 180 THEN COALESCE(

        END, 0) ELSE 0 END AS storage_calculated_cost_091_180_amount,

            WHEN b.purchase_contract_position_currency_code = 'USD' OR b.storage_cost_calculated_amount_local IS NULL
