#!/usr/bin/env python3

import argparse
import base64
import hashlib
import hmac
import json
import os
import secrets
import ssl
import sys
import time
import urllib.error
import urllib.request
from urllib.parse import urlsplit


class BotXError(Exception):
    pass


def _b64(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode()


def make_token(bot_id: str, secret_key: str, cts_host: str) -> str:
    """JWT HS256, которым бот подписывает каждый запрос. Живёт минуту."""
    now = int(time.time())
    header = {"alg": "HS256", "typ": "JWT"}
    payload = {
        "iss": bot_id,
        "aud": cts_host,
        "exp": now + 60,
        "nbf": now,
        "jti": secrets.token_hex(12),
        "iat": now,
        "version": 2,
    }
    signing_input = ".".join(
        _b64(json.dumps(part, separators=(",", ":")).encode()) for part in (header, payload)
    )
    signature = hmac.new(secret_key.encode(), signing_input.encode(), hashlib.sha256).digest()
    return f"{signing_input}.{_b64(signature)}"


class BotX:
    def __init__(self, cts_url: str, bot_id: str, secret_key: str, verify_ssl: bool = True):
        self.cts_url = cts_url.rstrip("/")
        self.bot_id = bot_id
        self.secret_key = secret_key
        self.host = urlsplit(self.cts_url).hostname or ""
        self.ssl_context = None if verify_ssl else ssl._create_unverified_context()

    def call(self, method: str, path: str, body=None):
        request = urllib.request.Request(
            self.cts_url + path,
            method=method,
            data=None if body is None else json.dumps(body).encode(),
            headers={
                "Authorization": "Bearer " + make_token(self.bot_id, self.secret_key, self.host),
                "Content-Type": "application/json",
            },
        )
        try:
            with urllib.request.urlopen(request, timeout=30, context=self.ssl_context) as response:
                raw = response.read()
        except urllib.error.HTTPError as exc:
            text = exc.read().decode("utf-8", "replace")[:500]
            raise BotXError(f"{method} {path} -> HTTP {exc.code}: {text}") from None
        except urllib.error.URLError as exc:
            raise BotXError(f"{method} {path} -> нет связи: {exc.reason}") from None

        try:
            return json.loads(raw)["result"]
        except (ValueError, KeyError, TypeError):
            raise BotXError(
                f"{method} {path} -> ответ не похож на BotX API: {raw[:200]!r}. "
                "Проверьте CTS_URL: нужен адрес сервера без пути.",
            ) from None

    def find_user(self, email: str):
        try:
            users = self.call("POST", "/api/v3/botx/users/by_email", {"emails": [email]})
        except BotXError as exc:
            if "HTTP 404" in str(exc):
                return None
            raise
        return users[0] if users else None

    def personal_chat(self, user_huid: str) -> str:
        for chat in self.call("GET", "/api/v3/botx/chats/list"):
            if chat.get("chat_type") == "chat" and user_huid in chat.get("members", []):
                return chat["group_chat_id"]
        created = self.call(
            "POST",
            "/api/v3/botx/chats/create",
            {"name": "chat", "chat_type": "chat", "members": [user_huid]},
        )
        return created["chat_id"]

    def send(self, chat_id: str, text: str) -> str:
        result = self.call(
            "POST",
            "/api/v4/botx/notifications/direct",
            {"group_chat_id": chat_id, "notification": {"status": "ok", "body": text}},
        )
        return result["sync_id"]


def main() -> int:
    parser = argparse.ArgumentParser(description="Сообщение пользователю eXpress по e-mail")
    parser.add_argument("email")
    parser.add_argument("text", default="Сообщение без текста сообщения")
    parser.add_argument("--cts-url", default="https://usr.al.team")
    parser.add_argument("--bot-id", default="6305cc5f-3543-5559-a84c-f898b8e2c0ce")
    parser.add_argument("--secret-key", default=os.environ.get("BOT_SECRET_KEY"))
    parser.add_argument("--insecure", action="store_true", help="не проверять сертификат сервера")
    args = parser.parse_args()

    missing = [
        name
        for name, value in (
            ("CTS_URL", args.cts_url),
            ("BOT_ID", args.bot_id),
            ("BOT_SECRET_KEY", args.secret_key),
        )
        if not value
    ]
    if missing:
        print("Не заданы: " + ", ".join(missing), file=sys.stderr)
        return 1

    botx = BotX(args.cts_url, args.bot_id, args.secret_key, verify_ssl=not args.insecure)
    try:
        user = botx.find_user(args.email)
        if user is None:
            print(f"Пользователь с адресом {args.email} не найден", file=sys.stderr)
            return 1
        chat_id = botx.personal_chat(user["user_huid"])
        sync_id = botx.send(chat_id, args.text)
    except BotXError as exc:
        print(f"Ошибка: {exc}", file=sys.stderr)
        return 1

    print(f"Отправлено: {user.get('name')} ({args.email}), чат {chat_id}, sync_id {sync_id}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
