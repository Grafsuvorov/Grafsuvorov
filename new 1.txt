отлично, сведу всё в три таблицы — как ты попросил:

Справочник проверок (где хранится сама проверка, её владелец и пороги).

Короткая сводка (по каждому запуску и проверке — агрегаты, статус).

Детальная (строки-нарушения с гибким payload_json).

Ниже — компактные DDL под Greenplum + пояснения каждого поля и примеры.

1) dq.registry — справочник проверок
CREATE SCHEMA IF NOT EXISTS dq;

-- маленькая и часто джоинится → REPLICATED
CREATE TABLE dq.registry (
  check_id         text PRIMARY KEY,      -- 'dq_sd_0001'
  name             text NOT NULL,         -- краткое имя проверки
  description      text,                  -- подробности логики
  category         text NOT NULL,         -- 'technical' | 'business' | 'freshness' | ...
  severity         text NOT NULL,         -- 'P1'..'P4'
  owner_team       text,                  -- команда-владелец
  owner_person     text,                  -- ответственный
  dataset          text,                  -- логическое направление (финансы/сбыт/…)
  schema_name      text NOT NULL,         -- где проверяем (схема)
  table_name       text NOT NULL,         -- где проверяем (таблица)

  -- Порог/критерий срабатывания сводного статуса
  threshold_type   text,                  -- 'abs' | 'pct' | 'dynamic'
  threshold_value  numeric,               -- напр. 0 (=нет ошибок) или 0.01 (=1%)
  warn_threshold   numeric,               -- опционально: порог для WARN (например 0.001)

  -- Как мы режем данные и когда проверяем
  partition_attr   text,                  -- название партиционного признака в источнике (например 'dt' или 'yyyymm')
  schedule         text,                  -- cron/rrule/описание расписания
  enabled          boolean DEFAULT true,  -- включена ли проверка

  -- Источник спецификации/генерации
  source_yaml_path text,                  -- путь к YAML (если spec-driven)

  created_at       timestamp DEFAULT now(),
  updated_at       timestamp DEFAULT now()
)
DISTRIBUTED REPLICATED;

Что означает каждое поле (коротко)

check_id — уникальный ID проверки, основной ключ в отчётности/алёртах.

name/description — человекочитаемое название/описание.

category — тип проверки (техническая, бизнесовая, свежесть и т.д.).

severity — критичность нарушения (влияет на приоритизацию).

owner_team/owner_person — кому писать, кто чинит.

dataset/schema_name/table_name — объект проверки.

threshold_type/value — как оценивать результат (абсолют/доля/динамика).

warn_threshold — «жёлтый» уровень (необязательно).

partition_attr — имя партиционного признака (например, «проверяем помесячно по yyyymm»).

schedule — когда гоняем.

enabled — можно массово отключить проверку.

source_yaml_path — если генерируешь SQL из YAML.

2) dq.summary — короткая сводка (агрегаты + статус)
-- много читаем агрегатами → AO Column
CREATE TABLE dq.summary_base (
  run_date       date NOT NULL,           -- дата запуска (партиция)
  run_id         text NOT NULL,           -- id прогона (для трассировки/идемпотентности)
  check_id       text NOT NULL,

  run_started_at timestamp,
  run_ended_at   timestamp,

  status         text,                    -- 'pass' | 'warn' | 'fail' | 'error'
  message        text,                    -- пояснение, почему warn/fail

  dataset        text,
  schema_name    text,
  table_name     text,

  partition_key  text,                    -- значение среза (напр. '2025-10' или '2025-11-19')
  data_version   text,                    -- версия/снапшот данных (batch_id/SCN/manifest)

  rows_tested    bigint,
  rows_failed    bigint,
  fail_pct       numeric,

  coverage_pct   numeric,                 -- если считаем охват
  freshness_sec  bigint,                  -- если проверка на свежесть

  severity       text,                    -- дубль из registry на момент запуска
  executor       text                     -- DAG/таска/инстанс
)
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED BY (run_date);

CREATE TABLE dq.summary
  (LIKE dq.summary_base INCLUDING DEFAULTS INCLUDING INDEXES)
PARTITION BY RANGE (run_date);

-- Пример партиции на месяц:
CREATE TABLE dq.summary_y2025m11 PARTITION OF dq.summary
  FOR VALUES FROM ('2025-11-01') TO ('2025-12-01')
  (LIKE dq.summary_base INCLUDING DEFAULTS INCLUDING INDEXES);

CREATE INDEX ON dq.summary (check_id, run_date);
CREATE INDEX ON dq.summary (status);

Поля (по смыслу)

run_date/run_id — когда и какой прогон.

check_id — какая проверка.

status/message — итог и краткое пояснение по порогам.

dataset/schema_name/table_name — контекст.

partition_key — значение контролируемого среза (в отличие от partition_attr в справочнике, которое — имя атрибута).

data_version — идентификатор версии данных (batch_id/SCN/manifest…), чтобы воспроизводить запуск.

rows_tested/rows_failed/fail_pct — метрики.

coverage_pct/freshness_sec — по желанию.

severity — для фильтрации без лишних JOIN.

executor — кто выполнял.

3) dq.detail — детальная (строки-нарушения с JSON)
CREATE TABLE dq.detail_base (
  run_date      date NOT NULL,            -- дата запуска (партиция)
  run_id        text NOT NULL,            -- id прогона
  check_id      text NOT NULL,

  schema_name   text NOT NULL,
  table_name    text NOT NULL,

  row_key       text NOT NULL,            -- бизнес-ключ нарушителя или стабильный хэш набора полей
  error_code    text NOT NULL,            -- тип нарушения (NULL_REQUIRED/UNIQUENESS/... )
  error_msg     text,                     -- коротко «что не так»

  payload_json  jsonb,                    -- гибкие детали (см. примеры ниже)

  severity      text,                     -- копия критичности на момент запуска
  partition_key text,                     -- значение среза (период/дата и т.п.)
  data_version  text,                     -- версия/снапшот данных

  found_at      timestamp DEFAULT now()   -- когда зафиксировали нарушение
)
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED BY (check_id, row_key);
