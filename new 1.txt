WITH RECURSIVE lineage AS (
    SELECT
        d.parent_table_id,
        d.parent_table,
        d.child_table_id,
        d.child_table,
        1 AS depth
    FROM tech_etl.table_dependencies d
    WHERE d.parent_table = 'dm.sb_wuc'

    UNION ALL

    SELECT
        d.parent_table_id,
        d.parent_table,
        d.child_table_id,
        d.child_table,
        l.depth + 1
    FROM tech_etl.table_dependencies d
    JOIN lineage l
      ON l.child_table_id = d.parent_table_id
)
SELECT
    l.depth,
    l.parent_table,
    l.child_table,
    pm.table_last_load AS parent_load,
    cm.table_last_load AS child_load,
    (cm.table_last_load - pm.table_last_load) AS lag
FROM lineage l
JOIN tech_etl.tables_meta pm ON pm.table_id = l.parent_table_id
JOIN tech_etl.tables_meta cm ON cm.table_id = l.child_table_id
ORDER BY depth, lag DESC;
