WITH hub_ext AS (
    SELECT
        h.transport_hub_code,
        h.address_code,
        a.country_code,
        ct.country_full_name,
        mr.market_region1_name,
        h.transport_hub_name_rus,
        h.transport_hub_name_eng
    FROM dict_dds.transport_hub h
    LEFT JOIN dict_dds.address a
        ON a.address_code = h.address_code
    LEFT JOIN dict_dds.country c
        ON c.country_code = a.country_code
    LEFT JOIN dict_dds.country_texts ct
        ON ct.country_code = a.country_code
       AND ct.language_code = 'R'
    LEFT JOIN dict_dds.market_region1_texts mr
        ON mr.market_region1_code = c.market_region1_code
       AND mr.language_code = 'R'
)

SELECT
    base.sales_bundle_code,
    base.plant_producer_delivery_code,
    base.plant_producer_delivery_position_code,
    base.initial_delivery_code,
    '10' AS initial_delivery_position_code,
    '10' AS sales_delivery_position_code,

    base.material_code,
    matet.material_name,
    base.material_code || '-' || matet.material_name AS material_search_name,

    base.dt_shipment,
    base.railcar_code,
    base.transport_bill_code,
    base.uni,

    base.material_shape_code,
    mst.material_shape_full_name AS material_shape_name,
    base.material_shape_code || '-' || mst.material_shape_full_name AS material_shape_search_name,

    base.port_of_discharge_code,
    port_hub.transport_hub_name_eng AS port_of_discharge_name,
    base.dt_arrival_to_russian_port,
    base.dt_storage_end_in_release,
    base.port_of_discharge_code || '-' || port_hub.transport_hub_name_eng AS port_of_discharge_search_name,

    base.transport_type_code,
    ipd.transport_transfer_type_name AS transport_type_name,
    base.transport_type_code || '-' || ipd.transport_transfer_type_name AS transport_type_search_name,

    base.etsng_code,
    ette.etsng_name,
    base.etsng_code || '-' || ette.etsng_name AS etsng_search_name,

    base.dt_report,
    base.sales_bundle_net_weight,
    base.sales_bundle_gross_weight,
    base.batch_code,
    base.dt_arrival,

    CASE
        WHEN wh_hub.market_region1_name = 'Турция' THEN 'Западная Азия'
        WHEN wh_hub.market_region1_name = 'Средний Восток и Афр' THEN 'Ближний Восток'
        WHEN wh_hub.market_region1_name IN ('Азия', 'Китай') THEN 'Восточная Азия'
        ELSE wh_hub.market_region1_name
    END AS region_of_remote_warehouse_name,

    (pas.plant_short_name || ' ' || pas.plant_full_name) AS receiving_plant_name,
    (base.receiving_plant_code || '-' || pas.plant_short_name || ' ' || pas.plant_full_name)
        AS receiving_plant_search_name,

    CASE
        WHEN wh_hub.country_full_name = 'Корея, республика' THEN 'Южная Корея'
        WHEN wh_hub.country_full_name = 'Тайвань (Китай)' THEN 'Тайвань'
        ELSE wh_hub.country_full_name
    END AS country_of_remote_warehouse_name,

    base.receiving_plant_code,
    base.sales_delivery_code,
    base.transportation_inbound_delivery_code,
    base.delivery_for_storage_calculation_code,
    base.transportation_inbound_delivery_position_code,
    base.delivery_position_for_storage_calculation_code,
    base.forwarder_code,
    base.storage_cost_calculation_type_name,

    base.transport_departure_hub_code,
    dep_hub.transport_hub_name_rus AS transport_departure_hub_name,

    base.transport_destination_hub_code,
    dest_hub.transport_hub_name_rus AS transport_destination_hub_name,

    base.transportation_outbound_delivery_code,
    base.transportation_outbound_delivery_position_code,
    base.is_final_transportation_stage_code,
    base.transportation_stage_code,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия', 'Логика Европа', 'Логика СВХ')
            THEN trst.transportation_stage_name
    END AS transportation_stage_name,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия', 'Логика Европа', 'Логика СВХ')
            THEN base.transportation_stage_code || '-' || trst.transportation_stage_name
    END AS transportation_stage_search_name,

    base.storage_calculation_bundle_quantity,

    COALESCE(
        CASE
            WHEN base.storage_cost_calculation_type_name IN ('Логика Азия', 'Логика Европа', 'Логика СВХ')
                THEN counter.counterparty_short_name
        END,
        'Не определено'
    ) AS forwarder_name,

    COALESCE(
        CASE
            WHEN base.storage_cost_calculation_type_name IN ('Логика Азия', 'Логика Европа', 'Логика СВХ')
                THEN base.forwarder_code || '-' || counter.counterparty_short_name
        END,
        'Не определено'
    ) AS forwarder_search_name,

    base.warehouse_code,

    CASE
        WHEN wh_hub.country_full_name = 'Россия'
            THEN wh_hub.transport_hub_name_rus
        ELSE wh_hub.transport_hub_name_eng
    END AS warehouse_name,

    base.warehouse_code || '-' ||
    CASE
        WHEN wh_hub.country_full_name = 'Россия'
            THEN wh_hub.transport_hub_name_rus
        ELSE wh_hub.transport_hub_name_eng
    END AS warehouse_search_name,

    base.delivery_in_final_release_code,
    base.dt_final_release,
    base.bill_of_lading_code,
    base.bill_of_lading_number,
    base.dt_bill_of_lading,
    base.incoterms_code,
    base.storage_cost_special_calculation_type_code,

    base.warehouse_storage_area_code,
    area_hub.transport_hub_name_rus AS warehouse_storage_area_name,
    base.warehouse_storage_area_code || '-' || area_hub.transport_hub_name_rus
        AS warehouse_storage_area_search_name,

    base.plant_code,
    base.unit_balance_code,
    base.creditor_in_shipment_instruction_code,

    base.dt_storage_start,
    base.dt_shipped_from_warehouse,
    base.dt_storage_end,

    base.service_number,
    base.storage_duration_total_calendar_days,
    base.shipment_instruction_number,
    base.dt_for_price_search_in_purchase_contract,
    base.purchase_contract_code,
    base.purchase_contract_position_code,
    base.purchase_contract_position_currency_code,

    base.service_code,
    zsrvpost.transportation_service_name AS service_name,
    base.service_code || '-' || zsrvpost.transportation_service_name AS service_search_name,

    base.purchase_contract_position_uom_code,

    addr_8.address_full_name AS creditor_in_shipment_instruction_name,
    base.creditor_in_shipment_instruction_code || '-' || addr_8.address_full_name
        AS creditor_in_shipment_instruction_search_name,

    base.storage_duration_free_by_contract_calendar_days,
    base.storage_duration_free_by_delivery_calendar_days,
    base.storage_duration_payable_calendar_days,

    base.remote_warehouse_code,

    CASE
        WHEN remote_hub.country_full_name = 'Россия'
            THEN remote_hub.transport_hub_name_rus
        ELSE remote_hub.transport_hub_name_eng
    END AS remote_warehouse_name,

    remote_hub.country_code AS country_of_remote_warehouse_code,

    base.dt_transportation_stage_start_p,
    base.dt_transportation_stage_start_r,
    base.dt_shipment_from_foreign_warehouse,

    base.storage_cost_calculated_amount,
    base.storage_cost_calculated_amount_local,
    base.storage_payable_001_030_days_gross_weight,
    base.storage_payable_031_060_days_gross_weight,
    base.storage_payable_061_090_days_gross_weight,
    base.storage_payable_091_180_days_gross_weight,
    base.storage_payable_181_365_days_gross_weight,
    base.storage_payable_over_365_days_gross_weight,

    base.is_stored_in_russian_port_code,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры', 'Логика РФ-балк')
            THEN 'Терминал порта'
        WHEN base.storage_cost_calculation_type_name IN ('Логика СВХ')
            THEN 'Склад СВХ'
        ELSE 'Не определено'
    END AS type_warehouse_name,

    CASE
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-контейнеры'
            THEN 'Контейнеры'
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-балк'
            THEN 'Балк'
        ELSE 'Не определено'
    END AS type_freight_name,

    (pas_1124.plant_short_name || ' ' || pas_1124.plant_full_name)
        AS metal_owner_for_reporting_name,

    base.storage_calculated_cost_001_030_amount,
    base.storage_calculated_cost_031_060_amount,
    base.storage_calculated_cost_061_090_amount,
    base.storage_calculated_cost_091_180_amount,
    base.storage_calculated_cost_181_365_amount,
    base.storage_calculated_cost_over_365_amount,
    base.metal_in_china_stock_balance_closing_weight
FROM dm_calc.storage_sales_bundles_amount base
    -- материалы
    LEFT JOIN dict_dds.material_texts matet
        ON matet.material_code = base.material_code
       AND matet.language_code = 'R'
    LEFT JOIN dict_dds.material_specification ms
        ON ms.material_code = base.material_code
    LEFT JOIN dict_dds.material_shape_texts mst
        ON mst.shape_code = ms.shape_code
       AND mst.language_code = 'R'

    -- услуги
    LEFT JOIN dict_dds.transportation_service_texts zsrvpost
        ON zsrvpost.transportation_service_code = base.service_code
       AND zsrvpost.language_code = 'R'

    -- ЕТСНГ
    LEFT JOIN dict_dds.etsng_texts ette
        ON ette.etsng_code = base.etsng_code
       AND ette.language_code = 'R'

    -- склады / хабы (через CTE hub_ext)
    LEFT JOIN hub_ext wh_hub
        ON wh_hub.transport_hub_code = base.warehouse_code
    LEFT JOIN hub_ext port_hub
        ON port_hub.transport_hub_code = base.port_of_discharge_code
    LEFT JOIN hub_ext dep_hub
        ON dep_hub.transport_hub_code = base.transport_departure_hub_code
    LEFT JOIN hub_ext dest_hub
        ON dest_hub.transport_hub_code = base.transport_destination_hub_code
    LEFT JOIN hub_ext area_hub
        ON area_hub.transport_hub_code = base.warehouse_storage_area_code
    LEFT JOIN hub_ext remote_hub
        ON remote_hub.transport_hub_code = base.remote_warehouse_code

    -- заводы
    LEFT JOIN dict_dds.plant_and_subsidiary pas
        ON pas.plant_code = base.receiving_plant_code

    -- собственник металла (завод из договора хранения)
    LEFT JOIN dds.delivery_document_position deldp
        ON deldp.delivery_code = base.delivery_for_storage_calculation_code
       AND deldp.delivery_position_line_item_code = '000010'
    LEFT JOIN dict_dds.plant_and_subsidiary pas_1124
        ON pas_1124.plant_code = deldp.plant_producer_code

    -- этап перевозки
    LEFT JOIN dict_dds.transportation_stage_texts trst
        ON trst.transportation_stage_code = base.transportation_stage_code
       AND trst.language_code = 'R'

    -- экспедитор
    LEFT JOIN dict_dds.counterparty counter
        ON counter.counterparty_code = base.forwarder_code

    -- кредитор + адрес
    LEFT JOIN dict_dds.counterparty counter_2
        ON counter_2.counterparty_code = base.creditor_in_shipment_instruction_code
    LEFT JOIN dict_dds.address addr_8
        ON addr_8.address_code = counter_2.address_code
       AND addr_8.international_display_format_code IS NULL

    -- тип ПС
    LEFT JOIN dict_dds.transport_transfer_type_texts ipd
        ON ipd.transport_transfer_type_code = base.transport_type_code
       AND ipd.language_code = 'R';
