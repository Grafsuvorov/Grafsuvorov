import requests

YOUTRACK_URL = ""
YOUTRACK_TOKEN = ""
ISSUE_ID = "DWH-10152"

headers = {
    "Authorization": f"Bearer {YOUTRACK_TOKEN}",
    "Accept": "application/json",
}

r = requests.get(
    f"{YOUTRACK_URL}/api/issues/{ISSUE_ID}",
    headers=headers,
    params={
        "fields": "idReadable,customFields(name,value(name))"
    },
    verify=False
)

data = r.json()

print("\n=== ВСЕ ПОЛЯ ===")
for f in data.get("customFields", []):
    print(f["name"], "=>", f.get("value"))

print("\n=== ИЩЕМ НАПРАВЛЕНИЕ ===")
found = False

for f in data.get("customFields", []):
    if f["name"] == "Направление":
        found = True
        print("FOUND:", f)

if not found:
    print("❌ Направление НЕ пришло из API")
