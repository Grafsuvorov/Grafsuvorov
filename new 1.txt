select * from dds.accounting_documents
where 
	is_active = true
	and (reference_procedure in( 'BKPF','BKPFF'))

select * from dds.accounting_documents
where 
	p.is_active = true
	and p.account_type in ('D', 'K')
	and (p.accounting_document_status_code is null
		or p.accounting_document_status_code = 'A');
