# -*- coding: utf-8 -*-
"""
Извлечение таблиц из tech_etl.tables_meta.query_insert_init с игнором временных таблиц:
- цель INSERT INTO (кроме temp)
- таблицы из FROM/JOIN (кроме temp и CTE)
Сохраняет результат в CSV и печатает итоги.

Требуются: sqlalchemy, psycopg2-binary, pandas
"""

import re
import sys
import pandas as pd
from sqlalchemy import create_engine, text

# ---- НАСТРОЙКИ ---------------------------------------------------------------
DB_URL = "postgresql+psycopg2://postgres:0506@localhost:5432/dwh"
OUTPUT_CSV = "tables_meta_dependencies.csv"

# Игнорируемые схемы (паттерны re.fullmatch по lower())
IGNORE_SCHEMAS = [
    r"pg_temp",           # иногда так
    r"pg_temp_\d+",       # чаще так
    r"pg_toast_temp_\d+",
]

# Игнорируемые имена таблиц (паттерны re.fullmatch по lower())
IGNORE_TABLE_PATTERNS = [
    r"#.*",               # sql-server style временные (если вдруг встретится в тексте)
    r"(tmp|temp|tt|t)_.+",# распространённые префиксы
]

# ---- Регексы и утилиты -------------------------------------------------------

RE_INSERT = re.compile(
    r"insert\s+into\s+([^\s(]+)",
    flags=re.IGNORECASE | re.MULTILINE,
)

RE_FROM_JOIN = re.compile(
    r"(?:from|join)\s+([^\s,()]+)",
    flags=re.IGNORECASE | re.MULTILINE,
)

# WITH cte_name AS ( ... ), cte2 AS ( ... )
RE_WITH_CTES = re.compile(
    r"\bwith\b\s+(.+?)\)\s*select\b",
    flags=re.IGNORECASE | re.DOTALL,
)

LEADING_KEYWORDS = {"only", "lateral"}

FUNCTION_PREFIXES = (
    "unnest(", "generate_series(", "regexp_split_to_table(",
    "json_each(", "jsonb_each(", "json_each_text(", "jsonb_each_text(",
    "json_to_record(", "jsonb_to_record(", "json_to_recordset(", "jsonb_to_recordset(",
)

def unquote_ident(s: str) -> str:
    s = re.split(r"::", s, maxsplit=1)[0]
    s = s.strip("() \t\n\r")
    if s.startswith('"') and s.endswith('"') and s.count('"') == 2:
        return s[1:-1]
    parts = s.split(".")
    clean_parts = []
    for p in parts:
        p = p.strip()
        if len(p) >= 2 and p[0] == '"' and p[-1] == '"':
            clean_parts.append(p[1:-1])
        else:
            clean_parts.append(p)
    return ".".join(clean_parts)

def split_schema_table(token: str):
    token = token.strip()
    token = unquote_ident(token)
    token = token.split()[0]  # отрезаем алиас
    if "." in token:
        schema, table = token.split(".", 1)
        return (schema, table)
    return (None, token)

def is_function_like(token: str) -> bool:
    t = token.strip().lower()
    return any(t.startswith(fp) for fp in FUNCTION_PREFIXES)

def is_subquery_token(token: str) -> bool:
    return token.strip().startswith("(")

def is_temp_schema(schema: str) -> bool:
    if not schema:
        return False
    s = schema.lower()
    return any(re.fullmatch(pat, s) for pat in IGNORE_SCHEMAS)

def is_temp_table_name(table: str) -> bool:
    if not table:
        return False
    t = table.lower()
    return any(re.fullmatch(pat, t) for pat in IGNORE_TABLE_PATTERNS)

def is_temp_like(schema: str, table: str) -> bool:
    return is_temp_schema(schema) or is_temp_table_name(table)

def collect_cte_names(sql_text: str):
    """
    Возвращает множество имён CTE из секции WITH ... AS (...), чтобы не считать их таблицами.
    Работает по простому разбору: берем всё между WITH и ближайшим ') select'.
    """
    ctes = set()
    text_lower = sql_text.lower()
    # Попробуем найти секции WITH ... AS ( ... ) [, ...] SELECT
    # Если такой «блок» есть несколько раз, возьмем первый — на практике обычно один на запрос.
    m = RE_WITH_CTES.search(text_lower + " select")  # защита на конец строки
    if not m:
        return ctes

    block = sql_text[m.start():m.end()]  # оригинальный регистр
    # Примерно распилим по запятым вне скобок: пойдём проще — найдём все шаблоны "<name> AS ("
    for name in re.findall(r'\b([a-zA-Z_][a-zA-Z0-9_$]*)\s+AS\s*\(', block, flags=re.IGNORECASE):
        ctes.add(unquote_ident(name))
    return ctes

# ---- Основная логика ---------------------------------------------------------

def main():
    engine = create_engine(DB_URL)

    with engine.connect() as conn:
        rp = conn.execute(text("SHOW search_path"))
        search_path = rp.scalar() or ""
    default_schema = None
    if search_path:
        for raw in [s.strip() for s in search_path.split(",")]:
            if raw and raw != "$user":
                default_schema = raw
                break

    df = pd.read_sql("SELECT * FROM tech_etl.tables_meta", con=engine)

    rows = []
    uniq_fullnames = set()
    dropped_temp = 0
    dropped_cte = 0

    for idx, row in df.iterrows():
        meta_id = row.get("id") if "id" in df.columns else idx
        obj_name = row.get("name") or row.get("table_name") or ""
        sql_text = row.get("query_insert_init")
        if not isinstance(sql_text, str) or not sql_text.strip():
            continue

        cte_names = collect_cte_names(sql_text)

        # ---- 1) Цель INSERT INTO
        m = RE_INSERT.search(sql_text)
        if m:
            raw_target = m.group(1)
            first_token = raw_target.strip()
            for kw in LEADING_KEYWORDS:
                if first_token.lower().startswith(kw + " "):
                    first_token = first_token[len(kw):].strip()
            schema, table = split_schema_table(first_token)
            if schema is None:
                schema = default_schema

            if table and not is_function_like(table):
                if is_temp_like(schema, table):
                    dropped_temp += 1
                else:
                    full = f'{schema+"." if schema else ""}{table}'
                    if full not in uniq_fullnames:
                        uniq_fullnames.add(full)
                    rows.append({
                        "meta_id": meta_id,
                        "meta_name": obj_name,
                        "source_type": "insert_target",
                        "schema": schema,
                        "table": table,
                        "full_name": full
                    })

        # ---- 2) Источники FROM/JOIN
        for m in RE_FROM_JOIN.finditer(sql_text):
            token = m.group(1)

            low = token.strip().lower()
            for kw in LEADING_KEYWORDS:
                if low.startswith(kw + " "):
                    token = token[len(kw):].strip()
                    break

            if is_subquery_token(token) or is_function_like(token):
                continue

            schema, table = split_schema_table(token)
            if not table:
                continue
            # CTE?
            if table and unquote_ident(table).lower() in {c.lower() for c in cte_names}:
                dropped_cte += 1
                continue

            if schema is None:
                schema = default_schema

            if is_temp_like(schema, table):
                dropped_temp += 1
                continue

            full = f'{schema+"." if schema else ""}{table}'
            if full not in uniq_fullnames:
                uniq_fullnames.add(full)
            rows.append({
                "meta_id": meta_id,
                "meta_name": obj_name,
                "source_type": "select_source",
                "schema": schema,
                "table": table,
                "full_name": full
            })

    out = pd.DataFrame(rows)

    uniq = None
    if not out.empty:
        uniq = (out[["schema", "table", "full_name"]]
                .drop_duplicates()
                .sort_values(["schema", "table"], na_position="last"))

    out.to_csv(OUTPUT_CSV, index=False, encoding="utf-8")
    print(f"[OK] Строк в детальном выводе: {len(out)}")
    print(f"[OK] Уникальных таблиц: {len(uniq) if uniq is not None else 0}")
    print(f"[INFO] Отфильтровано temp/pg_temp: {dropped_temp}, CTE: {dropped_cte}")
    if uniq is not None:
        print(uniq.head(30).to_string(index=False))
    print(f"[SAVE] CSV: {OUTPUT_CSV}")

if __name__ == "__main__":
    try:
        main()
    except Exception as e:
        print("[ERROR]", e, file=sys.stderr)
        sys.exit(1)
