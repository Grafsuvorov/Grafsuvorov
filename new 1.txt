with table_agg as(
	select
		tm.table_schema as schema_name,
		tm.table_name,
		count(distinct em.entity_name)  AS entity_count,
		json_agg(distinct em.entity_name order by em.entity_name)::jsonb AS entity_names,
		count(distinct tm.query_insert_init) as qii_diff_cnt
	from
	    tech_etl.tables_meta as tm
	    inner join tech_etl.entities_meta as em on
	        tm.entity_id = em.entity_id
	where
	    tm.flag_active
	    and (em.flag_active or em.on_new_framework)
	group by
		tm.table_schema,
		tm.table_name
), table_entity_unnest as (
select *
,jsonb_array_elements_text(ta.entity_names::jsonb) AS entity_name
from table_agg as ta
)
select 
	entity_name, 
	count(distinct schema_name||table_name ) as cn_tables_all,
	count(distinct schema_name||table_name) filter (where entity_count>1) as cn_tables_cross,	
	json_agg(distinct entity_name_list) as entity_names_cross
from table_entity_unnest as te
,lateral jsonb_array_elements_text(te.entity_names::jsonb) as entity_name_list
group by entity_name
