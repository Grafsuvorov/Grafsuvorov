
CREATE TABLE dc_dbt.model (
	model_name varchar NOT NULL,
	model_path varchar NOT NULL,
	model_type varchar NOT NULL,
	model_tags varchar NULL,
	schema_name varchar NULL,
	table_name varchar NULL,
	table_full_name varchar NULL,
	inserted_dttm timestamp NOT NULL,
	updated_dttm timestamp NOT NULL,
	deleted_flag bool NOT NULL
)
WITH (
	appendonly=true,
	blocksize=32768,
	compresstype=zstd,
	compresslevel=3,
	orientation=column
)
DISTRIBUTED BY (model_name);
