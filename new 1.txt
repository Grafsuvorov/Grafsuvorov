select
      n.nspname as schema_name,
      c.relname as table_name,
      c.oid,
      pg_get_userbyid(c.relowner) as owner_name,
      c.relkind,
      c.reltablespace
  from pg_class c
  join pg_namespace n on n.oid = c.relnamespace
  where n.nspname = 'your_schema'
    and c.relname = 'your_table';

  Для всех таблиц схемы:

  select
      n.nspname as schema_name,
      c.relname as table_name,
      pg_get_userbyid(c.relowner) as owner_name,
      pg_size_pretty(pg_total_relation_size(c.oid)) as total_size,
      case
          when coalesce(a.compresstype, '') <> '' then a.compresstype
          else 'none'
      end as compression_type,
      a.compresslevel,
      case
          when coalesce(a.compresstype, '') <> '' then 'yes'
          else 'no'
      end as has_compression
  from pg_class c
  join pg_namespace n on n.oid = c.relnamespace
  left join pg_appendonly a on a.relid = c.oid
  where n.nspname = 'your_schema'
    and c.relkind in ('r', 'p')
  order by pg_total_relation_size(c.oid) desc;
