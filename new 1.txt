DROP TABLE IF EXISTS tmp_s1c_wc_landing_uuid;
DROP TABLE IF EXISTS tmp_s1c_wc_processed_uuid;

-- 1. Сначала берём только uuid нужного потока из landing
CREATE TEMP TABLE tmp_s1c_wc_landing_uuid AS
SELECT DISTINCT
    t.uuid
FROM landing."INPUT_DATA_FROM_SAPXI_IN" t
WHERE t.flow_id = 'SI_WorkingCapital_AI'
DISTRIBUTED BY (uuid);

ANALYZE tmp_s1c_wc_landing_uuid;


-- 2. Находим, какие uuid уже были загружены в stg
-- Так мы не делаем NOT EXISTS на каждую строку landing напрямую
CREATE TEMP TABLE tmp_s1c_wc_processed_uuid AS
SELECT DISTINCT
    s.uuid
FROM stg.s1c_debt_for_working_capital s
JOIN tmp_s1c_wc_landing_uuid l
    ON l.uuid = s.uuid
DISTRIBUTED BY (uuid);

ANALYZE tmp_s1c_wc_processed_uuid;


-- 3. Забираем только новые XML
CREATE TEMP TABLE tmp_s1c_wc_src AS
SELECT
    t.uuid,
    t.dt_insert,
    t.document_xml,
    (xpath('Header/Sender/text()', t.document_xml))[1]::varchar AS sender,
    (xpath('Header/Receiver/text()', t.document_xml))[1]::varchar AS receiver,
    NULLIF(trim((xpath('Header/CreatedDateTime/text()', t.document_xml))[1]::varchar), '')::timestamp AS create_date_time
FROM landing."INPUT_DATA_FROM_SAPXI_IN" t
JOIN tmp_s1c_wc_landing_uuid l
    ON l.uuid = t.uuid
LEFT JOIN tmp_s1c_wc_processed_uuid p
    ON p.uuid = t.uuid
WHERE t.flow_id = 'SI_WorkingCapital_AI'
  AND p.uuid IS NULL
DISTRIBUTED BY (uuid);

ANALYZE tmp_s1c_wc_src;


-- 4. Раскрываем Body/Item отдельно
CREATE TEMP TABLE tmp_s1c_wc_items AS
SELECT
    s.uuid,
    s.dt_insert,
    s.sender,
    s.receiver,
    s.create_date_time,
    unnest(xpath('Body/Item', s.document_xml)) AS item_xml
FROM tmp_s1c_wc_src s
DISTRIBUTED BY (uuid);

ANALYZE tmp_s1c_wc_items;


-- 5. Вытаскиваем поля уровня Item и раскрываем entry
CREATE TEMP TABLE tmp_s1c_wc_entries AS
SELECT
    i.uuid,
    i.dt_insert,
    i.sender,
    i.receiver,
    i.create_date_time,

    NULLIF(trim((xpath('dtRp/text()', i.item_xml))[1]::varchar), '')::date AS dt_report,
    NULLIF(trim((xpath('dbCd/text()', i.item_xml))[1]::varchar), '') AS database_code_1c,
    NULLIF(trim((xpath('dbName/text()', i.item_xml))[1]::varchar), '') AS database_name_1c,
    NULLIF(trim((xpath('unitMd/text()', i.item_xml))[1]::varchar), '') AS unit_balance_mdm_code_1c,

    unnest(xpath('entry', i.item_xml)) AS entry_xml
FROM tmp_s1c_wc_items i
DISTRIBUTED BY (uuid);

ANALYZE tmp_s1c_wc_entries;
