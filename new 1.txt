Копировать код
CREATE TEMP TABLE euro_calc AS
SELECT
    *,
    CASE
        WHEN extract(day FROM dt_storage_end) < extract(day FROM dt_storage_start)
            THEN (extract(year FROM dt_storage_end) - extract(year FROM dt_storage_start)) * 12
               + (extract(month FROM dt_storage_end) - extract(month FROM dt_storage_start))
        ELSE  (extract(year FROM dt_storage_end) - extract(year FROM dt_storage_start)) * 12
               + (extract(month FROM dt_storage_end) - extract(month FROM dt_storage_start)) + 1
    END AS month_difference_2,
    CASE
        WHEN storage_cost_special_calculation_type_code='4'
            THEN ceil(storage_duration_total_calendar_days / 7::numeric)
        WHEN storage_cost_special_calculation_type_code='5'
            THEN ceil(storage_duration_total_calendar_days / 30::numeric)
    END AS special
FROM euro_rules;

CREATE TEMP TABLE for_1079_euro AS
SELECT *
FROM (
    SELECT
        *,
        SUM(sales_bundle_gross_weight) OVER (
            PARTITION BY warehouse_code, date_trunc('month', dt_storage_end)
        ) AS warehouse_weight_month_end,

        ROW_NUMBER() OVER (
            PARTITION BY rn
            ORDER BY order_euro_svh, kopos, kbetr, purchase_contract_code, position_line_item_code
        ) AS row_num
    FROM euro_calc
) t
WHERE row_num = 1;
