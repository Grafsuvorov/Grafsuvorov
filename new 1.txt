create temporary table pre_culc on commit drop as (

		select
		trht_3.transport_hub_name as warehouse_name, -- LE.001048
		base.dt_report, -- LE.001069
		for_1113.period_comparison_rule_code as period_comparison_rule_code, -- LE.001113
		for_1113.date_comparison::date as dt_comparison, -- LE.001125
		
		case
		when cotex.country_full_name = 'Корея, республика' then 'Южная Корея'
		when cotex.country_full_name = 'Тайвань (Китай)' then 'Тайвань'
		else cotex.country_full_name
		end as country_of_remote_warehouse_name, -- LE.001060
	
		case
		when t25a1.market_region1_name = 'Турция' then 'Западная Азия'
		when t25a1.market_region1_name = 'Средний Восток и Афр' then 'Ближний Восток'
		when t25a1.market_region1_name in ('Азия', 'Китай') then 'Восточная Азия'
		else t25a1.market_region1_name
		end as region_of_remote_warehouse_name, -- LE.001061
		
		coalesce(case when base.storage_cost_calculation_type_name in ('Логика Азия', 'Логика Европа', 'Логика СВХ') then counter.counterparty_short_name end, 'Не определено') as forwarder_name, -- LE.001041
		
		case
		when base.storage_cost_calculation_type_name in ('Логика РФ-контейнеры', 'Логика РФ-балк') then 'Терминал порта'
		when base.storage_cost_calculation_type_name in ('Логика СВХ') then 'Склад СВХ'
		else 'Не определено'
		end as type_warehouse_name, -- LE.001126
			
		case
		when base.storage_cost_calculation_type_name = 'Логика РФ-контейнеры' then 'Контейнеры'
		when base.storage_cost_calculation_type_name = 'Логика РФ-балк' then 'Балк'
		else 'Не определено'
		end as type_freight_name, -- LE.001127

		pas.plant_short_name || ' ' || pas.plant_full_name as receiving_plant_name, -- LE.001030
		-- pas.plant_full_name as receiving_plant_name, -- LE.001030
		-- ключи
		sum(base.sales_bundle_gross_weight) as sales_bundle_gross_weight, -- LE.001018
		sum(base.storage_payable_001_030_days_gross_weight + base.storage_payable_031_060_days_gross_weight + base.storage_payable_061_090_days_gross_weight + base.storage_payable_091_180_days_gross_weight +

		base.storage_payable_181_365_days_gross_weight + base.storage_payable_over_365_days_gross_weight) / sum(base.sales_bundle_gross_weight) * 100 as storage_payable_total_gross_weight_percentage, -- LE.001115
		sum(base.storage_calculated_cost_001_030_amount + base.storage_calculated_cost_031_060_amount + base.storage_calculated_cost_061_090_amount + base.storage_calculated_cost_091_180_amount +
		base.storage_calculated_cost_181_365_amount + base.storage_calculated_cost_over_365_amount) as storage_calculated_cost_total_amount, -- LE.001116

		sum(base.storage_payable_061_090_days_gross_weight) as storage_payable_061_090_days_gross_weight, -- LE.001097
