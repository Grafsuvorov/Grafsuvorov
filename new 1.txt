🟦 5. DIM WAREHOUSE_GEO
(вытаскиваем 1 строку на hub)
-----------------------------------------------------------
sql
Копировать код
DROP TABLE IF EXISTS dim_warehouse_geo;

CREATE TEMP TABLE dim_warehouse_geo
ON COMMIT DROP AS
SELECT
    hub.transport_hub_code,
    MIN(
        CASE WHEN ct.country_full_name = 'Россия' THEN 'R' ELSE 'E' END
    ) AS lang_code,
    MIN(
        CASE
            WHEN mr.market_region1_name = 'Турция' THEN 'Западная Азия'
            WHEN mr.market_region1_name = 'Средний Восток и Афр' THEN 'Ближний Восток'
            WHEN mr.market_region1_name IN ('Азия','Китай') THEN 'Восточная Азия'
            ELSE mr.market_region1_name
        END
    ) AS region_name,
    MIN(ct.country_full_name) AS country_name
FROM dict_dds.transport_hub hub
LEFT JOIN adrc adr
       ON adr.address_code = hub.address_code
LEFT JOIN dict_dds.country_texts ct
       ON ct.country_code = adr.country_code
      AND ct.language_code = 'R'
LEFT JOIN dict_dds.country c
       ON c.country_code = adr.country_code
LEFT JOIN dict_dds.market_region1_texts mr
       ON mr.market_region1_code = c.market_region1_code
      AND mr.language_code = 'R'
GROUP BY hub.transport_hub_code
DISTRIBUTED REPLICATED;

ANALYZE dim_warehouse_geo;

ON COMMIT DROP AS
SELECT
    deldp.delivery_code,
    MIN(pas.plant_short_name || ' ' || pas.plant_full_name) AS owner_name
FROM dds.delivery_document_position deldp
LEFT JOIN dict_dds.plant_and_subsidiary pas
       ON pas.plant_code = deldp.plant_producer_code
GROUP BY deldp.delivery_code
DISTRIBUTED REPLICATED;

CREATE TEMP TABLE dim_creditor
ON COMMIT DROP AS
SELECT
    counterparty.counterparty_code,
    MIN(addr.address_full_name) AS creditor_name
FROM dict_dds.counterparty counterparty
LEFT JOIN dict_dds.address addr
       ON addr.address_code = counterparty.address_code
      AND addr.international_display_format_code IS NULL
GROUP BY counterparty.counterparty_code
DISTRIBUTED REPLICATED;
