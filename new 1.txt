      AND htxt_wh.language_code      = wg.lang_code

       ON htxt_area.transport_hub_code = base.warehouse_storage_area_code
      AND htxt_area.language_code      = 'R'

       ON htxt_dep.transport_hub_code = base.transport_departure_hub_code
      AND htxt_dep.language_code      = 'R'

LEFT JOIN dim_hub_text htxt_dest
       ON htxt_dest.transport_hub_code = base.transport_destination_hub_code
      AND htxt_dest.language_code      = 'R'

LEFT JOIN dim_hub_text htxt_pd
       ON htxt_pd.transport_hub_code = base.port_of_discharge_code
      AND htxt_pd.language_code      = 'E'

LEFT JOIN dim_hub_text htxt_remote
       ON htxt_remote.transport_hub_code = base.remote_warehouse_code
      AND htxt_remote.language_code      = rg.lang_code

LEFT JOIN dict_dds.plant_and_subsidiary pas
       ON pas.plant_code = base.receiving_plant_code

LEFT JOIN dim_owner_plant op
       ON op.delivery_code = base.delivery_for_storage_calculation_code

LEFT JOIN dim_creditor ci
       ON ci.counterparty_code = base.creditor_in_shipment_instruction_code

LEFT JOIN dict_dds.counterparty counter
       ON counter.counterparty_code = base.forwarder_code

LEFT JOIN dict_dds.transportation_stage_texts trst
       ON trst.transportation_stage_code = base.transportation_stage_code
      AND trst.language_code = 'R'

LEFT JOIN dict_dds.transport_transfer_type_texts ipd
       ON ipd.transport_transfer_type_code = base.transport_type_code
      AND ipd.language_code = 'R';
