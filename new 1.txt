-- ============================================================================
-- NEW PIPELINE: Расчёт периодов и присоединение атрибутов
-- Ключевые идеи:
--   1) Сначала — ЛИШЬ ключи и даты (узкий слой), распараллеленный по ключам.
--   2) Интервалы через LEAD(), БЕЗ self-join (минимум движений и сортировок).
--   3) Материализуем "ядро" (keys+dates) в колоночную TEMP с нужным DISTRIBUTED BY.
--   4) Лишь затем — присоединяем "красоту": тексты/планы/методы (обычно REPLICATED).
-- ============================================================================

-- (по желанию) включить сессионную память под тяжелый шаг, если это принято у вас:
-- SET statement_mem = '2048MB';

-- 0) Базовый UNION ключевых "поворотных" дат из трех источников
WITH base_union AS (
    -- Берем только ключи + dt_from (поворотные начала интервалов)
    -- ВНИМАНИЕ: используйте реальные имена полей начала валидности (ниже предполагается dt_from)
    SELECT DISTINCT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        dt_from
    FROM dict_dds.fixed_asset_td

    UNION ALL
    SELECT DISTINCT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        dt_from
    FROM dict_dds.fixed_asset_depreciation_parameters_td

    UNION ALL
    SELECT DISTINCT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        dt_from
    FROM dict_dds.fixed_asset_non_liquid_technical_state_td
),

-- 1) Приводим к единому набору "поворотных" моментов по ключам
rng AS (
    SELECT
        unit_balance_code,
        asset_main_code,
        asset_sub_code,
        depreciation_area_code,
        -- дефолт, если где-то встречается null
        COALESCE(dt_from, DATE '1900-01-01')::date AS dt_valid_from,
        COALESCE(
            LEAD(dt_from) OVER (
                PARTITION BY unit_balance_code, asset_main_code, asset_sub_code, depreciation_area_code
                ORDER BY dt_from
            ) - INTERVAL '1 day',
            DATE '2299-12-31'
        )::date AS dt_valid_to
    FROM base_union
),

-- 2) Материализуем узкое "ядро" (ключи+даты) в TEMP-таблицу с правильной раздачей по сегментам
core_materialized AS (
    SELECT * FROM rng
)

-- Создаем TEMP таблицу (колонночная, сжатие zstd) и задаем DISTRIBUTED BY по ключам
-- ВАЖНО: Greenplum не позволяет задать распределение прямо из CTE,
-- поэтому создаем через CREATE TABLE AS с DISTRIBUTED BY:
-- (при повторных запусках можно DROP TABLE IF EXISTS)
;
DROP TABLE IF EXISTS tmp_fa_core;

CREATE TEMP TABLE tmp_fa_core
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
DISTRIBUTED BY (unit_balance_code, asset_main_code, asset_sub_code, depreciation_area_code)
AS
SELECT * FROM core_materialized;

ANALYZE tmp_fa_core;  -- статистика критична для хорошего плана

-- 3) Присоединяем "тяжёлые" таблицы по ключам и интервалам.
--    Ниже — пример, откорректируйте реальные имена колонок дат/валидности.
--    Старайтесь НЕ кастовать ключи (уберите ::text в joinах — выровняйте типы полей в схемах).
DROP TABLE IF EXISTS tmp_fa_joined;

CREATE TEMP TABLE tmp_fa_joined
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
DISTRIBUTED BY (unit_balance_code, asset_main_code, asset_sub_code, depreciation_area_code)
AS
SELECT
    c.unit_balance_code,
    c.asset_main_code,
    c.asset_sub_code,
    c.depreciation_area_code,
    c.dt_valid_from,
    c.dt_valid_to,

    -- Примеры числовых/флаговых атрибутов из тяжёлых таблиц:
    -- (Подставьте нужные поля из ваших источников)
    p.depreciation_method          AS dep_method_code,
    p.useful_life_months           AS useful_life_months,
    p.depreciation_rate            AS depreciation_rate,
    fa.capitalization_flag         AS capitalization_flag,
    nl.non_liquid_state_code       AS non_liquid_state_code

FROM tmp_fa_core c

LEFT JOIN dict_dds.fixed_asset_depreciation_parameters_td p
  ON  p.unit_balance_code      = c.unit_balance_code
  AND p.asset_main_code        = c.asset_main_code
  AND p.asset_sub_code         = c.asset_sub_code
  AND p.depreciation_area_code = c.depreciation_area_code
  -- интервал включенности: корректируйте согласно реальной логике валидности в источнике p
  AND c.dt_valid_from BETWEEN p.dt_from AND COALESCE(p.dt_to, DATE '2299-12-31')

LEFT JOIN dict_dds.fixed_asset_td fa
  ON  fa.unit_balance_code      = c.unit_balance_code
  AND fa.asset_main_code        = c.asset_main_code
  AND fa.asset_sub_code         = c.asset_sub_code
  AND fa.depreciation_area_code = c.depreciation_area_code
  AND c.dt_valid_from BETWEEN fa.dt_from AND COALESCE(fa.dt_to, DATE '2299-12-31')

LEFT JOIN dict_dds.fixed_asset_non_liquid_technical_state_td nl
  ON  nl.unit_balance_code      = c.unit_balance_code
  AND nl.asset_main_code        = c.asset_main_code
  AND nl.asset_sub_code         = c.asset_sub_code
  AND nl.depreciation_area_code = c.depreciation_area_code
  AND c.dt_valid_from BETWEEN nl.dt_from AND COALESCE(nl.dt_to, DATE '2299-12-31')
;

ANALYZE tmp_fa_joined;

-- 4) Присоединяем "красоту": тексты/планы/методы (обычно небольшие REPLICATED-таблицы).
--    Обратите внимание на фильтры селективности (например, language_code = 'R').
DROP TABLE IF EXISTS tmp_fa_enriched;

CREATE TEMP TABLE tmp_fa_enriched
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=5)
DISTRIBUTED BY (unit_balance_code, asset_main_code, asset_sub_code, depreciation_area_code)
AS
SELECT
    j.*,

    -- Примеры справочных текстов/наименований:
    dmt.method_name                  AS depreciation_method_name,
    vp.plan_name                     AS valuation_plan_name,
    cur.currency_name                AS currency_name

FROM tmp_fa_joined j

-- Пример: словарь методов амортизации
LEFT JOIN dict_dds.fixed_asset_depreciation_method_texts dmt
  ON dmt.depreciation_method = j.dep_method_code
 AND dmt.language_code = 'R'          -- селективный фильтр

-- Пример: словарь планов оценки
LEFT JOIN dict_dds.fixed_asset_valuation_plan_texts vp
  ON vp.plan_code = j.dep_method_code  -- пример, замените на реальный ключ
 AND vp.language_code = 'R'

-- Пример: справочник валют
LEFT JOIN dict_dds.currency_texts cur
  ON cur.currency_code = 'RUB'         -- пример, если нужна валюта из данных — замените
 AND cur.language_code = 'R'
;

ANALYZE tmp_fa_enriched;

LEAD() вместо self-join: в 1 проход даёт dt_valid_to — без двойных сортировок и огромных джоинов на себя.

DISTRIBUTED BY по ключам окна: WindowAgg распараллеливается; никакого singleton slice для окна.

Материализация “ядра”: мы сначала сужаем данные до ключей+дат, затем присоединяем тяжёлые таблицы интервалами (всё распараллелено общим ключом), и только потом добавляем лёгкие REPLICATED-словари.

Без кастов ::text в join: позволяет GP использовать нормальные hash-join и минимизировать перераздачи.

ANALYZE после ключевых шагов: оптимизатор выбирает адекватные планы.
