Ниже готовый Python-скрипт. Он:

  - берёт список таблиц по entity_id и schema
  - сверяет таблицу с backup-таблицей вида _backup_YYYY_MM_DD
  - исключает системные поля:
      - dttm_inserted
      - dttm_updated
      - job_name
      - deleted_flag
  - пишет прогресс в отдельную таблицу
  - пишет результаты в отдельную таблицу
  - делает commit после каждой таблицы

  Сначала создайте таблицы:

  create table if not exists tech_etl.entity_backup_check_progress (
      run_id text not null,
      entity_id integer not null,
      table_schema text not null,
      backup_date date not null,
      status text not null,
      total_tables integer not null default 0,
      processed_tables integer not null default 0,
      current_table text,
      started_at timestamp not null default now(),
      finished_at timestamp,
      error_message text
  )
  distributed randomly;

  create table if not exists tech_etl.entity_backup_check_results (
      run_id text not null,
      checked_at timestamp not null,
      backup_date date not null,
      entity_id integer not null,
      table_schema text not null,
      source_table text not null,
      backup_table text not null,
      status text not null,
      source_cnt bigint,
      backup_cnt bigint,
      diff_cnt bigint,
      error_message text
  )
  distributed randomly;

  Скрипт:

  #!/usr/bin/env python3
  import argparse
  import sys
  import uuid
  from datetime import date, datetime

  import psycopg2


  EXCLUDED_COLUMNS = {
      "dttm_inserted",
      "dttm_updated",
      "job_name",
      "deleted_flag",
  }


  def qident(name: str) -> str:
      return '"' + name.replace('"', '""') + '"'


  def fqtn(schema: str, table: str) -> str:
      return f"{qident(schema)}.{qident(table)}"


  def get_tables(conn, entity_id: int, table_schema: str):
      with conn.cursor() as cur:
          cur.execute(
              """
              select table_name
              from tech_etl.tables_meta
              where entity_id = %s
                and table_schema = %s
              order by table_name
              """,
              (entity_id, table_schema),
          )
          return [row[0] for row in cur.fetchall()]


  def table_exists(conn, table_schema: str, table_name: str) -> bool:
      with conn.cursor() as cur:
          cur.execute(
              """
              select 1
              from information_schema.tables
              where table_schema = %s
                and table_name = %s
              limit 1
              """,
              (table_schema, table_name),
          )
          return cur.fetchone() is not None


  def get_business_columns(conn, table_schema: str, table_name: str):
      with conn.cursor() as cur:
          cur.execute(
              """
              select column_name
              from information_schema.columns
              where table_schema = %s
                and table_name = %s
              order by ordinal_position
              """,
              (table_schema, table_name),
          )
          return [
              row[0]
              for row in cur.fetchall()
              if row[0] not in EXCLUDED_COLUMNS
          ]


  def get_count(conn, table_schema: str, table_name: str) -> int:
      sql = f"select count(*) from {fqtn(table_schema, table_name)}"
      with conn.cursor() as cur:
          cur.execute(sql)
          return cur.fetchone()[0]


  def get_diff_count(conn, table_schema: str, source_table: str, backup_table: str, columns):
      cols = ", ".join(qident(col) for col in columns)
      sql = f"""
          select count(*)
          from (
              (select {cols} from {fqtn(table_schema, source_table)}
               except all
               select {cols} from {fqtn(table_schema, backup_table)})
              union all
              (select {cols} from {fqtn(table_schema, backup_table)}
               except all
               select {cols} from {fqtn(table_schema, source_table)})
          ) q
      """
      with conn.cursor() as cur:
          cur.execute(sql)
          return cur.fetchone()[0]


  def upsert_progress(
      conn,
      run_id: str,
      entity_id: int,
      table_schema: str,
      backup_date: date,
      status: str,
      total_tables: int,
      processed_tables: int,
      current_table: str | None,
      error_message: str | None = None,
      finished: bool = False,
  ):
      with conn.cursor() as cur:
          cur.execute(
              "delete from tech_etl.entity_backup_check_progress where run_id = %s",
              (run_id,),
          )
          cur.execute(
              """
              insert into tech_etl.entity_backup_check_progress (
                  run_id,
                  entity_id,
                  table_schema,
                  backup_date,
                  status,
                  total_tables,
                  processed_tables,
                  current_table,
                  started_at,
                  finished_at,
                  error_message
              )
              values (
                  %s, %s, %s, %s, %s, %s, %s, %s,
                  coalesce(
                      (select min(started_at)
                       from tech_etl.entity_backup_check_progress
                       where run_id = %s),
                      now()
                  ),
                  %s,
                  %s
              )
              """,
              (
                  run_id,
                  entity_id,
                  table_schema,
                  backup_date,
                  status,
                  total_tables,
                  processed_tables,
                  current_table,
                  run_id,
                  datetime.now() if finished else None,
                  error_message,
              ),
          )


  def insert_result(
      conn,
      run_id: str,
      backup_date: date,
      entity_id: int,
      table_schema: str,
      source_table: str,
      backup_table: str,
      status: str,
      source_cnt=None,
      backup_cnt=None,
      diff_cnt=None,
      error_message=None,
  ):
      with conn.cursor() as cur:
          cur.execute(
              """
              insert into tech_etl.entity_backup_check_results (
                  run_id,
                  checked_at,
                  backup_date,
                  entity_id,
                  table_schema,
                  source_table,
                  backup_table,
                  status,
                  source_cnt,
                  backup_cnt,
                  diff_cnt,
                  error_message
              )
              values (%s, now(), %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
              """,
              (
                  run_id,
                  backup_date,
                  entity_id,
                  table_schema,
                  source_table,
                  backup_table,
                  status,
                  source_cnt,
                  backup_cnt,
                  diff_cnt,
                  error_message,
              ),
          )


  def main():
      parser = argparse.ArgumentParser()
      parser.add_argument("--host", required=True)
      parser.add_argument("--port", type=int, default=5432)
      parser.add_argument("--dbname", required=True)
      parser.add_argument("--user", required=True)
      parser.add_argument("--password", required=True)
      parser.add_argument("--entity-id", type=int, required=True)
      parser.add_argument("--table-schema", required=True)
      parser.add_argument("--backup-date", default=str(date.today()))
      args = parser.parse_args()

      backup_date = date.fromisoformat(args.backup_date)
      backup_suffix = backup_date.strftime("%Y_%m_%d")
      run_id = str(uuid.uuid4())

      conn = psycopg2.connect(
          host=args.host,
          port=args.port,
          dbname=args.dbname,
          user=args.user,
          password=args.password,
      )
      conn.autocommit = False

      try:
          tables = get_tables(conn, args.entity_id, args.table_schema)
          total = len(tables)

          upsert_progress(
              conn,
              run_id,
              args.entity_id,
              args.table_schema,
              backup_date,
              "running",
              total,
              0,
              None,
          )
          conn.commit()

          for idx, table_name in enumerate(tables, start=1):
              backup_table = f"{table_name}_backup_{backup_suffix}"
              source_name = f"{args.table_schema}.{table_name}"
              backup_name = f"{args.table_schema}.{backup_table}"

              print(f"[{idx}/{total}] checking {source_name}")
              upsert_progress(
                  conn,
                  run_id,
                  args.entity_id,
                  args.table_schema,
                  backup_date,
                  "running",
                  total,
                  idx - 1,
                  source_name,
              )
              conn.commit()

              try:
                  if not table_exists(conn, args.table_schema, backup_table):
                      insert_result(
                          conn,
                          run_id,
                          backup_date,
                          args.entity_id,
                          args.table_schema,
                          source_name,
                          backup_name,
                          "backup_not_found",
                      )
                      conn.commit()
                      continue

                  columns = get_business_columns(conn, args.table_schema, table_name)
                  if not columns:
                      insert_result(
                          conn,
                          run_id,
                          backup_date,
                          args.entity_id,
                          args.table_schema,
                          source_name,
                          backup_name,
                          "no_business_columns",
                      )
                      conn.commit()
                      continue

                  source_cnt = get_count(conn, args.table_schema, table_name)
                  backup_cnt = get_count(conn, args.table_schema, backup_table)
                  diff_cnt = get_diff_count(
                      conn,
                      args.table_schema,
                      table_name,
                      backup_table,
                      columns,
                  )

                  status = "ok" if source_cnt == backup_cnt and diff_cnt == 0 else "mismatch"

                  insert_result(
                      conn,
                      run_id,
                      backup_date,
                      args.entity_id,
                      args.table_schema,
                      source_name,
                      backup_name,
                      status,
                      source_cnt,
                      backup_cnt,
                      diff_cnt,
                  )
                  conn.commit()

              except Exception as exc:
                  conn.rollback()
                  insert_result(
                      conn,
                      run_id,
                      backup_date,
                      args.entity_id,
                      args.table_schema,
                      source_name,
                      backup_name,
                      "error",
                      error_message=str(exc)[:4000],
                  )
                  conn.commit()

              upsert_progress(
                  conn,
                  run_id,
                  args.entity_id,
                  args.table_schema,
                  backup_date,
                  "running",
                  total,
                  idx,
                  source_name,
              )
              conn.commit()

          upsert_progress(
              conn,
              run_id,
              args.entity_id,
              args.table_schema,
              backup_date,
              "finished",
              total,
              total,
              None,
              finished=True,
          )
          conn.commit()

          print(f"done, run_id={run_id}")
          return 0

      except Exception as exc:
          conn.rollback()
          try:
              upsert_progress(
                  conn,
                  run_id,
                  args.entity_id,
                  args.table_schema,
                  backup_date,
                  "failed",
                  0,
                  0,
                  None,
                  error_message=str(exc)[:4000],
                  finished=True,
              )
              conn.commit()
          except Exception:
              conn.rollback()
          print(f"failed, run_id={run_id}, error={exc}", file=sys.stderr)
          return 1

      finally:
          conn.close()


  if __name__ == "__main__":
      raise SystemExit(main())
