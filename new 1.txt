-- 1) CKMLPP: свернём в один ряд на период и нормализуем ключ
CREATE TEMP TABLE pp0
AS
SELECT
    (kalnr)::text           AS kalnr_text,
    bdatj::text             AS bdatj,
    poper::text             AS poper,
    to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD') AS dt_valid_from,
    SUM(umkumo) AS umkumo,          -- потоки → SUM
    SUM(abkumo) AS abkumo,          -- потоки → SUM
    SUM(zukumo) AS zukumo,          -- потоки → SUM
    MAX(lbkum)  AS lbkum            -- срез → MAX (если это конец периода)
FROM ods.ckmlpp_ral
WHERE untper = '000'
GROUP BY 1,2,3,4
DISTRIBUTED BY (kalnr_text);
ANALYZE pp0;

-- Индекс для джоинов по периоду
CREATE INDEX ON pp0 (kalnr_text, bdatj, poper);
CREATE INDEX ON pp0 (kalnr_text, dt_valid_from);

-- 2) CKMLCR текущий период
CREATE TEMP TABLE cr_curr
AS
SELECT
    (kalnr)::text AS kalnr_text,
    bdatj::text   AS bdatj,
    poper::text   AS poper,
    MAX(stprs)    AS stprs,
    MAX(peinh)    AS peinh,
    MAX(pvprs)    AS pvprs,
    MAX(abprd_o)  AS abprd_o,
    MAX(abprd_mo) AS abprd_mo,
    MAX(vprsv)    AS vprsv,
    MAX(salk3)    AS salk3
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY 1,2,3
DISTRIBUTED BY (kalnr_text);
ANALYZE cr_curr;

CREATE INDEX ON cr_curr (kalnr_text, bdatj, poper);

-- 3) CKMLCR следующий месяц (первое число)
CREATE TEMP TABLE cr_next
AS
SELECT
    (kalnr)::text AS kalnr_text,
    (to_date(bdatj||substr(poper,2,2)||'01','YYYYMMDD') + INTERVAL '1 month')::date AS dt_posting,
    MAX(salk3) AS salk3
FROM ods.ckmlcr_ral
WHERE untper = '000' AND curtp = '10'
GROUP BY 1,2
DISTRIBUTED BY (kalnr_text);
ANALYZE cr_next;

CREATE INDEX ON cr_next (kalnr_text, dt_posting);

-- 4) MLCD агрегаты
CREATE TEMP TABLE mlcd_agg
AS
SELECT
    (kalnr)::text AS kalnr_text,
    bdatj::text   AS bdatj,
    poper::text   AS poper,
    SUM(salk3)  AS salk3,
    SUM(estprd) AS estprd,
    SUM(mstprd) AS mstprd
FROM ods.mlcd_ral
WHERE curtp = '10'
  AND categ IN ('ZU','VP','PC')
GROUP BY 1,2,3
DISTRIBUTED BY (kalnr_text);
ANALYZE mlcd_agg;

CREATE INDEX ON mlcd_agg (kalnr_text, bdatj, poper);

-- 5) Границы действия по УНИКАЛЬНЫМ датам
WITH uniq_pp AS (
  SELECT DISTINCT kalnr_text, dt_valid_from FROM pp0
),
ranges AS (
  SELECT
    kalnr_text,
    dt_valid_from,
    COALESCE(
      LEAD(dt_valid_from) OVER (PARTITION BY kalnr_text ORDER BY dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
  FROM uniq_pp
)
SELECT
    r.kalnr_text                AS calculation_code,
    r.dt_valid_from,
    r.dt_valid_to,
    cr_curr.vprsv               AS price_valuation_type_code,
    cr_curr.stprs               AS standard_price_amount,
    cr_curr.peinh               AS price_unit_code,
    cr_curr.pvprs               AS moving_average_price_amount,

    /* rule3 */
    CASE WHEN (pp0.umkumo + pp0.abkumo + pp0.zukumo) = 0 THEN 0
         ELSE TRUNC(
                CAST(ROUND(
                      (cr_curr.stprs / NULLIF(cr_curr.peinh,0)) * (pp0.umkumo + pp0.abkumo)
                    + cr_curr.abprd_o + cr_curr.abprd_mo
                    + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                , 2) AS NUMERIC(40,20))
                / NULLIF((pp0.umkumo + pp0.abkumo + pp0.zukumo),0), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE WHEN (pp0.abkumo + pp0.umkumo + pp0.zukumo) <> 0
         THEN TRUNC(
                CAST(
                    cr_next.salk3 + COALESCE(mlcd_agg.salk3 + mlcd_agg.estprd + mlcd_agg.mstprd, 0)
                AS NUMERIC(40,20))
                / NULLIF((pp0.abkumo + pp0.umkumo + pp0.zukumo),0), 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE WHEN pp0.lbkum <> 0
         THEN TRUNC(CAST(cr_curr.salk3 AS NUMERIC(40,20)) / NULLIF(pp0.lbkum,0), 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    ckmlhd.price_rule_code
FROM ranges r
JOIN pp0
  ON pp0.kalnr_text   = r.kalnr_text
 AND pp0.dt_valid_from = r.dt_valid_from
LEFT JOIN cr_curr
  ON cr_curr.kalnr_text = pp0.kalnr_text
 AND cr_curr.bdatj      = pp0.bdatj
 AND cr_curr.poper      = pp0.poper
LEFT JOIN cr_next
  ON cr_next.kalnr_text = pp0.kalnr_text
 AND cr_next.dt_posting = pp0.dt_valid_from
LEFT JOIN mlcd_agg
  ON mlcd_agg.kalnr_text = pp0.kalnr_text
 AND mlcd_agg.bdatj      = pp0.bdatj
 AND mlcd_agg.poper      = pp0.poper
LEFT JOIN dds.material_ledger_header ckmlhd
  ON ckmlhd.calculation_code = pp0.kalnr_text
;
