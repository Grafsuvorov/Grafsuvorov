	a.dt,
	a.contract_number,
	c.relation_type_code
from
	dm.account_debt_for_working_capital as a
    left join dict_dds.counterparty as c on
	    a.counterparty_code = c.counterparty_code
where
	external_contract_source_table_name like 'sales%'
	and responsibility_center_code is null
group by
	a.dt,
	a.contract_number,
