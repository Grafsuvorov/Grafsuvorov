    sdgadg.SD_000041,
    sdgadg.SD_000042,
    sdgadg.SD_000482,

    sdgadg.SD_000067,

    sdgadg.SD_000048,
    sdgadg.SD_000049,
    sdgadg.SD_000024,

        ssms.port_of_discharge_in_foreign_port_code AS SD_000054,
        ssms.port_of_loading_code                 AS SD_000649,

        ssms.bill_of_lading_group_code            AS SD_000040,

-- 1.5 for_1055 — дата окончания бесплатного хранения (LE.001055)
CREATE TEMP TABLE for_1055 ON COMMIT DROP AS
SELECT
    vbss.delivery_code AS vbeln,
    MIN(vbsk.dt_end_of_free_storage) AS dt_end_of_free_storage
FROM dds.logistics_document_and_delivery_relationship vbss
LEFT JOIN dds.release vbsk
       ON vbsk.release_code = vbss.logistics_document_code
      AND vbsk.release_type_code = 'X'
LEFT JOIN dict_dds.settings_and_parameters_sap frel_kod
       ON vbsk.release_template_code = frel_kod.range_low_value
WHERE frel_kod.abap_program_code = '/RUSAL/SD2921M'
  AND frel_kod.parameter_code    = 'FREL_KOD'
  AND frel_kod.range_sign_code   = 'I'
  AND frel_kod.range_option_code = 'EQ'
  AND vbsk.release_code IS NOT NULL
  AND vbsk.dt_end_of_free_storage IS NOT NULL
GROUP BY vbss.delivery_code
DISTRIBUTED REPLICATED;

-- 1.7 for_1047_1 — склад хранения для РФ логики (оконка)
CREATE TEMP TABLE for_1047_1 ON COMMIT DROP AS
SELECT
    sales_bundle_code,
    transportation_stage_code,
    transport_route_departure_hub_code,
    transport_route_destination_hub_code
FROM (
    SELECT
        sales_bundle_code,
        transportation_stage_code,
        transport_route_departure_hub_code,
        transport_route_destination_hub_code,
        ROW_NUMBER() OVER (
            PARTITION BY sales_bundle_code
            ORDER BY transportation_stage_code DESC, delivery_code DESC
        ) AS rn
    FROM dds.sales_bundle_and_delivery_relationship
    WHERE is_deleted_code IS NULL
      AND transportation_stage_code IN ('10','15','16','25')
) x
WHERE x.rn = 1
DISTRIBUTED BY (sales_bundle_code);

CREATE TEMP TABLE base ON COMMIT DROP AS
WITH forwapd_1 AS (      -- приоритет иностранных складов/терминалов
    SELECT
        transport_hub_code,
        MIN(market_region1_code) AS market_region1_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE market_region1_code IN ('03','04')
      AND (is_terminal_code IS NOT NULL
           OR is_foreign_warehouse_code IS NOT NULL
           OR is_temporary_warehouse_code IS NOT NULL)
    GROUP BY transport_hub_code
),
forwapd_2 AS (           -- российский порт
    SELECT transport_hub_code
    FROM dict_dds.foreign_warehouse_priority_definition
    WHERE is_russian_port_code IS NOT NULL
),
stage_eu AS (
    SELECT range_low_value AS stage_code
    FROM dict_dds.settings_and_parameters_sap
    WHERE abap_program_code = '/RUSAL/SD2973M_2'
      AND parameter_code IN ('STAGE_EU')
      AND range_low_value IS NOT NULL
    GROUP BY range_low_value
),
stage_asia AS (
    SELECT range_low_value AS stage_code
    FROM dict_dds.settings_and_parameters_sap
    WHERE abap_program_code = '/RUSAL/SD2973M_2'
      AND parameter_code IN ('STAGE_A')
      AND range_low_value IS NOT NULL
    GROUP BY range_low_value
)
SELECT
    t4.sales_bundle_code                                     AS sales_bundle_code,           -- LE.001000
    lake.SD_000003                                           AS plant_producer_delivery_code,-- LE.001001
    lake.SD_000110                                           AS plant_producer_delivery_position_code, -- LE.001002
    lake.SD_000001                                           AS initial_delivery_code,       -- LE.001003
    lake.SD_000143                                           AS material_code,               -- LE.001012
    lake.SD_000010                                           AS dt_shipment,                 -- LE.001020
    lake.SD_000013                                           AS railcar_code,                -- LE.001032
    lake.SD_000014                                           AS transport_bill_code,         -- LE.001033
    lake.SD_000151                                           AS uni,                         -- LE.001034
    maspe.shape_code                                         AS material_shape_code,         -- LE.001036
    lake.SD_000028                                           AS transport_type_code,         -- LE.001050
    mater.etsng_code                                         AS etsng_code,                  -- LE.001066
    sabu.sales_bundle_net_weight / 1000::numeric             AS sales_bundle_net_weight,     -- LE.001018
    sabu.sales_bundle_gross_weight / 1000::numeric           AS sales_bundle_gross_weight,   -- LE.001019
    lake.SD_000004                                           AS batch_code,                  -- LE.001015
    t4.dt_transportation_stage_start_r                       AS dt_arrival,                  -- LE.001039
    dedopo_1.plant_producer_code                             AS receiving_plant_code,        -- LE.001029
    lake.delivery_number_sales                               AS sales_delivery_code,         -- LE.001005
    CASE
        WHEN f1.transport_hub_code IS NOT NULL THEN t4.delivery_code_le_p
    END                                                      AS transportation_inbound_delivery_code,  -- LE.001007
    CASE
        WHEN f1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN sabadr_03.delivery_code
        WHEN f1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN sabadr_04.delivery_code
        WHEN f2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
    END                                                      AS delivery_for_storage_calculation_code, -- LE.001016
    CASE
        WHEN f1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN NULL
        WHEN f1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN NULL
        WHEN f2.transport_hub_code IS NOT NULL THEN '1'
    END                                                      AS wwgsg_01,
    CASE
        WHEN f1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN NULL
        WHEN f1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN NULL
        WHEN f2.transport_hub_code IS NOT NULL THEN vbsk.dt_bill_of_lading
    END                                                      AS wwgsg_01_date,
    t4.transportation_inbound_delivery_position_code         AS transportation_inbound_delivery_position_code, -- LE.001008
    CASE
        WHEN f1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN t4.transportation_inbound_delivery_position_code
        WHEN f1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN t4.transportation_inbound_delivery_position_code
        ELSE '10'
    END                                                      AS delivery_position_for_storage_calculation_code, -- LE.001017
    COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)     AS forwarder_code,              -- LE.001040

    -- Логика региона хранения (LE.001064)
    CASE
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 = 'TL04'
            THEN 'Логика РФ-контейнеры'
        WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 <> 'TL04'
            THEN 'Логика РФ-балк'
        WHEN likp.delivery_type_code = 'ZLE'
             AND asia_log.range_low_value IS NOT NULL
             AND asia.counterparty_code IS NULL
            THEN 'Логика Азия'
        WHEN likp.delivery_type_code = 'ZLE'
             AND euro_1.counterparty_code IS NOT NULL
            THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE'
             AND euro_2.counterparty_code IS NOT NULL
             AND euro_log.range_low_value IS NOT NULL
            THEN 'Логика Европа'
        WHEN likp.delivery_type_code = 'ZLE'
             AND svh.counterparty_code IS NOT NULL
            THEN 'Логика СВХ'
        ELSE 'Логика не определена'
    END                                                      AS storage_cost_calculation_type_name, -- LE.001064

    sbudr_base.transport_route_departure_hub_code            AS transport_departure_hub_code,   -- LE.001021
    sbudr_base.transport_route_destination_hub_code          AS transport_destination_hub_code, -- LE.001023
    sbudr_base.is_final_transportation_stage_code            AS SD_000902,
    sbudr_base.transportation_stage_code                     AS SD_000903,
    for_LE_001035.cnt                                        AS SD_000905,
    lake.SD_000482,
    lake.SD_000041,
    lake.SD_000042,
    lake.SD_000047,
    lake.SD_000048,
    lake.SD_000054,
    lake.SD_000049,
    lake.SD_000067,
    lake.SD_000649,
    lake.SD_000044,
    lake.SD_000040,
    t4.delivery_code_le_r                                    AS SD_000516,
    t4.transportation_outbound_delivery_position_code        AS SD_000897,
    likp.weight_net                                          AS ntgew,
    likp.weight_net_with_wirerod                             AS btgew,
    t4.knote                                                 AS SD_000420,
    likp.dt_loaded                                           AS likp_lddat,
    CASE
        WHEN f1.market_region1_code = '03' AND sabadr_03.delivery_code IS NOT NULL THEN NULL
        WHEN f1.market_region1_code = '04' AND sabadr_04.delivery_code IS NOT NULL THEN NULL
        ELSE lake.SD_000024
    END                                                      AS dt_arrival_to_russian_port, -- LE.001059

    t4.dt_transportation_stage_start_p,
    COALESCE(t4.dt_transportation_stage_start_r, current_date) AS dt_transportation_stage_start_r,
    CONCAT(t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code) AS id_for_unique,

    COALESCE(sabadr_03.dt_transportation_stage_start,
             sabadr_04.dt_transportation_stage_start)        AS lddat_4798,
    COALESCE(sabadr_03.delivery_code, sabadr_04.delivery_code) AS vbeln_4798,
    LAG(COALESCE(sabadr_03.dt_transportation_stage_start,
                 sabadr_04.dt_transportation_stage_start))
        OVER (PARTITION BY t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code
              ORDER BY COALESCE(sabadr_03.dt_transportation_stage_start,
                                sabadr_04.dt_transportation_stage_start)) AS prev_lddat_4798,
    LEAD(COALESCE(sabadr_03.dt_transportation_stage_start,
                  sabadr_04.dt_transportation_stage_start))
        OVER (PARTITION BY t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code
              ORDER BY COALESCE(sabadr_03.dt_transportation_stage_start,
                                sabadr_04.dt_transportation_stage_start)) AS next_lddat_4798,

    for_1055.dt_end_of_free_storage                           AS dt_storage_end_in_release -- LE.001055

FROM dm_calc.sales_bundle_transport_hub_turnover_sdt0004 t4
LEFT JOIN lake
       ON t4.sales_bundle_code = lake.sales_bundle_code
LEFT JOIN dds.bill_of_lading vbsk
       ON vbsk.bill_of_lading_code      = lake.SD_000040
      AND vbsk.bill_of_lading_type_code = 'Y'
LEFT JOIN for_1055
       ON for_1055.vbeln = lake.delivery_number_sales
LEFT JOIN dict_dds.material mater
       ON mater.material_code = lake.SD_000143
LEFT JOIN dds.sales_bundle sabu
       ON sabu.sales_bundle_code = t4.sales_bundle_code
LEFT JOIN forwapd_1 f1
       ON f1.transport_hub_code = t4.knote
LEFT JOIN forwapd_2 f2
       ON f2.transport_hub_code = t4.knote

LEFT JOIN dds.sales_bundle_and_delivery_relationship sabadr_03
       ON sabadr_03.sales_bundle_code = t4.sales_bundle_code
      AND sabadr_03.transportation_stage_code IN (SELECT stage_code FROM stage_eu)
      AND sabadr_03.transport_route_destination_hub_code = t4.knote
      AND f1.market_region1_code = '03'
      AND sabadr_03.is_deleted_code IS NULL
      AND sabadr_03.dt_transportation_stage_start
            BETWEEN COALESCE(t4.dt_transportation_stage_start_p, current_date)
                AND COALESCE(t4.dt_transportation_stage_start_r, current_date)

LEFT JOIN dds.sales_bundle_and_delivery_relationship sabadr_04
       ON sabadr_04.sales_bundle_code = t4.sales_bundle_code
      AND sabadr_04.transportation_stage_code IN (SELECT stage_code FROM stage_asia)
      AND sabadr_04.transport_route_destination_hub_code = t4.knote
      AND f1.market_region1_code = '04'
      AND sabadr_04.is_deleted_code IS NULL
      AND sabadr_04.dt_transportation_stage_start
            BETWEEN COALESCE(t4.dt_transportation_stage_start_p, current_date)
                AND COALESCE(t4.dt_transportation_stage_start_r, current_date)

LEFT JOIN dds.sales_bundle_and_delivery_relationship sbudr_base
       ON sbudr_base.sales_bundle_code = t4.sales_bundle_code
      AND sbudr_base.delivery_code = t4.delivery_code_le_p
      AND sbudr_base.delivery_position_code = t4.transportation_inbound_delivery_position_code
      AND sbudr_base.is_deleted_code IS NULL

LEFT JOIN dds.delivery_document_position dedopo_1
       ON dedopo_1.delivery_code = lake.delivery_number_sales

LEFT JOIN dds.sales_document_counterparty_role ascr_1
       ON ascr_1.sales_document_code = t4.delivery_code_le_p
      AND ascr_1.counterparty_role_code IN ('XR')

LEFT JOIN dds.sales_document_counterparty_role ascr_2
       ON ascr_2.sales_document_code = t4.delivery_code_le_p
      AND ascr_2.counterparty_role_code IN ('ZU')

LEFT JOIN dds.delivery_document_header likp
       ON likp.delivery_code = CASE
                                   WHEN f1.market_region1_code = '03'
                                        AND sabadr_03.delivery_code IS NOT NULL
                                     THEN sabadr_03.delivery_code
                                   WHEN f1.market_region1_code = '04'
                                        AND sabadr_04.delivery_code IS NOT NULL
                                     THEN sabadr_04.delivery_code
                                   WHEN f2.transport_hub_code IS NOT NULL
                                     THEN lake.delivery_number_sales
                               END

LEFT JOIN asia
       ON asia.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN euro_1
       ON euro_1.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN euro_2
       ON euro_2.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)
LEFT JOIN svh
       ON svh.counterparty_code = COALESCE(ascr_1.supplier_code, ascr_2.supplier_code)

    transportation_stage_code,

    storage_duration_total_calendar_days,

    weight_for_1090_razdel,
    remote_warehouse_code,
    dt_transportation_stage_start_p,
    dt_transportation_stage_start_r,
    rn
)
SELECT
    b2.sales_bundle_code,
    b2.plant_producer_delivery_code,
    b2.plant_producer_delivery_position_code,
    b2.initial_delivery_code,
    b2.material_code,
    b2.dt_shipment,
    b2.railcar_code,
    b2.transport_bill_code,
    b2.uni,
    b2.material_shape_code,
    b2.dt_arrival_to_russian_port,
    b2.dt_storage_end_in_release,
    b2.transport_type_code,
    b2.etsng_code,
    d.dt_balance                         AS dt_report,  -- LE.001069
    b2.sales_bundle_net_weight,
    b2.sales_bundle_gross_weight,
    b2.batch_code,
    b2.dt_arrival,
    b2.receiving_plant_code,
    b2.sales_delivery_code,
    b2.transportation_inbound_delivery_code,
    b2.delivery_for_storage_calculation_code,
    b2.transportation_inbound_delivery_position_code,
    b2.delivery_position_for_storage_calculation_code,
    b2.forwarder_code,
    b2.storage_cost_calculation_type_name,
    b2.transport_departure_hub_code,
    b2.transport_destination_hub_code,
    b2.transportation_outbound_delivery_code,
    b2.transportation_outbound_delivery_position_code,
    b2.is_final_transportation_stage_code,
    b2.transportation_stage_code,
    b2.storage_calculation_bundle_quantity,
    b2.warehouse_code,
    b2.is_stored_in_russian_port_code,
    b2.delivery_in_final_release_code,
    b2.dt_final_release,
    b2.bill_of_lading_code,
    b2.bill_of_lading_number,
    b2.dt_bill_of_lading,
    b2.incoterms_code,
    b2.storage_cost_special_calculation_type_code,

    b2.unit_balance_code,
    b2.creditor_in_shipment_instruction_code,
    b2.dt_storage_start,
    b2.dt_shipped_from_warehouse,

    -- dt_storage_end (LE.001072)
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND b2.SD_000042 IS NOT NULL
             AND b2.SD_000042 <= d.dt_balance
          THEN b2.SD_000042
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
          THEN d.dt_balance
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
             AND b2.sbudr_2_dt_transportation_stage_start IS NOT NULL
             AND b2.sbudr_2_dt_transportation_stage_start <= d.dt_balance
          THEN b2.sbudr_2_dt_transportation_stage_start
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
          THEN d.dt_balance
    END AS dt_storage_end,

    -- storage_duration_total_calendar_days (LE.001086)
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND b2.SD_000042 IS NOT NULL
             AND b2.SD_000042 <= d.dt_balance
          THEN b2.SD_000042 - b2.dt_arrival_to_russian_port + 1
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
          THEN d.dt_balance - b2.dt_arrival_to_russian_port + 1
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
             AND b2.sbudr_2_dt_transportation_stage_start IS NOT NULL
             AND b2.sbudr_2_dt_transportation_stage_start <= d.dt_balance
          THEN b2.sbudr_2_dt_transportation_stage_start - b2.likp_lddat + 1
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
          THEN d.dt_balance - b2.likp_lddat + 1
    END AS storage_duration_total_calendar_days,

    b2.shipment_instruction_number,

    -- dt_for_price_search_in_purchase_contract (LE.001076)
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND b2.for_1076_1_lifnr IS NOT NULL
          THEN b2.dt_arrival_to_russian_port
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND b2.SD_000042 IS NOT NULL
             AND b2.SD_000042 <= d.dt_balance
          THEN b2.SD_000042
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
          THEN d.dt_balance
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
             AND b2.sbudr_2_dt_transportation_stage_start IS NOT NULL
             AND b2.sbudr_2_dt_transportation_stage_start <= d.dt_balance
          THEN b2.sbudr_2_dt_transportation_stage_start
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
          THEN d.dt_balance
        WHEN b2.storage_cost_calculation_type_name = 'Логика СВХ'
             AND b2.for_1076_2_counterparty_code IS NOT NULL
          THEN b2.likp_lddat
        WHEN b2.storage_cost_calculation_type_name = 'Логика СВХ'
             AND b2.sbudr_2_dt_transportation_stage_start IS NOT NULL
             AND b2.sbudr_2_dt_transportation_stage_start <= d.dt_balance
          THEN b2.sbudr_2_dt_transportation_stage_start
        WHEN b2.storage_cost_calculation_type_name = 'Логика СВХ'
          THEN d.dt_balance
    END AS dt_for_price_search_in_purchase_contract,

    -- port_of_discharge_code (LE.001044)
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
          THEN b2.SD_000649
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
             AND b2.SD_000049 < d.dt_balance
          THEN b2.SD_000054
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
          THEN b2.SD_000044
    END AS port_of_discharge_code,

    b2.service_number,
    b2.weight_for_1090_umno,
    b2.weight_for_1090_razdel,
    b2.remote_warehouse_code,
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
          THEN b2.dt_transportation_stage_start_p
    END AS dt_transportation_stage_start_p,
    CASE
        WHEN b2.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
          THEN b2.dt_arrival
    END AS dt_transportation_stage_start_r,
    b2.id_for_unique || d.dt_balance::text AS rn
FROM base_2 b2
JOIN date_for_vitr d
  ON d.dt_balance BETWEEN b2.prev_lddat_4798 AND b2.next_lddat_4798
WHERE (b2.wwgsg_01 IS NULL
       OR (b2.wwgsg_01 IS NOT NULL
           AND COALESCE(b2.wwgsg_01_date, '9999-12-31'::date) >= d.dt_balance))
  AND b2.rn = 1;
