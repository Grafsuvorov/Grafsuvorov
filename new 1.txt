CREATE TABLE IF NOT EXISTS dict_dds.calendar_periods (
    dt date,
    period_comparison_rule_code text,
    dt_comparison date
)
DISTRIBUTED BY (dt);

TRUNCATE TABLE dict_dds.calendar_periods;

INSERT INTO dict_dds.calendar_periods (dt, period_comparison_rule_code, dt_comparison)

-- 1. Предыдущий день
SELECT
    dt,
    '1' AS period_comparison_rule_code,
    (dt - INTERVAL '1 day')::date AS dt_comparison
FROM dict_dds.calendar
WHERE calendar_year BETWEEN 2000 AND 2050

UNION ALL

-- 2. Предыдущая неделя (понедельник → понедельник)
SELECT
    dt,
    '2',
    (dt - INTERVAL '1 week')::date
FROM dict_dds.calendar
WHERE day_of_week = 1
  AND calendar_year BETWEEN 2000 AND 2050

UNION ALL

-- 3. Предыдущий месяц (последний день месяца)
SELECT
    dt,
    '3',
    (date_trunc('month', dt)::date - 1)
FROM dict_dds.calendar
WHERE calendar_day = days_in_month
  AND calendar_year BETWEEN 2000 AND 2050

UNION ALL

-- 4. Предыдущий квартал (последний день квартала)
SELECT
    dt,
    '4',
    (dt + INTERVAL '1 day' - INTERVAL '3 month' - INTERVAL '1 day')::date
FROM dict_dds.calendar
WHERE date_part('quarter', dt) <> date_part('quarter', dt + INTERVAL '1 day')
  AND calendar_year BETWEEN 2000 AND 2050

UNION ALL

-- 5. Предыдущий год (31 декабря)
SELECT
    dt,
    '5',
    (dt - INTERVAL '1 year')::date
FROM dict_dds.calendar
WHERE calendar_month = 12
  AND calendar_day = 31
  AND calendar_year BETWEEN 2000 AND 2050;

CREATE TEMP TABLE pre_agg
AS
SELECT
    trht_3.transport_hub_name AS warehouse_name,
    base.dt_report,

    /* измерения */
    CASE
        WHEN cotex.country_full_name = 'Корея, республика' THEN 'Южная Корея'
        WHEN cotex.country_full_name = 'Тайвань (Китай)' THEN 'Тайвань'
        ELSE cotex.country_full_name
    END AS country_of_remote_warehouse_name,

    CASE
        WHEN t25a1.market_region1_name = 'Турция' THEN 'Западная Азия'
        WHEN t25a1.market_region1_name = 'Средний Восток и Афр' THEN 'Ближний Восток'
        WHEN t25a1.market_region1_name IN ('Азия','Китай') THEN 'Восточная Азия'
        ELSE t25a1.market_region1_name
    END AS region_of_remote_warehouse_name,

    COALESCE(
        CASE
            WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
            THEN counter.counterparty_short_name
        END,
        'Не определено'
    ) AS forwarder_name,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк') THEN 'Терминал порта'
        WHEN base.storage_cost_calculation_type_name = 'Логика СВХ' THEN 'Склад СВХ'
        ELSE 'Не определено'
    END AS type_warehouse_name,

    CASE
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-контейнеры' THEN 'Контейнеры'
        WHEN base.storage_cost_calculation_type_name = 'Логика РФ-балк' THEN 'Балк'
        ELSE 'Не определено'
    END AS type_freight_name,

    pas.plant_short_name || ' ' || pas.plant_full_name AS receiving_plant_name,

    /* метрики */
    SUM(base.sales_bundle_gross_weight) AS sales_bundle_gross_weight,

    SUM(
        base.storage_payable_001_030_days_gross_weight +
        base.storage_payable_031_060_days_gross_weight +
        base.storage_payable_061_090_days_gross_weight +
        base.storage_payable_091_180_days_gross_weight +
        base.storage_payable_181_365_days_gross_weight +
        base.storage_payable_over_365_days_gross_weight
    ) AS storage_payable_total_gross_weight,

    SUM(
        base.storage_calculated_cost_001_030_amount +
        base.storage_calculated_cost_031_060_amount +
        base.storage_calculated_cost_061_090_amount +
        base.storage_calculated_cost_091_180_amount +
        base.storage_calculated_cost_181_365_amount +
        base.storage_calculated_cost_over_365_amount
    ) AS storage_calculated_cost_total_amount

FROM userdata.storage_sales_bundles_amount base
LEFT JOIN dict_dds.transport_hub trhu ON trhu.transport_hub_code = base.warehouse_code
LEFT JOIN adrc adrc ON adrc.address_code = trhu.address_code
LEFT JOIN dict_dds.country_texts cotex ON cotex.country_code = adrc.country_code AND cotex.language_code = 'R'
LEFT JOIN dict_dds.transport_hub_texts trht_3
    ON trht_3.transport_hub_code = base.warehouse_code
   AND trht_3.language_code = CASE WHEN cotex.country_full_name = 'Россия' THEN 'R' ELSE 'E' END
LEFT JOIN dict_dds.plant_and_subsidiary pas ON pas.plant_code = base.receiving_plant_code
LEFT JOIN dict_dds.counterparty counter ON counter.counterparty_code = base.forwarder_code
LEFT JOIN dict_dds.country r_sd556 ON r_sd556.country_code = adrc.country_code
LEFT JOIN dict_dds.market_region1_texts t25a1
    ON t25a1.market_region1_code = r_sd556.market_region1_code
   AND t25a1.language_code = 'R'

GROUP BY
    warehouse_name,
    base.dt_report,
    country_of_remote_warehouse_name,
    region_of_remote_warehouse_name,
    forwarder_name,
    type_warehouse_name,
    type_freight_name,
    receiving_plant_name
DISTRIBUTED BY (warehouse_name, dt_report);

CREATE TEMP TABLE pre_culc
AS
SELECT
    a.*,
    p.period_comparison_rule_code,
    p.dt_comparison
FROM pre_agg a
JOIN dict_dds.calendar_periods p
  ON p.dt = a.dt_report
DISTRIBUTED BY (warehouse_name, dt_report);
