-- =========================================================
-- 0) Чистим на всякий случай
-- =========================================================
DROP TABLE IF EXISTS pp_base;
DROP TABLE IF EXISTS pp_base_keys;
DROP TABLE IF EXISTS pp_dates;
DROP TABLE IF EXISTS cr_curr_small;
DROP TABLE IF EXISTS cr_next;
DROP TABLE IF EXISTS mlcd_agg;
DROP TABLE IF EXISTS pp_fact;
DROP TABLE IF EXISTS fact_with_dates;

-- =========================================================
-- 1) База периодов из PP: (kalnr,bdatj,poper) + dt_valid_from
-- =========================================================
CREATE TEMP TABLE pp_base
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper, 2, 2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
FROM userdata.ckmlpp_ral p
WHERE p.untper = '000'
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE pp_base;

-- Мелкая таблица с ключами тройки — для раннего отсечения в CR/MLCD
CREATE TEMP TABLE pp_base_keys
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT kalnr, bdatj, poper
FROM pp_base
GROUP BY kalnr, bdatj, poper
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE pp_base_keys;

-- =========================================================
-- 2) CR текущего периода — сразу фильтруем по нужным тройкам
-- =========================================================
CREATE TEMP TABLE cr_curr_small
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT
    c.kalnr, c.bdatj, c.poper,
    c.vprsv, c.stprs, c.peinh, c.pvprs,
    COALESCE(c.abprd_o,0) AS abprd_o,
    COALESCE(c.abprd_mo,0) AS abprd_mo,
    COALESCE(c.salk3,0)   AS salk3
FROM userdata.ckmlcr_ral c
JOIN pp_base_keys k
  ON k.kalnr = c.kalnr
 AND k.bdatj = c.bdatj
 AND k.poper = c.poper
WHERE c.untper = '000'
  AND c.curtp  = '10'
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE cr_curr_small;

-- =========================================================
-- 3) CR на 1-е следующего месяца (дублей нет — без GROUP BY)
-- =========================================================
CREATE TEMP TABLE cr_next
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT
    c.kalnr,
    (make_date(c.bdatj::int, substring(c.poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    c.salk3
FROM cr_curr_small c
DISTRIBUTED BY (kalnr);
ANALYZE cr_next;

-- =========================================================
-- 4) MLCD — сразу фильтруем по нужным тройкам
--    (если всё-таки бывают дубли по тройке — включи агрегирование ниже)
-- =========================================================
CREATE TEMP TABLE mlcd_agg
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT
    m.kalnr, m.bdatj, m.poper,
    COALESCE(m.salk3,0)  AS salk3,
    COALESCE(m.estprd,0) AS estprd,
    COALESCE(m.mstprd,0) AS mstprd
FROM userdata.mlcd_ral m
JOIN pp_base_keys k
  ON k.kalnr = m.kalnr
 AND k.bdatj = m.bdatj
 AND k.poper = m.poper
WHERE m.curtp = '10'
  AND m.categ IN ('ZU','VP','PC')
DISTRIBUTED BY (kalnr, bdatj, poper);
ANALYZE mlcd_agg;

-- =========================================================
-- 5) DATES: вычисляем интервалы через LEAD() по уникальным датам
--    (GROUP BY вместо DISTINCT даёт шанс hash-агрегации)
-- =========================================================
CREATE TEMP TABLE pp_dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT kalnr, dt_valid_from
FROM pp_base
GROUP BY kalnr, dt_valid_from
DISTRIBUTED BY (kalnr);
ANALYZE pp_dates;

CREATE TEMP TABLE dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
ON COMMIT DROP AS
SELECT
    kalnr,
    dt_valid_from,
    COALESCE(
      LEAD(dt_valid_from) OVER (PARTITION BY kalnr ORDER BY dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
FROM pp_dates
DISTRIBUTED BY (kalnr);
ANALYZE dates;

-- =========================================================
-- 6) Собираем факт: PP + CR + MLCD (локальные LEFT JOIN по тройке)
-- =========================================================
CREATE TEMP TABLE pp_fact
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
AS
SELECT
    pp.kalnr, pp.bdatj, pp.poper, pp.dt_valid_from,
    pp.umkumo, pp.abkumo, pp.zukumo, pp.lbkum,
    cr.vprsv, cr.stprs, cr.peinh, cr.pvprs, cr.abprd_o, cr.abprd_mo,
    cr.salk3 AS salk3_curr,
    ma.salk3  AS mlcd_salk3,
    ma.estprd AS mlcd_estprd,
    ma.mstprd AS mlcd_mstprd
FROM pp_base pp
LEFT JOIN cr_curr_small cr
  ON cr.kalnr = pp.kalnr
 AND cr.bdatj = pp.bdatj
 AND cr.poper = pp.poper
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr
 AND ma.bdatj = pp.bdatj
 AND ma.poper = pp.poper
DISTRIBUTED BY (kalnr);
ANALYZE pp_fact;

-- =========================================================
-- 7) Подливаем даты и salk3 след. месяца (всё по kalnr)
-- =========================================================
CREATE TEMP TABLE fact_with_dates
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
AS
SELECT
    f.*,
    d.dt_valid_to,
    n.salk3 AS salk3_nextmonth
FROM pp_fact f
LEFT JOIN dates d
  ON d.kalnr = f.kalnr
 AND d.dt_valid_from = f.dt_valid_from
LEFT JOIN cr_next n
  ON n.kalnr = f.kalnr
 AND n.dt_posting = f.dt_valid_from
DISTRIBUTED BY (kalnr);
ANALYZE fact_with_dates;

-- =========================================================
-- 8) ФИНАЛ (LEFT JOIN локально по kalnr) — готовый результат
-- =========================================================
SELECT
    t.kalnr AS calculation_code,
    t.dt_valid_from,
    t.dt_valid_to,

    t.vprsv  AS price_valuation_type_code,
    t.stprs  AS standard_price_amount,
    t.peinh  AS price_unit_code,
    t.pvprs  AS moving_average_price_amount,

    /* rule3: приводим к numeric перед ROUND, чтобы избежать round(double precision, int) */
    CASE
      WHEN (t.umkumo + t.abkumo + t.zukumo) = 0 THEN 0
      ELSE TRUNC(
             CAST(
               ROUND((
                     (t.stprs / NULLIF(t.peinh,0)) * (t.umkumo + t.abkumo)
                   + COALESCE(t.abprd_o,0)
                   + COALESCE(t.abprd_mo,0)
                   + COALESCE(t.mlcd_salk3,0)
                   + COALESCE(t.mlcd_estprd,0)
                   + COALESCE(t.mlcd_mstprd,0)
               )::numeric, 2)
               AS NUMERIC(40,20)
             )
             / NULLIF((t.umkumo + t.abkumo + t.zukumo),0), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE
      WHEN (t.abkumo + t.umkumo + t.zukumo) <> 0
      THEN TRUNC(
             CAST(
                 COALESCE(t.salk3_nextmonth,0)
               + COALESCE(t.mlcd_salk3,0) + COALESCE(t.mlcd_estprd,0) + COALESCE(t.mlcd_mstprd,0)
             AS NUMERIC(40,20))
             / NULLIF((t.abkumo + t.umkumo + t.zukumo),0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE
      WHEN t.lbkum <> 0
      THEN TRUNC(CAST(COALESCE(t.salk3_curr,0) AS NUMERIC(40,20)) / NULLIF(t.lbkum,0), 15)
      ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount,

    h.price_rule_code
FROM fact_with_dates t
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = t.kalnr;
