do
$$
DECLARE    
	v_row_cnt int8;
	v_row_cnt_agg int8 = 0;
	v_calc_time interval ;
	v_start_dttm timestamp(0) ;
    v_key_min int ;
    v_key_max int ;
BEGIN
	v_start_dttm   := clock_timestamp()  ;
	raise notice '% START ', to_char(clock_timestamp(),'hh24:mi:ss');

	if not exists(select 1 from dds.fixed_asset_operations limit 1) then
		v_key_min:= 0;
		v_key_max:= date_part('year',now());
	else
		v_key_min:= date_part('year',now()) - 1;
		v_key_max:= date_part('year',now());
	end if;
	raise notice '% pg_temp.clean_delta v_key_min =  % ; v_key_max = % ', to_char(clock_timestamp(),'hh24:mi:ss'), v_key_min, v_key_max;


	if v_key_min > 0 then
		delete
		from
			dds.fixed_asset_operations
		where
			dt_posting >= make_date(v_key_min,1,1);
		get diagnostics v_row_cnt = row_count;
		raise notice '% Delete target (%)', to_char(clock_timestamp(),'hh24:mi:ss'), v_row_cnt;
		v_row_cnt_agg := v_row_cnt_agg + v_row_cnt;
	end if;


	drop table if exists pg_temp.anep_anea;
	create temp table pg_temp.anep_anea	with (appendonly=true,orientation=column,compresstype=zstd,	compresslevel=3)  as
	select
				coalesce(p.bukrs, a.bukrs) as bukrs,
				coalesce(p.anln1, a.anln1) as anln1,
				coalesce(p.anln2, a.anln2) as anln2,
				coalesce(p.gjahr, a.gjahr) as gjahr,
				coalesce(p.lnran, a.lnran) as lnran,
				coalesce(p.afabe, a.afabe) as afabe,
				coalesce(p.zzstgrd, a.zzstgrd) as zzstgrd,
				coalesce(p.zzstdat, a.zzstdat) as zzstdat,
				p.lnsan,
				p.bwasl,
				p.anbtr,
				a.erlbt,
				a.aufwv,
				a.nafav,
				a.nafal,
				a.safav,
				a.safal,
				a.aafav,
				a.aafal,
				a.aufnv,
				a.aufwl,
				case when sap.parameter_code in ('BWASL_CR', 'BWASL_VR') and sap.range_low_value is not null then 'X'
			end as is_reverse_valuation,
			case
				when sap.abap_program_code = '/RUSAL/FI_KHD' and sap.parameter_code = 'VIRT_BWA' and sap.range_low_value is not null then 'X'
			end as is_virtual_asset_movement
			from
				ods.anep_ral p
			full join ods.anea_ral a on
				p.BUKRS = a.BUKRS
				and p.GJAHR = a.GJAHR
				and p.ANLN1 = a.ANLN1
				and p.ANLN2 = a.ANLN2
				and p.LNRAN = a.LNRAN
				and p.AFABE = a.AFABE
			left join dict_dds.settings_and_parameters_sap sap on
			p.bwasl like replace(sap.range_low_value, '*', '%')
			--and sap.abap_program_code = '/RUSAL/FI_KHD'
			and sap.parameter_code in ('VIRT_BWA', 'BWASL_CR', 'BWASL_VR')
			and sap.range_sign_code is not null
	distributed by (bukrs, gjahr, anln1, anln2,  lnran, afabe);

	get diagnostics v_row_cnt = row_count;
	raise notice '% create pg_temp.anep_anea (%)', to_char(clock_timestamp(),'hh24:mi:ss'), v_row_cnt;
	
	insert into dds.fixed_asset_operations(
		unit_balance_code,
		asset_main_code,
		asset_sub_code,
		dt_posting_yyyy,
		asset_position_code,
		dt_posting,
		dt_reference,
		reference_asset_main_code,
		reference_asset_sub_code,
		business_transaction_code,
		reference_operation_code,
		reference_organization_unit_code,
		general_ledger_operation_type_code,
		dt_asset_document_created,
		valuation_area_code,
		red_reverse_reason_code,
		dt_red_reverse,
		red_reverse_code,
		is_virtual_asset_movement,
		asset_movement_type_code,
		valuation_area_currency_amount,
		revaluation_reserve_valuation_area_currency_amount,
		proportional_cumulative_revaluation_amount,
		asset_realization_revenue_amount,
		depreciation_typical_amount,
		depreciation_special_amount,
		depreciation_unplanned_amount,
		cumulative_acquisition_and_production_cost_amount,
		cumulative_depreciation_typical_amount,
		cumulative_depreciation_special_amount,
		cumulative_depreciation_unplanned_amount,
		valuation_area_currency_code,
		is_anlc,
		revaluation_reserve_valuation_area_curr_cumulative_amount
		)
	select
		fa.unit_balance_code,
		asset_main_code,
		asset_sub_code,
		dt_posting_yyyy,
		asset_position_code,
		dt_posting,
		dt_reference,
		reference_asset_main_code,
		reference_asset_sub_code,
		business_transaction_code,
		reference_operation_code,
		reference_organization_unit_code,
		general_ledger_operation_type_code,
		dt_asset_document_created,
		fa.valuation_area_code,
		red_reverse_reason_code,
		dt_red_reverse,
		red_reverse_code,
		is_virtual_asset_movement,
		asset_movement_type_code,
		valuation_area_currency_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2)))  as valuation_area_currency_amount,
		case when is_reverse_valuation = 'X' and is_virtual_asset_movement is null then valuation_area_currency_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) end as revaluation_reserve_valuation_area_currency_amount,
		proportional_cumulative_revaluation_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2)))  as proportional_cumulative_revaluation_amount,
		asset_realization_revenue_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2)))  as asset_realization_revenue_amount,
		depreciation_typical_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2)))  as depreciation_typical_amount,
		depreciation_special_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) as depreciation_special_amount,
		depreciation_unplanned_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2)))  as depreciation_unplanned_amount,
		cumulative_acquisition_and_production_cost_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) as cumulative_acquisition_and_production_cost_amount,
		cumulative_depreciation_typical_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) as cumulative_depreciation_typical_amount,
		cumulative_depreciation_special_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) as cumulative_depreciation_special_amount,
		cumulative_depreciation_unplanned_amount * (10 ^ (2 - coalesce(dp.decimal_place_number, 2))) as cumulative_depreciation_unplanned_amount,
		va.valuation_area_currency_code as valuation_area_currency_code,
		is_anlc,
		revaluation_reserve_valuation_area_curr_cumulative_amount
	from(
	---выбираем периоды
	with cte0 as (
		select
			unit_balance_code as bukrs,
			min(date_part('year',dt_end_of_month)) as gjahr
		from
			dm_calc.unit_balance_operating_periods
		group by
		unit_balance_code),
	---суммируем переоценку на каждом годе, складываем на год+1, по аналогии с расчетом anlc
	cte as (
		select
				bukrs,
				anln1,
				anln2,
				(gjahr+1) as gjahr,
			    afabe,
			case when max(is_reverse_valuation) = 'X' then sum(sum(coalesce(anbtr,0)) + sum(coalesce(aufwv,0))) over (partition by bukrs,
				anln1,
				anln2,
			    afabe order by gjahr+1 asc) else 0 end as revaluation_reserve_valuation_area_curr_cumulative_amount
			from pg_temp.anep_anea as anep_anea
			where (anbtr <> 0 or aufwv <> 0) and is_reverse_valuation = 'X'
			and gjahr <date_part('year',now())
		group by bukrs,
				anln1,
				anln2,
				gjahr+1,
			    afabe
	),
	---тк проводки могут быть не в каждом году, то генерируем все годы до текущего
	cte1 as (
		select c.bukrs,
			anln1,
			anln2,
			afabe,
			--revaluation_reserve_valuation_area_curr_cumulative_amount,
			 generate_series(c1.gjahr::int, date_part('year',now())::int) as gjahr
		from cte c
		join cte0 c1 on
			c1.bukrs = c.bukrs
		group by
			c.bukrs,
			anln1,
			anln2,
			afabe,
			generate_series(c1.gjahr::int, date_part('year',now())::int)
	),
	---копируем знаачение переоценки в периоды, когда не было проводок переоценки (с последнего года)
	cte2 as (
	select
		c.bukrs,
		c.anln1,
		c.anln2,
		c.gjahr,
		c.afabe,
		coalesce(revaluation_reserve_valuation_area_curr_cumulative_amount , val[array_upper(val,1)]::numeric(17,2)) as revaluation_reserve_valuation_area_curr_cumulative_amount
	from (
	select
	c.bukrs,
		c.anln1,
		c.anln2,
		c.gjahr,
		c.afabe,
		c1.revaluation_reserve_valuation_area_curr_cumulative_amount,
		array_agg(revaluation_reserve_valuation_area_curr_cumulative_amount::varchar) filter (where revaluation_reserve_valuation_area_curr_cumulative_amount is not null)
		over (partition by c.bukrs,
		c.anln1,
		c.anln2,
		c.afabe order by  c.gjahr  asc) as val
		from
		cte1 c
	left join cte as c1 on
		c.bukrs = c1.bukrs
		and c.gjahr=c1.gjahr
		and c.anln1= c1.anln1
		and c.anln2 = c1.anln2
		and c.afabe = c1.afabe) as c
	),
	---откидываем лишнее
		cte3 as (
		select
			cte.bukrs,
			cte.anln1,
			cte.anln2,
			cte.gjahr,
			cte.afabe,
			cte.revaluation_reserve_valuation_area_curr_cumulative_amount
		from cte2 as cte
		where
		 cte.revaluation_reserve_valuation_area_curr_cumulative_amount is not null
	
	)
		select
			k.bukrs as unit_balance_code,
			k.anln1 as asset_main_code,
			k.anln2 as asset_sub_code,
			k.gjahr as dt_posting_yyyy,
			k.lnran as asset_position_code,
			k.budat::date as dt_posting,
			k.bzdat::date as dt_reference,
			k.anlu1 as reference_asset_main_code,
			k.anlu2 as reference_asset_sub_code,
			k.glvor as business_transaction_code,
			k.awtyp as reference_operation_code,
			k.aworg as reference_organization_unit_code,
			k.vorgn as general_ledger_operation_type_code,
			concat(k.cpudt, ' ', k.cputm) as dt_asset_document_created,
			pa.afabe as valuation_area_code,
			pa.zzstgrd as red_reverse_reason_code,
			pa.zzstdat::date as dt_red_reverse,
			pa.lnsan as red_reverse_code,
			pa.is_virtual_asset_movement,
			pa.bwasl as asset_movement_type_code,
			coalesce(pa.anbtr,0) + coalesce(pa.aufwv,0)  as valuation_area_currency_amount,
			coalesce(pa.aufwl,0) + coalesce(pa.aufwv,0)  as proportional_cumulative_revaluation_amount,  -- ANEA-AUFWV + ANEA-AUFWL
			pa.erlbt  as asset_realization_revenue_amount,
			coalesce(pa.nafav,0) + coalesce(pa.nafal,0) + coalesce(pa.aufnv,0) as depreciation_typical_amount,
			coalesce(pa.safav,0) + coalesce(pa.safal,0) as depreciation_special_amount,
			coalesce(pa.aafav,0) + coalesce(pa.aafal,0) as depreciation_unplanned_amount,
			null::numeric(15,2) as cumulative_acquisition_and_production_cost_amount,
			null::numeric(15,2) as cumulative_depreciation_typical_amount,
			null::numeric(15,2) as cumulative_depreciation_special_amount,
			null::numeric(15,2) as cumulative_depreciation_unplanned_amount,
			null as is_anlc,
			pa.is_reverse_valuation,
			null as revaluation_reserve_valuation_area_curr_cumulative_amount
	
		from
			ods.anek_ral as k
		join pg_temp.anep_anea as pa on
			k.BUKRS = pa.BUKRS
			and k.GJAHR = pa.GJAHR
			and k.ANLN1 = pa.ANLN1
			and k.ANLN2 = pa.ANLN2
			and k.LNRAN = pa.LNRAN
		where k.gjahr between v_key_min and v_key_max
	
	union all
	
		select
			coalesce(c.bukrs, cte.bukrs) as unit_balance_code,
			coalesce(c.anln1, cte.anln1) as asset_main_code,
			coalesce(c.anln2,cte.anln2) as asset_sub_code,
			coalesce(c.gjahr, cte.gjahr) as dt_posting_yyyy,
			null as asset_position_code,
			concat(coalesce(c.gjahr,cte.gjahr),'-01-01')::date as dt_posting,
			null as dt_reference,
			null as reference_asset_main_code,
			null as reference_asset_sub_code,
			null as business_transaction_code,
			null as reference_operation_code,
			null as reference_organization_unit_code,
			null as general_ledger_operation_type_code,
			null as dt_asset_document_created,
			coalesce(c.afabe, cte.afabe)  as valuation_area_code,
			null as red_reverse_reason_code,
			null as dt_red_reverse,
			null as red_reverse_code,
			null as is_virtual_asset_movement,
			null as asset_movement_type_code,
			null::numeric(15,2) as valuation_area_currency_amount,
			null::numeric(15,2) as proportional_cumulative_revaluation_amount,
			null::numeric(15,2) as asset_realization_revenue_amount,
			null::numeric(15,2) as depreciation_typical_amount,
			null::numeric(15,2) as depreciation_special_amount,
			null::numeric(15,2) as depreciation_unplanned_amount,
			case when c.bukrs is not null then (coalesce(c.kansw,0) + coalesce(c.kaufw,0))else null end as cumulative_acquisition_and_production_cost_amount,
			case when c.bukrs is not null then (coalesce(c.knafa,0) + coalesce(c.kaufn,0))else null end as cumulative_depreciation_typical_amount,
			c.ksafa::numeric(15,2) as cumulative_depreciation_special_amount,
			c.kaafa::numeric(15,2) as cumulative_depreciation_unplanned_amount,
			'X' as is_anlc,
			null as is_reverse_valuation,
			revaluation_reserve_valuation_area_curr_cumulative_amount
		from
			ods.anlc_ral c
		full join cte3 cte on
			c.bukrs = cte.bukrs
			and	c.anln1 = cte.anln1
			and	c.anln2 = cte.anln2
			and	c.gjahr = cte.gjahr
			and	c.afabe = cte.afabe
		 where coalesce(c.gjahr, cte.gjahr) between v_key_min and v_key_max
		 	--and (coalesce(c.kansw,0) + coalesce(c.kaufw,0)) is not null
	 ) fa
	left join dict_dds.unit_balance_valuation_area va on
		 fa.unit_balance_code = va.unit_balance_code
		 and fa.valuation_area_code = va.valuation_area_code
	left join dict_dds.currency_decimal_place_ral as dp on
		dp.currency_code = va.valuation_area_currency_code;




	get diagnostics v_row_cnt = row_count;
	raise notice '% Insert target (%)', to_char(clock_timestamp(),'hh24:mi:ss'), v_row_cnt;
	v_row_cnt_agg := v_row_cnt_agg + v_row_cnt;


	if v_row_cnt_agg > 0 then
		analyze dds.fixed_asset_operations;
		raise notice '% analyze target', to_char(clock_timestamp(),'hh24:mi:ss');
	end if;

	v_calc_time :=  clock_timestamp() - v_start_dttm ;
	raise notice '% FINISH (v_start_dttm = % ;  v_calc_time = %.)', to_char(clock_timestamp(),'hh24:mi:ss'), v_start_dttm, v_calc_time;
end;
$$
