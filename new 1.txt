SELECT
    base.dt_report,
    COUNT(*) AS calendar_rows_joined
FROM userdata.storage_sales_bundles_amount base
JOIN (
    SELECT dt, '1' AS period_comparison_rule_code
    FROM dict_dds.calendar
    WHERE calendar_year BETWEEN 2000 AND 2050
) cal
    ON cal.dt = base.dt_report
WHERE base.dt_report = '2025-08-27'
GROUP BY base.dt_report;

✅ Вариант 2. Показать дубли напрямую (ещё нагляднее)
sql
Копировать код
SELECT
    cal.dt,
    COUNT(*) AS cnt
FROM (
    SELECT dt
    FROM dict_dds.calendar
    WHERE calendar_year BETWEEN 2000 AND 2050
) cal
WHERE cal.dt = '2025-08-27'
GROUP BY cal.dt;

✅ Вариант 3. «Доказательство в лоб» (лучшее для ревью)
Показываем, что факт дублируется до агрегации.

sql
Копировать код
SELECT
    base.sales_bundle_gross_weight,
    cal.period_comparison_rule_code
FROM userdata.storage_sales_bundles_amount base
JOIN (
    SELECT dt, '1' AS period_comparison_rule_code
    FROM dict_dds.calendar
    WHERE calendar_year BETWEEN 2000 AND 2050
) cal
    ON cal.dt = base.dt_report
WHERE base.dt_report = '2025-08-27'
  AND base.sales_bundle_gross_weight = 913.324;

💣 Один и тот же факт появился дважды ДО GROUP BY

🔥 Контрольный «убийственный» запрос
Этот можно просто кинуть в чат и всё.

sql
Копировать код
SELECT
    SUM(base.sales_bundle_gross_weight)                    AS sum_before_join,
    SUM(base.sales_bundle_gross_weight) * COUNT(cal.dt)    AS expected_after_join
FROM userdata.storage_sales_bundles_amount base
JOIN dict_dds.calendar cal
    ON cal.dt = base.dt_report
WHERE base.dt_report = '2025-08-27'
GROUP BY base.dt_report;
