CREATE TABLE tech_etl.yt_issue_comment (
	issue_id text NOT NULL,
	author text NULL,
	created_at timestamp NULL,
	comment_text text NULL,
	dttm_loaded timestamp DEFAULT now() NULL
)
DISTRIBUTED BY (issue_id);

CREATE TABLE tech_etl.yt_issue_custom_field (
	issue_id text NOT NULL,
	field_name text NOT NULL,
	field_value text NULL,
	dttm_loaded timestamp DEFAULT now() NULL
)
DISTRIBUTED BY (issue_id);

CREATE TABLE tech_etl.yt_issue_snapshot (
	issue_id text NOT NULL,
	summary text NULL,
	description text NULL,
	project_name text NULL,
	project_key text NULL,
	created_by text NULL,
	assignee text NULL,
	created_at timestamp NULL,
	updated_at timestamp NULL,
	resolved_at timestamp NULL,
	current_state text NULL,
	dttm_loaded timestamp DEFAULT now() NULL
)
DISTRIBUTED BY (issue_id);

CREATE TABLE tech_etl.yt_issue_timeline (
	issue_id text NOT NULL,
	ts timestamp NULL,
	author text NULL,
	event_type text NULL,
	field_name text NULL,
	value_from text NULL,
	value_to text NULL,
	dttm_loaded timestamp DEFAULT now() NULL
)
DISTRIBUTED BY (issue_id);

CREATE TABLE tech_etl.yt_issue_worklog (
	issue_id text NOT NULL,
	author text NULL,
	creator text NULL,
	work_date timestamp NULL,
	minutes int4 NULL,
	work_type text NULL,
	work_text text NULL,
	dttm_loaded timestamp DEFAULT now() NULL
)
DISTRIBUTED BY (issue_id);
