# etl_dep_monitor.py
# -*- coding: utf-8 -*-
"""
Единый мониторинг зависимостей из YAML → БД (Greenplum/Postgres).
Запуск без параметров: python etl_dep_monitor.py

Что делает:
  - сканирует YAML-файлы в YAML_ROOT
  - парсит table_schema/table_name/depends_on
  - строит граф зависимостей target -> sources
  - читает last_load из tech_etl.tables_meta
  - пишет результаты в тех. таблицы:
      tech_etl.dep_runs
      tech_etl.dep_edges_current  (TRUNCATE + INSERT)
      tech_etl.dep_edges_log
      tech_etl.dep_freshness
      tech_etl.dep_orphans
      tech_etl.dep_cycles
"""

import os
import sys
import csv
import datetime as dt
from typing import Dict, Tuple, Set, List, Optional

import yaml           # pip install pyyaml
import psycopg2       # pip install psycopg2-binary
from psycopg2 import sql

# =========================
# ХАРДКОД НАСТРОЕК
# =========================
YAML_ROOT = r"meta_info/database/greenplum/schema_name/tech_etl/etl_loads_entity"  # корень с YAML
GP_DSN    = "postgresql://user:password@host:5432/dbname"                           # DSN Greenplum/Postgres
# =========================

Node = Tuple[str, str]  # (schema, table)


# ---------- Утилиты пути ----------
def normalize_paths(base_dir: str, path: str) -> Tuple[str, str]:
    abs_path = os.path.abspath(path)
    rel_path = os.path.relpath(abs_path, os.path.abspath(base_dir))
    # единые слеши
    return rel_path.replace("\\", "/"), abs_path.replace("\\", "/")


# ---------- Скан YAML ----------
def scan_yaml_files(base_dir: str) -> List[str]:
    files = []
    for root, _, fs in os.walk(base_dir):
        for f in fs:
            if f.lower().endswith((".yaml", ".yml")):
                files.append(os.path.join(root, f))
    return files


def _s(x) -> Optional[str]:
    if x is None:
        return None
    x = str(x).strip()
    return x or None


def parse_yaml(path: str) -> Tuple[Optional[Node], Set[Node]]:
    """
    Ожидаем ключи:
      table_schema: dm
      table_name:   account_debt
      depends_on:
        dm_calc:
          - operating_periods_for_account_debt
        dm:
          - account_debt_for_working_capital
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            doc = yaml.safe_load(f)
    except Exception as e:
        print(f"[WARN] YAML parse failed {path}: {e}", file=sys.stderr)
        return None, set()

    if not isinstance(doc, dict):
        return None, set()

    schema = _s(doc.get("table_schema"))
    table = _s(doc.get("table_name"))
    tgt = (schema, table) if (schema and table) else None

    deps = set()
    dep_map = doc.get("depends_on")
    if isinstance(dep_map, dict):
        for dep_schema, names in dep_map.items():
            dep_schema_s = _s(dep_schema)
            if not dep_schema_s:
                continue
            if isinstance(names, list):
                for n in names:
                    n_s = _s(n)
                    if n_s:
                        deps.add((dep_schema_s, n_s))
            elif isinstance(names, str):
                n_s = _s(names)
                if n_s:
                    deps.add((dep_schema_s, n_s))
            elif isinstance(names, dict):
                for maybe_schema, lst in names.items():
                    ms = _s(maybe_schema)
                    if isinstance(lst, list):
                        for n in lst:
                            n_s = _s(n)
                            if ms and n_s:
                                deps.add((ms, n_s))
    return tgt, deps


def build_graph(yaml_dir: str):
    graph: Dict[Node, Set[Node]] = {}
    edges: List[dict] = []
    base = os.path.abspath(yaml_dir)

    for p in scan_yaml_files(yaml_dir):
        tgt, deps = parse_yaml(p)
        if not tgt:
            continue
        rel_path, abs_path = normalize_paths(base, p)

        graph.setdefault(tgt, set()).update(deps)
        for d in deps:
            edges.append({
                "tgt_schema": tgt[0],
                "tgt_table":  tgt[1],
                "dep_schema": d[0],
                "dep_table":  d[1],
                "source_path": rel_path,      # относительный путь от корня
                "source_path_abs": abs_path   # абсолютный путь (если колонка есть)
            })
    return graph, edges


# ---------- Метаданные last_load ----------
def load_last_load(conn) -> Dict[Node, Optional[dt.datetime]]:
    sql_text = "select table_schema, table_name, last_load from tech_etl.tables_meta"
    out: Dict[Node, Optional[dt.datetime]] = {}
    with conn.cursor() as cur:
        cur.execute(sql_text)
        for s, t, ll in cur.fetchall():
            out[(str(s).strip(), str(t).strip())] = ll
    return out


# ---------- Поиск циклов ----------
def find_cycles(graph: Dict[Node, Set[Node]]) -> List[List[Node]]:
    WHITE, GRAY, BLACK = 0, 1, 2
    color: Dict[Node, int] = {}
    parent: Dict[Node, Optional[Node]] = {}
    cycles: List[List[Node]] = []

    for n in list(graph.keys()):
        color[n] = WHITE
        parent[n] = None

    def dfs(u: Node):
        color[u] = GRAY
        for v in graph.get(u, set()):
            if v not in color:
                color[v] = WHITE
                parent[v] = None
                graph.setdefault(v, set())
            if color[v] == WHITE:
                parent[v] = u
                dfs(v)
            elif color[v] == GRAY:
                cyc = [v]
                cur = u
                while cur != v and cur is not None:
                    cyc.append(cur)
                    cur = parent.get(cur)
                cyc.reverse()
                if cyc and cyc not in cycles:
                    cycles.append(cyc)
        color[u] = BLACK

    for n in list(graph.keys()):
        if color.get(n, WHITE) == WHITE:
            dfs(n)
    return cycles


# ---------- Свежесть и осироты ----------
def compute_freshness(graph, last_load):
    rows = []
    for tgt, srcs in graph.items():
        tgt_ll = last_load.get(tgt)
        max_src = None
        for s in srcs:
            ll = last_load.get(s)
            if ll is not None and (max_src is None or ll > max_src):
                max_src = ll
        if max_src is None:
            status = "NO_SOURCES"
        else:
            status = "STALE_TARGET" if (tgt_ll is None or max_src > tgt_ll) else "OK"
        rows.append({
            "tgt_schema": tgt[0],
            "tgt_table":  tgt[1],
            "tgt_last_load": tgt_ll,
            "max_src_last_load": max_src,
            "freshness_status": status
        })
    return rows


def compute_orphans(graph, last_load):
    out = []
    for tgt, srcs in graph.items():
        for s in srcs:
            if s not in last_load:
                out.append({
                    "tgt_schema": tgt[0],
                    "tgt_table":  tgt[1],
                    "missing_dep_schema": s[0],
                    "missing_dep_table":  s[1]
                })
    return out


# ---------- Вспомогательное: есть ли колонка? ----------
def table_has_column(conn, schema: str, table: str, column: str) -> bool:
    q = """
    select 1
    from information_schema.columns
    where table_schema=%s and table_name=%s and column_name=%s
    """
    with conn.cursor() as cur:
        cur.execute(q, (schema, table, column))
        return cur.fetchone() is not None


# ---------- Запись в БД ----------
def db_write_all(dsn: str, yaml_dir: str, edges: List[dict],
                 fresh_rows: List[dict], orphan_rows: List[dict], cycles: List[List[Node]]):
    run_ts = dt.datetime.utcnow()
    run_id = run_ts.strftime("%Y-%m-%dT%H:%M:%SZ")

    with psycopg2.connect(dsn) as conn, conn.cursor() as cur:
        # audit run
        cur.execute(
            "INSERT INTO tech_etl.dep_runs(run_id, run_ts, yaml_dir) VALUES (%s,%s,%s)",
            (run_id, run_ts, yaml_dir)
        )

        # dep_edges_current: поддерживаем оба варианта — с и без source_path_abs
        has_abs = table_has_column(conn, "tech_etl", "dep_edges_current", "source_path_abs")

        cur.execute("TRUNCATE tech_etl.dep_edges_current")
        if has_abs:
            cur.executemany(
                """
                INSERT INTO tech_etl.dep_edges_current
                  (tgt_schema, tgt_table, dep_schema, dep_table, source_path, source_path_abs)
                VALUES (%s,%s,%s,%s,%s,%s)
                """,
                [
                    (
                        e["tgt_schema"], e["tgt_table"],
                        e["dep_schema"], e["dep_table"],
                        e["source_path"], e["source_path_abs"]
                    ) for e in edges
                ]
            )
        else:
            cur.executemany(
                """
                INSERT INTO tech_etl.dep_edges_current
                  (tgt_schema, tgt_table, dep_schema, dep_table, source_path)
                VALUES (%s,%s,%s,%s,%s)
                """,
                [
                    (
                        e["tgt_schema"], e["tgt_table"],
                        e["dep_schema"], e["dep_table"],
                        e["source_path"]
                    ) for e in edges
                ]
            )

        # dep_edges_log
        cur.executemany(
            """
            INSERT INTO tech_etl.dep_edges_log
              (run_id, run_ts, tgt_schema, tgt_table, dep_schema, dep_table, source_path)
            VALUES (%s,%s,%s,%s,%s,%s,%s)
            """,
            [
                (
                    run_id, run_ts,
                    e["tgt_schema"], e["tgt_table"],
                    e["dep_schema"], e["dep_table"],
                    e["source_path"]
                ) for e in edges
            ]
        )

        # dep_freshness
        cur.executemany(
            """
            INSERT INTO tech_etl.dep_freshness
              (run_id, run_ts, tgt_schema, tgt_table, tgt_last_load, max_src_last_load, freshness_status)
            VALUES (%s,%s,%s,%s,%s,%s,%s)
            """,
            [
                (
                    run_id, run_ts,
                    r["tgt_schema"], r["tgt_table"],
                    r["tgt_last_load"], r["max_src_last_load"],
                    r["freshness_status"]
                ) for r in fresh_rows
            ]
        )

        # dep_orphans
        cur.executemany(
            """
            INSERT INTO tech_etl.dep_orphans
              (run_id, run_ts, tgt_schema, tgt_table, missing_dep_schema, missing_dep_table)
            VALUES (%s,%s,%s,%s,%s,%s)
            """,
            [
                (
                    run_id, run_ts,
                    r["tgt_schema"], r["tgt_table"],
                    r["missing_dep_schema"], r["missing_dep_table"]
                ) for r in orphan_rows
            ]
        )

        # dep_cycles
        cyc_rows = []
        for i, cyc in enumerate(cycles, 1):
            path = " -> ".join([f"{s}.{t}" for s, t in cyc])
            cyc_rows.append((run_id, run_ts, i, len(cyc), path))
        if cyc_rows:
            cur.executemany(
                """
                INSERT INTO tech_etl.dep_cycles
                  (run_id, run_ts, cycle_id, cycle_len, path)
                VALUES (%s,%s,%s,%s,%s)
                """,
                cyc_rows
            )

        conn.commit()

    print(f"[OK] DB updated, run_id={run_id}")


# ---------- main ----------
def main():
    print(f"[INFO] YAML_ROOT: {YAML_ROOT}")
    print("[INFO] Сканирую YAML и строю граф…")
    graph, edges = build_graph(YAML_ROOT)
    print(f"[INFO] targets: {len(graph)}, edges: {len(edges)}")

    with psycopg2.connect(GP_DSN) as conn:
        last_load = load_last_load(conn)

    print("[INFO] Поиск циклов…")
    cycles = find_cycles(graph)

    print("[INFO] Свежесть и осироты…")
    fresh_rows = compute_freshness(graph, last_load)
    orphan_rows = compute_orphans(graph, last_load)

    print("[INFO] Запись в БД…")
    db_write_all(GP_DSN, YAML_ROOT, edges, fresh_rows, orphan_rows, cycles)


if __name__ == "__main__":
    main()
