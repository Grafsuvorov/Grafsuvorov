DROP TABLE IF EXISTS adrc;
CREATE TEMPORARY TABLE adrc ON COMMIT DROP AS
(
    -- Убираем дубли по address_code, получаем единственную страну
    SELECT
        address_code,
        MIN(country_code) AS country_code
    FROM dict_dds.address
    WHERE country_code IS NOT NULL
    GROUP BY address_code
)
DISTRIBUTED REPLICATED;


DROP TABLE IF EXISTS cotex;
CREATE TEMPORARY TABLE cotex ON COMMIT DROP AS
(
    SELECT
        a.country_code,
        th.transport_hub_code,

        -- Нормализация названий стран
        CASE
            WHEN ct.country_full_name = 'Корея, республика'    THEN 'Южная Корея'
            WHEN ct.country_full_name = 'Тайвань (Китай)'      THEN 'Тайвань'
            ELSE ct.country_full_name
        END AS country_of_remote_warehouse_name,        -- LE.001060

        -- Кейс языка для текстов хабов: Россия → R, всё остальное → E
        CASE
            WHEN ct.country_full_name = 'Россия' THEN 'R'
            ELSE 'E'
        END AS case_country_full_name
    FROM dict_dds.transport_hub       AS th
    LEFT JOIN adrc                    AS a
           ON a.address_code = th.address_code
    LEFT JOIN dict_dds.country_texts  AS ct
           ON ct.country_code   = a.country_code
          AND ct.language_code  = 'R'
)
DISTRIBUTED REPLICATED;


DROP TABLE IF EXISTS t25a1;
CREATE TEMPORARY TABLE t25a1 ON COMMIT DROP AS
(
    SELECT
        t.market_region1_code,
        t.language_code,
        CASE
            WHEN t.market_region1_name = 'Турция'
                THEN 'Западная Азия'
            WHEN t.market_region1_name = 'Средний Восток и Афр'
                THEN 'Ближний Восток'
            WHEN t.market_region1_name IN ('Азия', 'Китай')
                THEN 'Восточная Азия'
            ELSE t.market_region1_name
        END AS region_of_remote_warehouse_name          -- LE.001061
    FROM dict_dds.market_region1_texts AS t
    WHERE t.language_code = 'R'
)
DISTRIBUTED REPLICATED;
