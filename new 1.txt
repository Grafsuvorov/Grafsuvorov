CREATE TEMP TABLE for_1079_base ON COMMIT DROP AS
(
    SELECT
        for_1079_base_calc.purchase_contract_code,
        for_1079_base_calc.position_line_item_code,
        for_1079_base_calc.bukrs,
        for_1079_base_calc.lifnr,
        for_1079_base_calc.zzport,
        for_1079_base_calc.kdatb,
        for_1079_base_calc.kdate,
        for_1079_base_calc.werks,
        for_1079_base_calc.zzsrvpos,
        tmp_1.range_low_value AS tmp_1,
        tmp_2.range_low_value AS tmp_2,
        for_1079_base_calc.zzforma,
        for_1079_base_calc.zzet_tarif,
        for_1079_base_calc.sdabw,
        for_1079_base_calc.datbi,
        for_1079_base_calc.datab,
        for_1079_base_calc.dt_valid_to,
        for_1079_base_calc.dt_valid_from,
        for_1079_base_calc.purchase_document_subtype_code,
        for_1079_base_calc.kopos,
        for_1079_base_calc.konwa,
        for_1079_base_calc.zzterminal,
        for_1079_base_calc.vzzzone,
        COALESCE(for_1079_base_calc.price_scale_currency_code,
                 for_1079_base_calc.price_condition_position_currency_code) AS konws,
        for_1079_base_calc.service_code,
        for_1079_base_calc.price_uom_code,
        for_1079_base_calc.netpr,
        for_1079_base_calc.kbetr,

        -- order_rf
        (CASE WHEN for_1079_base_calc.zzport      IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzforma     IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzet_tarif  IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.sdabw       IS NULL THEN 1 ELSE 0 END) AS order_rf,

        -- order_asia
        (CASE WHEN for_1079_base_calc.zzport      IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzforma     IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzet_tarif  IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.sdabw       IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzterminal  IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.vzzzone     IS NULL THEN 1 ELSE 0 END) AS order_asia,

        -- order_euro_svh
        (CASE WHEN for_1079_base_calc.zzport      IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzforma     IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzet_tarif  IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.sdabw       IS NULL THEN 1 ELSE 0 END +
         CASE WHEN for_1079_base_calc.zzterminal  IS NULL THEN 1 ELSE 0 END) AS order_euro_svh,

        CASE WHEN curdpr.currency_code IS NOT NULL THEN 100 ELSE 1 END AS currency_dogovor
    FROM
    (
        SELECT
            ekpo.purchase_contract_code,
            ekpo.position_line_item_code,
            ekko.unit_balance_code                            AS bukrs,
            ekko.supplier_code                                AS lifnr,
            ekko.port_code                                    AS zzport,
            ekko.dt_valid_from                                AS kdatb,
            ekko.dt_valid_to                                  AS kdate,
            ekpo.plant_code                                   AS werks,
            ekpo.service_code                                 AS zzsrvpos,
            ekpo.shape_code                                   AS zzforma,
            ekpo.etsng_code                                   AS zzet_tarif,
            ekpo.transport_type_code                          AS sdabw,
            ekpo_price.dt_price_condition_valid_to            AS datbi,
            ekpo_price.dt_price_condition_valid_from          AS datab,
            ekko.dt_valid_to,
            ekko.dt_valid_from,
            ekko.purchase_document_subtype_code,
            ekpo_price.price_condition_position_code          AS kopos,
            ekpo_price.price_condition_position_currency_code,
            ekpo_price.price_scale_currency_code,
            ekpo.terminal_code                                AS zzterminal,
            ekpo.warehouse_storage_area_code                  AS vzzzone,
            ekpo.service_code,
            ekpo.price_uom_code,
            ekpo.price_vat_excluded_document_currency_amount  AS netpr,
            ekpo_price.price_rate                             AS kbetr,
            ekpo_price.is_price_condition_position_deleted_code,
            COUNT(*) OVER (PARTITION BY ekpo.purchase_contract_code,
                                      ekpo.position_line_item_code) AS cnt,
            ROW_NUMBER() OVER (
                PARTITION BY ekpo.purchase_contract_code,
                             ekpo.position_line_item_code,
                             ekpo_price.dt_price_condition_valid_from,
                             ekpo_price.dt_price_condition_valid_to,
                             ekpo_price.price_scale_basis_code
                ORDER BY ekpo_price.price_scale_basis_consecutive_code
            ) AS rwn
        FROM dds.purchase_contract_price_conditions ekpo_price
        JOIN dds.purchase_contract_header ekko
          ON ekko.purchase_contract_code = ekpo_price.purchase_contract_code
        JOIN dds.purchase_contract_position ekpo
          ON ekpo.purchase_contract_code = ekko.purchase_contract_code
         AND ekpo.position_line_item_code = ekpo_price.purchase_contract_position_code
        LEFT JOIN dict_dds.currency_decimal_place_ral curdpr
          ON curdpr.currency_code = COALESCE(ekpo_price.price_scale_currency_code,
                                             ekpo_price.price_condition_position_currency_code)
         AND curdpr.decimal_place_number = '0'
        WHERE ekko.purchase_document_subtype_code IN ('ZDTL','ZDT','ZDE','ZDP','ZDF','ZDS')
          AND ekko.paydox_contract_type_code IN ('1','6')
          AND ekpo.is_deleted IS NULL
          AND ekpo_price.is_price_condition_position_deleted_code IS NULL
          AND ekpo_price.price_condition_type_code = 'PB00'
          AND (ekpo_price.price_scale_basis_code = 'C'
               OR ekpo_price.price_scale_basis_code IS NULL)
    ) AS for_1079_base_calc
    LEFT JOIN dict_dds.settings_and_parameters_sap tmp_1
      ON tmp_1.abap_program_code = '/RUSAL/SD4553M'
     AND tmp_1.parameter_code = 'S_USLSTC'
     AND tmp_1.range_low_value = for_1079_base_calc.zzsrvpos
    LEFT JOIN dict_dds.settings_and_parameters_sap tmp_2
      ON tmp_2.abap_program_code = '/RUSAL/SD4553M'
     AND tmp_2.parameter_code = 'S_USLST'
     AND tmp_2.range_low_value LIKE 'EU%'
     AND tmp_2.range_low_value = for_1079_base_calc.zzsrvpos
    WHERE for_1079_base_calc.rwn = 1
)
DISTRIBUTED REPLICATED;

-- 5. Таблица для LE.001090 (ценовые шкалы по PB00)
------------------------------------------------------------

CREATE TEMP TABLE for_1090 ON COMMIT DROP AS
WITH tmp AS (
    SELECT
        pcpc.price_condition_code                AS knumh,
        pcpc.purchase_contract_code              AS evrtn,
        pcpc.purchase_contract_position_code     AS evrtp,
        pcpc.price_scale_basis_consecutive_code,
        pcpc.price_condition_position_code,
        pcpc.price_rate                          AS konp_kbetr,
        pcpc.price_scale_rate                    AS konm_kbetr,
        pcpc.price_scale_limit_from_quantity     AS kstbm,
        pcpc.price_scale_limit_to_quantity       AS kstbm_next,
        pcpc.price_scale_basis_code,
        pcpc.dt_price_condition_valid_to,
        pcpc.dt_price_condition_valid_from,
        COUNT(pcpc.price_scale_limit_from_quantity)
            OVER (PARTITION BY pcpc.purchase_contract_code,
                               pcpc.purchase_contract_position_code) AS cnt,
        ROW_NUMBER() OVER (
            PARTITION BY pcpc.purchase_contract_code,
                         pcpc.purchase_contract_position_code
            ORDER BY pcpc.price_scale_basis_consecutive_code
        ) AS rn,
        MIN(price_scale_limit_from_quantity)
            OVER (
                PARTITION BY pcpc.purchase_contract_code,
                             pcpc.purchase_contract_position_code,
                             pcpc.dt_price_condition_valid_from,
                             pcpc.dt_price_condition_valid_to
            ) AS min_price_scale_limit_from_quantity,
        (pcpc.price_scale_limit_from_quantity
         - LAG(pcpc.price_scale_limit_from_quantity) OVER (
               PARTITION BY pcpc.purchase_contract_code,
                            pcpc.purchase_contract_position_code
               ORDER BY pcpc.price_scale_basis_consecutive_code
           ))
        * LAG(pcpc.price_scale_rate) OVER (
               PARTITION BY pcpc.purchase_contract_code,
                            pcpc.purchase_contract_position_code
               ORDER BY pcpc.price_scale_basis_consecutive_code
           ) AS lagging
    FROM dds.purchase_contract_price_conditions pcpc
    WHERE pcpc.price_condition_type_code = 'PB00'
      AND (pcpc.price_scale_basis_code = 'C'
           OR pcpc.price_scale_basis_code IS NULL)
)
SELECT
    tmp.knumh,
    tmp.evrtn,
    tmp.evrtp,
    tmp.dt_price_condition_valid_to,
    tmp.dt_price_condition_valid_from,
    tmp.price_scale_basis_consecutive_code,
    tmp.price_condition_position_code,
    tmp.konp_kbetr,
    tmp.konm_kbetr,
    tmp.kstbm,
    tmp.kstbm_next,
    tmp.cnt,
    tmp.rn,
    tmp.price_scale_basis_code,
    tmp.min_price_scale_limit_from_quantity,
    SUM(tmp.lagging) OVER (
        PARTITION BY tmp.evrtn, tmp.evrtp
        ORDER BY tmp.price_scale_basis_consecutive_code
    ) AS presum
FROM tmp
DISTRIBUTED BY (evrtn);

                    f.kopos,
                    f.kbetr,
                    f.purchase_contract_code,
                    f.position_line_item_code
            ) AS row_num,
            f.konws,
            f.service_code,
            f.price_uom_code,
            f.currency_dogovor,
            f.datab,
            f.datbi
        FROM base_logic b
        JOIN for_1079_base f
          ON f.bukrs = b.unit_balance_code
         AND f.lifnr = b.forwarder_code
         AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
         AND f.werks = b.plant_code
         AND f.zzsrvpos = 'S49'
         AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
         AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
         AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.dt_valid_from AND f.dt_valid_to
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.datab AND f.datbi
         AND b.storage_cost_calculation_type_name IN ('Логика Азия')
         AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
         AND (f.vzzzone   = b.warehouse_storage_area_code OR f.vzzzone IS NULL)
         AND f.purchase_document_subtype_code = 'ZDTL'
    ) AS df
    WHERE df.row_num = 1

    UNION ALL

    --------------------------------------------------------
    -- СВХ (бывший for_1079_svh)
    --------------------------------------------------------
    SELECT
        df.rn,
        'SVH'::text AS logic_type,
        df.purchase_contract_code,
        df.position_line_item_code,
        df.konws,
        df.service_code,
        df.price_uom_code,
        df.currency_dogovor,
        df.datab,
        df.datbi,
        NULL::int4    AS month_difference_2,
        NULL::numeric AS special
    FROM (
        SELECT
            b.rn,
            f.purchase_contract_code,
            f.position_line_item_code,
            ROW_NUMBER() OVER (
                PARTITION BY b.rn
                ORDER BY
                    f.order_euro_svh,
                    f.kopos,
                    f.kbetr,
                    f.purchase_contract_code,
                    f.position_line_item_code
            ) AS row_num,
            f.konws,
            f.service_code,
            f.price_uom_code,
            f.currency_dogovor,
            f.datab,
            f.datbi
        FROM base_logic b
        JOIN for_1079_base f
          ON f.bukrs = '1120'
         AND f.lifnr = b.forwarder_code
         AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
         AND f.werks = '1121'
         AND f.zzsrvpos = 'S49'
         AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
         AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
         AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.dt_valid_from AND f.dt_valid_to
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.datab AND f.datbi
         AND b.storage_cost_calculation_type_name IN ('Логика СВХ')
         AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
         AND f.purchase_document_subtype_code = 'ZDTL'
    ) AS df
    WHERE df.row_num = 1

    UNION ALL

    --------------------------------------------------------
    -- Европа (бывший for_1079_euro)
    --------------------------------------------------------
    SELECT
        dfdf.rn,
        'EURO'::text AS logic_type,
        dfdf.purchase_contract_code,
        dfdf.position_line_item_code,
        dfdf.konws,
        dfdf.service_code,
        dfdf.price_uom_code,
        dfdf.currency_dogovor,
        dfdf.datab,
        dfdf.datbi,
        dfdf.month_difference_2,
        dfdf.special
    FROM (
        SELECT
            b.rn,
            f.purchase_contract_code,
            f.position_line_item_code,
            ROW_NUMBER() OVER (
                PARTITION BY b.rn
                ORDER BY
                    f.order_euro_svh,
                    f.kopos,
                    f.kbetr,
                    CASE
                        WHEN param_sap.range_low_value IS NOT NULL
                             AND param_srv_1.range_low_value IS NOT NULL THEN 1
                        WHEN param_sap.range_low_value IS NULL
                             AND param_srv_1.range_low_value IS NULL THEN 1
                        ELSE 0
                    END DESC,
                    f.purchase_contract_code,
                    f.position_line_item_code
            ) AS row_num,
            f.konws,
            f.service_code,
            f.price_uom_code,
            SUM(b.sales_bundle_gross_weight) OVER (
                PARTITION BY b.warehouse_code,
                             date_trunc('month', b.dt_storage_end)
            ) AS warehouse_weight_month_end,
            CASE
                WHEN EXTRACT(DAY FROM b.dt_storage_end)
                     < EXTRACT(DAY FROM b.dt_storage_start)
                    THEN
                        (EXTRACT(YEAR FROM b.dt_storage_end)
                         - EXTRACT(YEAR FROM b.dt_storage_start)) * 12
                      + (EXTRACT(MONTH FROM b.dt_storage_end)
                         - EXTRACT(MONTH FROM b.dt_storage_start))
                ELSE
                        (EXTRACT(YEAR FROM b.dt_storage_end)
                         - EXTRACT(YEAR FROM b.dt_storage_start)) * 12
                      + (EXTRACT(MONTH FROM b.dt_storage_end)
                         - EXTRACT(MONTH FROM b.dt_storage_start)) + 1
            END AS month_difference_2,
            f.netpr,
            CASE
                WHEN b.storage_cost_special_calculation_type_code = '4'
                    THEN ceil(b.storage_duration_total_calendar_days / 7::numeric)
                WHEN b.storage_cost_special_calculation_type_code = '5'
                    THEN ceil(b.storage_duration_total_calendar_days / 30::numeric)
            END AS special,
            f.currency_dogovor,
            f.datab,
            f.datbi
        FROM base_logic b
        LEFT JOIN dict_dds.settings_and_parameters_sap for_1082_1
            ON for_1082_1.abap_program_code = '/RUSAL/ZLE112M'
           AND for_1082_1.case_code = 'PERET_FRN'
           AND for_1082_1.parameter_code = 'S_LIFNRA'
           AND for_1082_1.range_low_value = b.forwarder_code
        LEFT JOIN dict_dds.settings_and_parameters_sap for_1082_2
            ON for_1082_2.abap_program_code = '/RUSAL/SD4553M'
           AND for_1082_2.parameter_code = 'S_FORMSTEU'
           AND for_1082_2.range_low_value = b.material_shape_code
        JOIN for_1079_base f
          ON b.storage_cost_calculation_type_name = 'Логика Европа'
         AND f.purchase_document_subtype_code = 'ZDTL'
         AND (
                (for_1082_1.range_low_value IS NOT NULL AND f.zzsrvpos = 'S49')
             OR (for_1082_1.range_low_value IS NULL
                 AND for_1082_2.range_low_value IS NOT NULL
                 AND f.tmp_1 IS NOT NULL)
             OR (for_1082_1.range_low_value IS NULL
                 AND for_1082_2.range_low_value IS NULL
                 AND f.tmp_2 IS NOT NULL)
             )
         AND f.bukrs = b.unit_balance_code
         AND f.lifnr = b.forwarder_code
         AND f.werks = b.plant_code
         AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
         AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
         AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
         AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.dt_valid_from AND f.dt_valid_to
         AND b.dt_for_price_search_in_purchase_contract
             BETWEEN f.datab AND f.datbi
         AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
        LEFT JOIN dict_dds.settings_and_parameters_sap for_1082_3
            ON f.tmp_2 IS NOT NULL
           AND for_1082_3.range_low_value LIKE 'EU%'
           AND for_1082_3.abap_program_code = '/RUSAL/SD4553M'
           AND for_1082_3.parameter_code = 'S_USLST'
        LEFT JOIN dict_dds.settings_and_parameters_sap param_sap
            ON param_sap.range_low_value = b.material_shape_code
           AND param_sap.abap_program_code = '/RUSAL/SD4553M'
           AND param_sap.parameter_code = 'FORMSTEU'
        LEFT JOIN dict_dds.settings_and_parameters_sap param_srv_1
            ON param_srv_1.range_low_value = for_1082_3.range_low_value
           AND param_srv_1.abap_program_code = '/RUSAL/SD4553M'
           AND param_srv_1.parameter_code = 'S_USLSTC'
    ) AS dfdf
    WHERE dfdf.row_num = 1
)
DISTRIBUTED BY (rn);

-- 8. logic_final: выбираем одну логику на RN по приоритету
--    RF_CONT > RF_BALK1 > RF_BALK2 > EURO > SVH > ASIA
------------------------------------------------------------

CREATE TEMP TABLE logic_final ON COMMIT DROP AS
SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    currency_dogovor,
    datab,
    datbi,
    month_difference_2,

FROM (
    SELECT
        lr.*,
        ROW_NUMBER() OVER (
            PARTITION BY lr.rn
            ORDER BY
                CASE lr.logic_type
                    WHEN 'RF_CONT'  THEN 1
                    WHEN 'RF_BALK1' THEN 2
                    WHEN 'RF_BALK2' THEN 3
                    WHEN 'EURO'     THEN 4
                    WHEN 'SVH'      THEN 5
                    WHEN 'ASIA'     THEN 6
                    ELSE 99
                END
        ) AS rn_priority
    FROM logic_raw lr
) s
WHERE rn_priority = 1
DISTRIBUTED BY (rn);
