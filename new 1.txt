    explain(verbose, format json)
   select base.*
FROM dm_calc.storage_sales_bundles_amount base
LEFT JOIN material_texts               mt  ON mt.material_code = base.material_code
LEFT JOIN material_shape               ms  ON ms.material_code = base.material_code
LEFT JOIN transportation_service_texts srv ON srv.transportation_service_code = base.service_code
LEFT JOIN etsng_texts                  et  ON et.etsng_code = base.etsng_code
LEFT JOIN warehouse_geo                wg  ON wg.transport_hub_code = base.warehouse_code
LEFT JOIN remote_geo                   rg  ON rg.transport_hub_code = base.remote_warehouse_code

-- PLANTS
LEFT JOIN dict_dds.plant_and_subsidiary pas ON pas.plant_code = base.receiving_plant_code

-- OWNER
LEFT JOIN owner_plant op ON op.delivery_code = base.delivery_for_storage_calculation_code

-- creditor
LEFT JOIN creditor_info ci ON ci.counterparty_code = base.creditor_in_shipment_instruction_code
LEFT JOIN dict_dds.counterparty counter ON counter.counterparty_code = base.forwarder_code

-- HUB TEXTS
LEFT JOIN hub_text htxt_wh     ON htxt_wh.transport_hub_code     = base.warehouse_code
                               AND htxt_wh.language_code         = wg.lang_code

LEFT JOIN hub_text htxt_area   ON htxt_area.transport_hub_code   = base.warehouse_storage_area_code
                               AND htxt_area.language_code       = 'R'

LEFT JOIN hub_text htxt_dep    ON htxt_dep.transport_hub_code    = base.transport_departure_hub_code
                               AND htxt_dep.language_code        = 'R'

LEFT JOIN hub_text htxt_dest   ON htxt_dest.transport_hub_code   = base.transport_destination_hub_code
                               AND htxt_dest.language_code       = 'R'

LEFT JOIN hub_text htxt_pd     ON htxt_pd.transport_hub_code     = base.port_of_discharge_code
                               AND htxt_pd.language_code         = 'E' -- usually English port names

LEFT JOIN hub_text htxt_remote ON htxt_remote.transport_hub_code = base.remote_warehouse_code
                               AND htxt_remote.language_code     = rg.lang_code

LEFT JOIN dict_dds.transportation_stage_texts trst 
       ON trst.transportation_stage_code = base.transportation_stage_code
      AND trst.language_code='R'

LEFT JOIN dict_dds.transport_transfer_type_texts ipd
       ON ipd.transport_transfer_type_code = base.transport_type_code
      AND ipd.language_code='R';
