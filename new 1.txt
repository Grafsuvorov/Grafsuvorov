#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import argparse
import json
import os
import re
from collections import Counter
from typing import Dict, List, Set, Tuple

try:
    from openpyxl import Workbook
    from openpyxl.utils import get_column_letter
except Exception:
    Workbook = None

# ---------- utils ----------
def _strip_quotes(name: str) -> str:
    return name.replace('"', '').replace("`", "").strip()

def _norm_ident(raw: str) -> str:
    s = _strip_quotes(raw)
    s = re.sub(r'\s*\.\s*', '.', s)
    s = re.sub(r'\s+', ' ', s).strip()
    return s.lower()

def _has_schema(obj: str) -> bool:
    return '.' in obj

# ---------- regex ----------
RE_INSERT_INTO = re.compile(r'(?is)\binsert\s+into\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_CREATE_TBL  = re.compile(r'(?is)\bcreate\s+table\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_CREATE_VIEW = re.compile(r'(?is)\bcreate\s+(?:or\s+replace\s+)?(?:materialized\s+)?view\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_MERGE_INTO  = re.compile(r'(?is)\bmerge\s+into\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_DELETE_FROM = re.compile(r'(?is)\bdelete\s+from\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')

# UPDATE tgt SET ... FROM src
RE_UPDATE_FROM = re.compile(
    r'(?is)\bupdate\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)\s+set\b.*?\bfrom\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)',
    re.S
)

# Источники: FROM/JOIN, но не 'IS DISTINCT FROM'
RE_FROM_JOIN = re.compile(r'(?is)(?<!\bdistinct\s)\b(?:from|join)\s+([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)')
RE_USING     = re.compile(r'(?is)\busing\s*\(\s*([a-z0-9_."`$]+\s*(?:\.\s*[a-z0-9_."`$]+)?)\s*\)')

# CTE
RE_WITH_CTE = re.compile(r'(?is)\bwith\s+(.+?)\bselect\b', re.S)
RE_CTE_NAME = re.compile(r'(?is)\b([a-z0-9_."`$]+)\s+as\s*\(')

# comments
RE_LINE_COMMENT  = re.compile(r'--.*?$', re.M)
RE_BLOCK_COMMENT = re.compile(r'/\*.*?\*/', re.S)

def _remove_comments(sql: str) -> str:
    sql = RE_BLOCK_COMMENT.sub(' ', sql)
    sql = RE_LINE_COMMENT.sub(' ', sql)
    return sql

def _collect_cte_names(sql: str) -> Set[str]:
    names: Set[str] = set()
    for with_block in RE_WITH_CTE.findall(sql):
        for m in RE_CTE_NAME.finditer(with_block):
            nm = _norm_ident(m.group(1))
            names.add(_strip_quotes(nm).split('.')[-1])
    return names

# ---------- parse one file ----------
def parse_sql(sql_text: str) -> Tuple[Set[str], Set[str]]:
    sql = _remove_comments(sql_text)
    cte_names = _collect_cte_names(sql)
    targets: Set[str] = set()
    sources: Set[str] = set()

    for rx in (RE_INSERT_INTO, RE_CREATE_TBL, RE_CREATE_VIEW, RE_MERGE_INTO, RE_DELETE_FROM):
        for m in rx.finditer(sql):
            obj = _norm_ident(m.group(1))
            if _has_schema(obj):
                targets.add(obj)

    for m in RE_FROM_JOIN.finditer(sql):
        tok = _norm_ident(m.group(1))
        if '(' in tok:    # функция/подзапрос
            continue
        if not _has_schema(tok):
            continue
        base = tok.split('.')[-1]
        if base in cte_names:
            continue
        sources.add(tok)

    for m in RE_USING.finditer(sql):
        tok = _norm_ident(m.group(1))
        if '(' in tok:
            continue
        if not _has_schema(tok):
            continue
        base = tok.split('.')[-1]
        if base in cte_names:
            continue
        sources.add(tok)

    for m in RE_UPDATE_FROM.finditer(sql):
        tgt = _norm_ident(m.group(1))
        src = _norm_ident(m.group(2))
        if _has_schema(tgt):
            targets.add(tgt)
        if _has_schema(src):
            base = src.split('.')[-1]
            if base not in cte_names:
                sources.add(src)

    sources -= targets   # не считаем self-source
    return targets, sources

# ---------- scan files ----------
def iter_proto_sql_files(root: str) -> List[str]:
    out = []
    for dirpath, _, filenames in os.walk(root):
        for fn in filenames:
            fl = fn.lower()
            if fl.startswith('proto_') and fl.endswith('.sql'):
                out.append(os.path.join(dirpath, fn))
    return sorted(out)

# ---------- lineage build ----------
def build_lineage(root: str) -> Dict[str, Dict]:
    lineage: Dict[str, Dict] = {}
    for path in iter_proto_sql_files(root):
        try:
            with open(path, 'r', encoding='utf-8', errors='ignore') as f:
                text = f.read()
        except Exception as e:
            print(f"[WARN] {path}: {e}")
            continue

        targets, sources = parse_sql(text)
        for tgt in targets:
            entry = lineage.setdefault(tgt, {"file": os.path.basename(path), "sources": set()})
            entry["file"] = entry.get("file") or os.path.basename(path)
            entry["sources"].update(sources)

    for tgt, data in lineage.items():
        data["sources"] = sorted(data["sources"])
    return lineage

# ---------- save JSON/CSV ----------
def save_json_csv(lineage: Dict[str, Dict], out_prefix: str) -> None:
    with open(f"{out_prefix}.json", 'w', encoding='utf-8') as jf:
        json.dump(lineage, jf, ensure_ascii=False, indent=2)

    with open(f"{out_prefix}.csv", 'w', encoding='utf-8') as cf:
        cf.write("target,source\n")
        for tgt, data in sorted(lineage.items()):
            if data["sources"]:
                for src in data["sources"]:
                    cf.write(f"{tgt},{src}\n")
            else:
                cf.write(f"{tgt},\n")

# ---------- Excel ----------
def _auto_width(ws):
    widths = {}
    for row in ws.iter_rows():
        for cell in row:
            val = "" if cell.value is None else str(cell.value)
            widths[cell.column_letter] = max(widths.get(cell.column_letter, 0), len(val))
    for col, w in widths.items():
        ws.column_dimensions[col].width = min(max(w + 2, 12), 80)

def save_excel(lineage: Dict[str, Dict], xlsx_path: str) -> None:
    if Workbook is None:
        print("[WARN] openpyxl не установлен — XLSX пропущен")
        return

    wb = Workbook()

    # 1) edges — по одной строке на каждую пару target-source
    ws_edges = wb.active
    ws_edges.title = "edges"
    ws_edges.append(["target", "source", "target_file"])
    for tgt, data in sorted(lineage.items()):
        file_name = data.get("file", "")
        if data["sources"]:
            for src in data["sources"]:
                ws_edges.append([tgt, src, file_name])
        else:
            ws_edges.append([tgt, "", file_name])
    _auto_width(ws_edges)

    # 2) targets_sources — то же, сгруппировано по target (без склейки в одну ячейку)
    ws_ts = wb.create_sheet("targets_sources")
    ws_ts.append(["target", "source"])
    for tgt, data in sorted(lineage.items()):
        if data["sources"]:
            for src in data["sources"]:
                ws_ts.append([tgt, src])
        else:
            ws_ts.append([tgt, ""])
    _auto_width(ws_ts)

    # 3) usage_summary — сводная по источникам
    ws_sum = wb.create_sheet("usage_summary")
    ws_sum.append(["schema", "table", "full_name", "used_as_source_count"])
    cnt = Counter()
    for data in lineage.values():
        cnt.update(data["sources"])
    for full, n in sorted(cnt.items(), key=lambda x: (-x[1], x[0])):
        sch, tbl = (full.split('.', 1) + [""])[:2] if '.' in full else ("", full)
        ws_sum.append([sch, tbl, full, n])
    _auto_width(ws_sum)

    wb.save(xlsx_path)
    print(f"[OK] XLSX → {xlsx_path}")

# ---------- CLI ----------
def main():
    ap = argparse.ArgumentParser(description="Extract lineage from proto_*.sql into CSV/JSON/XLSX")
    ap.add_argument("root", nargs="?", default=os.getcwd(), help="Корень проекта (по умолчанию — текущая папка)")
    ap.add_argument("--out", default="lineage", help="Префикс для JSON/CSV (по умолчанию: lineage)")
    ap.add_argument("--xlsx", default="lineage.xlsx", help="Путь к XLSX (по умолчанию: lineage.xlsx)")
    args = ap.parse_args()

    root = os.path.abspath(args.root)
    print(f"[INFO] root: {root}")
    lineage = build_lineage(root)

    save_json_csv(lineage, args.out)
    save_excel(lineage, args.xlsx)

    print(f"[DONE] Targets: {len(lineage)}")

if __name__ == "__main__":
    main()
