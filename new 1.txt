CREATE TEMP TABLE euro_wh_month_weight
ON COMMIT DROP AS
SELECT
    warehouse_code,
    date_trunc('month', dt_storage_end) AS dt_storage_end_month,
    SUM(sales_bundle_gross_weight) AS warehouse_weight_month_end
FROM dm_calc.storage_sales_bundles_weight
WHERE storage_cost_calculation_type_name = 'Логика Европа'
GROUP BY 1,2
DISTRIBUTED BY (warehouse_code);

DROP TABLE IF EXISTS for_1079_euro;

CREATE TEMP TABLE for_1079_euro
ON COMMIT DROP
AS
WITH base_euro AS (
    SELECT
        b.*,
        date_trunc('month', b.dt_storage_end) AS dt_storage_end_month
    FROM dm_calc.storage_sales_bundles_weight b
    WHERE b.storage_cost_calculation_type_name = 'Логика Европа'
),

candidates AS (
    SELECT
        b.rn,

        f.purchase_contract_code,
        f.position_line_item_code,

        f.konws,
        f.service_code,
        f.price_uom_code,

        w.warehouse_weight_month_end,

        -- количество месяцев хранения
        CASE
            WHEN extract(day from b.dt_storage_end) < extract(day from b.dt_storage_start)
            THEN (extract(year from b.dt_storage_end) - extract(year from b.dt_storage_start)) * 12
               + (extract(month from b.dt_storage_end) - extract(month from b.dt_storage_start))
            ELSE (extract(year from b.dt_storage_end) - extract(year from b.dt_storage_start)) * 12
               + (extract(month from b.dt_storage_end) - extract(month from b.dt_storage_start)) + 1
        END AS month_difference_2,

        f.netpr,

        -- спец. расчёты
        CASE
            WHEN b.storage_cost_special_calculation_type_code = '4'
                THEN ceil(b.storage_duration_total_calendar_days / 7::numeric)
            WHEN b.storage_cost_special_calculation_type_code = '5'
                THEN ceil(b.storage_duration_total_calendar_days / 30::numeric)
        END AS special,

        f.currency_dogovor,
        f.datab,
        f.datbi,

        -- поля приоритета
        f.order_euro_svh,
        f.kopos,
        f.kbetr,

        -- совпадение формы / услуги (без EU%-картезиана)
        CASE
            WHEN param_shape.range_low_value IS NOT NULL
             AND param_srv.range_low_value   IS NOT NULL THEN 1
            WHEN param_shape.range_low_value IS NULL
             AND param_srv.range_low_value   IS NULL     THEN 1
            ELSE 0
        END AS match_score

    FROM base_euro b

    -- корректный физический вес склада за месяц
    LEFT JOIN euro_wh_month_weight w
        ON w.warehouse_code = b.warehouse_code
       AND w.dt_storage_end_month = b.dt_storage_end_month

    -- --------------------------------------------------------
    -- параметры ветвления услуги
    -- --------------------------------------------------------
    LEFT JOIN dict_dds.settings_and_parameters_sap lifnr_param
        ON lifnr_param.abap_program_code = '/RUSAL/ZLE112M'
       AND lifnr_param.case_code = 'PERET_FRN'
       AND lifnr_param.parameter_code = 'S_LIFNRA'
       AND lifnr_param.range_low_value = b.forwarder_code

    LEFT JOIN dict_dds.settings_and_parameters_sap form_param
        ON form_param.abap_program_code = '/RUSAL/SD4553M'
       AND form_param.parameter_code = 'S_FORMSTEU'
       AND form_param.range_low_value = b.material_shape_code

    -- --------------------------------------------------------
    -- базовый договор
    -- --------------------------------------------------------
    JOIN for_1079_base f
        ON f.purchase_document_subtype_code = 'ZDTL'
       AND (
            (lifnr_param.range_low_value IS NOT NULL AND f.zzsrvpos = 'S49')
         OR (lifnr_param.range_low_value IS NULL AND form_param.range_low_value IS NOT NULL AND f.tmp_1 IS NOT NULL)
         OR (lifnr_param.range_low_value IS NULL AND form_param.range_low_value IS NULL     AND f.tmp_2 IS NOT NULL)
       )
       AND f.bukrs = b.unit_balance_code
       AND f.lifnr = b.forwarder_code
       AND f.werks = b.plant_code
       AND (f.zzport = b.port_of_discharge_code OR f.zzport IS NULL)
       AND (f.zzforma = b.material_shape_code OR f.zzforma IS NULL)
       AND (f.zzet_tarif = b.etsng_code OR f.zzet_tarif IS NULL)
       AND (f.sdabw = b.transport_type_code OR f.sdabw IS NULL)
       AND (f.zzterminal = b.warehouse_code OR f.zzterminal IS NULL)
       AND b.dt_for_price_search_in_purchase_contract BETWEEN f.dt_valid_from AND f.dt_valid_to
       AND b.dt_for_price_search_in_purchase_contract BETWEEN f.datab AND f.datbi

    -- --------------------------------------------------------
    -- совпадение формы и услуги
    -- --------------------------------------------------------
    LEFT JOIN dict_dds.settings_and_parameters_sap param_shape
        ON param_shape.abap_program_code = '/RUSAL/SD4553M'
       AND param_shape.parameter_code = 'FORMSTEU'
       AND param_shape.range_low_value = b.material_shape_code

    LEFT JOIN dict_dds.settings_and_parameters_sap param_srv
        ON f.tmp_2 IS NOT NULL
       AND param_srv.abap_program_code = '/RUSAL/SD4553M'
       AND param_srv.parameter_code = 'S_USLSTC'
       AND param_srv.range_low_value = f.zzsrvpos
),

ranked AS (
    SELECT
        c.*,
        row_number() OVER (
            PARTITION BY c.rn
            ORDER BY
                c.order_euro_svh,
                c.kopos,
                c.kbetr,
                c.match_score DESC,
                c.purchase_contract_code,
                c.position_line_item_code
        ) AS rn_best
    FROM candidates c
)

SELECT
    rn,
    purchase_contract_code,
    position_line_item_code,
    konws,
    service_code,
    price_uom_code,
    warehouse_weight_month_end,
    month_difference_2,
    netpr,
    special,
    currency_dogovor,
    datab,
    datbi
FROM ranked
WHERE rn_best = 1
DISTRIBUTED REPLICATED;
