create table if not exists dict_dds.material_texts(
	material_code 		varchar(54)  not null,
	language_code 		varchar(3)	 not null,
	
	material_name 		varchar(120) null,
	material_full_name 	varchar(780) null,
	
	dttm_inserted 		timestamp	 not null,
	dttm_updated 		timestamp  	 not null,
	job_name 			varchar(50)  not null default 'airflow',
	deleted_flag 		bool 		 not null default false
)
with (
    appendonly = true,
    orientation = column,
    compresstype = zstd,
    compresslevel = 3
)
distributed by (material_code, language_code);

from base
	left join dict_dds.plant_and_subsidiary as p
 		on base.plant_code = p.plant_code
	left join dict_dds.material_texts as mt
		on base.material_code = mt.material_code 
		and mt.language_code = 'R'
