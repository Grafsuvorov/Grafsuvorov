select
    general_ledger_account_code,
    unit_balance_code,
    accounting_document_code,
    fiscal_year,
    count(*) as cnt
from dm_calc.accounting_document_tax_classification
group by
    general_ledger_account_code,
    unit_balance_code,
    accounting_document_code,
    fiscal_year
having count(*) > 1
order by cnt desc
limit 100;
