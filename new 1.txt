CREATE TEMP TABLE warehouse_geo AS
SELECT DISTINCT
    h.transport_hub_code,
    c.country_full_name,
    mr.market_region1_name,
    CASE WHEN c.country_full_name='Россия' THEN 'R' ELSE 'E' END AS lang_code
FROM hub h
LEFT JOIN hub_country hc   ON hc.address_code = h.address_code
LEFT JOIN dict_dds.country c ON c.country_code = hc.country_code
LEFT JOIN dict_dds.market_region1_texts mr 
       ON mr.market_region1_code = c.market_region1_code
      AND mr.language_code='R'
DISTRIBUTED BY (transport_hub_code);
