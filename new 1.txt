-- =========================================
-- 0) Чистим временные объекты
-- =========================================
DROP TABLE IF EXISTS cr_next;
DROP TABLE IF EXISTS mlcd_agg;
DROP TABLE IF EXISTS pp_base;
DROP TABLE IF EXISTS cr_curr_small;
DROP TABLE IF EXISTS pp_join_keys;

-- ===========================================================
-- 1) CKMLCR (следующий месяц): salk3 на 1-е число следующего месяца
-- ===========================================================
CREATE TEMP TABLE cr_next ON COMMIT DROP AS
SELECT
    c.kalnr,
    (make_date(c.bdatj::int, substring(c.poper, 2, 2)::int, 1) + INTERVAL '1 month')::date AS dt_posting,
    MAX(c.salk3) AS salk3         -- на случай дублей
FROM ods.ckmlcr_ral c
WHERE c.untper = '000'
  AND c.curtp  = '10'
GROUP BY c.kalnr,
         (make_date(c.bdatj::int, substring(c.poper, 2, 2)::int, 1) + INTERVAL '1 month')::date
DISTRIBUTED BY (kalnr);
ANALYZE cr_next;

-- =====================================
-- 2) MLCD агрегаты по (kalnr,bdatj,poper)
-- =====================================
CREATE TEMP TABLE mlcd_agg ON COMMIT DROP AS
SELECT
    m.kalnr, m.bdatj, m.poper,
    SUM(m.salk3)  AS salk3,
    SUM(m.estprd) AS estprd,
    SUM(m.mstprd) AS mstprd
FROM ods.mlcd_ral m
WHERE m.curtp = '10'
  AND m.categ IN ('ZU','VP','PC')
GROUP BY m.kalnr, m.bdatj, m.poper
DISTRIBUTED BY (kalnr);
ANALYZE mlcd_agg;

-- =================================================
-- 3) pp_base: базовые поля + dt_valid_to
--    (формируем один раз, чтобы не делать self-join в финале)
-- =================================================
CREATE TEMP TABLE pp_base ON COMMIT DROP AS
WITH raw AS (
  SELECT
    p.kalnr,
    p.bdatj,
    p.poper,
    make_date(p.bdatj::int, substring(p.poper, 2, 2)::int, 1) AS dt_valid_from,
    p.umkumo, p.abkumo, p.zukumo, p.lbkum
  FROM ods.ckmlpp_ral p
  WHERE p.untper = '000'
),
dates AS (
  SELECT
    r.kalnr,
    r.dt_valid_from,
    COALESCE(
      LEAD(r.dt_valid_from) OVER (PARTITION BY r.kalnr ORDER BY r.dt_valid_from) - INTERVAL '1 day',
      DATE '2299-12-31'
    )::date AS dt_valid_to
  FROM (SELECT DISTINCT kalnr, dt_valid_from FROM raw) r
)
SELECT r.kalnr, r.bdatj, r.poper, r.dt_valid_from, d.dt_valid_to,
       r.umkumo, r.abkumo, r.zukumo, r.lbkum
FROM raw   r
JOIN dates d
  ON d.kalnr = r.kalnr AND d.dt_valid_from = r.dt_valid_from
DISTRIBUTED BY (kalnr);
ANALYZE pp_base;

-- ==========================================================
-- 4) Узкий CKMLCR: только нужные поля и только ключи из pp_base
-- ==========================================================
CREATE TEMP TABLE cr_curr_small ON COMMIT DROP AS
SELECT
  c.kalnr, c.bdatj, c.poper,
  c.vprsv, c.stprs, c.peinh, c.pvprs,
  c.abprd_o, c.abprd_mo,
  c.salk3
FROM ods.ckmlcr_ral c
JOIN (SELECT DISTINCT kalnr, bdatj, poper FROM pp_base) k
  ON  k.kalnr = c.kalnr
  AND k.bdatj = c.bdatj
  AND k.poper = c.poper
WHERE c.untper = '000'
  AND c.curtp  = '10'
DISTRIBUTED BY (kalnr);
ANALYZE cr_curr_small;

-- =================================================================
-- 5) Узкая стадия объединения источников (только ON, без USING)
-- =================================================================
CREATE TEMP TABLE pp_join_keys ON COMMIT DROP AS
SELECT
  pp.kalnr,
  pp.bdatj,
  pp.poper,
  pp.dt_valid_from,
  pp.dt_valid_to,
  pp.umkumo,
  pp.abkumo,
  pp.zukumo,
  pp.lbkum,

  cr.salk3  AS salk3_curr,     -- для rule2_next_date
  n.salk3   AS salk3_next,     -- для rule2_prev_date
  ma.salk3  AS ma_salk3,
  ma.estprd AS ma_estprd,
  ma.mstprd AS ma_mstprd
FROM pp_base pp
LEFT JOIN cr_curr_small cr
  ON cr.kalnr = pp.kalnr AND cr.bdatj = pp.bdatj AND cr.poper = pp.poper
LEFT JOIN cr_next n
  ON n.kalnr = pp.kalnr AND n.dt_posting = pp.dt_valid_from
LEFT JOIN mlcd_agg ma
  ON ma.kalnr = pp.kalnr AND ma.bdatj = pp.bdatj AND ma.poper = pp.poper
DISTRIBUTED BY (kalnr);
ANALYZE pp_join_keys;

-- ==========================
-- 6) Финал (SELECT / INSERT)
-- ==========================
-- Прогон как SELECT:
WITH final AS (
  SELECT
    k.kalnr       AS calculation_code,
    k.dt_valid_from,
    k.dt_valid_to,

    c.vprsv  AS price_valuation_type_code,
    c.stprs  AS standard_price_amount,
    c.peinh  AS price_unit_code,
    c.pvprs  AS moving_average_price_amount,

    /* rule3 */
    CASE WHEN (k.umkumo + k.abkumo + k.zukumo) = 0 THEN 0
         ELSE TRUNC(
                CAST(ROUND(
                      (c.stprs / NULLIF(c.peinh,0)) * (k.umkumo + k.abkumo)
                    + c.abprd_o + c.abprd_mo
                    + COALESCE(k.ma_salk3,0) + COALESCE(k.ma_estprd,0) + COALESCE(k.ma_mstprd,0)
                , 2) AS NUMERIC(40,20))
                / (k.umkumo + k.abkumo + k.zukumo), 15)
    END AS average_weighted_stock_price_rule3_amount,

    /* rule2_prev_date */
    CASE WHEN (k.abkumo + k.umkumo + k.zukumo) <> 0
         THEN TRUNC(
                CAST(
                    COALESCE(k.salk3_next,0)
                  + COALESCE(k.ma_salk3,0) + COALESCE(k.ma_estprd,0) + COALESCE(k.ma_mstprd,0)
                AS NUMERIC(40,20))
                / (k.abkumo + k.umkumo + k.zukumo), 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_prev_date_amount,

    /* rule2_next_date */
    CASE WHEN k.lbkum <> 0
         THEN TRUNC(CAST(COALESCE(k.salk3_curr,0) AS NUMERIC(40,20)) / k.lbkum, 15)
         ELSE 0
    END AS average_weighted_stock_price_rule2_next_date_amount
  FROM pp_join_keys k
  LEFT JOIN cr_curr_small c
    ON c.kalnr = k.kalnr AND c.bdatj = k.bdatj AND c.poper = k.poper
)
SELECT
  f.*,
  h.price_rule_code
FROM final f
LEFT JOIN dds.material_ledger_header h
  ON h.calculation_code = f.calculation_code
;
