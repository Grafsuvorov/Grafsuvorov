----------------------------------------------------------------------------------------------------------

drop table if exists bill_search;
create temporary table bill_search as( --on commit drop as (
with bill_step1 as (
	select
		a1.delivery_code,
		a2.bill_of_lading_code
	from
		(	select
				perw.delivery_code,
				sbdr1.sales_bundle_code
			from dds.transportation_service_invoice_subposition as perw
				left join dds.transportation_service_invoice_header as perh
					on perw.plant_code = perh.plant_code and
		   	   		   perw.transportation_service_invoice_code = perh.transportation_service_invoice_code
				left join dds.bill_of_lading as vbsk_src
					on vbsk_src.bill_of_lading_code = perw.bill_of_lading_code
				join dds.sales_bundle_and_delivery_relationship as sbdr1
					on sbdr1.delivery_code = perw.delivery_code
			where (vbsk_src.bill_of_lading_type_code = 'Й' or perw.bill_of_lading_code is null)
  	  		  and perh.dt_transportation_service_invoice >= '2023-01-01'
  	  		  and sbdr1.is_deleted_code is null
			group by
				perw.delivery_code,
				sbdr1.sales_bundle_code) as a1
	join
		(	select
				sbdr2.sales_bundle_code,
				sbdr2.delivery_code,
				vbsk_trg.bill_of_lading_code
			from dds.sales_bundle_and_delivery_relationship as sbdr2		--по химиям ищем полный список поставок
				join dds.delivery_document_header as likp
					on likp.delivery_code = sbdr2.delivery_code and
	   	   	   	   	   likp.delivery_type_code = 'EL'
	   			join dds.logistics_document_and_delivery_relationship as vbss
					on vbss.delivery_code = likp.delivery_code
				join dds.bill_of_lading as vbsk_trg
					on vbsk_trg.bill_of_lading_code = vbss.logistics_document_code and
	   	   	   		   vbsk_trg.bill_of_lading_type_code = 'Y'
	   	   	where sbdr2.is_deleted_code is null
	   		group by
	   			sbdr2.sales_bundle_code,

		bill_step1.delivery_code,

		bill_step2.delivery_code,
