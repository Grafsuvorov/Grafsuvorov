CREATE TEMP TABLE dict_counterparty_address AS
SELECT c.counterparty_code,
       a.address_full_name,
       a.international_display_format_code
FROM dict_dds.counterparty c
LEFT JOIN dict_dds.address a ON c.address_code = a.address_code
WHERE a.international_display_format_code IS NULL;
