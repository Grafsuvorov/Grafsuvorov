def get_token_api(login_url, login_payload):
    login = requests.post(login_url, data=json.dumps(login_payload), headers={"Content-Type": "application/json"})

    if login.status_code == 200:
        bearer_token = login.json()["token"]
        return bearer_token
    else:
        raise Exception(f"Failed to get data: {login.status_code}: {login.text}")

def date_range(start_date, end_date, delta):
    current_date = start_date
    while current_date < end_date:
        yield current_date, min(current_date + timedelta(days=delta -1), end_date)
        current_date += timedelta(days=delta)

def get_data_api(**kwargs):
    api_dt_start = kwargs['params']['api_dt_start']
    api_dt_end = kwargs['params']['api_dt_end']
    api_url = kwargs['api_url']

    start_date = datetime.strptime(api_dt_start, "%m.%d.%Y %H:%M:%S")
    end_date = datetime.strptime(api_dt_end, "%m.%d.%Y %H:%M:%S")

    if LOAD_TYPE == 'full':
        start_date = start_date
    else:
        start_date = end_date - timedelta(days=1)

    bearer_token = get_token_api(api_url, kwargs['payload'])

    logger.info("Getting hist data from API (60 days)")

    headers = {"Authorization": f"Bearer {bearer_token}"}
    df_list = []

    hist_start = start_date
    hist_end = min(start_date + timedelta(days=59), end_date)

    for chunk_start, chunk_end in date_range(hist_start, hist_end, 60):
        params = {
            'l_date_beg': chunk_start.strftime("%m.%d.%Y %H:%M:%S"),
            'l_date_end': chunk_end.strftime("%m.%d.%Y %H:%M:%S")
        }
        resp = requests.get(api_url, params=params, headers=headers)
        resp.raise_for_status()
        data = resp.json()
        df_list.append(pd.DataFrame(data))

    if hist_end < end_date:
        next_start = hist_end + timedelta(days=1)
        next_end = end_date
        logger.info("Getting data from API (45 days)")

        for chunk_start, chunk_end in date_range(next_start, next_end, 45):
            params = {
                'l_date_beg': chunk_start.strftime("%m-%d-%Y %H:%M:%S"),
                'l_date_end': chunk_end.strftime("%m-%d-%Y %H:%M:%S")
            }
            resp = requests.get(api_url, params=params, headers=headers)
            resp.raise_for_status()
            data = resp.json()
            df_list.append(pd.DataFrame(data))

    full_df = pd.concat(df_list, ignore_index=True)

    return full_df

def save_to_parquet(df):
    df = df[COLUMNS]
    buffer = BytesIO()
    df.to_parquet(buffer, engine='pyarrow', index=False)
    buffer.seek(0)

    if LOAD_TYPE == 'full':
        Variable.set('load_type', 'incremental')

    return buffer
