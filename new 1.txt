create or replace function tech_etl.check_entity_backup_tables(
      p_entity_id integer,
      p_table_schema text,
      p_backup_date date default current_date
  )
  returns table (
      source_table text,
      backup_table text,
      status text,
      source_cnt bigint,
      backup_cnt bigint,
      diff_cnt bigint
  )
  language plpgsql
  as
  $$
  declare
      r record;
      v_backup_suffix text := to_char(p_backup_date, 'YYYY_MM_DD');
      v_backup_table_name text;
      v_cols text;
      v_source_cnt bigint;
      v_backup_cnt bigint;
      v_diff_cnt bigint;
  begin
      for r in
          select t.table_name
          from tech_etl.tables_meta t
          where t.entity_id = p_entity_id
            and t.table_schema = p_table_schema
          order by t.table_name
      loop
          v_backup_table_name := format('%s_backup_%s', r.table_name, v_backup_suffix);

          if not exists (
              select 1
              from information_schema.tables
              where table_schema = p_table_schema
                and table_name = v_backup_table_name
          ) then
              source_table := format('%s.%s', p_table_schema, r.table_name);
              backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
              status := 'backup_not_found';
              source_cnt := null;
              backup_cnt := null;
              diff_cnt := null;
              return next;
              continue;
          end if;

          select string_agg(format('%I', c.column_name), ', ' order by c.ordinal_position)
          into v_cols
          from information_schema.columns c
          where c.table_schema = p_table_schema
            and c.table_name = r.table_name
            and c.column_name not in (
                'dttm_inserted',
                'dttm_updated',
                'job_name',
                'deleted_flag'
            );

          if v_cols is null then
              source_table := format('%s.%s', p_table_schema, r.table_name);
              backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
              status := 'no_business_columns';
              source_cnt := null;
              backup_cnt := null;
              diff_cnt := null;
              return next;
              continue;
          end if;

          execute format(
              'select count(*) from %I.%I',
              p_table_schema,
              r.table_name
          )
          into v_source_cnt;

          execute format(
              'select count(*) from %I.%I',
              p_table_schema,
              v_backup_table_name
          )
          into v_backup_cnt;

          execute format(
              $sql$
              select count(*)
              from (
                  (select %s from %I.%I
                   except all
                   select %s from %I.%I)
                  union all
                  (select %s from %I.%I
                   except all
                   select %s from %I.%I)
              ) q
              $sql$,
              v_cols, p_table_schema, r.table_name,
              v_cols, p_table_schema, v_backup_table_name,
              v_cols, p_table_schema, v_backup_table_name,
              v_cols, p_table_schema, r.table_name
          )
          into v_diff_cnt;

          source_table := format('%s.%s', p_table_schema, r.table_name);
          backup_table := format('%s.%s', p_table_schema, v_backup_table_name);
          source_cnt := v_source_cnt;
          backup_cnt := v_backup_cnt;
          diff_cnt := v_diff_cnt;

          if v_source_cnt = v_backup_cnt and v_diff_cnt = 0 then
              status := 'ok';
          else
              status := 'mismatch';
          end if;

          return next;
      end loop;
  end;
  $$;

  Запуск:

  select *
  from tech_etl.check_entity_backup_tables(35, 'dm');
