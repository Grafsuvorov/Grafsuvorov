explain(verbose, format json)
select
    b.dt_posting,
    b.material_code,
    b.material_uom_code,
    b.plant_code,
    b.warehouse_code,
    b.batch_code,
    b.valuation_type_code,
    b.stock_type_code,
    b.sales_document_code,
    b.sales_document_position_line_item_code,
    ws.wbs_element_number                                 as wbs_element_code,      -- нормализуем код WBS в номер
    b.supplier_code,
    b.customer_code,

    b.total_stock_quantity,
    b.unit_balance_currency_code,
    b.material_costing_system_code,
    b.price_rule_code,
    b.dt_posting_next,

    pas.plant_short_name                                   as plant_name,
    pas.valuation_area_code                                as valuation_assessment_code,

    w.warehouse_name                                       as warehouse_name,
    a.title_code                                           as warehouse_type_code,
    tt.title_name                                          as warehouse_type_name,
    mbw.storage_bin_code                                   as storage_bin_code,

    mt.material_full_name                                  as material_full_name,
    m.material_kubal_code                                  as material_kubal_code,
    m.is_batch_mandatory                                   as is_batch_mandatory,
    m.material_type_code                                   as material_type_code,
    m.material_group_code                                  as material_group_code,
    m.sector_code                                          as sector_code,
    m.material_category_code                               as material_category_code,

    case
        when m.is_batch_mandatory is not null
         and map.object_class_code is not null
         and mb.material_detailed_name is not null
        then mb.material_detailed_name
        else mt.material_name
    end                                                    as material_short_name,

    mu.material_alternative_uom_code                       as material_alternative_uom_code,
    ueft.uom_external_format_code                          as material_alternative_uom_name,

    mbp.tnved_code                                         as material_tnved_eas_code,
    mbp.excise_tax_foreign_trade_control_code              as material_okpd2_code,

    mbp.purchase_group_code                                as purchase_responsible_department_code,
    mbp.material_planning_controller_code                  as material_planning_controller_code,
    mbp.head_of_production_code                            as head_of_production_code,

    mtt.material_type_name                                 as material_type_name,
    mgt.material_group_short_name                           as material_group_name,
    pg.purchase_group_name                                 as purchase_responsible_department_name,
    mpc.material_planning_controller_name                  as material_planning_controller_name,

    t1.currency_code                                       as holding_currency_code,

    ws.wbs_element_short_name                              as wbs_element_short_name,

    -- unit_balance_code (как у вас — через valuation_area, не меняю)
    vaar.unit_balance_code                                 as unit_balance_code,

    mct.material_category_name                             as material_category_name,

    -- контрагенты (расширенные имена)
    c1.counterparty_full_name                              as supplier_extended_name,
    c2.counterparty_full_name                              as customer_extended_name,

    -- тип неликвида (код)
    case
        when b.wbs_element_code is not null and ws.project_type_code = '99'
            then substring(ws.wbs_element_code, 6, 1)
        when b.wbs_element_code is null and b.valuation_type_code like 'U%'
            then mb.nonliquid_asset_type_code
        else null
    end                                                    as non_liquid_asset_type_code,

    -- коэффициент ЕИ→альт.ЕИ
    mb.uom_to_alternative_uom_coefficient_amount           as uom_to_alternative_uom_coefficient_amount,

    -- наименование базовой ЕИ
    t006a.uom_external_format_code                         as base_uom_name,

    -- служебное: проектный тип из справочника WBS пригодится позже
    ws.project_type_code                                   as wbs_project_type_code
from dm_calc.material_stock_balance b
cross join tmp_1 t1
left join dict_dds.wbs_element_master_data_detail ws
  on ws.wbs_element_code = b.wbs_element_code
left join dict_dds.plant_and_subsidiary pas
  on pas.plant_code = b.plant_code
left join dict_dds.warehouse w
  on w.warehouse_code = b.warehouse_code
 and w.plant_code     = b.plant_code
left join (
    select paw.plant_code, paw.warehouse_code, max(paw.address_code) as address_code
    from dict_dds.plant_and_warehouse_address_definition paw
    group by paw.plant_code, paw.warehouse_code
) pawad
  on pawad.plant_code     = b.plant_code
 and pawad.warehouse_code = b.warehouse_code
left join dict_dds.address a
  on a.address_code = pawad.address_code
left join dict_dds.title_texts tt
  on tt.title_code = a.title_code
 and tt.language_code = 'R'
left join dict_dds.material_by_warehouse mbw
  on mbw.material_code  = b.material_code
 and mbw.plant_code     = b.plant_code
 and mbw.warehouse_code = b.warehouse_code
left join dict_dds.material_texts mt
  on mt.material_code = b.material_code
 and mt.language_code = 'R'
left join dict_dds.material m
  on m.material_code = b.material_code
left join dict_dds.map_object_to_class map
  on map.object_class_code = m.material_classification_system_code
 and map.object_or_class_code = 'O'
 and map.class_type_code = (
      select range_low_value
      from dict_dds.settings_and_parameters_sap
      where abap_program_code = '/RUSAL/CO360M' and parameter_code = 'KLART'
 )
left join dds.material_batch mb
  on mb.material_code = b.material_code
 and mb.batch_code    = b.batch_code
left join (
    select mum.material_code, mum.material_alternative_uom_code
    from dict_dds.material_unit_of_measure mum
    left join dict_dds.internal_characteristic ich
      on ich.internal_characteristic_code = mum.internal_characteristic_code
    where ich.internal_characteristic_title_code = 'COEF_TRANS'
) mu
  on mu.material_code = b.material_code
left join dict_dds.uom_external_format_texts ueft
  on ueft.uom_code = mu.material_alternative_uom_code
 and ueft.language_code = 'R'
left join dict_dds.material_by_plant mbp
  on mbp.material_code = b.material_code
 and mbp.plant_code    = b.plant_code
left join dict_dds.material_type_texts mtt
  on mtt.material_type_code = m.material_type_code
 and mtt.language_code = 'R'
left join dict_dds.material_group_texts mgt
  on mgt.material_group_code = m.material_group_code
 and mgt.language_code = 'R'
left join dict_dds.purchase_group pg
  on pg.purchase_group_code = mbp.purchase_group_code
left join dict_dds.material_planning_controller mpc
  on mpc.plant_code = b.plant_code
 and mpc.material_planning_controller_code = mbp.material_planning_controller_code
left join dict_dds.valuation_area vaar
  on vaar.valuation_area_code = b.plant_code          -- не меняю логику
left join dict_dds.material_category_main_data_texts mct
  on mct.material_category_code = m.material_category_code
 and mct.language_code = 'R'
left join dict_dds.counterparty c1
  on c1.counterparty_code = b.supplier_code
left join dict_dds.counterparty c2
  on c2.counterparty_code = b.customer_code
left join dict_dds.uom_external_format_texts t006a
  on t006a.uom_code = b.material_uom_code
 and t006a.language_code = 'R'
distributed by (material_costing_system_code);
