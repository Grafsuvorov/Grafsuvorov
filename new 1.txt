------------------------------------------------------------
-- 5. BASE_1_5: продолжение логики на основе base
------------------------------------------------------------

CREATE TEMP TABLE base_1_5 --ON COMMIT DROP
AS
SELECT
    base.sales_bundle_code,
    base.plant_producer_delivery_code,
    base.plant_producer_delivery_position_code,
    base.initial_delivery_code,
    base.material_code,
    base.dt_shipment,
    base.railcar_code,
    base.transport_bill_code,
    base.uni,
    base.material_shape_code,
    base.dt_arrival_to_russian_port,
    base.dt_storage_end_in_release,
    base.transport_type_code,
    base.etsng_code,
    base.sales_bundle_net_weight,
    base.sales_bundle_gross_weight,
    base.batch_code,
    base.dt_arrival,
    base.receiving_plant_code,
    base.sales_delivery_code,
    base.transportation_inbound_delivery_code,
    base.delivery_for_storage_calculation_code,
    base.transportation_inbound_delivery_position_code,
    base.delivery_position_for_storage_calculation_code,
    base.forwarder_code,
    base.storage_cost_calculation_type_name,
    base.transport_departure_hub_code,
    base.transport_destination_hub_code,
    base.SD_000049,
    base.SD_000044,
    base.wwgsg_01_date,
    base.id_for_unique,
    base.vbeln_4798,
    base.SD_000054,
    base.wwgsg_01,

    /* ------- Логика outbound -------- */
    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000516 END AS transportation_outbound_delivery_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000897 END AS transportation_outbound_delivery_position_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000902 END AS is_final_transportation_stage_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000903 END AS transportation_stage_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000905 END AS storage_calculation_bundle_quantity,

    /* ------- Склад -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND f1047.transportation_stage_code = '25'
            THEN f1047.transport_route_departure_hub_code
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             THEN f1047.transport_route_destination_hub_code
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
             THEN f1047_2.transport_route_destination_hub_code
    END AS warehouse_code,

    tr_hub.is_seaport AS is_stored_in_russian_port_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.sales_delivery_code END AS delivery_in_final_release_code,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000482 END AS dt_final_release,

    /* ------- Bill of lading -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN base.SD_000040
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
            THEN base.SD_000047
    END AS bill_of_lading_code,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN base.SD_000041
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
            THEN base.SD_000048
    END AS bill_of_lading_number,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN base.SD_000042
        WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа')
            THEN base.SD_000049
    END AS dt_bill_of_lading,

    CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
         THEN base.SD_000067 END AS incoterms_code,

    CASE WHEN base.storage_cost_calculation_type_name = 'Логика Европа'
         THEN trscm.calculation_method_code END AS storage_cost_special_calculation_type_code,

    CASE WHEN base.storage_cost_calculation_type_name = 'Логика Азия'
         THEN trwsa.warehouse_storage_area_code END AS warehouse_storage_area_code,

    /* ------- plant_code -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк','Логика СВХ')
            THEN '1121'
        WHEN base.storage_cost_calculation_type_name = 'Логика Европа'
            THEN COALESCE(f1078.plant_code, '1521')
        WHEN base.storage_cost_calculation_type_name = 'Логика Азия'
            THEN COALESCE(zlet.unit_balance_of_storage_agreement_code, zlet2.unit_balance_of_storage_agreement_code)
    END AS plant_code,

    /* ------- unit_balance_code -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк','Логика СВХ')
            THEN '1120'
        WHEN base.storage_cost_calculation_type_name IN ('Логика Европа','Логика Азия')
            THEN valu.unit_balance_code
    END AS unit_balance_code,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN COALESCE(ekpa.counterparty_code, vbpa.supplier_code)
    END AS creditor_in_shipment_instruction_code,

    /* ------- dt_storage_start -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN base.dt_arrival_to_russian_port
        ELSE base.likp_lddat
    END AS dt_storage_start,

    /* ------- dt_shipped_from_warehouse -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
            THEN base.SD_000042
        ELSE sbudr2.dt_transportation_stage_start
    END AS dt_shipped_from_warehouse,

    /* ------- shipment_instruction_number -------- */
    CASE WHEN base.delivery_for_storage_calculation_code = base.transportation_inbound_delivery_code
         THEN NULL ELSE vbak.zznoi END AS shipment_instruction_number,

    /* ------- веса -------- */
    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             THEN base.sales_bundle_net_weight
        WHEN base.storage_cost_calculation_type_name IN ('Логика Европа','Логика Азия','Логика СВХ')
             AND zle.lifnr IS NOT NULL
            THEN base.sales_bundle_net_weight
        ELSE base.sales_bundle_gross_weight
    END AS weight_for_1090_umno,

    CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             THEN base.ntgew
        WHEN base.storage_cost_calculation_type_name IN ('Логика Европа','Логика Азия','Логика СВХ')
             AND zle.lifnr IS NOT NULL
            THEN base.ntgew
        ELSE base.btgew
    END AS weight_for_1090_razdel,

    /* ============================================================
       🔥 ГЛАВНОЕ ИСПРАВЛЕНИЕ: prev / next lddat
       ============================================================ */

    base.lddat_4798,

    /* prev_lddat_4798 корректно: date + interval → date */
    CASE
        WHEN base.prev_lddat_4798 IS NOT NULL
            THEN (base.prev_lddat_4798 + INTERVAL '1 day')::date
        ELSE base.dt_transportation_stage_start_p::date
    END AS prev_lddat_4798,

    /* next_lddat_4798 корректно */
    CASE
        WHEN base.next_lddat_4798 IS NOT NULL
            THEN base.next_lddat_4798::date
        ELSE base.dt_transportation_stage_start_r::date
    END AS next_lddat_4798,

    /* ФИНАЛЬНЫЙ row_number */
    ROW_NUMBER() OVER (
        PARTITION BY base.id_for_unique,
                     CASE
                         WHEN base.prev_lddat_4798 IS NOT NULL
                             THEN (base.prev_lddat_4798 + INTERVAL '1 day')::date
                         ELSE base.dt_transportation_stage_start_p::date
                     END
        ORDER BY base.vbeln_4798 DESC,
                 sbudr2.dt_transportation_stage_start
    ) AS rn

FROM base
LEFT JOIN dict_dds.tech_zle112t37t3 zle
    ON zle.lifnr = base.forwarder_code
   AND zle.brutto IS NULL

LEFT JOIN for_1078 f1078
    ON f1078.counterparty_code = base.forwarder_code
   AND base.storage_cost_calculation_type_name = 'Логика Европа'

LEFT JOIN dedopo_2 d2
    ON d2.delivery_code = base.plant_producer_delivery_code

LEFT JOIN vbak
    ON vbak.vbeln = d2.sales_document_code

LEFT JOIN for_1047_1 f1047
    ON f1047.sales_bundle_code = base.sales_bundle_code

LEFT JOIN dds.sales_bundle_and_delivery_relationship f1047_2
    ON f1047_2.sales_bundle_code = base.sales_bundle_code
   AND f1047_2.delivery_code = base.delivery_for_storage_calculation_code
   AND f1047_2.is_deleted_code IS NULL
   AND base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')

LEFT JOIN trscm
    ON trscm.counterparty_code = base.forwarder_code

LEFT JOIN dict_dds.transportation_warehouse_storage_area trwsa
    ON trwsa.plant_code = base.receiving_plant_code
   AND trwsa.transport_hub_code = CASE
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             AND f1047.transportation_stage_code = '25'
             THEN f1047.transport_route_departure_hub_code
        WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
             THEN f1047.transport_route_destination_hub_code
        ELSE f1047_2.transport_route_destination_hub_code END
   AND trwsa.transportation_stage_code =
        CASE WHEN base.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа','Логика СВХ')
             THEN base.SD_000903 END

LEFT JOIN sbudr_1 sb1
    ON sb1.sales_bundle_code = base.sales_bundle_code
   AND sb1.delivery_code = base.delivery_for_storage_calculation_code

LEFT JOIN dds.sales_bundle_and_delivery_relationship sbudr2
    ON sbudr2.sales_bundle_code = base.sales_bundle_code
   AND sbudr2.batch_code = sb1.batch_code
   AND sbudr2.transportation_stage_code > sb1.transportation_stage_code
   AND sbudr2.transport_route_departure_hub_code = sb1.transport_route_destination_hub_code
   AND sbudr2.dt_transportation_stage_start >= base.likp_lddat
   AND sbudr2.is_deleted_code IS NULL

LEFT JOIN zlet43_1 zlet
    ON zlet.plant_code = base.receiving_plant_code
   AND zlet.counterparty_code = base.forwarder_code

LEFT JOIN dict_dds.map_trader_to_unit_balance_of_storage_agreement zlet2
    ON zlet2.plant_code = base.receiving_plant_code
   AND zlet2.counterparty_code IS NULL

LEFT JOIN dict_dds.plant_and_subsidiary pas
    ON pas.plant_code =
        CASE
            WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк','Логика СВХ')
                THEN '1121'
            WHEN base.storage_cost_calculation_type_name = 'Логика Европа'
                THEN COALESCE(f1078.plant_code, '1521')
            ELSE COALESCE(zlet.unit_balance_of_storage_agreement_code, zlet2.unit_balance_of_storage_agreement_code)
        END

LEFT JOIN dict_dds.valuation_area valu
    ON valu.valuation_area_code = pas.valuation_area_code

LEFT JOIN ekpa
    ON ekpa.objek = vbak.zznoi

LEFT JOIN dds.sales_document_counterparty_role vbpa
    ON vbpa.sales_document_code = vbak.zznoi
   AND vbpa.counterparty_role_code = 'ZJ'

LEFT JOIN dict_dds.transport_hub tr_hub
    ON tr_hub.transport_hub_code =
        CASE
            WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
                 AND f1047.transportation_stage_code = '25'
                THEN f1047.transport_route_departure_hub_code
            WHEN base.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк')
                THEN f1047.transport_route_destination_hub_code
        END

DISTRIBUTED BY (sales_bundle_code);
