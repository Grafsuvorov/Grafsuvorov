select
    table_schema,
    table_name,
    entity_name,
    depends_on
  from tech_etl.tables_meta
  where table_schema = 'dm_calc'
    and table_name = 'ТВОЯ_TARGET_ТАБЛИЦА';

  Проверка последних успешных загрузок source и target:

  select
    tm.table_schema,
    tm.table_name,
    max(l.loading_finish_dttm) as last_success_finish
  from tech_etl.log_objects_loading_history l
  join tech_etl.tables_meta tm
    on tm.table_id = l.object_id
  where l.object_type = 'table'
    and l.loading_state = 'SUCCESS'
    and (
      (tm.table_schema = 'sales_mm' and tm.table_name = 'ТВОЯ_SOURCE_ТАБЛИЦА')
      or
      (tm.table_schema = 'dm_calc' and tm.table_name = 'ТВОЯ_TARGET_ТАБЛИЦА')
    )
  group by tm.table_schema, tm.table_name
  order by tm.table_schema, tm.table_name;

  Проверка обоих table_id:

  select table_id, table_schema, table_name
  from tech_etl.tables_meta
  where (table_schema = 'sales_mm' and table_name = 'ТВОЯ_SOURCE_ТАБЛИЦА')
     or (table_schema = 'dm_calc' and table_name = 'ТВОЯ_TARGET_ТАБЛИЦА');
