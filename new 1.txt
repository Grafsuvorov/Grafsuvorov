#!/usr/bin/env python3
  import json
  import sys
  import urllib.request

  BASE_URL = "http://rgm-s-dwhapp01.hq.root.ad:15312"

  def main():
      if len(sys.argv) != 3:
          print("Usage: python get_keys.py <schema> <table>")
          sys.exit(1)

      schema = sys.argv[1]
      table = sys.argv[2]
      url = f"{BASE_URL}/api/card/{schema}/{table}"

      with urllib.request.urlopen(url) as resp:
          data = json.load(resp)

      result = {
          "schema": schema,
          "table": table,
          "key_attributes": data.get("key_attributes", []),
      }
      print(json.dumps(result, ensure_ascii=False, indent=2))

  if __name__ == "__main__":
      main()
