3. Сколько строк реально остаётся после отбора последней версии
with last_dt as (
    select
        "AUFNR",
        max(dt_insert) as max_dt_insert
    from stg."RAL_IS2_TORO2_ORD_HDR_OPR"
    group by 1
)
select count(*)
from stg."RAL_IS2_TORO2_ORD_HDR_OPR" t
join last_dt ld
  on t."AUFNR" = ld."AUFNR"
 and t.dt_insert = ld.max_dt_insert;
