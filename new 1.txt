CREATE TEMP TABLE asia_candidates
AS
SELECT
    b.rn,

    f.purchase_contract_code,
    f.position_line_item_code,

    f.konws,
    f.service_code,
    f.price_uom_code,
    f.currency_dogovor,
    f.datab,
    f.datbi,

    -- приоритеты (как в order_asia)
    f.order_asia,
    f.kopos,
    f.kbetr

FROM base_asia b
JOIN for_1079_base f
    ON f.purchase_document_subtype_code = 'ZDTL'
   AND f.zzsrvpos = 'S49'                     -- жёстко, как у тебя
   AND f.bukrs = b.unit_balance_code
   AND f.lifnr = b.forwarder_code
   AND f.werks = b.plant_code

   -- nullable-поля (оставляем, но уже на маленьком base)
   AND (f.zzport    = b.port_of_discharge_code     OR f.zzport IS NULL)
   AND (f.zzforma   = b.material_shape_code        OR f.zzforma IS NULL)
   AND (f.zzet_tarif= b.etsng_code                 OR f.zzet_tarif IS NULL)
   AND (f.sdabw     = b.transport_type_code        OR f.sdabw IS NULL)
   AND (f.zzterminal= b.warehouse_code             OR f.zzterminal IS NULL)
   AND (f.vzzzone   = b.warehouse_storage_area_code OR f.vzzzone IS NULL)

   -- закрытие периодов (КЛЮЧЕВО)
   AND b.dt_for_price_search_in_purchase_contract
       BETWEEN f.dt_valid_from AND f.dt_valid_to
   AND b.dt_for_price_search_in_purchase_contract
       BETWEEN f.datab AND f.datbi
DISTRIBUTED BY (rn);
📌 Ключевое отличие от старой Азии
→ оконки нет
→ join работает на резко меньшем base
→ OR уже не раздувает миллиард строк

CREATE TEMP TABLE asia_ranked
AS
SELECT
    c.*,
    row_number() OVER (
        PARTITION BY c.rn
        ORDER BY
            c.order_asia,
            c.kopos,
            c.kbetr,
            c.purchase_contract_code,
            c.position_line_item_code
    ) AS rn_best
FROM asia_candidates c
DISTRIBUTED BY (rn);
