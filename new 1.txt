import os, json, requests, urllib3
  urllib3.disable_warnings()

  issue = "DWH-10867"
  url = os.environ["YOUTRACK_URL"]
  token = os.environ["YOUTRACK_TOKEN"]

  fields = (
      "id,idReadable,"
      "customFields(id,name,$type,"
      "value(id,name,login,fullName,text,presentation))"
  )

  r = requests.get(
      f"{url}/api/issues/{issue}",
      headers={"Authorization": f"Bearer {token}", "Accept": "application/json"},
      params={"fields": fields},
      verify=False,
      timeout=60,
  )
  print(r.status_code)
  data = r.json()
  for cf in data.get("customFields", []):
      print(json.dumps(cf, ensure_ascii=False))
