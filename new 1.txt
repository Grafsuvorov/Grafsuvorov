-- ===========================
-- 1) Большой union заказов (распределение по key джойна)
-- ===========================
CREATE TEMP TABLE order_union (
    order_type_code      varchar(4),
    order_code           varchar(12),
    order_category_code  varchar
)
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED BY (order_code);

INSERT INTO order_union
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_toro
UNION ALL
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_production
UNION ALL
SELECT order_type_code, order_code, order_category_code FROM dict_dds.order_controlling;

ANALYZE order_union;

-- ===========================
-- 2) Маленькие справочники-параметры (джойним, а не EXISTS)
-- ===========================
CREATE TEMP TABLE params_z_matkl AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code='ZMM2968M_BW' AND parameter_code='Z_MATKL'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_bwrt AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code='/RUSAL/CO312K' AND parameter_code='BWRT_ZKP'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_auart AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code='/RUSAL/CO706M' AND parameter_code='AUART_BU'
DISTRIBUTED RANDOMLY;

CREATE TEMP TABLE params_autyp AS
SELECT range_low_value
FROM dict_dds.settings_and_parameters_sap
WHERE abap_program_code='/RUSAL/CO706M' AND parameter_code='AUTYP_BU'
DISTRIBUTED RANDOMLY;

ANALYZE params_z_matkl;
ANALYZE params_bwrt;
ANALYZE params_auart;
ANALYZE params_autyp;

-- ===========================
-- 3) Ядро движений (ТОЛЬКО нужные поля) + предрасчёт qty11/qty22_base
--    Важное: распределяем по (plant_code, material_code) — это ключ джойна на material_by_plant
-- ===========================
CREATE TEMP TABLE mg_core
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED BY (plant_code, material_code)
AS
SELECT
    date_trunc('month', mg.dt_posting) AS mon,
    mg.plant_code,
    mg.material_code,
    mg.uom_code,
    mg.order_code,

    -- предварительно считаем «подпись» количества для menge11
    CASE WHEN mg.debit_or_credit_code = 'H'
         THEN mg.quantity * -1
         ELSE mg.quantity
    END AS qty11_signed,

    -- базовый кандидат для menge22 (WE, receipt NULL, не в BWRT)
    CASE WHEN mg.operation_type_code = 'WE'
              AND mg.receipt_type_code IS NULL
              AND pbw.range_low_value IS NULL
         THEN CASE WHEN mg.debit_or_credit_code = 'H'
                   THEN mg.quantity * -1
                   ELSE mg.quantity
              END
         ELSE 0
    END AS qty22_base
FROM dds.material_movement_document mg
LEFT JOIN params_bwrt pbw
  ON pbw.range_low_value = mg.movement_type_code
-- фильтры вниз по потоку
WHERE mg.posting_rule_code IS NOT NULL
  AND mg.sales_document_number_of_valuated_stock_code IS NULL
  AND mg.wbs_element_of_valuated_stock_code = '00000000';

ANALYZE mg_core;

-- ===========================
-- 4) Достраиваем признаки по заказам одноразовыми лёгкими JOIN и агрегируем
--    (сначала сумма по mg_core + order_union/auart/autyp, потом — джойн к большим справочникам)
-- ===========================
CREATE TEMP TABLE mg_agg
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED BY (plant_code, material_code)
AS
SELECT
    c.mon,
    c.plant_code,
    c.material_code,
    c.uom_code,

    SUM(c.qty11_signed) AS menge11,

    -- обнуляем qty22_base если выполнено спец-условие по заказу:
    -- order_code NOT NULL AND auart MATCH AND autyp NOT MATCH
    SUM(
        CASE
            WHEN c.order_code IS NOT NULL
                 AND pauart.range_low_value IS NOT NULL
                 AND pautyp.range_low_value IS NULL
            THEN 0
            ELSE c.qty22_base
        END
    ) AS menge22
FROM mg_core c
LEFT JOIN order_union ou
  ON ou.order_code = c.order_code
LEFT JOIN params_auart pauart
  ON pauart.range_low_value = ou.order_type_code
LEFT JOIN params_autyp pautyp
  ON pautyp.range_low_value = ou.order_category_code
GROUP BY c.mon, c.plant_code, c.material_code, c.uom_code;

ANALYZE mg_agg;

-- ===========================
-- 5) Финальный шаг: подцепляем m-by-plant и material (после агрегации!) + фильтр Z_MATKL
-- ===========================
CREATE TEMP TABLE pt1_material_sum (
    spmon   varchar,
    werks   varchar,
    matnr   varchar,
    menge11 numeric(28,10),
    menge22 numeric(28,10),
    meins   varchar,
    ekgrp   varchar,
    dispo   varchar
)
WITH (appendonly=true, orientation=column, compresstype=zstd, compresslevel=3)
DISTRIBUTED RANDOMLY;

INSERT INTO pt1_material_sum
SELECT
    to_char(a.mon, 'YYYY-MM')                     AS spmon,
    a.plant_code                                  AS werks,
    a.material_code                               AS matnr,
    a.menge11,
    a.menge22,
    a.uom_code                                    AS meins,
    mbp.purchase_group_code                       AS ekgrp,
    mbp.material_planning_controller_code         AS dispo
FROM mg_agg a
JOIN dict_dds.material_by_plant mbp
  ON mbp.material_code = a.material_code
 AND mbp.plant_code    = a.plant_code
JOIN dict_dds.material m
  ON m.material_code = a.material_code
LEFT JOIN params_z_matkl pz
  ON pz.range_low_value = m.material_group_code
WHERE pz.range_low_value IS NULL;  -- эквивалент исходному фильтру

ANALYZE pt1_material_sum;
