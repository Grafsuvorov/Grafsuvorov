CREATE TEMP TABLE for_1055 ON COMMIT DROP AS
SELECT
  vbss.delivery_code AS vbeln,
  MIN(vbsk.dt_end_of_free_storage) AS dt_end_of_free_storage
FROM dds.logistics_document_and_delivery_relationship vbss
LEFT JOIN dds.release vbsk
  ON vbsk.release_code = vbss.logistics_document_code
 AND vbsk.release_type_code = 'X'
LEFT JOIN dict_dds.settings_and_parameters_sap frel_kod
  ON vbsk.release_template_code = frel_kod.range_low_value
WHERE frel_kod.abap_program_code = '/RUSAL/SD2921M'
  AND frel_kod.parameter_code = 'FREL_KOD'
  AND frel_kod.range_sign_code = 'I'
  AND frel_kod.range_option_code = 'EQ'
  AND vbsk.release_code IS NOT NULL
  AND vbsk.dt_end_of_free_storage IS NOT NULL
GROUP BY vbss.delivery_code
DISTRIBUTED REPLICATED;

-- =====================================================================
-- 3) Дедуп по сабам (EU/AS) и «лучший поставщик XR>ZU» (реплиц)
-- =====================================================================

CREATE TEMP TABLE sabadr_eu_best ON COMMIT DROP AS
SELECT *
FROM (
  SELECT
    s.*,
    ROW_NUMBER() OVER(
      PARTITION BY s.sales_bundle_code, s.transport_route_destination_hub_code
      ORDER BY s.dt_transportation_stage_start DESC NULLS LAST, s.delivery_code DESC
    ) AS rn
  FROM dds.sales_bundle_and_delivery_relationship s
  JOIN stage_eu se ON se.stage_code = s.transportation_stage_code
  WHERE s.is_deleted_code IS NULL
) q
WHERE rn = 1
DISTRIBUTED BY (sales_bundle_code);

CREATE TEMP TABLE sabadr_as_best ON COMMIT DROP AS
SELECT *
FROM (
  SELECT
    s.*,
    ROW_NUMBER() OVER(
      PARTITION BY s.sales_bundle_code, s.transport_route_destination_hub_code
      ORDER BY s.dt_transportation_stage_start DESC NULLS LAST, s.delivery_code DESC
    ) AS rn
  FROM dds.sales_bundle_and_delivery_relationship s
  JOIN stage_as sa ON sa.stage_code = s.transportation_stage_code
  WHERE s.is_deleted_code IS NULL
) q
WHERE rn = 1
DISTRIBUTED BY (sales_bundle_code);

CREATE TEMP TABLE supplier_final ON COMMIT DROP AS
SELECT sales_document_code, supplier_code
FROM (
  SELECT
    sdr.sales_document_code,
    sdr.supplier_code,
    ROW_NUMBER() OVER(
      PARTITION BY sdr.sales_document_code
      ORDER BY (sdr.counterparty_role_code = 'XR') DESC, sdr.counterparty_role_code
    ) AS rn
  FROM dds.sales_document_counterparty_role sdr
  WHERE sdr.counterparty_role_code IN ('XR','ZU')
) x
WHERE rn = 1
DISTRIBUTED REPLICATED;

DROP TABLE IF EXISTS base;
CREATE TEMP TABLE base ON COMMIT DROP AS
WITH bounds AS (
  SELECT (SELECT MIN(dt_balance) FROM date_for_vitr) AS min_dt,
         (SELECT MAX(dt_balance) FROM date_for_vitr) AS max_dt
),
prep AS (
  SELECT
    t4.sales_bundle_code,
    lake.SD_000003 AS plant_producer_delivery_code,
    lake.SD_000110 AS plant_producer_delivery_position_code,
    lake.SD_000001 AS initial_delivery_code,
    lake.SD_000143 AS material_code,
    lake.SD_000010 AS dt_shipment,
    lake.SD_000013 AS railcar_code,
    lake.SD_000014 AS transport_bill_code,
    lake.SD_000151 AS uni,
    lake.SD_000028 AS transport_type_code,
    lake.SD_000004 AS batch_code,
    lake.SD_000024 AS dt_arrival_rus_port_raw,

    t4.dt_transportation_stage_start_p,
    t4.dt_transportation_stage_start_r,
    t4.knote,
    t4.transportation_inbound_delivery_position_code,
    t4.transportation_outbound_delivery_position_code,
    t4.delivery_code_le_p,
    t4.delivery_code_le_r,
    lake.delivery_number_sales,

    fw1.market_region1_code,
    (fw2.transport_hub_code IS NOT NULL)::int AS is_rus_port,

    eu.delivery_code  AS vbeln_eu,
    asia.delivery_code AS vbeln_as,

    eu.dt_transportation_stage_start   AS sab_eu_dt_start,
    asia.dt_transportation_stage_start AS sab_as_dt_start,

    CASE
      WHEN fw1.market_region1_code = '03' AND eu.delivery_code IS NOT NULL  THEN eu.delivery_code
      WHEN fw1.market_region1_code = '04' AND asia.delivery_code IS NOT NULL THEN asia.delivery_code
      WHEN fw2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
    END AS delivery_for_join,

    CASE
      WHEN fw1.market_region1_code IN ('03','04') AND (eu.delivery_code IS NOT NULL OR asia.delivery_code IS NOT NULL) THEN NULL
      WHEN fw2.transport_hub_code IS NOT NULL THEN '1'
    END AS wwgsg_01,

    CASE
      WHEN fw1.market_region1_code IN ('03','04') AND (eu.delivery_code IS NOT NULL OR asia.delivery_code IS NOT NULL) THEN NULL::date
      WHEN fw2.transport_hub_code IS NOT NULL THEN vbsk.dt_bill_of_lading
    END AS wwgsg_01_date,

    sf.supplier_code AS supplier_code_final,

    COALESCE(eu.dt_transportation_stage_start, asia.dt_transportation_stage_start) AS lddat_4798,
    COALESCE(eu.delivery_code, asia.delivery_code)                                 AS vbeln_4798

  FROM dm_calc.sales_bundle_transport_hub_turnover_sdt0004 t4
  LEFT JOIN lake ON lake.sales_bundle_code = t4.sales_bundle_code AND lake.rn = 1
  LEFT JOIN forwapd_1 fw1 ON fw1.transport_hub_code = t4.knote
  LEFT JOIN forwapd_2 fw2 ON fw2.transport_hub_code = t4.knote

  LEFT JOIN sabadr_eu_best eu
         ON eu.sales_bundle_code = t4.sales_bundle_code
        AND eu.transport_route_destination_hub_code = t4.knote
        AND eu.dt_transportation_stage_start BETWEEN t4.dt_transportation_stage_start_p
                                               AND COALESCE(t4.dt_transportation_stage_start_r, CURRENT_DATE)

  LEFT JOIN sabadr_as_best asia
         ON asia.sales_bundle_code = t4.sales_bundle_code
        AND asia.transport_route_destination_hub_code = t4.knote
        AND asia.dt_transportation_stage_start BETWEEN t4.dt_transportation_stage_start_p
                                                 AND COALESCE(t4.dt_transportation_stage_start_r, CURRENT_DATE)

  LEFT JOIN dds.bill_of_lading vbsk
         ON vbsk.bill_of_lading_code = lake.SD_000040
        AND vbsk.bill_of_lading_type_code = 'Y'

  LEFT JOIN supplier_final sf ON sf.sales_document_code = t4.delivery_code_le_p
  CROSS JOIN bounds b
  WHERE t4.dt_transportation_stage_start_p <= b.max_dt
    AND (t4.dt_transportation_stage_start_r IS NULL OR t4.dt_transportation_stage_start_r >= b.min_dt)
    AND t4.dt_transportation_stage_start_p <> COALESCE(t4.dt_transportation_stage_start_r, DATE '2299-12-31')
    AND t4.transport_route_destination_hub_code LIKE 'G%'
    AND (
      CASE
        WHEN fw1.market_region1_code = '03' AND eu.delivery_code IS NOT NULL THEN eu.delivery_code
        WHEN fw1.market_region1_code = '04' AND asia.delivery_code IS NOT NULL THEN asia.delivery_code
        WHEN fw2.transport_hub_code IS NOT NULL THEN lake.delivery_number_sales
      END
    ) IS NOT NULL
)
SELECT
  p.sales_bundle_code,
  lake.SD_000003 AS plant_producer_delivery_code,
  lake.SD_000110 AS plant_producer_delivery_position_code,
  lake.SD_000001 AS initial_delivery_code,
  lake.SD_000143 AS material_code,
  lake.SD_000010 AS dt_shipment,
  lake.SD_000013 AS railcar_code,
  lake.SD_000014 AS transport_bill_code,
  lake.SD_000151 AS uni,
  maspe.shape_code AS material_shape_code,
  lake.SD_000028 AS transport_type_code,
  mater.etsng_code AS etsng_code,
  sabu.sales_bundle_net_weight / 1000.0 AS sales_bundle_net_weight,
  sabu.sales_bundle_gross_weight / 1000.0 AS sales_bundle_gross_weight,
  lake.SD_000004 AS batch_code,
  t4.dt_transportation_stage_start_r AS dt_arrival,

  dedopo_1.plant_producer_code AS receiving_plant_code,
  lake.delivery_number_sales    AS sales_delivery_code,

  CASE WHEN fw1.transport_hub_code IS NOT NULL THEN t4.delivery_code_le_p END AS transportation_inbound_delivery_code,

  CASE
    WHEN fw1.market_region1_code = '03' AND p.vbeln_eu IS NOT NULL  THEN p.vbeln_eu
    WHEN fw1.market_region1_code = '04' AND p.vbeln_as IS NOT NULL THEN p.vbeln_as
    WHEN p.is_rus_port = 1 THEN lake.delivery_number_sales
  END AS delivery_for_storage_calculation_code,

  CASE
    WHEN p.market_region1_code IN ('03','04') AND (p.vbeln_eu IS NOT NULL OR p.vbeln_as IS NOT NULL) THEN NULL
    WHEN p.is_rus_port = 1 THEN '1'
  END AS wwgsg_01,

  CASE
    WHEN p.market_region1_code IN ('03','04') AND (p.vbeln_eu IS NOT NULL OR p.vbeln_as IS NOT NULL) THEN NULL::date
    WHEN p.is_rus_port = 1 THEN p.wwgsg_01_date
  END AS wwgsg_01_date,

  t4.transportation_inbound_delivery_position_code AS transportation_inbound_delivery_position_code,
  CASE
    WHEN p.market_region1_code IN ('03','04') AND (p.vbeln_eu IS NOT NULL OR p.vbeln_as IS NOT NULL)
      THEN t4.transportation_inbound_delivery_position_code
    ELSE '10'
  END AS delivery_position_for_storage_calculation_code,

  p.supplier_code_final AS forwarder_code,

  CASE
    WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 = 'TL04' THEN 'Логика РФ-контейнеры'
    WHEN likp.delivery_type_code <> 'ZLE' AND lake.SD_000028 <> 'TL04' THEN 'Логика РФ-балк'
    WHEN likp.delivery_type_code  = 'ZLE' AND asia_log.range_low_value IS NOT NULL AND asia.counterparty_code IS NULL THEN 'Логика Азия'
    WHEN likp.delivery_type_code  = 'ZLE' AND euro_1.counterparty_code IS NOT NULL THEN 'Логика Европа'
    WHEN likp.delivery_type_code  = 'ZLE' AND euro_2.counterparty_code IS NOT NULL AND euro_log.range_low_value IS NOT NULL THEN 'Логика Европа'
    WHEN likp.delivery_type_code  = 'ZLE' AND svh.counterparty_code  IS NOT NULL THEN 'Логика СВХ'
    ELSE 'Логика не определена'
  END AS storage_cost_calculation_type_name,

  sbudr_base.transport_route_departure_hub_code AS transport_departure_hub_code,
  sbudr_base.transport_route_destination_hub_code AS transport_destination_hub_code,
  sbudr_base.is_final_transportation_stage_code  AS SD_000902,
  sbudr_base.transportation_stage_code           AS SD_000903,
  for_LE_001035.cnt                              AS SD_000905,

  lake.SD_000482, lake.SD_000041, lake.SD_000042, lake.SD_000047, lake.SD_000048,
  lake.SD_000054, lake.SD_000049, lake.SD_000067, lake.SD_000649, lake.SD_000044, lake.SD_000040,

  t4.delivery_code_le_r                             AS SD_000516,
  t4.transportation_outbound_delivery_position_code AS SD_000897,
  likp.weight_net                                   AS ntgew,
  likp.weight_net_with_wirerod                      AS btgew,
  t4.knote                                          AS SD_000420,
  likp.dt_loaded                                    AS likp_lddat,

  CASE
    WHEN p.market_region1_code IN ('03','04') AND (p.vbeln_eu IS NOT NULL OR p.vbeln_as IS NOT NULL) THEN NULL
    ELSE lake.SD_000024
  END AS dt_arrival_to_russian_port,

  t4.dt_transportation_stage_start_p,
  COALESCE(t4.dt_transportation_stage_start_r, CURRENT_DATE) AS dt_transportation_stage_start_r,
  CONCAT(t4.sales_bundle_code, t4.delivery_code, t4.delivery_position_code) AS id_for_unique,

  p.lddat_4798,
  p.vbeln_4798,

  for_1055.dt_end_of_free_storage AS dt_storage_end_in_release

FROM prep p
JOIN dm_calc.sales_bundle_transport_hub_turnover_sdt0004 t4
  ON t4.sales_bundle_code = p.sales_bundle_code
LEFT JOIN lake ON lake.sales_bundle_code = p.sales_bundle_code AND lake.rn = 1
LEFT JOIN dds.sales_bundle sabu ON sabu.sales_bundle_code = p.sales_bundle_code
LEFT JOIN dict_dds.material mater ON mater.material_code = lake.SD_000143
LEFT JOIN dict_dds.material_specification maspe ON maspe.material_code = lake.SD_000143
LEFT JOIN dds.delivery_document_position dedopo_1 ON dedopo_1.delivery_code = lake.delivery_number_sales

LEFT JOIN dds.sales_bundle_and_delivery_relationship sbudr_base
  ON sbudr_base.sales_bundle_code = t4.sales_bundle_code
 AND sbudr_base.delivery_code      = t4.delivery_code_le_p
 AND sbudr_base.delivery_position_code = t4.transportation_inbound_delivery_position_code
 AND sbudr_base.is_deleted_code IS NULL

LEFT JOIN dds.delivery_document_header likp ON likp.delivery_code = p.delivery_for_join

LEFT JOIN asia      ON asia.counterparty_code   = p.supplier_code_final
LEFT JOIN euro_1    ON euro_1.counterparty_code = p.supplier_code_final
LEFT JOIN euro_2    ON euro_2.counterparty_code = p.supplier_code_final
LEFT JOIN svh       ON svh.counterparty_code    = p.supplier_code_final
LEFT JOIN asia_log  ON asia_log.range_low_value = p.supplier_code_final
LEFT JOIN euro_log  ON euro_log.range_low_value = p.supplier_code_final

LEFT JOIN for_LE_001035 ON for_LE_001035.delivery_code = t4.delivery_code_le_p
                        AND for_LE_001035.batch_code   = lake.SD_000004
LEFT JOIN for_1055 ON for_1055.vbeln = lake.delivery_number_sales
LEFT JOIN forwapd_1 fw1 ON fw1.transport_hub_code = t4.knote
DISTRIBUTED BY (sales_bundle_code, plant_producer_delivery_code);

      AND b15.for_1076_2_counterparty_code IS NOT NULL
      THEN b15.likp_lddat
    WHEN b15.storage_cost_calculation_type_name = 'Логика СВХ'
      AND b15.sbudr_2_dt_transportation_stage_start IS NOT NULL
      AND b15.sbudr_2_dt_transportation_stage_start <= d.dt_balance
      THEN b15.sbudr_2_dt_transportation_stage_start
    WHEN b15.storage_cost_calculation_type_name = 'Логика СВХ'
      THEN d.dt_balance
  END AS dt_for_price_search_in_purchase_contract,

  CASE
    WHEN b15.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк') THEN b15.SD_000649
    WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') AND b15.SD_000049 < d.dt_balance THEN b15.SD_000054
    WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') THEN b15.SD_000044
  END AS port_of_discharge_code,

  CASE
    WHEN b15.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика Азия','Логика СВХ') THEN 'S49'
    WHEN b15.storage_cost_calculation_type_name = 'Логика РФ-балк' AND stmatl.storage_method_code = '1' THEN 'S49/3'
    WHEN b15.storage_cost_calculation_type_name = 'Логика РФ-балк' THEN 'S49/2'
    WHEN b15.storage_cost_calculation_type_name = 'Логика Европа' AND f1082_1.range_low_value IS NOT NULL THEN f1082_2.res
    WHEN b15.storage_cost_calculation_type_name = 'Логика Европа' THEN f1082_3.res
  END AS service_number,

  b15.weight_for_1090_umno,
  b15.weight_for_1090_razdel,
  b15.SD_000420,
  b15.id_for_unique,

  ROW_NUMBER() OVER(
    PARTITION BY b15.id_for_unique, d.dt_balance
    ORDER BY b15.lddat_4798 DESC NULLS LAST, b15.vbeln_4798 DESC NULLS LAST
  ) AS unique_row

FROM base_1_5 b15
JOIN date_for_vitr d
  ON d.dt_balance BETWEEN b15.dt_transportation_stage_start_p AND b15.dt_transportation_stage_start_r

LEFT JOIN dict_dds.location_sales losa
  ON b15.storage_cost_calculation_type_name = 'Логика РФ-балк'
 AND losa.transport_hub_code = CASE
   WHEN b15.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк') THEN b15.SD_000649
   WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') AND b15.SD_000049 < d.dt_balance THEN b15.SD_000054
   WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') THEN b15.SD_000044
 END

LEFT JOIN dict_dds.storage_method_at_tsw_location stmatl
  ON b15.storage_cost_calculation_type_name = 'Логика РФ-балк'
 AND stmatl.tsw_location_code = losa.location_code
 AND stmatl.material_shape_code = CASE
   WHEN b15.storage_cost_calculation_type_name IN ('Логика РФ-контейнеры','Логика РФ-балк') THEN b15.SD_000649
   WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') AND b15.SD_000049 < d.dt_balance THEN b15.SD_000054
   WHEN b15.storage_cost_calculation_type_name IN ('Логика Азия','Логика Европа') THEN b15.SD_000044
 END

LEFT JOIN for_1082_1 f1082_1
  ON b15.storage_cost_calculation_type_name = 'Логика Европа'
 AND f1082_1.range_low_value = b15.material_shape_code
LEFT JOIN for_1082_2 f1082_2
  ON b15.storage_cost_calculation_type_name = 'Логика Европа' AND f1082_2.rn = 1
LEFT JOIN for_1082_3 f1082_3
  ON b15.storage_cost_calculation_type_name = 'Логика Европа' AND f1082_3.rn = 1

LEFT JOIN dict_dds.transportation_warehouse_storage_area trwsa
  ON trwsa.plant_code = b15.receiving_plant_code
 AND trwsa.transport_hub_code = b15.warehouse_code
 AND trwsa.transportation_stage_code = b15.transportation_stage_code

WHERE
  (b15.wwgsg_01 IS NULL OR (b15.wwgsg_01 IS NOT NULL AND COALESCE(b15.wwgsg_01_date, DATE '9999-12-31') >= d.dt_balance))
  AND (b15.lddat_4798 <= d.dt_balance OR b15.wwgsg_01 IS NOT NULL)
DISTRIBUTED BY (sales_bundle_code);
